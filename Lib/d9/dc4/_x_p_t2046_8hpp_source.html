43 /** Constructor for XPT2046.
44 * @param config Configuration structure with pin handles.
45 * @param parameters Calibration and settings parameters.

47 XPT2046(const Config & config, const Parameters & parameters)
48 : interface{
49 ExtInterrupt{config.nirq},
50 PinOutput{config.mosi},
51 PinOutput{config.sck},
52 PinOutput{config.nss},
53 PinInput{config.miso}},

55 position{-1, -1},
56 isPressed(false),

59 const int32_t yRange = parameters.yMax - parameters.yMin;
60 assert(xRange > 0);
61 assert(yRange > 0);
62 assert(parameters.width > 0);
63 assert(parameters.height > 0);

65 scaleMultiplierX = static_cast<int32_t>((static_cast<int64_t>(parameters.width) << 16) / xRange);

68 interface.nirq.bind([this](const Lib::HAL::GPIO::ExtInterrupt *) { queueButtonPressed.overwriteFromISR(true); });
69 interface.sck.write(1);
70 interface.nss.write(1);

90 /** Reads the current touch position.

152 /** Handles the initial touch press event and debouncing. */
153 void handleTouchPress() {
154 isPressed = false;
155 bool pinState = false;
156 queueButtonPressed.receive(pinState, portMAX_DELAY);
157 RTOS::delay(5);
158
159 integratorX.resetWindow();
160 integratorY.resetWindow();
161 position.x = -1;
162 position.y = -1;
163
164 timeoutPinDebounce.set(parameters.debounceMs);
165 while (!timeoutPinDebounce.isExpired()) {
166 if (interface.nirq.read()) timeoutPinDebounce.set(parameters.debounceMs);
167 RTOS::Task::delay(1);
168 }
169 }

171 /** Writes a byte to SPI.
172 * @param data The byte to write.
173 */
174 inline void writeSpiByte(uint8_t data) {
175 for (uint8_t bit = 8; bit > 0; --bit) {
176 interface.mosi.write(data & 0x80);
177 data <<= 1;
178 interface.sck.write(0);
179 __NOP();
180 interface.sck.write(1);
181 __NOP();
182 }
183 }
184
185 /** Reads ADC value from specified channel.
186 * @param command The command byte to send.
187 * @return The 12-bit ADC value.
188 */
189 inline uint16_t readAdc(uint8_t command) {
190 uint16_t adc = 0;
191 interface.nss.write(0);
192 writeSpiByte(command);
193 for (uint8_t bit = 12; bit > 0; --bit) {
194 interface.sck.write(0);
195 __NOP();
196 adc = (adc << 1) | interface.miso.read();
197 interface.sck.write(1);
198 __NOP();
199 }
200 interface.nss.write(1);
201 return adc;
202 }

204 /** Applies rotation transformation to position. */
205 inline void applyRotation() {
206 const auto width = static_cast<int32_t>(parameters.width);
207 const auto height = static_cast<int32_t>(parameters.height);
208 switch (rotation) {
209 case 1:
210 position.x = width - position.x;
211 break;
212 case 2:
213 position.x = width - position.x;
214 position.y = height - position.y;
215 break;
216 case 3:
217 position.y = height - position.y;
218 break;
219 case 4:
220 position.x = width - position.x;
221 position.y = height - position.y;
222 break;
223 case 0:
224 default:
225 break;
226 }
227 }

229 /** Reads and integrates ADC values.
230 * @return True if integration window is full, false otherwise.
231 */
232 inline bool readAndIntegrateAdc() {
233 rawPosition.x = integratorX.processValue(readAdc(CMD_READ_X));
234 rawPosition.y = integratorY.processValue(readAdc(CMD_READ_Y));
235 if (!integratorX.isWindowFull()) {
236 RTOS::Task::delay(5);
237 return false;
238 }
239 return true;
240 }
241
242 /** Clamps and scales raw position to display coordinates. */
243 inline void clampAndScalePosition() {
244 if (rawPosition.x < parameters.xMin) {
245 rawPosition.x = parameters.xMin;
246 } else if (rawPosition.x > parameters.xMax) {
247 rawPosition.x = parameters.xMax;
248 }
249
250 if (rawPosition.y < parameters.yMin) {
251 rawPosition.y = parameters.yMin;
252 } else if (rawPosition.y > parameters.yMax) {
253 rawPosition.y = parameters.yMax;
254 }
255
256 rawPosition.x -= parameters.xMin;
257 rawPosition.y -= parameters.yMin;
258 position.x = static_cast<int32_t>((static_cast<int64_t>(rawPosition.x) * scaleMultiplierX) >> 16);
259 position.y = static_cast<int32_t>((static_cast<int64_t>(rawPosition.y) * scaleMultiplierY) >> 16);
260 }
261
262 /** Processes touch data and updates position. */
263 inline void processTouch() {
264 if (!readAndIntegrateAdc()) return;
265 clampAndScalePosition();
266 isPressed = true;
267 applyRotation();
268 RTOS::Task::delay(40);
269 }
270
271 /** Handles touch while pressed. */
272 inline void handleTouchWhilePressed() {
273 while (!interface.nirq.read()) {
274 processTouch();
275 }
276 }
277 };
