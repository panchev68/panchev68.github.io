60 /** Process a new sample and update peak detection state.
61 * This method accepts a single integer sample and updates the internal
62 * sliding window and peak metrics. It is non-blocking and deterministic.
63 * @param sample Input sample value.
64 */
65 void addSample(int sample) {
66 constexpr size_t FIRST_INDEX = 0;
67 constexpr size_t LAST_INDEX = WINDOW_SIZE - 1;
68 constexpr size_t MIDDLE_INDEX = 1;
69 constexpr int SMOOTHING_FACTOR = 2;
70
71 if (sampleCount < WINDOW_SIZE) {
72 window[sampleCount++] = sample;
73 if (sampleCount < WINDOW_SIZE) return;
74 } else {
75 // Shift window left by one position (optimized for 3-element array)
76 window[FIRST_INDEX] = window[MIDDLE_INDEX];
77 window[MIDDLE_INDEX] = window[LAST_INDEX];
78 window[LAST_INDEX] = sample;
79 }
80
81 const int diff = window[LAST_INDEX] - window[FIRST_INDEX];
82 peakValue = diff + peakValue / SMOOTHING_FACTOR;
83
84 if (peakValue > risingPeak.value || risingPeak.position == 0)
85 risingPeak = PeakInfo{peakValue, sampleCount - 1};
86 if (peakValue < fallingPeak.value || fallingPeak.position == 0)
87 fallingPeak = PeakInfo{peakValue, sampleCount - 1};
88 }
