Lib::HTTP::Client

Demonstrates a GSM modem adapted to 'HttpTransport', an HTTPS POST through 'WolfSSLTransport' and a plain HTTP GET.

Demonstrates a GSM modem adapted to 'HttpTransport', an HTTPS POST through 'WolfSSLTransport' and a plain HTTP GET.'GsmStream' gives the modem the two methods the concept requires; it inherits nothing, and its 'connect()' and 'disconnect()' are its own, since neither 'Client' nor 'WolfSSLTransport' opens or closes the link. The modem driver has no TCP API of its own, so the adapter drives the TCP AT commands through the public 'sendCommand()', 'sendCommandWithData()', 'writeCommand()' and 'waitForLine()' calls; the command syntax and the '+CIPRXGET' framing are written for the A76xx TCP/IP commands and must be checked against the module manual before use. For HTTPS the TLS session runs over that stream and the client runs over the session; for plain HTTP the client uses the stream directly. The host name appears twice for different reasons: 'open()' checks it against the server certificate, the request sends it in the Host header. 'response.body' views into 'buffer', so it is valid only while 'buffer' lives. The requests block the calling task; call them from a task, never from an ISR.

Important
Connect the raw link before 'open()' and disconnect it after 'close()': the handshake and the close notification both travel over it.
Every request sends 'Connection: close': reconnect the link, and for HTTPS rerun 'open()', before the next request.

Usage

HTTPS POST and plain HTTP GET over the modem

#include <Lib/BSP/GSM/A7670E.hpp>
#include <Lib/HTTP/Client.hpp>
#include <algorithm>
#include <array>
#include <chrono>
#include <cstddef>
#include <cstdint>
#include <optional>
#include <span>
#include <string_view>
using Modem = Lib::BSP::GSM::A7670E;
// Adapter giving the modem the HttpTransport shape. connect() and disconnect() are its own.
class GsmStream final {
public:
explicit GsmStream(Modem & modem) noexcept :
modem(modem) {
}
[[nodiscard]] bool connect(std::string_view host, uint16_t port) noexcept {
Modem::CommandBuilder command{commandBuffer};
command.append("AT+CIPOPEN=0,\"TCP\",\"");
command.append(host);
command.append("\",");
command.appendNumber(port);
return command.isComplete() && (modem.sendCommand(command.view(), CONNECT_TIMEOUT) == Modem::Result::OK);
}
void disconnect() noexcept {
static_cast<void>(modem.sendCommand("AT+CIPCLOSE=0", CLOSE_TIMEOUT));
}
[[nodiscard]] bool send(std::span<const uint8_t> data, size_t & sentLength) noexcept {
sentLength = 0U;
Modem::CommandBuilder command{commandBuffer};
command.append("AT+CIPSEND=0,");
command.appendNumber(static_cast<uint32_t>(data.size()));
if (!command.isComplete() || (modem.sendCommandWithData(command.view(), ">", data, SEND_TIMEOUT) != Modem::Result::OK)) {
return false;
}
sentLength = data.size(); // The whole block is accepted or the call fails.
return true;
}
[[nodiscard]] bool receive(std::span<uint8_t> buffer, size_t & receivedLength) noexcept {
receivedLength = 0U;
Modem::CommandBuilder command{commandBuffer};
command.append("AT+CIPRXGET=2,0,");
command.appendNumber(static_cast<uint32_t>(buffer.size()));
if (!command.isComplete() || !modem.writeCommand(command.view())) {
return false;
}
std::array<char, LINE_SIZE> line{};
const std::optional<size_t> length = modem.waitForLine(RECEIVE_PREFIX, line, RECEIVE_TIMEOUT);
if (!length) {
return true; // Nothing has arrived yet: 0 bytes with true.
}
// The line carries the requested and the confirmed length; the confirmed one is the byte count that follows.
const std::string_view fields = std::string_view{line.data(), *length}.substr(RECEIVE_PREFIX.size());
const size_t comma = fields.find(',');
const std::optional<uint32_t> available = (comma == std::string_view::npos) ? std::nullopt : Modem::Parse::parseNumber(fields.substr(comma + 1U));
if (!available) {
return false;
}
receivedLength = modem.readData(buffer.first(std::min<size_t>(*available, buffer.size())), RECEIVE_TIMEOUT);
static_cast<void>(modem.waitForLine("OK", line, RECEIVE_TIMEOUT));
return true;
}
private:
static constexpr std::chrono::milliseconds CONNECT_TIMEOUT{10000};
static constexpr std::chrono::milliseconds CLOSE_TIMEOUT{5000};
static constexpr std::chrono::milliseconds SEND_TIMEOUT{10000};
static constexpr std::chrono::milliseconds RECEIVE_TIMEOUT{500};
static constexpr std::string_view RECEIVE_PREFIX = "+CIPRXGET: 2,0,";
static constexpr size_t LINE_SIZE = 64U;
Modem & modem;
std::array<char, 160U> commandBuffer{};
};
static_assert(Lib::HTTP::HttpTransport<GsmStream>);
extern const std::span<const uint8_t> CLOUD_CA_CERTIFICATE; // PEM, embedded in the firmware.
constexpr std::string_view HOST = "api.example.com";
bool sendTelemetry(Modem & modem, std::span<const uint8_t> json) {
GsmStream stream(modem);
if (!stream.connect(HOST, 443U)) {
return false;
}
bool isAccepted = false;
Lib::HTTP::WolfSSLTransport<GsmStream> tls;
if (tls.open(stream, HOST, CLOUD_CA_CERTIFICATE)) {
Lib::HTTP::Client client(tls);
std::array<uint8_t, 512> buffer{};
Lib::HTTP::Response response;
const Lib::HTTP::Status status = client.post(HOST, "/v1/telemetry", "application/json", json, buffer, response);
isAccepted = (status == Lib::HTTP::Status::SUCCESS) && (response.statusCode == 200U);
tls.close(); // TLS first, the TCP link below still carries its close notification.
}
stream.disconnect();
return isAccepted;
}
uint16_t fetchStatus(Modem & modem) {
GsmStream stream(modem);
if (!stream.connect(HOST, 80U)) {
return 0U;
}
Lib::HTTP::Client client(stream); // Plain HTTP: the raw stream directly.
std::array<uint8_t, 512> buffer{};
Lib::HTTP::Response response;
const Lib::HTTP::Status status = client.get(HOST, "/v1/status", buffer, response);
stream.disconnect();
return (status == Lib::HTTP::Status::SUCCESS) ? response.statusCode : static_cast<uint16_t>(0U);
}
