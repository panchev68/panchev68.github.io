Lib::BSP::GSM::A7670E

Powers the modem, waits for the network, reads the signal, time and position, stores a setting in the modem flash and powers it off again.

Powers the modem, waits for the network, reads the signal, time and position, stores a setting in the modem flash and powers it off again.The driver does not know the board: the application passes the serial port, the 'PWRKEY', 'RESET', 'STATUS' and optional 'DTR' pins and a callback that switches the modem supply in one 'Config'. The functions are grouped: 'power()' for the supply, 'network()' for registration, time and position, and 'files()' for whole files. 'File' gives handle-based access for larger or partial transfers. A missing 'DTR' pin is passed as 'nullptr' and only disables UART sleep.

Important
Use the modem from one task only, and call 'power().off()' before cutting the supply by other means: it detaches from the network and lets the file system finish its writes.
Warning
'files().write()' erases and rewrites the file in the modem flash. Write only when the content changes.

Power cycle, network report and stored setting

#include <Lib/BSP/GSM/A7670E.hpp>
#include <array>
#include <chrono>
#include <cstdint>
#include <optional>
#include <string_view>
extern "C" UART_HandleTypeDef huart3; // Generated by CubeMX in main.c.
namespace {
using Modem = Lib::BSP::GSM::A7670E;
// Hardware of the board; the modem driver only holds references to it.
Lib::HAL::StreamSerialPort port({.handle = &huart3, .bufferSize = 512U});
Lib::HAL::GPIO::Output powerKey{PIN(MODEM_PWRKEY)};
Lib::HAL::GPIO::Output reset{PIN(MODEM_RESET)};
Lib::HAL::GPIO::Input status{PIN(MODEM_STATUS)};
Lib::HAL::GPIO::Output supplyEnable{PIN(MODEM_PWR_EN)};
// The driver calls this to switch the regulator of the modem.
void setSupply(bool isOn) {
supplyEnable.write(isOn);
}
Modem modem{{
.port = port,
.powerKey = powerKey,
.reset = reset,
.status = status,
.dtr = nullptr, // 'DTR' is not wired in this example, so UART sleep is unavailable.
.setSupply = setSupply,
}};
constexpr std::string_view SETTINGS_PATH = "C:/settings.bin";
} // namespace
void report() {
if (!modem.power().on()) {
return; // The modem did not answer; the supply is cut again by the driver.
}
if (modem.network().waitForRegistration(std::chrono::seconds{60})) {
const std::optional<uint8_t> signal = modem.network().getSignalQuality();
const std::optional<uint32_t> utc = modem.network().getTime();
const std::optional<Modem::Location> position = modem.network().getLocation();
// Use signal, utc and position here, then send the report with modem.sendCommandWithData() or HTTP commands.
static_cast<void>(signal);
static_cast<void>(utc);
static_cast<void>(position);
}
// A small setting survives power loss of the whole board; write it only when it changes.
const std::array<uint8_t, 4U> settings{1U, 2U, 3U, 4U};
std::array<uint8_t, 4U> stored{};
if ((modem.files().read(SETTINGS_PATH, stored) != stored.size()) || (stored != settings)) {
if (!modem.files().write(SETTINGS_PATH, settings)) {
// The modem refused the file; keep working with the value in RAM.
}
}
// Handle-based access: append a record to a log file.
Modem::File log{modem};
if (log.open("C:/log.bin", Modem::File::Mode::APPEND)) {
static_cast<void>(log.write(settings));
log.close();
}
modem.power().off(); // Detaches from the network, then cuts the supply.
}
