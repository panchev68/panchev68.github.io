RTOS::TickCounter

TickCounter Usage Example

This example demonstrates how to use the TickCounter class to measure elapsed time in a FreeRTOS task. It shows retrieving the current tick count, calculating elapsed ticks, and converting between milliseconds and ticks.

Example usage:
#include "Lib/RTOS/TickCounter.hpp"
#include "FreeRTOS.h"
#include "task.h"
void someTaskFunction(void *pvParameters) {
// Get the starting tick count
RTOS::TickCounter::TickType startTicks = RTOS::TickCounter::get();
// Simulate some work (e.g., delay for 100 ms)
vTaskDelay(RTOS::TickCounter::msToTicks(100));
// Get the current tick count
RTOS::TickCounter::TickType currentTicks = RTOS::TickCounter::get();
// Calculate elapsed ticks
RTOS::TickCounter::TickType elapsedTicks = RTOS::TickCounter::getElapsedTicksSince(startTicks);
// Convert elapsed ticks back to milliseconds
uint32_t elapsedMs = RTOS::TickCounter::ticksToMs(elapsedTicks);
// Use elapsedMs for further processing (e.g., logging or decision making)
// ...
// Infinite loop for task
for (;;) {
vTaskDelay(RTOS::TickCounter::msToTicks(1000)); // Delay 1 second
}
}
