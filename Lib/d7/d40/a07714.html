PGA460TPWRQ1 usage

How to read the distance from the PGA460 ultrasonic sensor with "Lib::BSP::Ultrasonic::PGA460TPWRQ1<IT>".

The driver does not own the UART port; pass the "Lib::HAL::UARTPort<DriverMode::IT>" wired to the PGA460 RXD/TXD pins. "readDistance()" blocks the calling task for about "listenDuration" plus two UART transfers and yields via "RTOS::delay()", so call it from a task, never from an ISR. The driver is not thread-safe: use one instance from a single task, or guard it with a mutex.

Hardware assumptions

  • Two-wire UART on RXD/TXD; the IO pin is unused (pulled up), so transmitted bytes are not echoed back.
  • TEST pin tied to GND at power-up, so the digital I/O runs at 3.3 V.
  • Factory "UART_ADDR" of 0; set "Config::uartAddress" only if the EEPROM address was reprogrammed.

Single measurement

#include <Lib/BSP/Ultrasonic/PGA460TPWRQ1.hpp>
void measureOnce(Lib::HAL::UARTPort<Lib::HAL::DriverMode::IT> & uartPort) noexcept {
Lib::BSP::Ultrasonic::PGA460TPWRQ1<Lib::HAL::DriverMode::IT> sensor({
.uart = uartPort,
.timeout = std::chrono::milliseconds(50),
.listenDuration = std::chrono::microseconds(15000),
.ambientTemperatureCelsius = 22.0F,
});
if (!sensor.open()) {
return;
}
if (const std::optional<float> distanceMeters = sensor.readDistance(); distanceMeters.has_value()) {
// Use *distanceMeters.
}
static_cast<void>(sensor.close());
}

Periodic measurement task

The first reading after power-up may be rejected because the diagnostic byte still carries a stale error flag; treat an empty result as a skipped sample, not a fatal error.

#include <Lib/BSP/Ultrasonic/PGA460TPWRQ1.hpp>
#include <RTOS.hpp>
class LevelMonitor final {
public:
explicit LevelMonitor(Lib::HAL::UARTPort<Lib::HAL::DriverMode::IT> & uartPort) noexcept :
sensor({
.uart = uartPort,
.timeout = std::chrono::milliseconds(50),
.listenDuration = std::chrono::microseconds(15000),
}),
task(RTOS::Task::Config("Level", 512, RTOS::Task::Priority::NORMAL), [this]() { run(); }) {
}
private:
static constexpr std::chrono::milliseconds PERIOD{1000};
static constexpr uint32_t MAX_CONSECUTIVE_FAILURES = 5;
Lib::BSP::Ultrasonic::PGA460TPWRQ1<Lib::HAL::DriverMode::IT> sensor; // Constructed before the task starts using it.
RTOS::Task task; // Must stay the last member.
void run() noexcept {
if (!sensor.open()) {
assert(!"LevelMonitor: failed to open sensor UART");
return;
}
uint32_t failures = 0;
while (true) {
if (const std::optional<float> distanceMeters = sensor.readDistance(); distanceMeters.has_value()) {
failures = 0;
// Publish *distanceMeters.
} else if (++failures >= MAX_CONSECUTIVE_FAILURES) {
failures = 0;
// Report a sensor fault.
}
RTOS::delay(PERIOD);
}
}
};

Non-default UART address

Lib::BSP::Ultrasonic::PGA460TPWRQ1<Lib::HAL::DriverMode::IT> sensor({
.uart = uartPort,
.timeout = std::chrono::milliseconds(50),
.listenDuration = std::chrono::microseconds(15000),
.uartAddress = 3, // Must match UART_ADDR programmed in the device EEPROM.
});
