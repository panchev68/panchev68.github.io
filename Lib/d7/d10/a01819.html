Lib::Helper::Convert::HexHelper

Example showing common usage of the lightweight conversion helpers provided by Lib::Helper::Convert::HexHelper. The example demonstrates converting a nibble to its ASCII hex representation, packing a 0..99 value into an 8-bit BCD, packing a decimal value into a 32-bit BCD, and converting an ASCII hex digit back to its numeric nibble.

#include <cstdint>
#include <iostream>
#include "Lib/Helper/Convert/HexHelper.hpp"
using Lib::Helper::Convert::HexHelper;
void example_HexHelper_usage() {
// Convert low nibble 0xA to ASCII 'A'
const char c = HexHelper::toAscii(0xAU);
std::cout << "Nibble 0xA -> " << c << "\n"; // Expected: A
// Pack decimal 42 into 8-bit BCD (0x42)
const uint8_t bcd8 = HexHelper::toBcd8(42U);
std::cout << "BCD8(42) -> 0x" << std::hex << int(bcd8) << std::dec << "\n"; // Expected: 0x42
// Pack decimal 1234 into 32-bit packed BCD
const uint32_t bcd32 = HexHelper::toBcd(1234U);
std::cout << "BCD32(1234) -> 0x" << std::hex << bcd32 << std::dec << "\n"; // Expected: 0x1234
// Convert ASCII 'F' to numeric nibble (15); a non-hex character yields HexHelper::INVALID_NIBBLE
const uint8_t nibble = HexHelper::fromAscii('F');
std::cout << "'F' -> " << int(nibble) << "\n"; // Expected: 15
}
