RTOS::TickService

TickService Example

This example demonstrates how to use the TickService to handle periodic tick events.

Example usage:

Create a client that inherits from TickService::ClientBase and implement handleTickEvent.

#include "TickService.hpp"
class MyTickClient : public RTOS::TickService::ClientBase {
public:
void handleTickEvent() override {
// Handle the tick event here
// For example, toggle an LED or update a counter
}
};
int main() {
MyTickClient client;
RTOS::TickService::instance().start(100); // Start with 100ms interval
// The service will run indefinitely
return 0;
}
