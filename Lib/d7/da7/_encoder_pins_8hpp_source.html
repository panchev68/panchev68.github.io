74 /** Reads debounced button state.
75 * @param debounce Debounce period in ms.
76 * @return `true` if pressed, `false` otherwise.
77 * @pre `debounce` should be reasonable (typically 10-50 ms).
78 */
79 bool getButtonState(uint32_t debounce = 20) {
80 if (debounce == 0) return !button.read();
81
82 bool state = button.read();
83 for (uint32_t i = 0; i < debounce; ++i) {
84 const bool currentState = button.read();
85 if (currentState != state) {
86 state = currentState;
87 i = 0;
88 }
89 vTaskDelay(1);
90 }
91 return !state;
92 }

94 /** Handles pin change event.
95 * @param pinNumber Pin number that triggered the event.
96 * @return `true` if handled, `false` otherwise.
97 * @pre Event queue must be opened.
98 */
99 bool handlePinChangedEvent(uint16_t pinNumber) {
100 // Use instance members instead of function-static variables so multiple
101 // EncoderPins instances behave independently and state is testable.
102 if (pinNumber == button.getPinNumber()) {
103 write(button.read() ? EventState::BUT_PULL : EventState::BUT_PUSH);
104 return true;
105 }
106 if (pinNumber == channel1.getPinNumber()) {
107 channel1.isEventPending = true;
108 if (channel2.isEventPending) {
109 channel2.isEventPending = false;
110 write(rotationDirection == channel1.read() ? EventState::CCW : EventState::CW);
111 }
112 return true;
113 }
114 if (pinNumber == channel2.getPinNumber()) {
115 channel2.isEventPending = true;
116 if (channel1.isEventPending) {
117 channel1.isEventPending = false;
118 rotationDirection = channel2.read();
119 }
120 return true;
121 }
122 return false;
123 }

176 /** Reads an event from the queue with timeout.
177 * @param timeout Timeout in ms.
178 * @return Event state.

180 EventState read(uint32_t timeout) {
181 EventState result;
182 return isOpen() && queue.receive(result, timeout) ? result : EventState::NONE;
183 }
184
185 /** Writes an event state to the queue.
186 * @param state Event state.
187 * @pre Queue must be opened.
188 */
189 void write(EventState state) {
190 if (!isOpen()) return;
191
192 if (state != lastState) {
193 lastState = state;
194 rotationEventCounter = 0;
195
196 if (state == EventState::BUT_PUSH) {
197 buttonStartTime = xTaskGetTickCount();
198 } else if (state == EventState::BUT_PULL) {
199 buttonHoldPeriod = xTaskGetTickCount() - buttonStartTime;
200 }
201 }
202
203 const bool isRotationEvent = (state == EventState::CCW || state == EventState::CW);
204 if (isRotationEvent && (++rotationEventCounter > pulseDivider)) {
205 rotationEventCounter = 0;
206 if (isReverse) {
207 state = (state == EventState::CCW) ? EventState::CW : EventState::CCW;
208 }
209 queue.sendToBackFromISR(state);
210 }
211 }
