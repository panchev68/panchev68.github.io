24 * Manages rotation and button events from a rotary encoder peripheral.

129 /** Reads debounced button state directly from GPIO.
130 *
131 * Performs debouncing by reading the button state repeatedly until it stabilizes
132 * for the specified debounce period. This is a blocking, synchronous operation
133 * suitable for initialization or standalone button state queries.
134 *
135 * @param debounce Debounce stability period in milliseconds (0 disables debouncing, default 20 ms).
136 *
137 * @return `true` if button is physically pressed (active low), `false` otherwise.
138 *
139 * @note This is a blocking function that delays during the debounce period.
140 * @note Suitable for initialization but not recommended for runtime event handling.
141 * @pre debounce should be reasonable (typically 10-50 ms for mechanical buttons).
142 */
143 bool getButtonState(uint32_t debounce = 20) {
144 if (debounce == 0) return !button.read();
145 bool state = button.read();
146 for (uint32_t i = 0; i < debounce; ++i) {
147 const bool currentState = button.read();
148 if (currentState != state) {
149 state = currentState;
150 i = 0;
151 }
152 vTaskDelay(1);
153 }
154 return !state;
155 }

157 /** Handles GPIO pin change event from interrupt handler.
158 *
159 * Processes GPIO pin change events and generates appropriate encoder events
160 * (rotation direction or button press/release). This method is called from an ISR
161 * and implements quadrature decoding for rotation detection. Button events are
162 * generated immediately; rotation events are filtered through the pulse divider.
163 *
164 * @param pinNumber GPIO pin number that triggered the interrupt event.
165 *
166 * @return `true` if the pin number matches an encoder pin, `false` if unrecognized.
167 *
168 * @pre Event queue must be opened before calling this function.
169 * @note Instance members are used for state tracking to support multiple
170 * independent EncoderPins instances and enable testability.
171 * @note Safe to call from ISR context; uses ISR-safe queue operations.
172 */
173 bool handlePinChangedEvent(uint16_t pinNumber) {
174 const uint16_t btnPin = button.getPinNumber();
175 if (pinNumber == btnPin) {
176 write(button.read() ? EventState::BUT_PULL : EventState::BUT_PUSH);
177 return true;
178 }
179 const uint16_t ch1Pin = channel1.getPinNumber();
180 if (pinNumber == ch1Pin) {
181 channel1.isEventPending = true;
182 if (channel2.isEventPending) {
183 channel2.isEventPending = false;
184 write(rotationDirection == channel1.read() ? EventState::CCW : EventState::CW);
185 }
186 return true;
187 }
188 const uint16_t ch2Pin = channel2.getPinNumber();
189 if (pinNumber == ch2Pin) {
190 channel2.isEventPending = true;
191 if (channel1.isEventPending) {
192 channel1.isEventPending = false;
193 rotationDirection = channel2.read();
194 }
195 return true;
196 }
197 return false;
198 }

280 /** Reads an event from the queue with timeout.
281 *
282 * Attempts to receive an event from the RTOS queue. Returns immediately if the queue
283 * is closed, providing a default NONE event indicating no activity.
284 *
285 * @param timeout Timeout in milliseconds for the receive operation.
286 *
287 * @return Event state from the queue, or NONE if timeout expires or queue is closed.
288 */
289 EventState read(uint32_t timeout) {
290 EventState result;
291 return isOpen() && queue.receive(result, timeout) ? result : EventState::NONE;
292 }

294 /** Writes an encoder event to the queue after processing.
295 *
296 * Processes the event state, applies pulse dividing for rotation events, and queues
297 * the result for delivery to the application. Handles button timing for hold duration
298 * tracking and applies rotation direction reversal if configured. Only queues rotation
299 * events that exceed the pulse divider threshold.
300 *
301 * @param state Event state to process and potentially queue.
302 *
303 * @pre Queue must be opened before calling this function.
304 *
305 * @note This method is safe to call from ISR context; uses ISR-safe queue operations.
306 */
307 void write(EventState state) {
308 if (!isOpen()) {
309 return;
310 }
311 if (state != lastState) {
312 lastState = state;
313 rotationEventCounter = 0;
314 if (state == EventState::BUT_PUSH) {
315 buttonStartTime = xTaskGetTickCount();
316 } else if (state == EventState::BUT_PULL) {
317 buttonHoldPeriod = xTaskGetTickCount() - buttonStartTime;
318 }
319 }
320 const bool isRotationEvent = (state == EventState::CCW || state == EventState::CW);
321 if (isRotationEvent && (++rotationEventCounter > pulseDivider)) {
322 rotationEventCounter = 0;
323 if (isReverse) {
324 state = (state == EventState::CCW) ? EventState::CW : EventState::CCW;
325 }
326 queue.sendToBackFromISR(state);
327 }
328 }
