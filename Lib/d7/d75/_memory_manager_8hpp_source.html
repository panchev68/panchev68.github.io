66 /** Allocates memory from FreeRTOS heap.
67 * This overload calls the central implementation and discards the status code.
68 * @param size Number of bytes to allocate.
69 * @return Pointer to the user area (not including internal header) on success, or `nullptr` on failure.
70 * @pre `size` must be a valid allocation size as checked by `isValidSize`.
71 * @post On success the returned pointer must be freed with `deallocate()`.
72 * @note Use the overload taking a `Status &` if you need the detailed failure reason.
73 */
74 static void * allocate(size_t size) {
75 Status dummyStatus;
76 return allocateImpl(size, &dummyStatus);
77 }

79 /** Allocates memory from FreeRTOS heap with status code.
80 * Central entry point for allocations. The returned pointer points to the user area (header is internal).
81 * @param size Number of bytes to allocate.

83 * @return Pointer to user memory on success, or `nullptr` on failure. On failure `status` contains the reason.

117 /** Checks if size is valid for allocation.
118 * Validates requested allocation size against internal header overhead and configured total heap (if available).
119 * @param size Number of bytes requested by the caller.
120 * @return `true` if the requested size is non-zero, does not overflow when adding the internal header, and
121 * (if `configTOTAL_HEAP_SIZE` is defined) does not exceed the configured heap minus header.
122 * @pre This function is `constexpr` and may be evaluated at compile time when possible.
123 */
124 static constexpr bool isValidSize(size_t size) {
125 // Ensure non-zero, fits with header, and does not exceed configured heap minus header.
126 if (size == 0) return false;
127 if (size > (kMaxSize - kHeaderSize)) return false;
128#if defined(configTOTAL_HEAP_SIZE)
129 const size_t totalHeap = static_cast<size_t>(configTOTAL_HEAP_SIZE);
130 const size_t maxAlloc = (totalHeap > kHeaderSize) ? (totalHeap - kHeaderSize) : 0u;
131 return size <= maxAlloc;
132#else
133 return true;
134#endif
135 }

137 /** Gets heap usage as a percentage (0-100).
138 * Uses the configured `configTOTAL_HEAP_SIZE` (if defined) and the FreeRTOS free-heap query to compute usage.
139 * @return Heap usage percent in whole numbers (0-100). Returns 0 when `configTOTAL_HEAP_SIZE` is not defined.
140 * @note The calculation avoids division by zero and clamps the result to 100.
141 */
142 static uint8_t getHeapUsagePercent() {
143#if defined(configTOTAL_HEAP_SIZE)
144 const size_t totalHeap = static_cast<size_t>(configTOTAL_HEAP_SIZE);
145 if (totalHeap == 0) return 0;
146 const size_t freeHeap = getFreeHeapSize();
147 const size_t usedHeap = (totalHeap > freeHeap) ? (totalHeap - freeHeap) : 0u;
148 const uint64_t percent = (static_cast<uint64_t>(usedHeap) * 100ull) / static_cast<uint64_t>(totalHeap);
149 return static_cast<uint8_t>(percent > 100ull ? 100u : percent);
150#else
151 return 0;
152#endif
153 }

173 /** Centralized allocation implementation.
174 * Returns pointer to user area or nullptr and sets status if provided.
175 */
176 static void * allocateImpl(size_t size, Status * status) {
177 if (!isValidSize(size)) {
178 if (status) { *status = (size == 0) ? Status::INVALID_SIZE : Status::SIZE_TOO_LARGE; }
179#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__) || defined(__DOXYGEN__)
180 diagnostic.failedAllocations++;
181#endif
182 return nullptr;
183 }
184
185 const size_t totalSize = size + kHeaderSize;
186 void * raw = pvPortMalloc(totalSize);
187 if (!raw) {
188 if (status) *status = Status::OUT_OF_MEMORY;
189 handleAllocationFailure();
190#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__) || defined(__DOXYGEN__)
191 diagnostic.failedAllocations++;
192#endif
193 return nullptr;
194 }
195
196 auto * hdr = static_cast<AllocationHeader *>(raw);
197 hdr->size = size;
198 hdr->magic = ALLOC_HEADER_MAGIC;
199
200 if (status) *status = Status::SUCCESS;
201
202#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__) || defined(__DOXYGEN__)
203 recordAlloc(size);
204#endif
205
206 return static_cast<void *>(static_cast<uint8_t *>(raw) + kHeaderSize);
207 }
208
209 /** Centralized deallocation implementation.
210 * Frees memory and sets status if provided.
211 */
212 static void deallocateImpl(void * pointer, Status * status) {
213 if (!pointer) {
214 if (status) *status = Status::NULL_POINTER;
215 return;
216 }
217
218 auto * raw = static_cast<uint8_t *>(pointer) - kHeaderSize;
219 auto * hdr = reinterpret_cast<AllocationHeader *>(raw);
220
221 if (hdr->magic == ALLOC_HEADER_MAGIC) {
222 const size_t allocSize = hdr->size;
223#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__)
224 recordDealloc(allocSize);
225#endif
226 vPortFree(raw);
227 if (status) *status = Status::SUCCESS;
228 } else {
229 // Best-effort free for pointers not allocated by this manager.
230#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__)
231 // Update diagnostics conservatively.
232 if (diagnostic.active > 0) diagnostic.active--;
233 diagnostic.deallocated++;
234#endif
235 vPortFree(raw);
236 if (status) *status = Status::SUCCESS;
237 }
238 }

240#if defined(MEMORY_MANAGER_DIAG) || defined(__INTELLISENSE__) || defined(__DOXYGEN__)
241 /** Records an allocation in diagnostics.
242 * @param size Size of the allocation in bytes.
243 */
244 static void recordAlloc(size_t size) {
245 diagnostic.allocated++;
246 diagnostic.active++;
247 // Safe add: avoid overflow
248 if (diagnostic.currentUsage <= (kMaxSize - size)) {
249 diagnostic.currentUsage += size;
250 if (diagnostic.currentUsage > diagnostic.peakUsage) diagnostic.peakUsage = diagnostic.currentUsage;
251 } else {
252 diagnostic.currentUsage = kMaxSize;
253 diagnostic.peakUsage = diagnostic.currentUsage;
254 }
255 }
256
257 static void recordDealloc(size_t size) {
258 if (diagnostic.currentUsage >= size) diagnostic.currentUsage -= size;
259 else diagnostic.currentUsage = 0;
260 diagnostic.deallocated++;
261 if (diagnostic.active > 0) diagnostic.active--;
262 }
263#endif

304 * using RTOS::MemoryManager::Status;
305 *
306 * // Allocate 128 bytes
307 * Status status;
308 * void *ptr = MemoryManager::allocate(128, status);
309 * if (status == Status::SUCCESS && ptr) {
310 * std::printf("Allocated 128 bytes at %p\n", ptr);
311 * } else {
312 * std::printf("Allocation failed: %u\n", static_cast<unsigned>(status));
313 * }

318 * std::printf("Zero-size allocation failed as expected: %u\n", static_cast<unsigned>(status));
319 * }
320 *
321 * // Deallocate the first allocation
322 * MemoryManager::deallocate(ptr, status);

324 * std::printf("Deallocated memory at %p\n", ptr);
325 * }

327 * #if defined(MEMORY_MANAGER_DIAG)

329 * const auto &diag = MemoryManager::getDiagnostics();

337 * @endcode
