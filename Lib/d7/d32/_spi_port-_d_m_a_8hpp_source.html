1/** DMA-driven SPI port specialization for HAL layer.
2 *
3 * DMA-driven, asynchronous SPI operations specialization of `SpiPort` for the
4 * HAL layer. This header provides the `SpiPort<DriverMode::DMA>` template
5 * specialization which implements non-blocking transmit, receive and
6 * full-duplex transfers backed by the HAL DMA API and uses the base class
7 * event mechanism to signal completion to callers.
8 *
9 * Purpose and usage:
10 * - Use this specialization when high-throughput, low-CPU SPI transfers are
11 * required (bulk transfers, large buffers, offloaded IO via DMA).
12 * - Requires the HAL SPI module and DMA support enabled in the build
13 * configuration (see `HAL_SPI_MODULE_ENABLED`).
14 *
15 * Dependencies:
16 * - `SpiPortBase.hpp` for common SPI port plumbing and event synchronization.
17 * - HAL DMA callbacks and SPI handle provided by the platform HAL.
18 *
19 * @date 17 Jan 2026
20 * @version 3.04
21 * @file SpiPort-DMA.hpp
22 */

32 /** SpiPort specialization for DMA-driven asynchronous communication.
33 *
34 * Provides non-blocking SPI transmit, receive and full-duplex operations
35 * using HAL DMA APIs. Completion events are signaled via the base class
36 * event mechanism.
37 */
38 template <> class SpiPort<DriverMode::DMA> : protected SpiPortBase {
39 public:
40 /** Construct a DMA-mode SPI port.
41 *
42 * @param handle Pointer to `SPI_HandleTypeDef` provided by HAL.
43 * @pre `handle` must be a valid, initialized SPI handle.
44 */
45 explicit SpiPort(SPI_HandleTypeDef * handle) noexcept : SpiPortBase(handle) {}
46
47 /** Destructor for DMA-mode SPI port. */
48 virtual ~SpiPort() noexcept = default;
49
50 /** Return the underlying HAL SPI handle.
51 *
52 * @return Pointer to `SPI_HandleTypeDef` managed by the base class.
53 */
54 virtual SPI_HandleTypeDef * getHandle() const noexcept override {
55 return SpiPortBase::getHandle();
56 }
57
58 /** Start a DMA-based SPI reception and wait for completion event.
59 *
60 * @param buffer Destination buffer where received bytes will be stored.
61 * @param size Number of bytes to receive (must be > 0).
62 * @param timeout RTOS ticks to wait for completion.
63 * @return `true` if DMA transfer started successfully and completion event occurred, `false` otherwise.
64 * @pre `buffer` must point to valid memory and `size` must be positive.
65 */
66 virtual bool receiveData(void * buffer, size_t size, uint32_t timeout) noexcept override {
67 (void)(timeout);
68 return HAL_SPI_Receive_DMA(handle.getHandle(), static_cast<uint8_t *>(buffer), size) == HAL_OK &&
69 waitForEvent(RX_COMPLETED, timeout);
70 }
71
72 /** Start a DMA-based SPI transmission and wait for completion event.
73 *
74 * @param buffer Source buffer containing bytes to transmit.
75 * @param size Number of bytes to transmit (must be > 0).
76 * @param timeout RTOS ticks to wait for completion.
77 * @return `true` if DMA transfer started successfully and completion event occurred, `false` otherwise.
78 * @pre `buffer` must point to valid memory and `size` must be positive.
79 */
80 virtual bool sendData(const void * buffer, size_t size, uint32_t timeout) noexcept override {
81 (void)(timeout);
82 return HAL_SPI_Transmit_DMA(handle.getHandle(), static_cast<const uint8_t *>(buffer), size) == HAL_OK &&
83 waitForEvent(TX_COMPLETED, timeout);
84 }
85
86 /** Start a DMA-based full-duplex SPI transfer and wait for completion event.
87 *
88 * @param txBuffer Source buffer with data to transmit.
89 * @param rxBuffer Destination buffer where received bytes will be stored.
90 * @param size Number of bytes to exchange in both directions (must be > 0).
91 * @param timeout RTOS ticks to wait for completion.
92 * @return `true` if DMA transfer started successfully and completion event occurred, `false` otherwise.
93 * @pre `txBuffer` and `rxBuffer` must point to valid memory and `size` must be positive.
94 */
95 virtual bool sendReceiveData(const void * txBuffer, void * rxBuffer, size_t size, uint32_t timeout) noexcept override {
96 (void)(timeout);
97 return HAL_SPI_TransmitReceive_DMA(handle.getHandle(), static_cast<const uint8_t *>(txBuffer),
98 static_cast<uint8_t *>(rxBuffer), size) == HAL_OK &&
99 waitForEvent(TX_RX_COMPLETED, timeout);
100 }
101 };
