42 * Extends `SerialPort` to provide USB serial port repeater functionality, including open/close,

79 /** Open the serial port repeater.
80 *
81 * @return `Status::OK` on success, `Status::ERROR` on failure.
82 */
83 virtual Status open() override {
84 if (isOpenFlag) return Status::ERROR;
85 if (SerialPort::open() != Status::OK) return Status::ERROR;
86 RTOS::delay(USB_PORT_OPEN_DELAY_MS);
87 enableUSBPort(true);
88 USBD_Interface_fops_FS.Init = []() -> int8_t { return instance->initCallback(); };
89 USBD_Interface_fops_FS.DeInit = []() -> int8_t { return instance->deinitCallback(); };
90 USBD_Interface_fops_FS.Control = [](uint8_t command, uint8_t * buffer, uint16_t length) -> int8_t {
91 return instance->controlCallback(command, buffer, length);
92 };
93 USBD_Interface_fops_FS.Receive = [](uint8_t * buffer, uint32_t * length) { return instance->receiveCallback(buffer, length); };
94 MX_USB_DEVICE_Init();
95 isOpenFlag = true;
96 return Status::OK;
97 }
98
99 /** Close the serial port repeater. */
100 virtual void close() override {
101 if (!isOpenFlag) return;
102 enableUSBPort(false);
103 SerialPort::close();
104 isOpenFlag = false;
105 }
106

110 */

165 /** USB CDC initialization callback.
166 *
167 * @return `USBD_OK` on success.
168 */
169 int8_t initCallback() {
170 USBD_CDC_SetTxBuffer(&hUsbDeviceFS, UserTxBufferFS, 0);
171 USBD_CDC_SetRxBuffer(&hUsbDeviceFS, UserRxBufferFS);
172 isConnectedFlag = true;
173 if (onConnected) onConnected(true);
174 return USBD_OK;
175 }
176
177 /** USB CDC deinitialization callback.
178 *
179 * @return `USBD_OK` on success.
180 */
181 int8_t deinitCallback() {
182 isConnectedFlag = false;
183 if (onConnected) onConnected(false);
184 return USBD_OK;
185 }
186
187 /** USB CDC control callback.
188 *
189 * @param command CDC command.
190 * @param buffer Data buffer.
191 * @param length Buffer length.
192 * @return `USBD_OK` on success.
193 */
194 int8_t controlCallback(uint8_t command, uint8_t * buffer, uint16_t length) {
195 struct LineCoding {
196 uint32_t baudRate; /*!< Baud rate. */
197 uint8_t charFormat; /*!< Character format. */
198 uint8_t parityType; /*!< Parity type. */
199 uint8_t dataBits; /*!< Number of data bits. */
200 };
201
202 static LineCoding lineCoding{115200, 0, 0, 8};
203 switch (command) {
204 case CDC_SET_LINE_CODING:
205 memcpy(&lineCoding, buffer, sizeof(LineCoding));
206 SerialPort::setBaudRate(lineCoding.baudRate);
207 break;
208 case CDC_GET_LINE_CODING:
209 memcpy(buffer, &lineCoding, sizeof(LineCoding));
210 break;
211 default:
212 break;
213 }
214 return USBD_OK;
215 }
216
217 /** USB CDC receive callback.
218 *
219 * @param buffer Received data buffer.
220 * @param length Pointer to received data length.
221 * @return `USBD_OK` if successful, `USBD_FAIL` otherwise.
222 */
223 int8_t receiveCallback(uint8_t * buffer, uint32_t * length) {
224 return SerialPort::send(buffer, static_cast<size_t>(*length)) == Status::OK ? USBD_OK : USBD_FAIL;
225 }
226 };
