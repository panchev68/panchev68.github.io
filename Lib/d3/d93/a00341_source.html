1/** Device Manager Base Class

3 * This file defines a base template class for managing devices in the HAL layer. It provides
4 * functionality to add, find, and remove devices using handles, serving as a foundation for
5 * specific device managers in embedded systems.

25 /** Base Device Manager Template
26 * This template class provides a singleton base for managing devices of type T_DEVICE using
27 * handles of type T_HANDLE. It maintains a map of handles to devices for efficient lookup
28 * and management in the HAL layer.
29 * @tparam T_DEVICE The type of device being managed.
30 * @tparam T_HANDLE The type of handle used for device identification.
31 */
32 template <typename T_DEVICE, typename T_HANDLE> class DeviceManagerBase {
33 public:
34 /** Returns the singleton instance of the device manager.
35 * Provides access to the single instance of the device manager.
36 * @return Reference to the singleton instance.
37 */
38 static DeviceManagerBase & instance() {
39 static DeviceManagerBase inst;
40 return inst;
41 }
42
43 /** Adds a device item to the manager.
44 * Inserts the device into the internal map using its handle as key.
45 * @param item Pointer to the device to add.
46 * @return True if the device was added successfully, false if it already exists.
47 */
48 virtual bool addItem(T_DEVICE * item) {
49 assert(item);
50 auto hdl = item->getHandle();
51 assert(hdl);
52 return deviceMap.emplace(hdl, item).second;
53 }
54
55 /** Finds a device by its handle.
56 * Searches the internal map for a device associated with the given handle.
57 * @param hdl Pointer to the handle to search for.
58 * @return Pointer to the device if found, nullptr otherwise.
59 */
60 virtual T_DEVICE * findItem(T_HANDLE * hdl) const {
61 assert(hdl);
62 auto it = deviceMap.find(hdl);
63 return it != deviceMap.cend() ? it->second : nullptr;
64 }
65
66 /** Removes a device item from the manager.
67 * Removes the device from the internal map if it exists and matches the provided item.
68 * @param item Pointer to the device to remove.
69 * @return True if the device was removed successfully, false otherwise.
70 */
71 virtual bool removeItem(T_DEVICE * item) {
72 assert(item);
73 auto hdl = item->getHandle();
74 assert(hdl);
75 auto it = deviceMap.find(hdl);
76 return it != deviceMap.cend() && it->second == item ? (deviceMap.erase(it), true) : false;
77 }
78
79 private:
80 std::unordered_map<T_HANDLE *, T_DEVICE *> deviceMap; /*!< Map of handles to devices. */
81
82 DeviceManagerBase() = default;
83
84 virtual ~DeviceManagerBase() = default;
85
86 DeviceManagerBase(const DeviceManagerBase &) = delete;
87
88 DeviceManagerBase(DeviceManagerBase &&) = delete;
89
90 DeviceManagerBase & operator=(const DeviceManagerBase &) = delete;
91
92 DeviceManagerBase & operator=(DeviceManagerBase &&) = delete;
93 };
