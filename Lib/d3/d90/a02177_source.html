56 /** Starts the CAN service.
57 * Initializes the queue, filter, port, and starts the task.
58 * @param nodeId The node ID for this service.
59 * @return true if started successfully, false otherwise.
60 */
61 bool start(uint8_t nodeId) {
62 id = nodeId;
63 if (!queueRxMessage.open()) {
64 assert(false);
65 return false;
66 }
67 static Filter filter(nodeId);
68 filter.setId(0);
69 filter.setMask(0);
70 filter.setBank(0);
71 if (!port.setFilter(filter)) {
72 assert(false);
73 return false;
74 }
75 if (!port.open()) {
76 assert(false);
77 return false;
78 }
79 return task.start([this]() { taskFn(); });
80 }
