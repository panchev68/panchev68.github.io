30 class Helper {

32 /** Union for conversion between uint16_t and byte array.
33 *
34 * Provides efficient access to individual bytes within a 16-bit value.
35 */
36 union U16 {
37 uint16_t value; /*!< 16-bit value. */
38 uint8_t v[2]; /*!< 2-byte array. */

43 } bytes; /*!< Low/high byte struct. */
44 };
45
46 /** Union for conversion between uint32_t and word/byte arrays.
47 *
48 * Provides efficient access to individual bytes and words within a 32-bit value.
49 */
50 union U32 {
51 uint32_t value; /*!< 32-bit value. */
52 uint16_t w[2]; /*!< 2-word array. */
53 uint8_t v[4]; /*!< 4-byte array. */
54
55 struct {
56 uint16_t lw; /*!< Low word. */
57 uint16_t hw; /*!< High word. */
58 } words; /*!< Low/high word struct. */
59
60 struct {
61 uint8_t lb; /*!< Low byte. */

63 uint8_t ub; /*!< Upper byte. */
64 uint8_t mb; /*!< Middle byte. */
65 } bytes; /*!< Byte struct. */
66
67 struct {
68 U16 low; /*!< Low word as U16. */
69 U16 high; /*!< High word as U16. */
70 } uwords; /*!< U16 word struct. */
71 };
72
73 /** Union for conversion between uint64_t and dword/word/byte arrays.
74 *
75 * Provides efficient access to individual bytes, words, and dwords within a 64-bit value.
76 */
77 union U64 {
78 uint64_t value; /*!< 64-bit value. */
79 uint32_t d[2]; /*!< 2-dword array. */
80 uint16_t w[4]; /*!< 4-word array. */
81 uint8_t v[8]; /*!< 8-byte array. */
82
83 struct {
84 uint32_t ld; /*!< Low dword. */
85 uint32_t hd; /*!< High dword. */
86 } dwords; /*!< Low/high dword struct. */
87
88 struct {
89 uint16_t lw; /*!< Low word. */
90 uint16_t hw; /*!< High word. */
91 uint16_t uw; /*!< Upper word. */
92 uint16_t mw; /*!< Middle word. */
93 } words; /*!< Word struct. */
94 };
95 };
