32 /** Reads data from the EEPROM.
33 * @param addr Starting address to read from.
34 * @param[out] buffer Buffer to store the read data.
35 * @param size Number of bytes to read.
36 * @return `true` if successful, `false` if parameters are invalid or the transfer fails.
37 * @pre Device must be opened and ready.
38 * @pre `buffer` must not be `nullptr`.
39 * @pre `size` must be greater than 0.
40 * @pre `addr + size` must not exceed chip size.
41 */
42 bool read(uint32_t addr, void * buffer, size_t size) override {
43 if (!validateTransfer(addr, buffer, size)) return false;
44 return read8(addr, buffer, size);
45 }
46
47 /** Writes data to the EEPROM.
48 * @param addr Starting address to write to.
49 * @param buffer Buffer containing the data to write.
50 * @param size Number of bytes to write.
51 * @return `true` if successful, `false` if parameters are invalid or the transfer fails.
52 * @pre Device must be opened and ready.
53 * @pre `buffer` must not be `nullptr`.
54 * @pre `size` must be greater than 0.
55 * @pre `addr + size` must not exceed chip size.
56 */
57 bool write(uint32_t addr, const void * buffer, size_t size) override {
58 if (!validateTransfer(addr, buffer, size)) return false;
59 return write8(addr, buffer, size);
60 }
61private:
62 /** Validates that the buffer is not null and the transfer fits inside the chip. */
63 static bool validateTransfer(uint32_t addr, const void * buffer, size_t size) noexcept {
64 return buffer != nullptr && isAddressRangeValid(addr, size);
65 }
66
67 /** Validates that the address range is contained within the chip. */
68 static constexpr bool isAddressRangeValid(uint32_t addr, size_t size) noexcept {
69 const size_t start = static_cast<size_t>(addr);
70 return size > 0 && size <= CHIP_SIZE && start < CHIP_SIZE && (start + size) <= CHIP_SIZE;
71 }
