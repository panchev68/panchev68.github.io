154 static bool parseIpString(const std::string_view & str, std::array<uint8_t, 4> * address) noexcept {
155 if (str.empty() || str.size() > 15 || str.find(':') != std::string_view::npos) return false;
156 std::array<uint8_t, 4> temp{};
157 size_t pos = 0;
158 for (size_t i = 0; i < 4; ++i) {
159 if (pos >= str.size()) return false;
160 size_t found = (i < 3) ? str.find('.', pos) : std::string_view::npos;
161 if (i < 3 && (found == std::string_view::npos || found == pos)) return false;
162 uint32_t value = 0;
163 size_t end = (i < 3) ? found : str.size();
164 for (size_t j = pos; j < end; ++j) {
165 char c = str[j];
166 if (c < '0' || c > '9') return false;
167 value = value * 10 + static_cast<uint32_t>(c - '0');
168 if (value > 255) return false;
169 }
170 if (pos == end) return false;
171 temp[i] = static_cast<uint8_t>(value);
172 pos = (i < 3) ? found + 1 : str.size();
173 }
174 if (str.find('.', pos) != std::string_view::npos) return false;
175 if (address) *address = temp;
176 return true;
177 }
