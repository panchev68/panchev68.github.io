69 /** Starts the DTH component.
70 * @return `true` if started successfully, `false` otherwise.
71 */
72 virtual bool start() {
73 if (!semaphoreDataReception.create()) {
74 assert(false);
75 return false;
76 }
77 instance = this;
78 const auto result = HAL_TIM_RegisterCallback(config.timHandle.handle, HAL_TIM_IC_CAPTURE_CB_ID, handleTimerInterrupt);
79 if (result != HAL_OK) {
80 assert(false);
81 return false;
82 }
83 task.start([this] { taskEntry(); });
84 return true;
85 }
86
87 /** DTH task function. */
88 virtual void taskEntry() = 0;

90 /** Writes GPIO pin state.
91 * @param state Pin state (GPIO_PIN_RESET or GPIO_PIN_SET).
92 */
93 void writePinState(GPIO_PinState state) {
94 HAL_GPIO_WritePin(config.pinHandle.getPort(), config.pinHandle.getPinNumber(), state);
95 }
96
97 /** Sends startup impulse to DTH sensor.
98 * @param impulseDuration Impulse duration in ms.
99 */
100 void startupImpulse(uint32_t impulseDuration) {
101 configurePin(GPIO_MODE_OUTPUT_PP, GPIO_PULLUP, GPIO_SPEED_FREQ_LOW);
102 writePinState(GPIO_PIN_RESET);
103 vTaskDelay(impulseDuration);
104 writePinState(GPIO_PIN_SET);
105 configurePin(GPIO_MODE_AF_PP, GPIO_PULLUP, GPIO_SPEED_FREQ_LOW, GPIO_AF1_TIM1);
106 }
107
108 /** Ends data reception. */
109 void endReceivedData() {
110 configurePin(GPIO_MODE_OUTPUT_PP, GPIO_PULLUP, GPIO_SPEED_FREQ_LOW);
111 writePinState(GPIO_PIN_SET);
112 }
113
114 /** Begins DMA data reception.
115 * @param data Data buffer.
116 * @param length Buffer length.
117 * @return `true` if started successfully, `false` otherwise.
118 */
119 bool beginReceiveData(uint32_t * data, size_t length) {
120 const auto result = HAL_TIM_IC_Start_DMA(config.timHandle.handle, config.timHandle.channel, data, length);
121 // assert(result == HAL_OK);
122 return result == HAL_OK;
123 }

125 /** Waits for data reception completion.
126 * @param timeout Timeout in ms.
127 * @return `true` if completed, `false` otherwise.
128 */
129 inline bool waitForComplete(uint32_t timeout) { return semaphoreDataReception.take(timeout); }
130
131 private:
132 RTOS::Semaphore semaphoreDataReception; /**< Semaphore for data reception. */
133 RTOS::Task task; /**< DTH task. */

136 /** Timer ISR callback.
137 * @param handle Timer handle pointer.
138 */
139 static void handleTimerInterrupt(TIM_HandleTypeDef * handle) {
140 if (instance == nullptr) return;
141 if (handle != instance->config.timHandle.handle) return;
142 if (!instance->semaphoreDataReception.giveFromISR()) assert(false);
143 }
144
145 /** Configures GPIO pin.
146 * @param mode GPIO mode.
147 * @param pull GPIO pull config.
148 * @param speed GPIO speed.
149 * @param alternate Alternate function (optional).
150 */
151 void configurePin(uint32_t mode, uint32_t pull, uint32_t speed, uint32_t alternate = 0) {
152 GPIO_InitTypeDef gpioInit{};
153 gpioInit.Pin = config.pinHandle.getPinNumber();
154 gpioInit.Mode = mode;
155 gpioInit.Pull = pull;
156 gpioInit.Speed = speed;
157 gpioInit.Alternate = alternate;
158 HAL_GPIO_Init(config.pinHandle.getPort(), &gpioInit);
159 }
