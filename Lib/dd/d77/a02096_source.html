28 /** Abstract base class that defines the interface and shared behavior for
29 * hardware-specific display drivers.
30 *
31 * Subclasses implement device initialization and pixel transfer logic
32 * specific to their hardware platform. The class centralizes transfer-state
33 * management and provides helper utilities used by the BSP display layer.
34 *
35 * @note Subclasses must implement the setup() and transmitBlock() pure
36 * virtual methods.
37 */
38 class DriverBase {
39 protected:
40 static constexpr int MAX_TRANSFER_HEIGHT = 240; /*!< Maximum permitted bottom coordinate (y + height - 1) for a transfer block. */
41 static inline DriverBase * instance = nullptr; /*!< Pointer to the active DriverBase instance. */
42
43 /** Constructs a DriverBase instance.
44 *
45 * The constructor registers the created instance as the global instance
46 * pointer that can be accessed by external C callbacks.
47 *
48 * @post The global instance pointer refers to this object.
49 */
50 DriverBase() { instance = this; }
51
52 /** Destroys the DriverBase instance. */
53 virtual ~DriverBase() = default;
54
55 /** Initializes the display driver.
56 *
57 * Subclass implementations must perform device and peripheral
58 * configuration in this method. Typical actions include GPIO, clock,
59 * DMA setup, and other hardware initialization required by the
60 * specific display driver.
61 *
62 * @return True when initialization completes successfully; false on error.
63 */
64 virtual bool setup() = 0;
65
66 /** Transmits a rectangular pixel block to the display.
67 *
68 * Subclass implementations must initiate the transfer of the rectangle
69 * described by the provided buffer and coordinates. The transfer may be
70 * asynchronous; implementations should set the transfer-active flag
71 * when starting an asynchronous transfer and clear it when the transfer
72 * completes.
73 *
74 * @param buffer Pointer to the pixel data to transmit.
75 * @param x X coordinate of the rectangle origin in display pixels.
76 * @param y Y coordinate of the rectangle origin in display pixels.
77 * @param width Width of the rectangle in pixels.
78 * @param height Height of the rectangle in pixels.
79 *
80 * @pre The rectangle (x, y, width, height) shall lie within the
81 * display bounds and height shall be <= MAX_TRANSFER_HEIGHT.
82 */
83 virtual void transmitBlock(const void * buffer, uint32_t x, uint32_t y, size_t width, size_t height) = 0;
84
85 /** Handles a transfer-complete event.
86 *
87 * This method is invoked when the underlying DMA or transfer engine
88 * signals that the current transfer has finished. The default
89 * implementation clears the transfer-active flag. Hardware drivers may
90 * override this method to perform additional bookkeeping.
91 *
92 * The external C callback DisplayDriver_TransferCompleteCallback() shall
93 * call this method when a transfer completes.
94 */
95 virtual void handleTransferCompleteEvent() {
96 setTransferActive(false);
97 }
98
99 /** Validates whether a block with the given bottom coordinate may be transferred.
100 *
101 * @param bottom Bottom coordinate (y + height - 1) of the block to validate.
102 *
103 * @return True when the block bottom is within the allowed transfer height;
104 * false otherwise.
105 */
106 virtual bool canTransferBlock(int16_t bottom) const { return bottom <= MAX_TRANSFER_HEIGHT; }
107
108 /** Queries whether a transfer is currently active.
109 *
110 * @return True when a transfer is in progress; false otherwise.
111 */
112 virtual bool getTransferActive() const { return isTransferActive; }
113
114 /** Sets the transfer-active flag.
115 *
116 * @param value True to mark transfers as active; false to mark them as inactive.
117 */
118 void setTransferActive(bool value) { isTransferActive = value; }
119
120 private:
121 bool isTransferActive = false; /*!< Indicates whether a transfer is in progress. */
122 };
