27 /** PWM controlled backlight with auto-dimming support.
28 *
29 * Backlight encapsulates a TIM peripheral PWM channel and provides a
30 * configurable software dimmer that smoothly fades between values and can
31 * reduce brightness after a period of inactivity. The class uses the RTOS
32 * TickService::ClientBase interface to receive periodic tick events for
33 * timing fades and sleep timeout handling.
34 *
35 * Responsibilities:
36 * - Controlling PWM brightness via a TIM peripheral.
37 * - Smoothly fading brightness over a configured time interval.
38 * - Implementing inactivity-based automatic dimming.
39 * - Coordinating with the RTOS tick service for timing events.
40 *
41 * @see RTOS::TickService::ClientBase
42 */
43 class Backlight : public RTOS::TickService::ClientBase {
44 public:
45 /** Backlight configuration parameters.
46 *
47 * Defines the TIM peripheral, channel, and soft-dimming parameters.
48 *
49 * @note All fields must remain valid during the Backlight lifetime.
50 */
51 struct Config {
52 TIM_HandleTypeDef * timHandle; /*!< Pointer to HAL TIM handle for PWM. */
53 uint32_t timChannel; /*!< TIM channel (TIM_CHANNEL_1..4) for PWM. */
54 uint32_t fadeStep; /*!< Fade step interval in milliseconds. */
55 uint32_t sleepTimeout{60000}; /*!< Inactivity timeout (ms) before dimming. */
56 uint8_t dimmerMinValue{25}; /*!< Minimum brightness (PWM duty). */
57 uint8_t dimmerMaxValue{100}; /*!< Maximum brightness (PWM duty). */
58 };
59
60 /** Internal dimmer helper.
61 *
62 * Implements timed fade behaviour and inactivity-based sleep dimming.
63 * Dimmer is a nested class with access to the Backlight parent.
64 *
65 * @see Backlight
66 */
67 class Dimmer final {
68 friend class Backlight;
69
70 public:
71 /** Create a Dimmer bound to the given Backlight parent.
72 *
73 * @param parent Pointer to the owning Backlight instance (non-null).
74 *
75 * @pre parent is a valid non-null pointer.
76 */
77 explicit Dimmer(Backlight * parent)
78 : parent(parent),
79 config(parent->config),
80 isEnabled(true),
81 minValue(config.dimmerMinValue),
82 maxValue(config.dimmerMaxValue),
83 currentValue(0) {
84 assert(parent);
85 fadeTimeout.set(config.fadeStep);
86 }
87
88 /** Destructor. */
89 ~Dimmer() = default;
90
91 /** Restart the dimmer to the configured maximum value.
92 *
93 * Also restarts the sleep timeout if configured.
94 *
95 * @post Dimmer targets maximum brightness and inactivity timer is reset.
96 */
97 void restart() {
98 if (isEnabled) {
99 parent->targetValue = maxValue;
100 if (config.sleepTimeout != 0) sleepTimeout.set(config.sleepTimeout);
101 }
102 }
103
104 /** Enable the dimmer and restart fading.
105 *
106 * @post Dimmer is active and begins fade-in to maximum brightness.
107 */
108 void start() {
109 isEnabled = true;
110 restart();
111 }
112
113 /** Disable the dimmer; inactivity will not reduce brightness.
114 *
115 * @post Dimmer is inactive; sleep timeout is ignored.
116 */
117 void stop() { isEnabled = false; }
118
119 /** Notify the dimmer of user activity to defer the sleep timeout.
120 *
121 * @post Sleep timeout is reset if enabled.
122 */
123 void signalActivity() {
124 if (!isEnabled || config.sleepTimeout == 0) return;
125 sleepTimeout.set(config.sleepTimeout);
126 }
127
128 private:
129 Backlight * parent; /*!< Owning Backlight instance. */
130 const Config & config; /*!< Reference to parent configuration. */
131 RTOS::Timeout fadeTimeout; /*!< Timeout scheduling fade steps. */
132 RTOS::Timeout sleepTimeout; /*!< Timeout for inactivity-based dimming. */
133 uint8_t currentValue; /*!< Current PWM duty value in CCR. */
134 uint8_t minValue; /*!< Cached minimum brightness value. */
135 uint8_t maxValue; /*!< Cached maximum brightness value. */
136 bool isEnabled; /*!< Dimmer enabled state. */
137
138 /** Handle RTOS tick events to progress fades and monitor sleep timeout. */
139 void handleTickEvent() {
140 if (fadeTimeout.isExpired()) {
141 fadeTimeout.set(config.fadeStep);
142 if (currentValue < parent->targetValue) {
143 ++currentValue;
144 if (parent->timerCCR) *parent->timerCCR = currentValue;
145 } else if (currentValue > parent->targetValue) {
146 --currentValue;
147 if (parent->timerCCR) *parent->timerCCR = currentValue;
148 }
149 }
150 if (isEnabled && sleepTimeout.isExpired()) {
151 sleepTimeout.disable();
152 parent->targetValue = minValue;
153 }
154 }
155 };

158 /** Construct a Backlight controller.
159 *
160 * Starts the TIM PWM channel immediately. The provided Config must contain
161 * a valid TIM handle and channel. Configuration is validated with asserts.
162 *
163 * @param config Configuration parameters (copied).
164 *
165 * @pre config.timHandle is valid and properly initialized.
166 * @pre config.fadeStep > 0.
167 * @pre config.dimmerMinValue <= config.dimmerMaxValue.
168 *
169 * @post PWM is started and dimmer is ready for use.
170 *
171 * @throws assert() if configuration is invalid.
172 */
173 explicit Backlight(const Config & config)
174 : config(config),
175 targetValue(0),
176 timerCCR(resolveTimerCCR(config)),
177 dimmer(this) {
178 assert(config.timHandle);
179 assert(config.fadeStep > 0);
180 assert(config.dimmerMinValue <= config.dimmerMaxValue);
181 assert(timerCCR != nullptr);
182 if (HAL_TIM_PWM_Start(config.timHandle, config.timChannel) != HAL_OK) {
183 assert(false);
184 }
185 }
186
187 /** Destructor - stops PWM output. */
188 virtual ~Backlight() {

190 assert(false);
191 }
192 }

194 /** Get the currently targeted brightness value.
195 *
196 * Note: the actual applied PWM may be in transition during fades.
197 *
198 * @return Target brightness value (clamped to configured min/max).
199 */
200 virtual uint8_t get() const { return targetValue; }
201
202 /** Set a new target brightness.
203 *
204 * The value is clamped to the configured min/max range and the dimmer
205 * is signalled so that fading and inactivity timer are handled accordingly.
206 *
207 * @param newValue Desired brightness value.
208 *
209 * @post targetValue is updated and dimmer signals activity.
210 */
211 virtual void set(uint8_t newValue) {
212 targetValue = std::clamp(newValue, config.dimmerMinValue, config.dimmerMaxValue);
213 dimmer.signalActivity();
214 }

217 *
218 * @param newValue Desired brightness value.
219 * @return Reference to this Backlight instance.

221 * @see set()
222 */
223 Backlight & operator=(uint8_t newValue) {

226 }

228 private:
229 const Config config; /*!< Immutable configuration copy for this instance. */
230 volatile uint32_t * timerCCR; /*!< Pointer to the timer CCR register for PWM duty. */
231 uint8_t targetValue; /*!< Currently requested brightness (clamped). */
232 Dimmer dimmer; /*!< Internal dimmer for fades and sleep handling. */
233
234 /** Resolve the pointer to the selected TIM CCR register.
235 *
236 * Maps the configured TIM channel to the correct CCR register of the
237 * TIM peripheral instance. Returns nullptr and asserts on invalid input.
238 *
239 * @param config Configuration with timHandle and timChannel.
240 * @return Pointer to the CCR register for the selected channel.
241 *
242 * @pre config.timHandle is valid.
243 * @pre config.timChannel is TIM_CHANNEL_1, TIM_CHANNEL_2, TIM_CHANNEL_3, or TIM_CHANNEL_4.
244 */
245 volatile uint32_t * resolveTimerCCR(const Config & config) {
246 assert(config.timHandle);
247 assert(config.timHandle->Instance);
248 switch (config.timChannel) {
249 case TIM_CHANNEL_1:
250 return &config.timHandle->Instance->CCR1;
251 case TIM_CHANNEL_2:
252 return &config.timHandle->Instance->CCR2;
253 case TIM_CHANNEL_3:
254 return &config.timHandle->Instance->CCR3;
255 case TIM_CHANNEL_4:
256 return &config.timHandle->Instance->CCR4;
257 default:
258 assert(false);
259 return nullptr;
260 }
261 }

263 /** Handle RTOS tick events by delegating to the dimmer. */
264 void handleTickEvent() override { dimmer.handleTickEvent(); }
