118 /** Register a tick event client.

120 * @return Status of the operation.
121 * @pre client must not be nullptr.

123 Status registerClient(ClientBase * client) {
124 assert(client != nullptr);
125 if (!client) return Status::INVALID;
126
127 MutexGuard guard(mutex);
128 if (std::find(clients.begin(), clients.end(), client) != clients.end()) return Status::FAILURE;
129
130 if (!isRunning()) start();

133 return Status::SUCCESS;

136 /** Unregister a tick event client.
137 * @param client Pointer to the client to unregister.
138 * @return Status of the operation.
139 * @pre client must not be nullptr.
140 */
141 Status unregisterClient(ClientBase * client) {
142 assert(client != nullptr);
143 if (!client) return Status::INVALID;
144
145 MutexGuard guard(mutex);
146 const auto it = std::find(clients.begin(), clients.end(), client);
147 if (it == clients.end()) return Status::FAILURE;
148
149 clients.erase(it);
150 if (getClientCount() == 0) stop();
151 return Status::SUCCESS;
152 }

166 /** Start the tick service with a specified interval.
167 * @param interval Tick interval in milliseconds (default: 1).
168 * @return Status of the operation.
169 * @pre interval must be > 0 and <= 1000.
170 */
171 Status start() {
172 if (isRunning()) return Status::ALREADY_RUNNING;
173 const TickType_t targetTicks = tickInterval;
174 const auto taskFunc = [this, targetTicks]() { runTickLoop(targetTicks); };
175 return (task.start(taskFunc) != nullptr) ? Status::SUCCESS : Status::FAILURE;
176 }
177
178 /** Stop the tick service.
179 * @return Status of the operation.
180 */
181 Status stop() {
182 if (!isRunning()) return Status::NOT_RUNNING;
183 task.stop();
184 return Status::SUCCESS;
185 }

200 /** Run the tick loop for the RTOS task.
201 * @param targetTicks Target ticks for the interval.
202 */
203 void runTickLoop(TickType_t targetTicks) {
204 while (true) {
205 const TickType_t startTick = TickCounter::get();
206 {
207 MutexGuard guard(mutex);
208 for (auto * client : clients) {
209 if (client) {
210 if (client->updatePeriodCounter > 0) client->updatePeriodCounter--;
211 else {
212 client->updatePeriodCounter = client->updatePeriodValue;
213 client->handleTickEvent();
214 }
215 }
216 }
217 }
218 const TickType_t elapsed = TickCounter::getElapsedTicksSince(startTick);
219 if (elapsed < targetTicks) Task::delay(targetTicks - elapsed);
220 }
221 }
