35 /** TickService.
36 * Manages periodic tick events and notifies registered clients.
37 * The TickService class is a singleton that runs an RTOS task managing periodic tick events.
38 * Registered clients inherit from ClientBase and receive tick notifications based on their
39 * configured update periods. The service is thread-safe and automatically starts/stops
40 * based on client registration. Tick events are dispatched at configurable intervals,
41 * allowing multiple clients to operate at different periods within a single task.
42 */

122 /** Register tick event client.
123 *
124 * Adds a client to the managed list and starts the tick task if this is the first client.
125 *
126 * @param client Pointer to the ClientBase instance to register.
127 * @return Status code indicating success or error.
128 *
129 * @pre client must not be nullptr.
130 */
131 Status registerClient(ClientBase * client) noexcept {
132 if (client == nullptr) {
133 assert(false);
134 return Status::INVALID_CLIENT;
135 }
136 MutexGuard guard(mutex);
137 if (std::find(clients.begin(), clients.end(), client) != clients.end()) {
138 return Status::CLIENT_ALREADY_REGISTERED;
139 }
140 const bool wasEmpty = clients.empty();
141 clients.push_back(client);
142 if (wasEmpty) TaskWorker::start();
143 return Status::SUCCESS;
144 }
145
146 /** Unregister tick event client.
147 *
148 * Removes a client from the managed list and stops the tick task if no clients remain.
149 *
150 * @param client Pointer to the ClientBase instance to unregister.
151 * @return Status code indicating success or error.
152 *
153 * @pre client must not be nullptr.
154 */
155 Status unregisterClient(ClientBase * client) noexcept {
156 if (client == nullptr) {
157 assert(false);
158 return Status::INVALID_CLIENT;
159 }
160 MutexGuard guard(mutex);
161 const auto it = std::find(clients.begin(), clients.end(), client);
162 if (it == clients.end()) {
163 return Status::CLIENT_NOT_REGISTERED;
164 }
165 clients.erase(it);
166 if (clients.empty()) TaskWorker::stop();
167 return Status::SUCCESS;
168 }

170 /** Task function for handling tick events.
171 *
172 * Executes in the context of the TickService task and iterates over registered clients,
173 * decrementing their update period counters and invoking their handleTickEvent() method
174 * when the counter reaches zero. Thread-safe access is ensured via mutex.
175 */
176 void taskFunction() noexcept override {
177 while (true) {
178 MutexGuard guard(mutex);
179 for (auto client : clients) {
180 if (--client->updatePeriodCounter <= 0) {
181 client->updatePeriodCounter = client->updatePeriodValue;
182 client->handleTickEvent();
183 }
184 }
185 RTOS::Task::delay(1);
186 }
187 }
