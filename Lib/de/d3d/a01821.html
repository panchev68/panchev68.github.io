Lib::Helper::Crc16

Computing CRC-16 Checksum for Data Integrity

This example demonstrates how to use the Crc16 class to compute a CRC-16-CCITT checksum for a data buffer and a vector of bytes. The checksum can be used to verify data integrity in communication protocols or storage systems.

Example usage:
#include "crc16.hpp"
#include <iostream>
#include <vector>
int main() {
// Example data buffer
const uint8_t data[] = {0x01, 0x02, 0x03, 0x04};
size_t dataSize = sizeof(data);
// Compute CRC-16 from buffer
Lib::Helper::Crc16 crcBuffer(data, dataSize);
uint16_t checksumBuffer = crcBuffer;
std::cout << "CRC-16 for buffer: 0x" << std::hex << checksumBuffer << std::endl;
// Example vector of bytes
std::vector<uint8_t> dataVector = {0x05, 0x06, 0x07, 0x08};
// Compute CRC-16 from vector
Lib::Helper::Crc16 crcVector(dataVector);
uint16_t checksumVector = crcVector;
std::cout << "CRC-16 for vector: 0x" << std::hex << checksumVector << std::endl;
// Using the alias
crc16 crcAlias(data, dataSize);
uint16_t checksumAlias = crcAlias;
std::cout << "CRC-16 using alias: 0x" << std::hex << checksumAlias << std::endl;
return 0;
}
