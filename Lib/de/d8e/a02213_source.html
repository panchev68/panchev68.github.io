24 /** Singleton class for managing system reset states and operations.
25 *
26 * Provides methods to read the current reset state, trigger system resets,
27 * and access reset information in a thread-safe manner.
28 */
29 class Reset {
30 public:
31 /** Enumeration of possible reset states. */
32 enum class State : uint8_t {
33 UNKNOWN = 0, /*!< Unknown or undetermined reset state. */
34 LOW_POWER_RESET, /*!< Reset due to low power mode. */
35 WINDOW_WATCHDOG_RESET, /*!< Reset triggered by window watchdog. */
36 INDEPENDENT_WATCHDOG_RESET, /*!< Reset triggered by independent watchdog. */
37 SOFTWARE_RESET, /*!< Software-initiated reset. */
38 POWER_ON_POWER_DOWN_RESET, /*!< Reset due to power-on or power-down. */
39 EXTERNAL_RESET_PIN_RESET, /*!< Reset via external reset pin. */
40 BROWNOUT_RESET /*!< Reset due to brownout detection. */
41 };

82 /** Read the reset state from hardware flags.
83 *
84 * @return The determined reset state.
85 */
86 static State readState() {
87 struct FlagState {
88 uint32_t flag;
89 State state;
90 };
91
92 /* Array mapping reset flags to states. */
93 static constexpr std::array<FlagState, 7> flagStates = {{
94
95 {RCC_FLAG_LPWRRST, State::LOW_POWER_RESET},
96 {RCC_FLAG_WWDGRST, State::WINDOW_WATCHDOG_RESET},
97 {RCC_FLAG_IWDGRST, State::INDEPENDENT_WATCHDOG_RESET},
98 {RCC_FLAG_SFTRST, State::SOFTWARE_RESET},
99 {RCC_FLAG_PORRST, State::POWER_ON_POWER_DOWN_RESET},
100 {RCC_FLAG_PINRST, State::EXTERNAL_RESET_PIN_RESET},
101 {RCC_FLAG_BORRST, State::BROWNOUT_RESET}
102
103 }};
104
105 for (const auto & fs : flagStates) {
106 if (__HAL_RCC_GET_FLAG(fs.flag)) return fs.state;
107 }
108 return State::UNKNOWN;
109 }
110
111 /** Constructor initializing the reset state. */
112 explicit Reset() : currentState(readState()) { __HAL_RCC_CLEAR_RESET_FLAGS(); }
113

115

117 };
