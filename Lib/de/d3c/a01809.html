Lib::Core::Delegate

Simple usage examples for Lib::Core::Delegate demonstrating construction from a lambda, assignment from a functor, and invocation.

Simple usage examples for Lib::Core::Delegate demonstrating construction from a lambda, assignment from a functor, and invocation. The examples use int(int) as a representative callable signature.

#include <Lib/Core/Delegate.hpp>
#include <iostream>
struct Adder {
int base;
Adder(int b) : base(b) {}
int operator()(int x) const { return base + x; }
};
void delegateExample() {
// Construct from a lambda
Lib::Core::Delegate<int(int)> d = [](int x) { return x * 2; };
std::cout << d(5) << "\n"; // prints 10
// Reassign from a functor
d = Adder{3};
std::cout << d(5) << "\n"; // prints 8
// Assign a free function
auto square = [](int v) { return v * v; };
d = square;
std::cout << d(4) << "\n"; // prints 16
}
