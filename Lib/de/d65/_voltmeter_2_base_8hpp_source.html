76 /** Handles periodic updates.

78 void handleTickEvent() override {
79 if (!timeout.isExpired()) return;
80 timeout.set(updatePeriod);
81 uint16_t adcValue = 0;
82 if (!adcPort.read(adcValue, 0)) return;
83 adcValue = integrator.processValue(adcValue);
84 if (!integrator.isWindowFull()) {
85 adcPort.start();
86 return;
87 }
88 float newVoltage = interpolator.interpolate(adcValue);
89 if (newVoltage != currentVoltage) {
90 currentVoltage = newVoltage;
91 handleValueChanged(newVoltage);
92 }
93 adcPort.start();
