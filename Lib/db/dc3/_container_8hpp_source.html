25 *
26 * This class provides methods for adding, removing, and processing pointers to objects of type T.

46 /** Adds a pointer to the container if not already present.
47 *
48 * Checks if the pointer is null or already in the container. If valid and not present, adds it.
49 *
50 * @param item The pointer to add.
51 * @return True if added, false if null or already exists.
52 */
53 bool add(T * item) {
54 if (item == nullptr) return false;
55 if (std::find(items.begin(), items.end(), item) != items.end()) return false;
56 items.emplace_back(item);
57 return true;
58 }
59
60 /** Adds multiple pointers from an initializer list.
61 *
62 * Attempts to add each pointer in the list. Returns true only if all were added successfully.
63 *
64 * @param itemList The list of pointers to add.
65 * @return True if all added, false if any failed.
66 */
67 bool add(const std::initializer_list<T *> & itemList) {
68 bool allAdded = true;
69 for (auto * item : itemList) allAdded &= add(item);
70 return allAdded;
71 }

73 /** Removes a pointer from the container.
74 *
75 * Finds and removes the specified pointer if it exists.
76 *
77 * @param item The pointer to remove.
78 * @return True if removed, false if not found.
79 */
80 bool remove(T * item) {
81 auto it = std::find(items.begin(), items.end(), item);
82 if (it == items.end()) return false;
83 items.erase(it);
84 return true;
85 }

89 * Removes all items from the container.

91 void clear() { items.clear(); }

93 /** Processes each item with a callback function.
94 *
95 * Iterates over all pointers and calls the provided callback for each, passing an optional context.
96 * Asserts if callback is null.
97 *
98 * @param callback The function to call for each item.
99 * @param context Optional user-defined context passed to the callback.
100 */
101 void processItems(void (*callback)(T *, void *), void * context = nullptr) {
102 if (!callback) assert(false);
103 for (auto * item : items) callback(item, context);
104 }
