Lib::Helper::JaggedArray

Jagged Array Usage Example.

Jagged Array Usage Example.This example demonstrates how to use the JaggedArray class to manage a collection of rows, where each row is a JaggedArrayRow containing elements. It shows adding rows, accessing elements, and iterating over the structure.

Example usage:
#include "Lib/Helper/JaggedArray.hpp"
#include <iostream>
int main() {
// Create some data elements
int a = 1, b = 2, c = 3;
int d = 4, e = 5;
int f = 6;
// Create rows
Lib::Helper::JaggedArrayRow<int> row1;
row1.addItems({&a, &b, &c});
Lib::Helper::JaggedArrayRow<int> row2;
row2.addItems({&d, &e});
Lib::Helper::JaggedArrayRow<int> row3;
row3.addItem(&f);
// Create jagged array and add rows
Lib::Helper::JaggedArray<int> jagged;
jagged.addItems({&row1, &row2, &row3});
// Access and print elements
for (size_t i = 0; i < jagged.getSize(); ++i) {
std::cout << "Row " << i << ": ";
auto& row = jagged[i];
for (size_t j = 0; j < row.getSize(); ++j) {
std::cout << row[j] << " ";
}
std::cout << std::endl;
}
return 0;
}
