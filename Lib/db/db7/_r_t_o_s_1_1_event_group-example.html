RTOS::EventGroup

Example usage of EventGroup

Demonstrates how to create and use the EventGroup class for event bit management in FreeRTOS.

Example code:

#include "Lib/RTOS/EventGroup.hpp"
#include <iostream>
int main() {
// Create an event group with initial bits set
auto eventGroup = RTOS::EventGroup::createWithBits(RTOS::EventBits().set(0).set(2));
if (!eventGroup) {
std::cerr << "Failed to create event group!" << std::endl;
return -1;
}
// Set a bit
eventGroup->setBit(1);
// Wait for bits 0 and 1 to be set
RTOS::EventBits waitBits;
waitBits.set(0);
waitBits.set(1);
RTOS::EventBits result = eventGroup->waitForBits(waitBits, RTOS::EventGroup::ClearMode::CLEAR, RTOS::EventGroup::WaitMode::ALL, 1000);
if ((result & waitBits) == waitBits) {
std::cout << "Both bits 0 and 1 are set!" << std::endl;
} else {
std::cout << "Timeout waiting for bits." << std::endl;
}
return 0;
}
