47 * @param dataLength Number of data bytes (0-8).
48 * @param isExtendedId `true` for extended ID, `false` for standard ID.

52 explicit CanTxHeader(uint32_t messageId, size_t dataLength = 0, bool isExtendedId = false, bool isRemote = false, bool enableTimestamp = false) {
53 assert(dataLength <= MAX_DATA_LENGTH);
54 CAN_TxHeaderTypeDef::StdId = isExtendedId ? 0U : messageId;
55 CAN_TxHeaderTypeDef::ExtId = isExtendedId ? messageId : 0U;
56 CAN_TxHeaderTypeDef::IDE = isExtendedId ? CAN_ID_EXT : CAN_ID_STD;
57 CAN_TxHeaderTypeDef::RTR = isRemote ? CAN_RTR_REMOTE : CAN_RTR_DATA;
58 CAN_TxHeaderTypeDef::TransmitGlobalTime = enableTimestamp ? ENABLE : DISABLE;
59 CAN_TxHeaderTypeDef::DLC = static_cast<uint32_t>(dataLength);
60 }
