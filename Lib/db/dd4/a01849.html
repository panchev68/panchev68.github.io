Lib::Helper::TimeZone

Time zone conversion example.

This example demonstrates basic usage of the TimeZone class: construction, reading the configured offset, and conversion of a local calendar time to UTC minutes.

Example usage:
#include "Lib/Helper/TimeZone.hpp"
static void useTimeZoneExample() {
// Eastern European Time (EET, UTC+2).
Lib::Helper::TimeZone tz("EET", 120);
// Convert 2026-01-16 10:30 local time to UTC minutes.
const time_t offsetSeconds = static_cast<time_t>(tz.getOffsetMinutes()) * 60;
const int utcMinutes = tz.convertLocalToUtcMinutes(2026, 1, 16, 10, 30, offsetSeconds);
(void)utcMinutes;
}
