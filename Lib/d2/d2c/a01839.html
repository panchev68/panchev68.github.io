Lib::Helper::MoonPhase

Moon Phase Calculation Example.

Moon Phase Calculation Example.This example demonstrates how to use the MoonPhase class to calculate and display the moon phase for a specific date. It shows creating a MoonPhase object, calculating the phase, and retrieving the phase name and index.

Example usage:
#include "Lib/Helper/MoonPhase.hpp"
#include "Lib/Helper/DateTime.hpp"
#include <iostream>
int main() {
// Create a date
Lib::Helper::DateTime::Date date{2023, 10, 15}; // October 15, 2023
// Create MoonPhase object and calculate
Lib::Helper::MoonPhase moonPhase(date);
// Alternatively, create default and calculate
Lib::Helper::MoonPhase anotherPhase;
uint32_t index = anotherPhase.calculate(date);
// Get phase information
std::string_view name = moonPhase.name();
uint32_t phaseIndex = moonPhase.index();
// Display results
std::cout << "Date: " << date.year << "-" << date.month << "-" << date.day << std::endl;
std::cout << "Moon Phase: " << name << " (Index: " << phaseIndex << ")" << std::endl;
return 0;
}
