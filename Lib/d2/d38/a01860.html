Lib::Helper::DateTime

Working with Date and Time in Embedded Systems

This example demonstrates how to use the DateTime, Date, and Time classes for handling timestamps, validation, string conversion, and packing/unpacking for storage in embedded systems.

Example usage:
#include "DateTime.hpp"
#include <iostream>
int main() {
// Create a Time object
Lib::Helper::DateTime::Time time(14, 30, 45);
if (time.isValid()) {
std::cout << "Time: " << time.toString() << std::endl;
}
// Create a Date object
Lib::Helper::DateTime::Date date(2023, 10, 15);
if (date.isValid()) {
std::cout << "Date: " << date.toString() << std::endl;
auto dayOfWeek = date.getDayOfWeek();
std::cout << "Day of week: " << Lib::Helper::DateTime::Date::DayOfWeek::getLongString(dayOfWeek) << std::endl;
}
// Create a DateTime object
Lib::Helper::DateTime dateTime(date, time);
if (dateTime.isValid()) {
std::cout << "DateTime: " << dateTime.toString() << std::endl;
}
// Pack and unpack DateTime
uint32_t packed = Lib::Helper::DateTime::pack(dateTime);
Lib::Helper::DateTime unpacked = Lib::Helper::DateTime::unpack(packed);
std::cout << "Unpacked DateTime: " << unpacked.toString() << std::endl;
// Convert to tm struct
tm tmStruct = dateTime;
std::cout << "tm year: " << tmStruct.tm_year << std::endl;
return 0;
}
