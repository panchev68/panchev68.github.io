66 /** Opens the serial port and starts communication tasks.
67 * Initializes the UART ports, starts the receive and transmit tasks, and sets the reset pin.
68 * @return Status::OK if successful, Status::ERROR otherwise.

70 virtual Status open() {
71 if (!messageBuffer.open(USB_MSG_BUF_SIZE)) return Status::ERROR;
72 rxPort->init();
73 txPort->init();

133 /** Task for receiving data from UART.

136 void receiveTask() {
137 std::array<uint8_t, USB_MSG_BUF_SIZE> buffer{};
138 size_t bufferIndex = 0;
139 uint8_t value = 0;
140 for (;;) {
141 if (rxPort->receive(&value, sizeof(uint8_t), 1) == sizeof(uint8_t)) {
142 if (bufferIndex < buffer.size()) {
143 buffer[bufferIndex++] = value;
144 } else {
145 bufferIndex = 0;
146 }
147 } else if (bufferIndex > 0) {
148 postReceivedCallback(buffer.data(), bufferIndex);
149 bufferIndex = 0;
150 }
151 }
152 }

157 void transmitTask() {
158 std::array<uint8_t, USB_MSG_BUF_SIZE> buffer{};
159 for (;;) {
160 auto length = messageBuffer.receive(buffer.data(), buffer.size(), portMAX_DELAY);
161 if (!txPort->send(buffer.data(), length, 100)) { assert(false); }

163 }
