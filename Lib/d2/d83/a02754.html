Lib::HTTP::Client

Demonstrates a GSM modem stream adapted to 'HttpTransport', an HTTPS POST through 'WolfSSLTransport' and a plain HTTP GET.

Demonstrates a GSM modem stream adapted to 'HttpTransport', an HTTPS POST through 'WolfSSLTransport' and a plain HTTP GET.'GsmStream' gives the modem's data-mode stream the two methods the concept requires; it inherits nothing, and its 'connect()'/'disconnect()' are its own, since neither 'Client' nor 'WolfSSLTransport' opens or closes the link. For HTTPS the TLS session runs over that stream and the client runs over the session; for plain HTTP the client uses the stream directly. The host name appears twice for different reasons: 'open()' checks it against the server certificate, the request sends it in the Host header. 'response.body' views into 'buffer', so it is valid only while 'buffer' lives.

Important
Connect the raw link before 'open()' and disconnect it after 'close()': the handshake and the close notification both travel over it.
Every request sends 'Connection: close': reconnect the link, and for HTTPS rerun 'open()', before the next request.
See also
Lib::HTTP::Client, Lib::HTTP::WolfSSLTransport, Lib::HTTP::HttpTransport, Lib::HTTP::Status
Example usage:
#include <Lib/HTTP/Client.hpp>
#include <array>
#include <cstddef>
#include <cstdint>
#include <span>
#include <string_view>
// Adapter giving the modem's data-mode stream the HttpTransport shape. connect()/disconnect() are its own.
class GsmStream final {
public:
explicit GsmStream(Modem & modem) noexcept :
modem(modem) {
}
[[nodiscard]] bool connect(std::string_view host, uint16_t port) noexcept {
return modem.openTcp(host, port);
}
void disconnect() noexcept {
modem.closeTcp();
}
[[nodiscard]] bool send(std::span<const uint8_t> data, size_t & sentLength) noexcept {
return modem.write(data, sentLength);
}
[[nodiscard]] bool receive(std::span<uint8_t> buffer, size_t & receivedLength) noexcept {
return modem.read(buffer, receivedLength); // 0 bytes with true: nothing yet.
}
private:
Modem & modem;
};
static_assert(Lib::HTTP::HttpTransport<GsmStream>);
extern const std::span<const uint8_t> CLOUD_CA_CERTIFICATE; // PEM, embedded in the firmware.
constexpr std::string_view HOST = "api.example.com";
bool sendTelemetry(Modem & modem, std::span<const uint8_t> json) {
GsmStream stream(modem);
if (!stream.connect(HOST, 443U)) {
return false;
}
bool isAccepted = false;
Lib::HTTP::WolfSSLTransport<GsmStream> tls;
if (tls.open(stream, HOST, CLOUD_CA_CERTIFICATE)) {
Lib::HTTP::Client client(tls);
std::array<uint8_t, 512> buffer{};
Lib::HTTP::Response response;
const Lib::HTTP::Status status = client.post(HOST, "/v1/telemetry", "application/json", json, buffer, response);
isAccepted = (status == Lib::HTTP::Status::SUCCESS) && (response.statusCode == 200U);
tls.close(); // TLS first, the TCP link below still carries its close notification.
}
stream.disconnect();
return isAccepted;
}
uint16_t fetchStatus(Modem & modem) {
GsmStream stream(modem);
if (!stream.connect(HOST, 80U)) {
return 0U;
}
Lib::HTTP::Client client(stream); // Plain HTTP: the raw stream directly.
std::array<uint8_t, 512> buffer{};
Lib::HTTP::Response response;
const Lib::HTTP::Status status = client.get(HOST, "/v1/status", buffer, response);
stream.disconnect();
return (status == Lib::HTTP::Status::SUCCESS) ? response.statusCode : 0U;
}
