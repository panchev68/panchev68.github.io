46 /** Computes the hash of a data buffer.
47 *
48 * @param buffer The data buffer to hash.
49 * @param size The size of the buffer in bytes.
50 * @return The computed 32-bit hash value.
51 */
52 static uint32_t hash(const void * buffer, size_t size) {
53 if (buffer == nullptr || size < 4U) { return 0U; }
54 const auto * ptr = static_cast<const std::uint8_t *>(buffer);
55 constexpr uint32_t HASH_C = 9U;
56 uint32_t hashA = static_cast<uint32_t>(size);
57 uint32_t hashB = hashA * 5U;
58 uint32_t hashD = hashB;
59 uint32_t hashC = HASH_C;
60 const size_t midOff = (static_cast<size_t>(size) >> 1U) & ~static_cast<size_t>(3U);
61 hashA += fetch32(ptr);
62 hashB += fetch32(ptr + size - 4U);
63 hashC += fetch32(ptr + midOff);
64 return finalizeHash(murmurMix(hashC, murmurMix(hashB, murmurMix(hashA, hashD))));
65 }

87 hash ^= hash >> 16U;
88 hash *= 0x85ebca6bU;

95 /** Applies MurmurHash mixing to the value and hash.
96 *
97 * @param value The value to mix.
98 * @param hash The current hash value.
99 * @return The mixed hash value.

101 static uint32_t murmurMix(uint32_t value, uint32_t hash) {
102 constexpr uint32_t C1 = 0xcc9e2d51U;
103 constexpr uint32_t C2 = 0x1b873593U;
104 value *= C1;
105 value = (value << 17U) | (value >> 15U);
106 value *= C2;
107 hash ^= value;
108 hash = (hash << 19U) | (hash >> 13U);
109 return hash * 5U + 0xe6546b64U;
110 }
