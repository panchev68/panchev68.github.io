26 /** Configuration class for OTP memory management.
27 * Provides an interface for reading, writing, and validating configuration data stored in OTP memory.
28 */
29 class Configuration {
30 static constexpr uint32_t VALID_MASK = 0x10000000; /*!< Mask to check if the configuration is valid (bit 28 must be 0). */
31 static constexpr uint32_t EMPTY_VALUE = 0xFFFFFFFF; /*!< Value indicating empty configuration (all bits set). */
32
33 public:
34 /** Status enum for operation results. */
35 enum class Status {
36 OK, /*!< Operation completed successfully. */
37 ERROR /*!< Operation failed. */
38 };
39
40 using Bitset = std::bitset<31>; /*!< Bitset type for configuration data. */
41
42 const uint32_t flashAddress; /*!< Absolute address in OTP memory. */
43
44 /** Constructor initializes OTP memory address from offset.
45 * @param offset Offset from the beginning of OTP memory.
46 * @pre `offset` must be less than the size of the OTP memory region.
47 */
48 explicit Configuration(uint32_t offset) : flashAddress(Lib::HAL::OTP::beginAddr() + offset) {
49 assert(offset < (Lib::HAL::OTP::endAddr() - Lib::HAL::OTP::beginAddr()));
50 }
51
52 /** Virtual destructor for polymorphism support. */
53 virtual ~Configuration() = default;
54
55 /** Reads configuration bitset from OTP memory.
56 * @return Bitset representing configuration data.
57 */
58 Bitset read() const { return Bitset(readValue(flashAddress)); }
59
60 /** Writes configuration bitset to OTP memory with validation.
61 * @param value Bitset to write.
62 * @return Status of the write operation.
63 */
64 Status write(const Bitset & value) {
65 auto & otp = Lib::HAL::OTP::instance();
66 auto unlocked = otp.unlock();
67 assert(unlocked);
68 const uint32_t val = value.to_ulong() & ~VALID_MASK;
69 bool result = otp.write(flashAddress, val);
70 if (!result) result = otp.write(flashAddress, val);
71 assert(otp.lock());
72 return result ? Status::OK : Status::ERROR;
73 }
74
75 /** Checks if configuration is empty (all bits set).
76 * @return `true` if configuration is empty, `false` otherwise.
77 */
78 virtual bool isEmpty() const { return readValue(flashAddress) == EMPTY_VALUE; }
79
80 /** Checks if configuration is valid (bit 28 is 0).
81 * @return `true` if configuration is valid, `false` otherwise.
82 */
83 virtual bool isValid() const { return (readValue(flashAddress) & VALID_MASK) == 0; }
84
85 private:
86 /** Reads 32-bit value from OTP memory at specified address.
87 * @param flashAddr Address to read from.
88 * @return 32-bit value from OTP.
89 */
90 uint32_t readValue(uint32_t flashAddr) const { return Lib::HAL::OTP::instance().read<uint32_t>(flashAddr); }
91 };
