28 /** TickCounter utility class.
29 *
30 * Provides static methods to retrieve the current FreeRTOS tick count from both task and ISR
31 * contexts. This class cannot be instantiated or destroyed.
32 */
33 class TickCounter {
34 public:
35 using TickType = TickType_t; /*!< RTOS tick type alias for FreeRTOS tick values. */
36
37 /** Get current FreeRTOS tick count.
38 *
39 * Retrieves the current FreeRTOS tick count from task context.
40 * This method should be called from task code, not from ISR context.
41 *
42 * @return Current tick count.
43 *
44 * @note Use getFromISR() when calling from interrupt service routines.
45 */
46 static inline TickType get() { return xTaskGetTickCount(); }
47
48 /** Get current FreeRTOS tick count from ISR context.
49 *
50 * Retrieves the current FreeRTOS tick count from interrupt service routine context.
51 * This method is safe to call from ISR handlers without risking task scheduler suspension.
52 *
53 * @return Current tick count.
54 *
55 * @note Use get() when calling from task code.
56 */
57 static inline TickType getFromISR() { return xTaskGetTickCountFromISR(); }
58
59 /** Get elapsed ticks since a given tick count.
60 *
61 * Calculates the number of ticks that have elapsed since the provided start tick count,
62 * handling wrap-around correctly for FreeRTOS tick counters.
63 *
64 * @param startTick The tick count from which to calculate elapsed time.
65 * @return Elapsed tick count since `startTick`.
66 */
67 static inline TickType getElapsedTicksSince(TickType startTick) { return static_cast<TickType>(get() - startTick); }
68
69 /** Get elapsed ticks since a given tick count from ISR context.
70 *
71 * Calculates the number of ticks that have elapsed since the provided start tick count,
72 * handling wrap-around correctly for FreeRTOS tick counters in ISR context.
73 *
74 * @param startTick The tick count from which to calculate elapsed time.
75 * @return Elapsed tick count since `startTick`.
76 */
77 static inline TickType getElapsedTicksSinceFromISR(TickType startTick) { return static_cast<TickType>(getFromISR() - startTick); }
78
79 /** Convert milliseconds to ticks.
80 *
81 * Converts a time value in milliseconds to FreeRTOS tick count.
82 *
83 * @param ms Time in milliseconds.
84 * @return Equivalent tick count.
85 */
86 static inline TickType msToTicks(uint32_t ms) { return static_cast<TickType>((static_cast<uint64_t>(ms) * configTICK_RATE_HZ + 999ULL) / 1000ULL); }
87
88 /** Convert ticks to milliseconds.
89 *
90 * Converts a FreeRTOS tick count to time in milliseconds.
91 *
92 * @param ticks Tick count.
93 * @return Equivalent time in milliseconds.
94 */
95 static inline uint32_t ticksToMs(TickType ticks) { return static_cast<uint32_t>((static_cast<uint64_t>(ticks) * 1000ULL) / configTICK_RATE_HZ); }
96
97 private:
98 /** Deleted constructor.
99 *
100 * This class cannot be instantiated as it provides only static utility methods.
101 */
102 TickCounter() = delete;
103
104 /** Deleted destructor.
105 *
106 * This class cannot be destroyed as it provides only static utility methods.
107 */
108 ~TickCounter() = delete;
109 };
