RTOS::MessageBuffer

Example usage of RTOS::MessageBuffer for sending and receiving messages.

This example demonstrates how to open a FreeRTOS message buffer using the RTOS::MessageBuffer class, send a message from a task, and receive it in another task. The example prints the sent and received messages to the standard output.

Example usage:
#include <cstdio>
#include "Lib/RTOS/MessageBuffer.hpp"
void senderTask(void *) {
RTOS::MessageBuffer messageBuffer;
const size_t bufferSize = 128;
if (!messageBuffer.open(bufferSize)) {
printf("Failed to open message buffer\n");
return;
}
const char *text = "Hello, FreeRTOS!";
size_t sent = messageBuffer.send(text, strlen(text) + 1, 100);
printf("Sent %u bytes: %s\n", static_cast<unsigned>(sent), text);
vTaskDelete(nullptr);
}
void receiverTask(void *) {
RTOS::MessageBuffer messageBuffer;
const size_t bufferSize = 128;
if (!messageBuffer.open(bufferSize)) {
printf("Failed to open message buffer\n");
return;
}
char buffer[128] = {};
size_t received = messageBuffer.receive(buffer, sizeof(buffer), 100);
printf("Received %u bytes: %s\n", static_cast<unsigned>(received), buffer);
vTaskDelete(nullptr);
}
int main() {
xTaskCreate(senderTask, "Sender", 256, nullptr, 1, nullptr);
xTaskCreate(receiverTask, "Receiver", 256, nullptr, 1, nullptr);
vTaskStartScheduler();
return 0;
}
