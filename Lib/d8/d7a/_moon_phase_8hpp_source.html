49 /** Calculate moon phase.
50 * Computes the moon phase index and name based on the input date.
51 * @param date The date for calculation.
52 * @return The phase index.
53 */
54 uint32_t calculate(const Date & date) noexcept {
55 constexpr double DAYS_PER_YEAR = 365.25;
56 constexpr double DAYS_PER_MONTH = 30.6;
57 constexpr double JULIAN_EPOCH_OFFSET = 694039.09;
58 constexpr double LUNAR_CYCLE_DAYS = 29.5305882;
59 constexpr int MARCH_MONTH = 3;
60 constexpr int MONTHS_IN_YEAR = 12;
61 constexpr int PHASE_COUNT = 8;
62 constexpr double ROUNDING_OFFSET = 0.5;
63
64 const int year = date.year;
65 const int month = date.month;
66 const int day = date.day;
67
68 // Adjust year and month for astronomical calculations (March = month 1)
69 const int adjustedYear = (month < MARCH_MONTH) ? (year - 1) : year;
70 const int adjustedMonth = (month < MARCH_MONTH) ? (month + MONTHS_IN_YEAR) : month;
71
72 // Calculate Julian day number relative to epoch
73 double julianDay = DAYS_PER_YEAR * adjustedYear + DAYS_PER_MONTH * (adjustedMonth + 1) + day - JULIAN_EPOCH_OFFSET;
74 julianDay /= LUNAR_CYCLE_DAYS;
75
76 // Extract fractional part and compute phase index
77 const double fractionalPart = julianDay - static_cast<int>(julianDay);
78 const int idx = static_cast<int>(fractionalPart * PHASE_COUNT + ROUNDING_OFFSET) & (PHASE_COUNT - 1);
79
80 phaseIndex = static_cast<uint32_t>(idx);
81 phaseName = PHASE_NAMES[idx];
82 return phaseIndex;
83 }
84

86 * @return The name of the current moon phase.

89

98 explicit constexpr operator uint32_t() const noexcept { return phaseIndex; }
99
100 /** Conversion to const char*.
101 * @return The phase name.
102 */
103 explicit constexpr operator const char *() const noexcept { return phaseName; }
104
105 private:

108 };
