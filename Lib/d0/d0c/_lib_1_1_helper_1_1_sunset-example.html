Lib::Helper::Sunset

Calculating Sunrise and Sunset Times

This example demonstrates how to use the Sunset class to calculate sunrise and sunset times for a specific geographic location and date. It shows setting the position, date, and computing various solar events including official, civil, nautical, and astronomical sunrise/sunset times. Additionally, it illustrates moon phase calculation for the given date.

Example usage:
cpp
#include "Lib/Helper/Sunset.hpp"
#include "Lib/Helper/DateTime.hpp"
#include <iostream>
int main() {
// Create a Sunset instance for a location (e.g., Sofia, Bulgaria)
Lib::Helper::Sunset sunset(42.6977, 23.3219, 2.0); // Latitude, Longitude, UTC offset
// Set the current date (e.g., January 16, 2026)
Lib::Helper::DateTime::Date date{2026, 1, 16};
sunset.setCurrentDate(date);
// Calculate sunrise and sunset times
double sunrise = sunset.calculateSunrise();
double sunset = sunset.calculateSunset();
// Convert to Time objects for display
Lib::Helper::DateTime::Time sunriseTime = Lib::Helper::Sunset::convertToTime(sunrise);
Lib::Helper::DateTime::Time sunsetTime = Lib::Helper::Sunset::convertToTime(sunset);
// Calculate moon phase
int moonPhase = sunset.calculateCurrentMoonPhase();
// Output the results
std::cout << "Sunrise: " << static_cast<int>(sunriseTime.hour) << ":"
<< static_cast<int>(sunriseTime.minute) << std::endl;
std::cout << "Sunset: " << static_cast<int>(sunsetTime.hour) << ":"
<< static_cast<int>(sunsetTime.minute) << std::endl;
std::cout << "Moon Phase Index: " << moonPhase << std::endl;
return 0;
}
