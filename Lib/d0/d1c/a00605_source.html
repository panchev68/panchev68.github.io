27 /** ProduceDate class.
28 *
29 * Manages production date storage and retrieval from OTP memory.
30 *
31 * The ProduceDate class encapsulates operations to read and write production
32 * dates stored in OTP memory at a specified offset. It provides validation,
33 * error handling, and ensures that writes occur only to empty locations.
34 *
35 * Role within the System module: Handles immutable production date information.
36 *
37 * Important design decisions: Uses OTP for permanent storage, includes status
38 * enumeration for operation results, and validates dates before writing.
39 *
40 * @note Writes are allowed only if the location is empty to prevent overwriting.
41 */
42 class ProduceDate {
43 public:
44 /** Date type alias.

46 * Alias for the Date type from DateTime utilities.

48 using Date = Lib::Helper::DateTime::Date;

56 /** Status enumeration for operation results.
57 *
58 * Defines the possible outcomes of read/write operations.
59 */
60 enum class Status {
61 OK, /*!< Operation successful. */
62 ERROR /*!< Operation failed. */
63 };
64
65 /** Empty value constant.
66 *
67 * Represents an uninitialized or empty OTP value.
68 */
69 static constexpr uint32_t EMPTY_VALUE = 0xFFFFFFFF;

71 /** Constructor for ProduceDate.
72 *
73 * Initializes the ProduceDate with an offset into OTP memory.
74 *
75 * @param offset Offset from the start of OTP memory for this date storage.
76 */
77 explicit ProduceDate(uint32_t offset) : flashAddress(Lib::HAL::OTP::beginAddr() + offset) {
78 assert(offset < (Lib::HAL::OTP::endAddr() - Lib::HAL::OTP::beginAddr()));
79 }

81 /** Read production date from OTP.
82 *
83 * Reads the packed date value from OTP, unpacks it, validates it, and
84 * returns the date if valid.
85 *
86 * @param result Reference to store the read date if successful.
87 * @return Status of the read operation (OK if successful, ERROR otherwise).
88 */
89 Status read(Date & result) const {
90 uint32_t value = Lib::HAL::OTP::instance().read<uint32_t>(flashAddress);
91 if (value == EMPTY_VALUE) return Status::ERROR;
92 DateTime dateTime(DateTime::unpack(value));
93 if (!dateTime.isValid()) return Status::ERROR;
94 result = dateTime.date;
95 return Status::OK;
96 }
97
98 /** Write production date to OTP.
99 *
100 * Writes the given date to OTP if the location is empty and the date is valid.
101 * Performs OTP unlock/lock operations for writing.
102 *
103 * @param value The date to write.
104 * @return Status of the write operation (OK if successful, ERROR otherwise).
105 */
106 Status write(const Date & value) {
107 if (!isEmpty()) return Status::ERROR;
108 if (!value.isValid()) return Status::ERROR;
109 auto & otp = Lib::HAL::OTP::instance();
110 if (!otp.unlock()) {
111 assert(false);
112 return Status::ERROR;
113 }
114 DateTime dateTime;
115 dateTime.date = value;
116 uint32_t packed = DateTime::pack(dateTime);
117 bool writeResult = otp.write(flashAddress, packed);
118 if (!writeResult) writeResult = otp.write(flashAddress, packed);
119 assert(writeResult);
120 if (!otp.lock()) {
121 assert(false);
122 return Status::ERROR;
123 }
124 return writeResult ? Status::OK : Status::ERROR;
125 }
126
127 /** Check if OTP location is empty.
128 *
129 * Determines whether the OTP location contains the empty value.
130 *
131 * @return true if empty, false otherwise.
132 */
133 bool isEmpty() const { return Lib::HAL::OTP::instance().read<uint32_t>(flashAddress) == EMPTY_VALUE; }
134
135 private:
136 /** Flash address for OTP storage.
137 *
138 * The calculated address in OTP memory where the date is stored.
139 */
140 const uint32_t flashAddress;
141 };
