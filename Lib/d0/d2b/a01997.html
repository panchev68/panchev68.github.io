Lib::USB::CDCDevice

Demonstrates a USB virtual COM port that echoes every received byte back to the host.

Demonstrates a USB virtual COM port that echoes every received byte back to the host.The device answers the enumeration and the CDC class requests in the USB interrupt, so the host sees the port as soon as start() returns. Clock, pins and the USB interrupt stay in the CubeMX code (HAL_PCD_MspInit() and the interrupt calling HAL_PCD_IRQHandler()). read() and write() are for task context only. write() fails early until the host has selected the configuration, so a task should wait for getIsConfigured() first.

See also
Lib::USB::CDCDevice::Config, Lib::USB::CDCDevice::LineCoding
Example usage:
#include <Lib/USB/CDCDevice.hpp>
#include <RTOS.hpp>
#include <array>
#include <chrono>
using namespace std::chrono_literals;
extern "C" PCD_HandleTypeDef hpcd_USB_OTG_FS; // CubeMX handle of the USB controller.
class UsbEcho {
public:
bool begin() noexcept {
if (!device.start()) {
return false;
}
task.setup({"UsbEcho", 256, RTOS::Task::Priority::NORMAL});
return task.start([this] { taskEntry(); }) != nullptr;
}
bool end() noexcept {
const bool isStopped = task.stop();
device.stop();
return isStopped;
}
private:
Lib::HAL::USBDevicePort<Lib::HAL::DriverMode::IT> port{&hpcd_USB_OTG_FS};
Lib::USB::CDCDevice device{{
.port = port,
.vendorId = 0x0483U,
.productId = 0x5740U,
.releaseNumber = 0x0100U,
.manufacturer = "Senstate",
.product = "Virtual COM Port",
.serialNumber = "0001",
}};
RTOS::Task task;
void taskEntry() noexcept {
std::array<uint8_t, 64> buffer;
while (true) {
if (!device.getIsConfigured()) {
RTOS::delay(100ms);
continue;
}
const size_t count = device.read(buffer, 500ms);
if (count > 0U) {
static_cast<void>(device.write(std::span<const uint8_t>(buffer.data(), count), 100ms));
}
}
}
};
