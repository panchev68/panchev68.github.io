36 /** STM32 RTC peripheral device abstraction.
37 * Provides thread-safe access to the STM32 RTC peripheral, supporting date/time read/write and backup register
38 * operations. Integrates with RTOS mutex and backup SRAM domain for safe concurrent access.
39 */
40 class RTClock : Handle<RTC_HandleTypeDef> {
41 using DateTime = Lib::Helper::DateTime;
42 static constexpr uint32_t NUMBER_OF_REGISTERS = 20; /*!< Number of RTC backup registers. */
43 RTOS::Mutex mutex; /*!< RTOS mutex for thread safety. */
44
45 public:
46 /** Constructor of the RTClock class.
47 * @param handle Pointer to RTC_HandleTypeDef.
48 */
49 explicit RTClock(RTC_HandleTypeDef * handle) : Handle(handle) { assert(handle); }
50
51 /** Destructor of the RTClock class. */
52 virtual ~RTClock() = default;

54 /** Initialize RTC device and enable backup SRAM. */
55 void setup() noexcept {
56 assert(!mutex.isCreated());
57 if (!mutex.create()) assert(0);
58 BackupRamDomain::instance().enable();
59 }
60
61 /** Release device resources. */
62 void release() noexcept { mutex.destroy(); }
63
64 /** Read current date and time from RTC.
65 * @return DateTime structure with current date and time.
66 */
67 DateTime read() noexcept {
68 RTC_TimeTypeDef time{};
69 RTC_DateTypeDef date{};
70 MutexGuard guard(mutex);
71 if (HAL_RTC_GetTime(getHandle(), &time, FORMAT_BIN) != HAL_OK) assert(0);
72 if (HAL_RTC_GetDate(getHandle(), &date, FORMAT_BIN) != HAL_OK) assert(0);
73 return {{date.Date, date.Month, date.Year + DateTime::Date::FIRST_YEAR},
74 {time.Hours, time.Minutes, time.Seconds}};
75 }
76
77 /** Write date and time to RTC.
78 * @param dateTime DateTime structure to write.
79 */
80 void write(const DateTime & dateTime) noexcept {
81 RTC_TimeTypeDef time{};
82 RTC_DateTypeDef date{};
83 time.Hours = dateTime.time.hours;
84 time.Minutes = dateTime.time.minutes;
85 time.Seconds = dateTime.time.seconds;
86 date.Date = dateTime.date.day;
87 date.Month = dateTime.date.month;
88 date.Year = dateTime.date.year - DateTime::Date::FIRST_YEAR;
89 MutexGuard guard(mutex);
90 if (HAL_RTC_SetTime(getHandle(), &time, FORMAT_BIN) != HAL_OK) assert(0);
91 if (HAL_RTC_SetDate(getHandle(), &date, FORMAT_BIN) != HAL_OK) assert(0);
92 }
93
94 /** Read RTC backup register.
95 * @param index Register index.
96 * @param[out] value Output value.
97 * @return RTClockStatus result code.
98 */
99 RTClockStatus readRegister(size_t index, uint32_t & value) noexcept {
100 if (index >= NUMBER_OF_REGISTERS) return RTClockStatus::INVALID_INDEX;
101 MutexGuard guard(mutex);
102 value = HAL_RTCEx_BKUPRead(getHandle(), static_cast<uint32_t>(index));
103 return RTClockStatus::OK;
104 }
105
106 /** Write RTC backup register.
107 * @param index Register index.
108 * @param value Value to write.
109 * @return RTClockStatus result code.
110 */
111 RTClockStatus writeRegister(size_t index, uint32_t value) noexcept {
112 if (index >= NUMBER_OF_REGISTERS) return RTClockStatus::INVALID_INDEX;
113 MutexGuard guard(mutex);
114 HAL_RTCEx_BKUPWrite(getHandle(), static_cast<uint32_t>(index), value);
115 return RTClockStatus::OK;
116 }
