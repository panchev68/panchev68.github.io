1/** ADC Transparent Port Base Class.
2 *
3 * This file defines the AdcTransparentPortBase class, which serves as a base class for transparent ADC port implementations in the HAL layer.
4 * It provides asynchronous ADC conversion handling using RTOS queues, callback management, and device lifecycle operations.
5 * The class integrates with the DeviceManager for registration and callback handling, ensuring thread-safe ADC operations.
6 *
7 * Key features include:
8 * - Asynchronous ADC result retrieval via RTOS queues.
9 * - Interrupt-driven callback handling for ADC events.
10 * - Automatic registration with the DeviceManager for lifecycle management.
11 *
12 * Important dependencies include AdcPortBase for basic ADC functionality, DeviceManagerBase for device management,
13 * RTOS for queue operations, and HAL ADC callbacks for interrupt handling.
14 *
15 * Special instructions: Ensure USE_HAL_ADC_REGISTER_CALLBACKS is set to 1 in the HAL configuration.
16 * This class is designed for use in embedded systems with RTOS support.
17 *
18 * @note This base class requires derived classes to implement start(), stop(), and handleInterruptEvent() methods.
19 *
20 * @attention Ensure proper ADC channel configuration and handle initialization before using instances of this class.
21 *
22 * @version 3.04
23 * @date 18 Jan 2026
24 * @file AdcTransparentPortBase.hpp
25 */

43 /** Base class for transparent ADC port implementations.
44 *
45 * This class provides a foundation for ADC ports that handle conversions asynchronously using RTOS queues.
46 * It manages ADC channel setup, result queuing, and interrupt event handling through a nested DeviceManager.
47 * Derived classes must implement start(), stop(), and handleInterruptEvent() to customize ADC behavior.
48 *
49 * The class ensures thread-safe operations by using RTOS queues for result communication and HAL callbacks for interrupts.
50 *
51 * @tparam Channel Type alias for AdcChannel.
52 */
53 class AdcTransparentPortBase : protected AdcPortBase {
54 friend class DeviceManagerBase<AdcTransparentPortBase, ADC_HandleTypeDef>;

62 protected:

65 /** Constructor for AdcTransparentPortBase.
66 *
67 * Initializes the ADC transparent port with the specified ADC handle and channel configuration.
68 * Calls the base class constructor to set up the ADC port.

70 * @param handle Pointer to the ADC_HandleTypeDef structure.

73 explicit AdcTransparentPortBase(ADC_HandleTypeDef * handle, const AdcChannel::Config & channelConfig) : AdcPortBase(handle, channelConfig) {}

75 /** Sets up the ADC transparent port.
76 *
77 * Registers the device with the DeviceManager, opens the result queue, and sets up the ADC channel.
78 * Asserts on failure to ensure proper initialization.
79 */
80 void setup() override {
81 if (!DeviceManager::instance().addItem(this)) assert(false);
82 if (!queueResult.open()) assert(false);
83 if (!AdcChannel::setup(getHandle())) assert(false);
84 }

88 * Unregisters the device from the DeviceManager and closes the result queue.
89 * Asserts on failure to ensure proper cleanup.

91 void release() override {
92 if (!DeviceManager::instance().removeItem(this)) assert(false);
93 queueResult.close();
94 }

112 /** Reads an ADC result from the queue with timeout.

151 /** Nested class for managing ADC transparent port devices.

158 /** Adds an ADC transparent port device to the manager.
159 *
160 * Registers the device with the base manager and sets up ADC callbacks if the handle is valid.
161 * Asserts on failure to ensure proper registration.
162 *
163 * @param device Pointer to the AdcTransparentPortBase device to add.
164 * @return true if the device was added successfully, false otherwise.
165 */
166 bool addItem(AdcTransparentPortBase * device) override {
167 if (DeviceManagerBase::addItem(device) && device->getHandle()) {
168 registerCallbacks(device->getHandle());
169 return true;
170 }
171 assert(false);
172 return false;
173 }

177 * Unregisters ADC callbacks and removes the device from the base manager.

179 *
180 * @param device Pointer to the AdcTransparentPortBase device to remove.
181 * @return true if the device was removed successfully, false otherwise.
182 */
183 bool removeItem(AdcTransparentPortBase * device) override {
184 assert(device);

186 return DeviceManagerBase::removeItem(device);

203 /** Handles ADC callback events.
204 *
205 * Template method that forwards ADC callback events to the associated device.
206 * Retrieves the device associated with the handle and calls its handleInterruptEvent method.
207 *
208 * @tparam T_ID The callback ID type.
209 * @param handle Pointer to the ADC handle triggering the event.
210 */
211 template <HAL_ADC_CallbackIDTypeDef T_ID> static void handleEvent(ADC_HandleTypeDef * handle) {
212 if (auto adcPort = instance().getDevice(handle)) adcPort->handleInterruptEvent(T_ID);
213 }
214
215 /** Registers ADC callback handlers.
216 *
217 * Sets up HAL ADC callbacks for various ADC events using predefined handler pairs.
218 * Asserts on failure to register any callback.
219 *
220 * @param handle Pointer to the ADC handle to register callbacks for.
221 */
222 static void registerCallbacks(ADC_HandleTypeDef * handle) {
223 assert(handle);
224 using HandlerPair = std::pair<HAL_ADC_CallbackIDTypeDef, pADC_CallbackTypeDef>;
225 constexpr HandlerPair HANDLERS[] = {{HAL_ADC_CONVERSION_COMPLETE_CB_ID, handleEvent<HAL_ADC_CONVERSION_COMPLETE_CB_ID>},
226 {HAL_ADC_CONVERSION_HALF_CB_ID, handleEvent<HAL_ADC_CONVERSION_HALF_CB_ID>}, {HAL_ADC_ERROR_CB_ID, handleEvent<HAL_ADC_ERROR_CB_ID>},
227 {HAL_ADC_INJ_CONVERSION_COMPLETE_CB_ID, handleEvent<HAL_ADC_INJ_CONVERSION_COMPLETE_CB_ID>},
228 {HAL_ADC_LEVEL_OUT_OF_WINDOW_1_CB_ID, handleEvent<HAL_ADC_LEVEL_OUT_OF_WINDOW_1_CB_ID>},
229 {HAL_ADC_MSPINIT_CB_ID, handleEvent<HAL_ADC_MSPINIT_CB_ID>}, {HAL_ADC_MSPDEINIT_CB_ID, handleEvent<HAL_ADC_MSPDEINIT_CB_ID>}};
230 for (const auto & handler : HANDLERS)
231 if (HAL_ADC_RegisterCallback(handle, handler.first, handler.second) != HAL_OK) assert(false);
232 }
233
234 /** Unregisters ADC callback handlers.
235 *
236 * Removes HAL ADC callbacks for all predefined event IDs.
237 * Asserts on failure to unregister any callback.
238 *
239 * @param handle Pointer to the ADC handle to unregister callbacks for.
240 */
241 static void unregisterCallbacks(ADC_HandleTypeDef * handle) {
242 assert(handle);
243 constexpr HAL_ADC_CallbackIDTypeDef IDS[] = {HAL_ADC_CONVERSION_COMPLETE_CB_ID, HAL_ADC_CONVERSION_HALF_CB_ID, HAL_ADC_ERROR_CB_ID,
244 HAL_ADC_INJ_CONVERSION_COMPLETE_CB_ID, HAL_ADC_LEVEL_OUT_OF_WINDOW_1_CB_ID, HAL_ADC_MSPINIT_CB_ID, HAL_ADC_MSPDEINIT_CB_ID};
245 for (auto id : IDS)
246 if (HAL_ADC_UnRegisterCallback(handle, id) != HAL_OK) assert(false);
247 }
248 };
