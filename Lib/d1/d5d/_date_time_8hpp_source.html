115 THU, /*!< Thursday. */
116 FRI, /*!< Friday. */

121 /** Get day of week for a date.
122 * Uses a compact arithmetic algorithm with precomputed month offsets
123 * to map a calendar date to a day index in the range [0, 6], where
124 * 0 = Monday and 6 = Sunday.
125 * @param date Date to check.
126 * @return DayOfWeek enum value.
127 */
128 static constexpr Name get(const Date & date) {
129 if (!date.isValid()) return Name::MON;
130 const uint32_t y = date.year - (date.month < 3U ? 1U : 0U);
131 const uint8_t mo = monthOffsets[date.month - 1U];
132 const uint32_t wi = (y + y / 4U - y / 100U + y / 400U + mo + date.day) % 7U;
133 return static_cast<Name>((wi + 6U) % 7U);
134 }
135
136 /** Get short string for day of week.
137 * @param date Date to check.
138 * @return Short string (e.g., "MON").
139 */
140 const static constexpr char * getShortString(Name name) {
141 const size_t index = static_cast<size_t>(name);
142 return daysOfWeekShort[index];
143 }
144
145 /** Get long string for day of week.
146 * @param date Date to check.
147 * @return Long string (e.g., "Monday").
148 */
149 const static constexpr char * getLongString(Name name) {
150 const size_t index = static_cast<size_t>(name);
151 return daysOfWeekLong[index];
152 }
153
154 private:
155 static constexpr std::array<const char *, 7> daysOfWeekShort{"MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN"}; /*!< Short day-of-week strings. */
156 static constexpr std::array<const char *, 7> daysOfWeekLong{"Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"}; /*!< Long day-of-week strings. */

301 /** Pack DateTime into 32-bit integer.
302 * @param dateTime DateTime to pack.
303 * @return Packed value.
304 */
305 static constexpr uint32_t pack(const DateTime & dateTime) {
306 const uint32_t y = (dateTime.date.year >= Date::FIRST_YEAR) ? (dateTime.date.year - Date::FIRST_YEAR) : 0U;
307 return (dateTime.time.seconds & 0x3FU) | ((dateTime.time.minutes & 0x3FU) << 6U) | ((dateTime.time.hours & 0x1FU) << 12U) | ((dateTime.date.day & 0x1FU) << 17U) | ((dateTime.date.month & 0x0FU) << 22U) | ((y & 0x3FU) << 26U);
308 }
309
310 /** Unpack 32-bit integer to DateTime.
311 * @param value Packed value.
312 * @return DateTime object.
313 */
314 static constexpr DateTime unpack(uint32_t value) {
315 return DateTime(Date(Date::FIRST_YEAR + ((value >> 26U) & 0x3FU), (value >> 22U) & 0x0FU, (value >> 17U) & 0x1FU), Time((value >> 12U) & 0x1FU, (value >> 6U) & 0x3FU, value & 0x3FU));
316 }
