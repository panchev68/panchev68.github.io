Lib::HAL::I2CPort

I2C Port Usage Example

This example demonstrates how to initialize and use the I2CPort class for I2C communication with an external device, such as reading and writing data to an EEPROM.

Example usage:

Initialize the I2C port, check device readiness, and perform memory read/write operations.

#include "Lib/HAL/core/I2C/I2CPort.hpp"
// Assume hi2c1 is initialized HAL I2C handle
extern I2C_HandleTypeDef hi2c1;
int main() {
Lib::HAL::I2CPort i2cPort(&hi2c1);
if (!i2cPort.init()) {
// Handle initialization failure
return -1;
}
const uint8_t deviceAddr = 0x50; // Example EEPROM address
if (!i2cPort.isReady(deviceAddr)) {
// Device not ready
return -1;
}
uint8_t dataToWrite[4] = {0x01, 0x02, 0x03, 0x04};
if (!i2cPort.send(deviceAddr, Lib::HAL::I2CPort::MemAddrSize::ADDR_16BITS, 0x0000, dataToWrite, sizeof(dataToWrite))) {
// Write failed
return -1;
}
uint8_t dataToRead[4];
if (!i2cPort.receive(deviceAddr, Lib::HAL::I2CPort::MemAddrSize::ADDR_16BITS, 0x0000, dataToRead, sizeof(dataToRead))) {
// Read failed
return -1;
}
// Data read successfully
return 0;
}
