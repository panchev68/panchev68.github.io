24 /** Buffer class for managing dynamic byte arrays.
25 *
26 * The Buffer class provides a convenient and safe way to handle dynamic byte buffers.
27 * It supports creation from existing data, resizing, element access, and conversion to raw pointers.
28 * Memory is managed automatically using std::vector.

30 class Buffer {
31 public:

39 *

43 explicit Buffer(uint8_t * data, size_t size) : bufferData(data, data + size), isLocal(false) {}
44
45 /** Constructs a buffer from existing const data.
46 *
47 * @param data Pointer to the data.
48 * @param size Size of the data.
49 */
50 explicit Buffer(const uint8_t * data, size_t size) : bufferData(data, data + size), isLocal(false) {}
51
52 /** Constructs a buffer with a specified size.
53 *
54 * @param size Initial size of the buffer.
55 */
56 explicit Buffer(size_t size) : bufferData(size), isLocal(true) {}
57
58 /** Destroys the buffer. */
59 ~Buffer() = default;

61 /** Sets the buffer size and initializes with zeros.
62 *
63 * @param size New size of the buffer.

65 void setSize(size_t size) {
66 bufferData.assign(size, 0);
67 isLocal = true;
68 }
69
70 /** Sets the buffer to reference existing data.
71 *
72 * @param data Pointer to the existing data.
73 * @param size Size of the data.
74 * @return Pointer to the buffer data.

76 uint8_t * setExisting(uint8_t * data, size_t size) {
77 assert(data);
78 bufferData.assign(data, data + size);

80 return bufferData.data();

83 /** Accesses an element by index (mutable).

85 * @param index Index of the element.
86 * @return Reference to the element.

88 uint8_t & operator[](size_t index) {
89 assert(index < bufferData.size());

93 /** Accesses an element by index (const).
94 *
95 * @param index Index of the element.

98 const uint8_t & operator[](size_t index) const {
99 assert(index < bufferData.size());

102
103 /** Clears the buffer by setting all bytes to zero. */
104 void clear() {
105 if (!bufferData.empty()) {
106 std::memset(bufferData.data(), 0, bufferData.size());
107 }
108 }
109
110 /** Converts to mutable uint8_t pointer. */
111 operator uint8_t *() const { return const_cast<uint8_t *>(bufferData.data()); }
112
113 /** Converts to const uint8_t pointer. */
114 operator const uint8_t *() const { return bufferData.data(); }
115

119 /** Gets the size of the buffer.

121 * @return Size of the buffer.

124
125 /** Creates a buffer from existing data.

131 static Buffer make(uint8_t * data, size_t size) {

143 return Buffer(size);
144 }
145
146 private:

148 bool isLocal; /*!< Flag indicating if the buffer is locally managed. */
