Lib::BSP::INA219

Demonstrates two Lib::BSP::INA219 monitors sharing one I2C bus and polled from a single task.

Demonstrates two Lib::BSP::INA219 monitors sharing one I2C bus and polled from a single task.The battery channel reports signed current, so power turns negative while charging. The motor channel uses Lib::BSP::INA219::Polarity::POSITIVE with a 20 mA offset to hide noise around zero. A reading comes back as std::nullopt on a bus failure or when the shunt ADC saturates. Both monitors are initialized before the task starts and then used only from that task, since the driver is not thread-safe. With 128-sample averaging each result takes tens of milliseconds, so polling faster than a few times per second gains nothing.

Usage

Battery and motor channels polled from one task

Lib::BSP::INA219::init() configures both monitors before the task starts. Lib::BSP::INA219::readVoltage() and Lib::BSP::INA219::readCurrent() each return std::optional, so check for a value before using it.

#include <Lib/BSP/CurrentAndPowerMonitor/INA219.hpp>
#include <Lib/HAL/I2C.hpp>
#include <Lib/RTOS/FreeRTOS/Kernel.hpp>
#include <Lib/RTOS/FreeRTOS/Task.hpp>
#include <chrono>
#include <optional>
extern "C" I2C_HandleTypeDef hi2c1; // Generated by CubeMX in main.c.
using namespace std::chrono_literals;
class PowerMonitor final {
public:
using Monitor = Lib::BSP::INA219<Lib::HAL::DriverMode::BLOCKING>;
bool begin() noexcept {
if (!battery.init() || !motor.init()) {
return false;
}
task.setup({"Power", 256, RTOS::Task::Priority::LOW});
return task.start([this] { taskEntry(); }) != nullptr;
}
private:
Lib::HAL::I2CPort<Lib::HAL::DriverMode::BLOCKING> i2cPort{&hi2c1}; ///< Bus shared by both monitors.
Monitor battery{{
.i2cPort = i2cPort,
.i2cAddr = 0x40,
.shuntResistorOhms = 0.1F,
}}; ///< Charge and discharge: signed current.
Monitor motor{{
.i2cPort = i2cPort,
.i2cAddr = 0x41,
.shuntResistorOhms = 0.01F,
.correction = {.polarity = Monitor::Polarity::POSITIVE, .offsetAmperes = 0.02F},
}}; ///< Load only: magnitude above a 20 mA noise floor.
RTOS::Task task;
void taskEntry() noexcept {
while (true) {
const std::optional<float> volts = battery.readVoltage();
const std::optional<float> amperes = battery.readCurrent();
if (volts && amperes) {
const float watts = *volts * *amperes; // Negative while charging.
}
if (const std::optional<float> motorAmperes = motor.readCurrent(); !motorAmperes) {
// Bus failure or shunt saturated: the true current is above the +-320 mV range.
}
RTOS::delay(250ms);
}
}
};
