80 /** Resets LAN9252 chip and waits for ready state.
81 *
82 * @return `true` if reset successful, `false` otherwise.
83 * @post Chip state is set to `READY` if successful, `ERROR` otherwise.
84 */
85 bool resetChip() {
86 static constexpr uint32_t RESET_LOW_MS = 50;
87 static constexpr uint32_t RESET_TIMEOUT_MS = 1000;
88 static constexpr uint32_t READY_DELAY_MS = 100;
89
90 pinReset.write(false);
91 vTaskDelay(RESET_LOW_MS);
92 pinReset.write(true);
93
94 RTOS::Timeout timeout;
95 timeout.set(RESET_TIMEOUT_MS);
96
97 while (!timeout.isExpired()) {
98 if (verifyChipId()) {
99 chipState = State::READY;
100 vTaskDelay(READY_DELAY_MS);
101 return true;
102 }
103 }
104
105 chipState = State::ERROR;
106 return false;
107 }

175 * @note LAN9252 PRAM - used for process data exchange in EtherCAT.

215 /** Reads CSR register.
216 *
217 * @note LAN9252 CSR - used for configuration and status registers.
218 *
219 * @param address Register address.
220 * @param[out] buffer Destination buffer.
221 * @param length Number of bytes to read.
222 * @pre `buffer` must not be `nullptr`.
223 * @pre `length` must be greater than 0.
224 */
225 void readCSR(uint16_t address, void * buffer, uint8_t length) {
226 Helper::U32 param{0};
227 param.v[0] = loByte(address);
228 param.v[1] = hiByte(address);
229 param.v[2] = length;
230 param.v[3] = Registers::ESC_READ_BYTE;
231
232 writeU32(Registers::ESC_CSR_CMD_REG, param.value);
233 while (readU32(Registers::ESC_CSR_CMD_REG) & Registers::ESC_CSR_BUSY) {}
234
235 param.value = readU32(Registers::ESC_CSR_DATA_REG);
236 std::memcpy(buffer, param.v, length);
237 }

239 /** Writes CSR register.
240 *
241 * @note LAN9252 CSR - used for configuration and status registers.
242 *
243 * @param address Register address.
244 * @param buffer Source buffer.
245 * @param length Number of bytes to write.
246 * @pre `buffer` must not be `nullptr`.
247 * @pre `length` must be greater than 0.

249 void writeCSR(uint16_t address, const uint8_t * buffer, uint8_t length) {

254 param.v[0] = loByte(address);

260 while (readU32(Registers::ESC_CSR_CMD_REG) & Registers::ESC_CSR_BUSY) {}
261 }

263 /** Reads from PRAM.
264 *
265 * Reads data from the LAN9252's Process RAM (PRAM) area.
266 *
267 * @note LAN9252 PRAM - used for process data exchange in EtherCAT.
268 *
269 * @param address Register address.
270 * @param[out] buffer Destination buffer.
271 * @param length Number of bytes to read.
272 * @pre `buffer` must not be `nullptr`.
273 * @pre `length` must be greater than 0.
274 */
275 void readPRAM(uint16_t address, uint8_t * buffer, uint16_t length) {
276 Helper::U64 param{0};
277 writeU32(Registers::PRAM_READ_CMD_REG, Registers::PRAM_RW_ABORT_MASK);
278 while (readU32(Registers::PRAM_READ_CMD_REG) & Registers::PRAM_RW_BUSY_8B) {}
279 param.w[0] = address;
280 param.w[1] = length;
281 param.w[2] = 0;
282 param.w[3] = 0x8000;
283 writeBuffer(Registers::PRAM_READ_ADDR_LEN_REG, reinterpret_cast<uint8_t *>(&param.value), 8);
284 while (!(readU32(Registers::PRAM_READ_CMD_REG) & Registers::IS_PRAM_SPACE_AVBL_MASK)) {}
285 param.value = readU32(Registers::PRAM_READ_FIFO_REG);
286 uint8_t bytePos = address & 0x03;
287 uint8_t partLen = std::min<uint8_t>(4 - bytePos, length);
288 std::memcpy(buffer, &param.v[bytePos], partLen);
289 length -= partLen;
290 uint16_t i = partLen;
291 if (length) {
292 spi.select();
293 spi.write(Registers::CMD_FAST_READ);
294 sendAddress(Registers::PRAM_READ_FIFO_REG);
295 spi.write(Registers::CMD_FAST_READ_DUMMY);
296 while (length) {
297 param.value = readBurstMode();
298 partLen = std::min<uint16_t>(4, length);
299 std::memcpy(buffer + i, &param, partLen);
300 i += partLen;
301 length -= partLen;
302 }
303 spi.deselect();
304 }
305 }

307 /** Writes to PRAM.
308 *
309 * Writes data to the LAN9252's Process RAM (PRAM) area.
310 *
311 * @note LAN9252 PRAM - used for process data exchange in EtherCAT.
312 *
313 * @param buffer Source buffer.
314 * @param address Register address.
315 * @param length Number of bytes to write.
316 * @pre `buffer` must not be `nullptr`.
317 * @pre `length` must be greater than 0.
318 */
319 void writePRAM(const uint8_t * buffer, uint16_t address, uint16_t length) {
320 Helper::U64 param{0};
321 writeU32(Registers::PRAM_WRITE_CMD_REG, Registers::PRAM_RW_ABORT_MASK);
322 while (readU32(Registers::PRAM_WRITE_CMD_REG) & Registers::PRAM_RW_BUSY_8B) {}
323 param.w[0] = address;
324 param.w[1] = length;
325 param.w[2] = 0;
326 param.w[3] = 0x8000;
327 writeBuffer(Registers::PRAM_WRITE_ADDR_LEN_REG, reinterpret_cast<uint8_t *>(&param.value), 8);
328 while (!(readU32(Registers::PRAM_WRITE_CMD_REG) & Registers::IS_PRAM_SPACE_AVBL_MASK)) {}
329 uint8_t available = readU32(Registers::PRAM_WRITE_CMD_REG) & Registers::PRAM_SPACE_AVBL_COUNT_MASK;
330 uint8_t bytePos = address & 0x03;
331 uint8_t partLen = std::min<uint8_t>(4 - bytePos, length);
332 param.value = 0;
333 std::memcpy(&param.v[bytePos], buffer, partLen);
334 writeU32(Registers::PRAM_WRITE_FIFO_REG, param.value);
335 available--;
336 length -= partLen;
337 uint16_t i = partLen;
338 if (length) {
339 spi.select();
340 spi.write(Registers::CMD_SERIAL_WRITE);
341 sendAddress(Registers::PRAM_WRITE_FIFO_REG);
342 while (length) {
343 partLen = std::min<uint16_t>(4, length);
344 param.value = 0;
345 std::memcpy(&param, buffer + i, partLen);
346 writeBurstMode(param.value);
347 i += partLen;
348 length -= partLen;
349 available--;
350 }
351 spi.deselect();
352 }
353 }

367 /** Reads data in burst mode.
368 *
369 * @note Burst mode - reads 4 bytes in sequence without re-sending address/command.
370 *
371 * @return 32-bit value read.
372 * @pre SPI must be selected.
373 */
374 uint32_t readBurstMode() {
375 Helper::U32 result;
376 std::array<uint8_t, 4> readBytes{0};
377 spi.readWrite(nullptr, 0, readBytes.data(), readBytes.size());
378 result.bytes.lb = readBytes[0];
379 result.bytes.hb = readBytes[1];
380 result.bytes.ub = readBytes[2];
381 result.bytes.mb = readBytes[3];
382 return result.value;
383 }
