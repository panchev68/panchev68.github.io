Lib::Helper::Convert::Bcd

Demonstrates converting BCD values to string, ASCII digit and numeric decimal value.

Demonstrates converting BCD values to string, ASCII digit and numeric decimal value.This example shows typical uses of Bcd::toString, Bcd::toAsc, Bcd::toHex and Bcd::isValid.

#include <cstdint>
#include <iostream>
#include "Lib/Helper/Convert/Bcd.hpp"
using Lib::Helper::Convert::Bcd;
int main() {
// Example 1: Convert 0x1234 (BCD -> "1234"); the result is a fixed NUL-terminated buffer
const uint16_t bcdValue = 0x1234;
const Bcd::Digits digits = Bcd::toString(bcdValue);
std::cout << "toString(0x1234) = " << digits.data() << '\n';
// Example 2: Single digit to ASCII
const char ch = Bcd::toAsc(0x05);
std::cout << "toAsc(0x05) = " << ch << '\n';
// Example 3: BCD to numeric decimal value
const uint16_t num = Bcd::toHex(0x0102); // 0x01 0x02 -> 102
std::cout << "toHex(0x0102) = " << num << '\n';
// Example 4: Malformed input is rejected with a sentinel
const bool isValid = Bcd::isValid(0x12A4); // false, nibble 0xA is no decimal digit
const bool isRejected = (Bcd::toHex(0x12A4) == Bcd::INVALID_VALUE); // true (asserts in debug builds)
(void)isValid; (void)isRejected;
return 0;
}
