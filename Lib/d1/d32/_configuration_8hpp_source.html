16 /** Configuration class for OTP memory management.
17 * Provides an interface for reading, writing, and validating configuration data stored in OTP memory.
18 */
19 class Configuration {
20 static constexpr uint32_t VALID_MASK = 0x10000000; /*!< Mask to check if the configuration is valid (bit 28 must be 0). */
21 static constexpr uint32_t EMPTY_VALUE = 0xFFFFFFFF; /*!< Value indicating empty configuration (all bits set). */

23 public:
24 /** Status enum for operation results. */

30 using Bitset = std::bitset<31>; /*!< Bitset type for configuration data. */

32 const uint32_t flashAddress; /*!< Absolute address in OTP memory. */

34 /** Constructor of Configuration class.
35 * @param offset Offset from the beginning of OTP memory.
36 * @pre `offset` must be less than the size of the OTP memory region.
37 */
38 explicit Configuration(uint32_t offset) : flashAddress(Lib::HAL::OTP::beginAddr() + offset) {
39 assert(offset < (Lib::HAL::OTP::endAddr() - Lib::HAL::OTP::beginAddr()));
40 }
41
42 /** Destructor of Configuration class. */
43 virtual ~Configuration() = default;
44
45 /** Reads the configuration bitset from OTP memory.
46 * @return Bitset representing the configuration data.
47 */
48 Bitset read() const { return Bitset(readValue(flashAddress)); }
49
50 /** Writes the configuration bitset to OTP memory.
51 * @param value Bitset to write.
52 * @return Status of the write operation.
53 */
54 Status write(const Bitset & value) {
55 auto & otp = Lib::HAL::OTP::instance();
56 bool unlocked = otp.unlock();
57 assert(unlocked);
58 auto writeToOtp = [&otp](uint32_t addr, uint32_t val) -> bool { return otp.write(addr, val & ~VALID_MASK); };
59
60 bool result = writeToOtp(flashAddress, value.to_ulong());
61 if (!result) result = writeToOtp(flashAddress, value.to_ulong());
62 bool locked = otp.lock();
63 assert(locked);
64 return result ? Status::OK : Status::ERROR;
65 }
66
67 /** Checks if the configuration is empty.
68 * @return `true` if configuration is empty, `false` otherwise.
69 */
70 virtual bool isEmpty() const { return readValue(flashAddress) == EMPTY_VALUE; }
71
72 /** Checks if the configuration is valid.
73 * @return `true` if configuration is valid, `false` otherwise.
74 */
75 virtual bool isValid() const { return (readValue(flashAddress) & VALID_MASK) == 0; }
76
77 private:
78 /** Reads a 32-bit value from OTP memory.
79 * @param flashAddr Address to read from.
80 * @return 32-bit value read from OTP.
81 */
82 uint32_t readValue(uint32_t flashAddr) const { return Lib::HAL::OTP::instance().read<uint32_t>(flashAddr); }
83 };
