RTOS::StreamBuffer

Example usage of RTOS::StreamBuffer for byte stream communication.

Demonstrates creating a stream buffer, sending and receiving data with timeouts, and checking buffer state in a FreeRTOS-based embedded system.

Example usage:

The example creates a stream buffer with 256 bytes capacity and trigger level of 10, then demonstrates sending and receiving data with state checks.

#include "Lib/RTOS/StreamBuffer.hpp"
void producerTask(void* params) {
RTOS::StreamBuffer* buffer = static_cast<RTOS::StreamBuffer*>(params);
const char data[] = "Hello, World!";
if (buffer->hasSpace()) {
size_t sent = buffer->send(data, sizeof(data), 100);
if (sent > 0) {
// Data sent successfully
}
}
}
void consumerTask(void* params) {
RTOS::StreamBuffer* buffer = static_cast<RTOS::StreamBuffer*>(params);
char data[64] = {0};
if (buffer->hasData()) {
size_t received = buffer->receive(data, sizeof(data), 100);
if (received > 0) {
// Process received data
}
}
}
int main() {
RTOS::StreamBuffer buffer;
if (buffer.create(256, 10)) {
// Create tasks
xTaskCreate(producerTask, "Producer", 256, &buffer, 1, nullptr);
xTaskCreate(consumerTask, "Consumer", 256, &buffer, 1, nullptr);
RTOS::Kernel::startScheduler();
}
return 0;
}
