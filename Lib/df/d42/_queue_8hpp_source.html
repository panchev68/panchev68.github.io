57 /** RAII wrapper for FreeRTOS queues.
58 *
59 * Provides type-safe operations for queue management, including creation, destruction,
60 * sending, receiving, and querying. Supports both dynamic and static allocation.
61 *
62 * The Queue class template provides a modern C++ interface to FreeRTOS queues with:
63 * - Compile-time type safety and size validation
64 * - Automatic resource management through RAII
65 * - Support for both heap-based (dynamic) and user-provided (static) storage
66 * - Timeout-based operations for flexible blocking behavior
67 * - ISR-safe variants for interrupt context usage
68 * - Comprehensive queue state inspection methods
69 *
70 * Usage patterns:
71 * - Use dynamic allocation for simple cases with heap availability
72 * - Use static allocation for deterministic memory usage or memory-constrained systems
73 * - Choose appropriate timeouts based on real-time requirements
74 * - Use ISR variants only from interrupt service routines
75 *
76 * @tparam T_ELEMENT_TYPE Element type, must be trivially copyable.
77 * @tparam QUEUE_LENGTH Maximum number of elements in the queue.
78 *
79 * @note Element type must satisfy std::is_trivially_copyable for safe memcpy operations.
80 * @note Queue length is validated at compile-time to be greater than zero.
81 *
82 * @attention Queue operations are not inherently thread-safe; use appropriate synchronization.
83 * @attention ISR operations should complete quickly to avoid blocking other interrupts.
84 */
85 template <typename T_ELEMENT_TYPE, size_t QUEUE_LENGTH = 1> class Queue {
86 static_assert(std::is_trivially_copyable<T_ELEMENT_TYPE>::value, "Element type must be trivially copyable");
87 static_assert(QUEUE_LENGTH > 0, "Queue length must be greater than zero");

139 /** Move assignment transfers queue ownership.
140 *

142 * The current queue is closed before the transfer.

144 * @param other Queue to move from. Becomes invalid after move.

658 * if (queue.receive(received)) {
659 * std::cout << "Received value: " << received << std::endl;

663 * return 0;
664 * }

667 * This example shows how to open a queue, send and receive an integer value, and print the queue state.
668 *
669 * ---
670 *
671 * ## Expected output:
672 * @code
673 * Sent value: 42
674 * Received value: 42
675 * Messages waiting: 0
