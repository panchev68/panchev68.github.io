34 /** Protocol helper for building and parsing XCOM frames.
35 *
36 * The `Protocol` class provides static methods to construct and decode framed messages
37 * using start delimiter, escaping, length prefix, and CRC16 checksum validation.
38 * Frames are built by prepending a header (delimiter, length, command) and appending
39 * a two-byte CRC16 checksum, then escaping special bytes. Parsing reverses this process
40 * with full validation of frame integrity.
41 *
42 * @note All byte values matching `START_DELIMITER` or `ESC_SUBST` outside the initial
43 * delimiter are escaped by replacing with `ESC_SUBST` followed by the byte XORed with
44 * `PRECODE_MASK`.
45 */
46 class Protocol {
47 public:
48 /** Status codes returned by parsing operations. */
49 enum class Status {
50 OK, /*!< Frame parsed successfully. */
51 ERROR /*!< Frame parsing failed due to invalid format or checksum. */
52 };
53
54 static constexpr uint8_t START_DELIMITER = 0x6e; /*!< Frame start delimiter value. */
55 static constexpr uint8_t ESC_SUBST = 0x6d; /*!< Escape substitution byte. */
56 static constexpr uint8_t PRECODE_MASK = 0x20; /*!< Mask applied to escaped bytes. */
57 static constexpr uint8_t MAX_PAYLOAD_SIZE = 255; /*!< Maximum allowed payload size in bytes. */
58 static constexpr size_t MAX_FRAME_SIZE = 300; /*!< Maximum allowed raw frame size. */
59
60 /** Representation of a protocol frame.
61 *
62 * Contains the command byte and an optional payload vector. The frame structure
63 * is transmitted as: [command][payload...] following the header and checksum.
64 */
65 class Frame {
66 public:
67 uint8_t command = 0; /*!< Command identifier byte (0-255). */
68 std::vector<uint8_t> payload; /*!< Payload data bytes (0-255 bytes max). */
69 };

71 /** Build a framed and escaped byte vector from a `Frame` instance.
72 *
73 * Constructs a complete frame by prepending the header (start delimiter, payload
74 * length, command) and appending a CRC16 checksum, then escaping all special bytes.
75 * The resulting frame is ready for transmission over a serial or similar interface.
76 *
77 * @param frame Source `Frame` containing command and payload data.
78 * @return Escaped frame bytes ready for transmission, including all delimiters, length, command, payload, and checksum.
79 *
80 * @pre `frame.payload.size() <= MAX_PAYLOAD_SIZE`; larger payloads are silently truncated.
81 * @post Result contains escaped representation of header + payload + CRC16 checksum.
82 */
83 static std::vector<uint8_t> build(const Frame & frame) {
84 std::vector<uint8_t> rawFrame;
85 size_t payloadSize = std::min(frame.payload.size(), static_cast<size_t>(MAX_PAYLOAD_SIZE));
86 rawFrame.reserve(sizeof(FrameHeader) + payloadSize + 2);
87 rawFrame.push_back(START_DELIMITER);
88 rawFrame.push_back(static_cast<uint8_t>(payloadSize));
89 rawFrame.push_back(frame.command);
90 rawFrame.insert(rawFrame.end(), frame.payload.begin(), frame.payload.begin() + payloadSize);
91 uint16_t crc = calcChkSum(rawFrame);
92 rawFrame.push_back(static_cast<uint8_t>(crc & 0xFF));
93 rawFrame.push_back(static_cast<uint8_t>(crc >> 8));
94 std::vector<uint8_t> escapedFrame;
95 escapedFrame.reserve(rawFrame.size() + 4);
96 escapedFrame.push_back(START_DELIMITER);
97 for (size_t index = 1; index < rawFrame.size(); ++index) {
98 uint8_t value = rawFrame[index];
99 if (isEsc(value)) {
100 escapedFrame.push_back(ESC_SUBST);
101 value ^= PRECODE_MASK;
102 }
103 escapedFrame.push_back(value);
104 }
105 return escapedFrame;
106 }

108 /** Parse an incoming frame buffer into a `Frame` object.
109 *
110 * Validates the input buffer structure, un-escapes all special bytes, extracts the
111 * command and payload, and verifies the CRC16 checksum. The frame must start with
112 * `START_DELIMITER` and contain properly formatted header and checksum.
113 *
114 * @param inputFrame Raw input bytes to parse (must start with `START_DELIMITER`).
115 * @param[out] outputFrame Decoded output `Frame` object on successful parse.
116 * @return `Status::OK` if frame parsed and verified successfully, `Status::ERROR` if validation fails.
117 *
118 * @pre `inputFrame` must not be empty and must start with `START_DELIMITER`.
119 * @post `outputFrame.command` and `outputFrame.payload` are populated on success.
120 */
121 static Status parse(const std::vector<uint8_t> & inputFrame, Frame & outputFrame) {
122 if (inputFrame.size() < sizeof(FrameHeader) + 2 || inputFrame[0] != START_DELIMITER || inputFrame.size() > MAX_FRAME_SIZE) { return Status::ERROR; }
123 std::vector<uint8_t> decodedFrame;
124 decodedFrame.reserve(inputFrame.size());
125 decodedFrame.push_back(START_DELIMITER);
126 bool isEscape = false;
127 for (size_t i = 1; i < inputFrame.size(); ++i) {
128 uint8_t value = inputFrame[i];
129 if (isEscape) {
130 value ^= PRECODE_MASK;
131 isEscape = false;
132 } else if (value == ESC_SUBST) {
133 isEscape = true;
134 continue;
135 }
136 decodedFrame.push_back(value);
137 }
138 if (decodedFrame.size() < sizeof(FrameHeader) + 2) { return Status::ERROR; }
139 uint8_t payloadLength = decodedFrame[offsetof(FrameHeader, length)];
140 size_t payloadOffset = sizeof(FrameHeader);
141 size_t crcOffset = payloadOffset + payloadLength;
142 if (crcOffset + 2 != decodedFrame.size() || payloadLength > MAX_PAYLOAD_SIZE) { return Status::ERROR; }
143 uint16_t crcCalc = calcChkSum(decodedFrame);
144 uint16_t crcFrame;
145 std::memcpy(&crcFrame, &decodedFrame[crcOffset], 2);
146 if (crcCalc != crcFrame) { return Status::ERROR; }
147 outputFrame.command = decodedFrame[offsetof(FrameHeader, command)];
148 outputFrame.payload.assign(decodedFrame.begin() + payloadOffset, decodedFrame.begin() + payloadOffset + payloadLength);
149 return Status::OK;
150 }

152 /** Check whether a byte must be escaped.
153 *
154 * Determines if a byte value matches one of the special protocol bytes (`START_DELIMITER`
155 * or `ESC_SUBST`) which require escaping during frame construction.
156 *
157 * @param value Byte value to check for special meaning.
158 * @return `true` if the byte is `START_DELIMITER` or `ESC_SUBST`, `false` otherwise.
159 */
160 static constexpr bool isEsc(uint8_t value) { return value == START_DELIMITER || value == ESC_SUBST; }
161
162 private:
163 /** Packed frame header as present on the wire before escaping.
164 *
165 * The header layout is: start delimiter, payload length, command byte.
166 * This structure defines the fixed portion of every frame transmitted.
167 */
168 __packed struct FrameHeader {
169 uint8_t startDelimiter; /*!< Start delimiter byte. */
170 uint8_t length; /*!< Payload length in bytes. */
171 uint8_t command; /*!< Command identifier. */
172 };

174 /** Calculate CRC16 checksum for a frame buffer.
175 *
176 * Computes the CRC16 checksum over the header and payload only (excluding the
177 * trailing CRC bytes themselves). The CRC is computed using the `crc16()` helper
178 * function from `Lib::Helper`.
179 *
180 * @param frameData Byte buffer containing at least a header and payload.
181 * @return Computed CRC16 value, or 0 if the buffer is invalid or too small.
182 *
183 * @pre `frameData.size() >= sizeof(FrameHeader)`; otherwise returns 0.
184 * @post Computed checksum covers bytes from index 0 to (sizeof(FrameHeader) + payloadLength - 1).
185 */
186 static uint16_t calcChkSum(const std::vector<uint8_t> & frameData) {
187 if (frameData.size() < sizeof(FrameHeader)) return 0;
188 uint8_t length = frameData[offsetof(FrameHeader, length)];
189 size_t dataLength = sizeof(FrameHeader) + length;
190 if (dataLength > frameData.size()) return 0;
191 return crc16(frameData.data(), dataLength);
192 }
193 };
