36 * Provides common functionality for interacting with I2C EEPROM devices over an I2C interface.

89 /** Opens the I2C connection and validates device presence.

93 *
94 * @return `true` if connection established and device is present, `false` otherwise.

235 /** Internal helper for writing data to the EEPROM with page-aligned optimization.
236 *
237 * Handles page-aligned write operations to optimize EEPROM write performance.
238 * Writes are split at page boundaries to comply with EEPROM page write protocol.
239 * Includes a 5ms delay between page writes as per EEPROM write cycle timing requirements.
240 *
241 * @param addr Memory address to write to.
242 * @param buffer Pointer to the buffer containing data to write.
243 * @param size Number of bytes to write.
244 * @param memAddSize Memory address size (I2C_MEMADD_SIZE_8BIT or I2C_MEMADD_SIZE_16BIT).
245 *
246 * @return `true` if write operation succeeds for all pages, `false` otherwise.
247 *
248 * @pre Device must be ready (isReady == true).
249 * @pre Address range must be valid (addr + size <= chipSize).
250 */
251 bool writeInternal(uint32_t addr, const void * buffer, size_t size, uint16_t memAddSize) {
252 assert(isReady && (addr + size) <= chipSize);
253 if (!isReady || (addr + size) > chipSize) return false;
254
255 auto * data = static_cast<const uint8_t *>(buffer);
256 uint32_t currentAddr = addr;
257 size_t remaining = size;
258
259 while (remaining > 0) {
260 const size_t pageOffset = currentAddr % pageSize;
261 const size_t writeLen = std::min(remaining, static_cast<size_t>(pageSize - pageOffset));
262
263 if (!i2cPort->send(deviceAddr, memAddSize, currentAddr, const_cast<uint8_t *>(data), writeLen, I2C_TIMEOUT)) {
264 assert(false);
265 return false;
266 }
267
268 vTaskDelay(5);
269 currentAddr += writeLen;
270 data += writeLen;
271 remaining -= writeLen;
272 }
273 return true;
274 }
