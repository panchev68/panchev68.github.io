38 /** Specialization of the Timer template for ENCODER mode.
39 *
40 * This class provides encoder-specific timer functionality, including value change notification and encoder mode initialization.
41 */
42 template <> class Timer<TimerMode::ENCODER> : public TimerBase {
43 public:
44 /** Maximum counter value for encoder mode. */
45 static constexpr uint32_t MAX_COUNTER_VALUE = 0xFFFF;
46
47 /** Callback invoked when the encoder value changes. */
48 std::function<void(uint32_t newValue)> onValueChanged;
49
50 /** Constructs an encoder timer object with the specified hardware handle.
51 *
52 * @param handle Pointer to the hardware timer handle (TIM_HandleTypeDef).
53 */
54 explicit Timer(TIM_HandleTypeDef * handle) noexcept : TimerBase(handle) {}
55
56 /** Virtual destructor for the encoder timer object. */
57 virtual ~Timer() noexcept override = default;
58
59 /** Opens and initializes the encoder timer in encoder mode.
60 *
61 * Sets the auto-reload register to the maximum counter value and starts the encoder in interrupt mode.
62 *
63 * @return `true` if the timer was successfully opened and started, `false` otherwise.
64 */
65 virtual bool open() noexcept override {
66 return TimerBase::open() &&
67 (getHandle()->Instance->ARR = MAX_COUNTER_VALUE, true) &&
68 HAL_TIM_Encoder_Start_IT(getHandle(), TIM_CHANNEL_ALL) == HAL_OK;
69 }
70
71 /** Closes and deinitializes the encoder timer.
72 *
73 * Stops the encoder in interrupt mode and closes the timer base.
74 *
75 * @return `true` if the timer was successfully stopped and closed, `false` otherwise.
76 */
77 virtual bool close() noexcept override {
78 return HAL_TIM_Encoder_Stop_IT(getHandle(), TIM_CHANNEL_ALL) == HAL_OK && TimerBase::close();
79 }
80
81 private:
82 /** Handles timer events and invokes the value change callback if appropriate.
83 *
84 * @param id Callback identifier for the timer event.
85 */
86 void eventHandler(HAL_TIM_CallbackIDTypeDef id) noexcept override {
87 if (id == HAL_TIM_IC_CAPTURE_CB_ID && onValueChanged) onValueChanged(getHandle()->Instance->CNT);
88 }
89 };
