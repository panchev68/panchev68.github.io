45 /** External interrupt class for GPIO pins with event-driven callback support.
46 *
47 * This class extends Input to provide external interrupt functionality
48 * for GPIO pins. It allows binding callback functions to interrupt events
49 * and manages event handling through a centralized EventManager singleton.
50 *
51 * Role of the type within the GPIO module as an interrupt-capable input device.
52 *
53 * Important design decisions include using singleton EventManager for centralized
54 * event dispatching, callback-based event handling, and automatic registration/unregistration.
55 *
56 * Usage example:
57 * @code
58 * ExtInterrupt extInt(handle);
59 * extInt.bind([](const ExtInterrupt * interrupt) {
60 * // Handle the interrupt event
61 * });
62 * @endcode
63 */
64 class ExtInterrupt : public Input {
65 public:
66 /** Gets a pointer to the pin number associated with this interrupt.
67 *
68 * @return Pointer to the pin number (uint16_t *).
69 */
70 uint16_t * getPointerToPinNumber() {
71 return Input::getHandle().getPointerToPinNumber();
72 }
73
74 /** Event manager singleton for external interrupt event dispatching.
75 *
76 * Manages a registry of ExtInterrupt instances and dispatches interrupt
77 * events to the appropriate handler based on pin number. Operates as a
78 * centralized event hub for the interrupt subsystem.
79 *
80 * Role of the type within the interrupt module as a singleton event dispatcher.
81 *
82 * Important design decisions include using singleton pattern for global access,
83 * unordered_map for efficient device lookup, and linear search for pin-based lookup.
84 */
85 class EventManager {
86 public:
87 /** Gets the singleton instance of EventManager.
88 *
89 * @return Reference to the singleton EventManager instance.
90 * @post Returns the singleton instance.
91 */
92 static EventManager & instance() {
93 static EventManager instance;
94 return instance;
95 }
96
97 /** Registers an ExtInterrupt device for event handling.
98 *
99 * Adds the device to the internal event map using its pin number
100 * as the lookup key.
101 *
102 * @param device Pointer to the ExtInterrupt device to register.
103 * @return True if registered successfully; false if already registered.
104 * @pre device != nullptr and device->getPointerToPinNumber() is valid.
105 * @post Device is registered for event handling.
106 */
107 bool addItem(ExtInterrupt * device) {
108 auto result = map.emplace(device, *(device->getPointerToPinNumber()));
109 return result.second;
110 }
111
112 /** Unregisters an ExtInterrupt device from event handling.
113 *
114 * Removes the device from the internal event map.
115 *
116 * @param device Pointer to the ExtInterrupt device to unregister.
117 * @return True if unregistered successfully; false if not found.
118 * @pre device was previously registered.
119 * @post Device is no longer registered for event handling.
120 */
121 bool removeItem(ExtInterrupt * device) {
122 return map.erase(device) > 0;
123 }
124
125 /** Dispatches an interrupt event to the appropriate handler.
126 *
127 * Locates the ExtInterrupt device corresponding to the pin number
128 * and invokes its event handler.
129 *
130 * @param pinNumber The GPIO pin number that triggered the interrupt.
131 * @pre A device matching pinNumber must be registered.
132 * @post The corresponding device's event handler is invoked.
133 */
134 void handleEvent(uint16_t pinNumber) {
135 auto device = findItem(pinNumber);
136 assert(device != nullptr);
137 if (device != nullptr) {
138 device->handleEvent();
139 }
140 }
141
142 private:
143 std::unordered_map<ExtInterrupt *, uint16_t> map; /*!< Maps device instances to pin numbers. */
144
145 /** Finds an ExtInterrupt device by pin number.
146 *
147 * Performs a linear search through the device map to locate the
148 * ExtInterrupt instance corresponding to a specific pin number.
149 *
150 * @param pinNumber The GPIO pin number to search for.
151 * @return Pointer to the matching ExtInterrupt device, or nullptr
152 * if no device is registered for the pin number.
153 * @pre pinNumber is a valid GPIO pin number.
154 * @post Returns the device pointer or nullptr.
155 */
156 ExtInterrupt * findItem(uint16_t pinNumber) const {
157 for (const auto & entry : map) {
158 if (entry.second == pinNumber) {
159 return entry.first;
160 }
161 }
162 return nullptr;
163 }
164 };
165
166 /** Constructor.
167 *
168 * Initializes the interrupt device with the provided GPIO handle and
169 * validates that the handle is properly configured.
170 *
171 * @param handle The GPIO handle for the pin to configure as an interrupt.
172 * @pre handle.getPort() must be valid (non-nullptr).
173 * @pre handle.getPinNumber() must be less than 16.
174 * @post The interrupt device is initialized with no bound callback.
175 */
176 explicit ExtInterrupt(const Handle & handle)
177 : Input(handle), onEventFunction{nullptr} {
178 assert(handle.getPort());
179 assert(handle.getPinNumber() < 16);
180 }

183 *
184 * Automatically unbinds any registered callback and unregisters
185 * the device from the EventManager.

187 * @post The callback is unbound and device is unregistered.

189 virtual ~ExtInterrupt() {

191 }

193 /** Reads the current state of the GPIO pin.
194 *
195 * Returns the logical state of the pin without triggering an interrupt.
196 *
197 * @return True if the pin reads high; false if the pin reads low.
198 * @pre The GPIO pin is properly configured.
199 * @post Returns the current pin state.
200 */
201 bool read() const {
202 return Input::read();
203 }
204
205 /** Registers a callback function for the interrupt event.
206 *
207 * Binds the provided callback to this interrupt instance and registers
208 * the instance with the EventManager. The callback is invoked when the
209 * interrupt is triggered.
210 *
211 * @param function The callback function to bind. The callback receives
212 * a pointer to the triggering ExtInterrupt instance as its argument.
213 * @return True if binding was successful; false if already bound.
214 * @pre function must be a valid callable (may be nullptr to skip binding).
215 * @post If successful, the callback will be called on interrupt events.
216 */
217 bool bind(const std::function<void(const ExtInterrupt *)> & function) {
218 if (!EventManager::instance().addItem(this)) return false;
219 this->onEventFunction = function;
220 return true;
221 }
222
223 /** Unregisters the callback function from the interrupt event.
224 *
225 * Removes the callback and unregisters this instance from the
226 * EventManager. After unbinding, interrupt events will not trigger
227 * the callback.
228 *
229 * @return True if unbinding was successful; false if not bound.
230 * @post The callback is cleared and the device is unregistered.
231 */
232 bool unbind() {
233 onEventFunction = nullptr;
234 if (!EventManager::instance().removeItem(this)) return false;
235 return true;
236 }
237
238 private:
239 std::function<void(const ExtInterrupt *)> onEventFunction; /*!< Registered callback function invoked on interrupt events. */
240
241 /** Invokes the registered callback function for the interrupt event.
242 *
243 * Called by EventManager when the interrupt is triggered. Executes
244 * the bound callback if one is registered.
245 *
246 * @return True if a callback was invoked; false if no callback is bound.
247 * @pre Called from interrupt context through EventManager.
248 * @post The callback function is executed if bound.
249 */
250 bool handleEvent() const {
251 if (onEventFunction) {
252 onEventFunction(this);
253 return true;
254 }
255 return false;
256 }
257 };
