34 /** Returns the elapsed ticks since a given tick count.

36 * handling wrap-around correctly for FreeRTOS tick counters.
37 * @param startTick The tick count from which to calculate elapsed time.
38 * @return Elapsed tick count.

40 static inline TickType getElapsedTicksSince(TickType startTick) noexcept {
41 const TickType now = get();

43 }

45 /** Returns the elapsed ticks since a given tick count from ISR context.

60 static inline TickType msToTicks(uint32_t ms) noexcept {
