30 /** Tick counter utility class.
31 * Provides static methods for tick counting, time conversion, and elapsed/remaining time
32 * calculations using FreeRTOS tick counters.
33 */
34 class TickCounter {

54 /** Convert milliseconds to ticks.
55 * @param ms Milliseconds to convert.
56 * @return Number of ticks corresponding to the given milliseconds.
57 */
58 static TickType msToTicks(uint32_t ms) noexcept {
59 if (ms == 0) return 0;
60#ifdef pdMS_TO_TICKS
61 return static_cast<TickType>(pdMS_TO_TICKS(ms));
62#else
63 constexpr uint32_t tickPeriodMs = getTickPeriodMs();
64 assert(ms <= std::numeric_limits<uint32_t>::max() - tickPeriodMs + 1);
65 return static_cast<TickType>((ms + tickPeriodMs - 1) / tickPeriodMs);
66#endif
67 }
68
69 /** Convert ticks to milliseconds.
70 * @param ticks Number of ticks.
71 * @return Milliseconds corresponding to the given ticks.
72 */
73 static uint32_t ticksToMs(TickType ticks) noexcept {
74 const uint32_t tickPeriodMs = getTickPeriodMs();
75 const uint64_t result = static_cast<uint64_t>(ticks) * tickPeriodMs;
76 return (result > std::numeric_limits<uint32_t>::max()) ? std::numeric_limits<uint32_t>::max() : static_cast<uint32_t>(result);
77 }
78
79 /** Convert seconds to ticks.
80 * @param seconds Seconds to convert.
81 * @return Number of ticks corresponding to the given seconds.
82 */
83 static TickType secondsToTicks(uint32_t seconds) noexcept {
84 if (seconds == 0) return 0;
85 const uint64_t ms = static_cast<uint64_t>(seconds) * MS_PER_SEC;
86 if (ms > std::numeric_limits<uint32_t>::max()) return getMaxDelay();
87 return msToTicks(static_cast<uint32_t>(ms));
88 }
89
90 /** Convert ticks to seconds.
91 * @param ticks Number of ticks.
92 * @return Seconds corresponding to the given ticks.
93 */
94 static uint32_t ticksToSeconds(TickType ticks) noexcept {
95 const uint64_t ms = static_cast<uint64_t>(ticksToMs(ticks));
96 return static_cast<uint32_t>(ms / MS_PER_SEC);
97 }
98
99 /** Get tick period in milliseconds.
100 * @return Tick period in milliseconds.

102 static constexpr uint32_t getTickPeriodMs() {
103 static_assert(configTICK_RATE_HZ != 0, "configTICK_RATE_HZ must not be zero");
104 return static_cast<uint32_t>(1000U / configTICK_RATE_HZ);
105 }

107 /** Get tick rate in Hz.
108 * @return Tick rate in Hz.

112 /** Get maximum delay in ticks.
113 * @return Maximum delay in ticks.

115 static constexpr TickType getMaxDelay() { return portMAX_DELAY; }

117 /** Get elapsed ticks between two tick counts.
118 * @param startTicks Start tick count.
119 * @param endTicks End tick count.
120 * @return Elapsed ticks between start and end, handling overflow.
121 */
122 static TickType getElapsedTicks(TickType startTicks, TickType endTicks) noexcept {
123 return (endTicks >= startTicks) ? (endTicks - startTicks) : (std::numeric_limits<TickType>::max() - startTicks + endTicks + 1);
124 }
125
126 /** Get elapsed milliseconds between two tick counts.
127 * @param startTicks Start tick count.
128 * @param endTicks End tick count.
129 * @return Elapsed milliseconds between start and end.
130 */
131 static uint32_t getElapsedMs(TickType startTicks, TickType endTicks) noexcept {
132 return ticksToMs(getElapsedTicks(startTicks, endTicks));
133 }

135 /** Get elapsed milliseconds since a given tick count.

139 static uint32_t getElapsedMsSince(TickType startTicks) noexcept {

145 * @return Elapsed ticks since start.

147 static TickType getElapsedTicksSince(TickType startTicks) noexcept {

151 /** Check if a timeout (in milliseconds) has occurred since a given tick count.
152 * @param startTicks Start tick count.
153 * @param timeoutMs Timeout in milliseconds.
154 * @return `true` if timeout has occurred, `false` otherwise.
155 */
156 static bool hasTimedOut(TickType startTicks, uint32_t timeoutMs) noexcept {
157 if (timeoutMs == 0) return true;
158 return getElapsedMsSince(startTicks) >= timeoutMs;
159 }
160
161 /** Check if a timeout (in ticks) has occurred since a given tick count.
162 * @param startTicks Start tick count.
163 * @param timeoutTicks Timeout in ticks.
164 * @return `true` if timeout has occurred, `false` otherwise.
165 */
166 static bool hasTimedOutTicks(TickType startTicks, TickType timeoutTicks) noexcept {
167 if (timeoutTicks == 0) return true;
168 return getElapsedTicksSince(startTicks) >= timeoutTicks;
169 }
170
171 /** Get remaining time in milliseconds before timeout.
172 * @param startTicks Start tick count.
173 * @param timeoutMs Timeout in milliseconds.
174 * @return Remaining milliseconds before timeout, or 0 if already timed out.
175 */
176 static uint32_t getRemainingTimeMs(TickType startTicks, uint32_t timeoutMs) noexcept {
177 if (timeoutMs == 0) return 0;
178 const uint32_t elapsed = getElapsedMsSince(startTicks);
179 return (elapsed >= timeoutMs) ? 0 : (timeoutMs - elapsed);
180 }
181
182 /** Get remaining ticks before timeout.
183 * @param startTicks Start tick count.
184 * @param timeoutTicks Timeout in ticks.
185 * @return Remaining ticks before timeout, or 0 if already timed out.
186 */
187 static TickType getRemainingTicks(TickType startTicks, TickType timeoutTicks) noexcept {
188 if (timeoutTicks == 0) return 0;
189 const TickType elapsed = getElapsedTicksSince(startTicks);
190 return (elapsed >= timeoutTicks) ? 0 : (timeoutTicks - elapsed);
191 }
