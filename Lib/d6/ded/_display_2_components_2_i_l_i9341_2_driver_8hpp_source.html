120 nssPin.write(1);

138 /** Transmits a block of pixel data to the display at specified coordinates.
139 *
140 * Configures the display address window to the specified region and transfers
141 * pixel data via SPI DMA. Pixel data is expected in RGB565 format.
142 *
143 * @param pixelBuffer Pointer to pixel data in RGB565 format.
144 * @param x Horizontal position (column) of the block.
145 * @param y Vertical position (row) of the block.
146 * @param width Block width in pixels.
147 * @param height Block height in pixels.
148 * @pre pixelBuffer must be non-null and valid.
149 * @pre width and height must be non-zero.
150 * @pre width * height must not exceed 65535 (16-bit limit).
151 * @note This method initiates DMA transfer; completion is notified via callback.
152 * @see setAddressWindow
153 */
154 virtual void transmitBlock(const void * pixelBuffer, uint32_t x, uint32_t y, size_t width, size_t height) override {
155 assert(pixelBuffer);
156 assert(width);
157 assert(height);
158 assert(width * height <= 0xffff);
159 const auto * pixelData = static_cast<const uint16_t *>(pixelBuffer);
160 auto * spiHandle = spiPort.getHandle();
161 configureSpi(spiHandle, SPI_DATASIZE_8BIT);
162 setAddressWindow(x, y, width, height);
163 sendCommand(Registers::RAMWR);
164 configureSpi(spiHandle, SPI_DATASIZE_16BIT);
165 setTransferActive(1);
166 spiPort.sendBlock(pixelData, width * height);
167 }

169 /** Configures the display rotation mode.
170 *
171 * Sets the display memory access control register (MADCTL) to enable or disable
172 * 90-degree rotation of the display output.
173 *
174 * @param isRotated True to enable 90-degree rotation, false for normal orientation.
175 * @pre Display must be initialized via setup().
176 * @note Changes take effect immediately on next pixel transmission.
177 */
178 virtual void setRotation(bool isRotated) {
179 sendCommand(Registers::MADCTL);
180 sendByte(isRotated ? 0b10111000 : 0b01111000);
181 }

321 bool sendData(const void * dataBuffer, size_t bufferSize, uint32_t timeout = 0) override {
322 (void)timeout;
323 const bool result = HAL_SPI_Transmit_DMA(
324 getHandle(),
325 static_cast<uint8_t *>(const_cast<void *>(dataBuffer)),

337 /** Configures SPI data size and reinitializes peripheral if needed.
338 *
339 * Dynamically adjusts SPI data size between 8-bit and 16-bit modes based on
340 * communication requirements (8-bit for commands, 16-bit for pixel data).
341 *
342 * @param spiHandle SPI peripheral handle.
343 * @param dataSize SPI data size (SPI_DATASIZE_8BIT or SPI_DATASIZE_16BIT).
344 */
345 void configureSpi(SPI_HandleTypeDef * spiHandle, uint32_t dataSize) {
346 if (spiHandle->Init.DataSize != dataSize) {
347 spiHandle->Init.DataSize = dataSize;
348 HAL_SPI_Init(spiHandle);
349 }

352 /** Sets the display address window (drawing area) for pixel transmission.
353 *
354 * Configures column (X) and page (Y) address registers to define the rectangular
355 * region where subsequent pixel data will be written.
356 *
357 * @param x Starting column address.
358 * @param y Starting row address.
359 * @param width Window width in pixels.
360 * @param height Window height in pixels.
361 * @see setPosition
362 */
363 void setAddressWindow(uint32_t x, uint32_t y, size_t width, size_t height) {
364 setPosition(Registers::CASET, x, x + width - 1);
365 setPosition(Registers::PASET, y, y + height - 1);
366 }
367
368 /** Sets address range in a specific position register (column or page).
369 *
370 * Configures either column (CASET) or row (PASET) address registers with start
371 * and end addresses to define the drawing region boundaries.
372 *
373 * @param registerAddress Register address (CASET for columns, PASET for rows).
374 * @param startAddress Start position value.
375 * @param endAddress End position value.
376 */
377 void setPosition(uint8_t registerAddress, uint32_t startAddress, uint32_t endAddress) {
378 const uint8_t dataBuffer[4] = {HI_BYTE(startAddress), LO_BYTE(startAddress), HI_BYTE(endAddress), LO_BYTE(endAddress)};
379 sendCommand(registerAddress);
380 for (const uint8_t dataByte : dataBuffer) {
381 spiPort.sendByte(dataByte);
382 }
383 }

385 /** Sends a command byte to the ILI9341 controller via SPI.
386 *
387 * Sets Data/Command pin to command mode (low), transmits the command byte,
388 * and returns to data mode (high).
389 *
390 * @param commandByte Command code to transmit.
391 */
392 void sendCommand(uint8_t commandByte) {
393 dcPin.write(0);
394 sendByte(commandByte);
395 dcPin.write(1);
396 }

398 /** Sends a single data byte via SPI to the display controller. */

401 /** Performs hardware reset of the ILI9341 display controller.
402 *
403 * Executes a power-on reset sequence: asserts reset signal (low),
404 * waits for specified duration, deasserts reset signal (high),
405 * and waits for device recovery and stabilization.
406 *
407 * @see RESET_DELAY_LOW, RESET_DELAY_HIGH

409 void resetDevice() {
