56 /** Initializes the ILI9341 display controller.
57 * Performs hardware reset, applies initialization sequence, and enables display output.
58 * @return true if initialization succeeded, false otherwise.
59 */
60 bool setup() override {
61 constexpr uint8_t INIT_SEQUENCE[] = {
62 4, Registers::POWCTRB, 0x00, 0xc3, 0x30,
63 5, Registers::POWONSEQCTRL, 0x64, 0x03, 0x12, 0x81,
64 4, Registers::DRVTCTRA, 0x85, 0x10, 0x79,
65 6, Registers::POWCTRA, 0x39, 0x2c, 0x00, 0x34, 0x02,
66 2, Registers::PUMPCRTL, 0x20,
67 3, Registers::DRVTCTRB, 0x00, 0x00,
68 2, Registers::PWCTR1, 0x22,
69 2, Registers::PWCTR2, 0x11,
70 3, Registers::VMCTR1, 0x3a, 0x1c,
71 2, Registers::VMCTR2, 0xa9,
72 2, Registers::MADCTL, 0b01111000,
73 2, Registers::PIXFMT, 0x55,
74 3, Registers::FRMCTR1, 0x00, 0x1b,
75 3, Registers::DFUNCTR, 0x0a, 0xa2,
76 3, 0xf6, 0x01, 0x1d,
77 2, Registers::ENBLABLE3G, 0x00,
78 2, Registers::GAMMASET, 0x01,
79 16, Registers::GMCTRP1, 0x0f, 0x3f, 0x2f, 0x0c, 0x10, 0x0a, 0x53, 0xd5, 0x40, 0x0a, 0x13, 0x03, 0x08, 0x03, 0x00,
80 16, Registers::GMCTRN1, 0x00, 0x00, 0x10, 0x03, 0x0f, 0x05, 0x2c, 0xa2, 0x3f, 0x05, 0x0e, 0x0c, 0x37, 0x3c, 0x0F,
81 1, Registers::INVON,
82 1, Registers::SLPOUT,
83 0};
84 csPin.write(1);
85 spiPort.init();
86 setBacklight(BACKLIGHT_OFF); // Ensure backlight is off during initialization
87 vTaskDelay(INIT_DELAY_SHORT); // Wait for power stabilization
88 resetDevice(); // Perform hardware reset
89
90 csPin.write(0); // Select the device
91 // Send initialization commands
92 for (const uint8_t * seq = INIT_SEQUENCE; *seq; seq += seq[0] + 1) {
93 const uint8_t dataCount = seq[0] - 1;
94 sendCommand(seq[1]);
95 for (uint8_t i = 0; i < dataCount; ++i) sendByte(seq[2 + i]);
96 }
97 vTaskDelay(INIT_DELAY_LONG); // Wait for internal processing
98 sendCommand(Registers::DISPON); // Turn on the display
99 vTaskDelay(INIT_DELAY_FINAL); // Wait for display to stabilize
100 setBacklight(BACKLIGHT_ON); // Enable backlight
101 return true;
102 }

104 /** Transmits a rectangular block of pixel data to the display.
105 * @param buffer Pointer to pixel data buffer.
106 * @param x X-coordinate of block start.
107 * @param y Y-coordinate of block start.
108 * @param width Width of the block in pixels.
109 * @param height Height of the block in pixels.
110 */
111 void transmitBlock(const void * pixelBuffer, uint32_t x, uint32_t y, size_t width, size_t height) override {
112 assert(pixelBuffer);
113 assert(width);
114 assert(height);
115 assert(width * height <= 0xffff);
116 const auto * pixelData = static_cast<const uint16_t *>(pixelBuffer);
117 auto * spiHandle = spiPort.getHandle();
118 configureSpi(spiHandle, SPI_DATASIZE_8BIT);
119 setAddressWindow(x, y, width, height);
120 sendCommand(Registers::RAMWR);
121 configureSpi(spiHandle, SPI_DATASIZE_16BIT);
122 setTransferActive(1);
123 spiPort.sendBlock(pixelData, width * height);
124 }

146 /** Sends a block of pixel data via DMA.
147 * @param buffer Pointer to pixel data buffer.
148 * @param size Number of pixels (16-bit values).
149 * @return true if DMA transmission started, false otherwise.
150 */
151 bool sendBlock(const uint16_t * dataBuffer, size_t pixelCount) {
152 assert(dataBuffer);
153 assert(pixelCount);
154 return sendData(dataBuffer, pixelCount);
155 }
156
157 /** Sends a single byte via SPI synchronously in blocking mode.
158 * @param value Byte value to send.
159 * @return true if transmission completed, false otherwise.
160 */
161 bool sendByte(uint8_t dataByte) {
162 const bool result = (HAL_SPI_Transmit(getHandle(), &dataByte, sizeof(uint8_t), SPI_TIMEOUT) == HAL_OK);
163 assert(result);
164 return result;
165 }
166
167 /** Handles SPI DMA transfer complete event. */
168 void handleTxCompleteEvent() { parent->handleTransferCompleteEvent(); }
169
170 private:
171 Driver<Controller::ILI9341> * parent; /*!< Parent driver instance pointer. */
172
173 /** Sends data buffer using DMA via SPI.
174 * @param buffer Pointer to data buffer.
175 * @param bufferSize Size of buffer in bytes.
176 * @param timeout Timeout value (unused for DMA operations).
177 * @return true if DMA transmission started, false otherwise.
178 */
179 bool sendData(const void * dataBuffer, size_t bufferSize, uint32_t timeout = 0) override {
180 (void)timeout; // DMA operations do not use timeout
181 const bool result = HAL_SPI_Transmit_DMA(
182 getHandle(),
183 static_cast<uint8_t *>(const_cast<void *>(dataBuffer)),
184 static_cast<uint16_t>(bufferSize)) == HAL_OK;
185 assert(result);
186 return result;
187 }

199 void configureSpi(SPI_HandleTypeDef * spiHandle, uint32_t dataSize) {

201 spiHandle->Init.DataSize = dataSize;

206 /** Sets the address window for pixel data transfer.
207 * @param x X-coordinate of window start.
208 * @param y Y-coordinate of window start.
209 * @param width Width of the window in pixels.
210 * @param height Height of the window in pixels.
211 */
212 void setAddressWindow(uint32_t x, uint32_t y, size_t width, size_t height) {
213 setPosition(Registers::CASET, x, x + width - 1);
214 setPosition(Registers::PASET, y, y + height - 1);
215 }
216
217 /** Sends position command and column/row address data.
218 * @param reg Register address (CASET or PASET).
219 * @param start Start address value.
220 * @param end End address value.
221 */
222 void setPosition(uint8_t registerAddress, uint32_t startAddress, uint32_t endAddress) {
223 const uint8_t dataBuffer[4] = {HI_BYTE(startAddress), LO_BYTE(startAddress), HI_BYTE(endAddress), LO_BYTE(endAddress)};
224 sendCommand(registerAddress);
225 for (const uint8_t dataByte : dataBuffer) {
226 spiPort.sendByte(dataByte);
227 }
228 }

230 /** Sets display orientation mode.
231 * @param rotate true for rotated orientation, false for default.
232 */
233 void setRotation(bool isRotated) {
234 sendCommand(Registers::MADCTL);
235 sendByte(isRotated ? 0b10111000 : 0b01111000);
236 }
