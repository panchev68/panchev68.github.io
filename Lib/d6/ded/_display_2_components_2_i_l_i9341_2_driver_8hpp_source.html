74 /** Initializes the ILI9341 display controller.
75 *
76 * Performs complete hardware initialization including power sequencing, interface
77 * configuration, gamma correction, and display activation. Sends initialization
78 * sequence to the device via SPI with appropriate timing delays.
79 *
80 * @return True if initialization completed successfully, false otherwise.
81 * @pre SPI interface and GPIO pins must be configured.
82 * @post Display is ready for pixel data transmission and rotation control.
83 * @note This method blocks for the duration of initialization delays.
84 */
85 virtual bool setup() noexcept override {

87 4, Registers::POWCTRB, 0x00, 0xc3, 0x30,
88 5, Registers::POWONSEQCTRL, 0x64, 0x03, 0x12, 0x81,
89 4, Registers::DRVTCTRA, 0x85, 0x10, 0x79,

91 2, Registers::PUMPCRTL, 0x20,
92 3, Registers::DRVTCTRB, 0x00, 0x00,
93 2, Registers::PWCTR1, 0x22,
94 2, Registers::PWCTR2, 0x11,
95 3, Registers::VMCTR1, 0x3a, 0x1c,
96 2, Registers::VMCTR2, 0xa9,
97 2, Registers::MADCTL, 0b01111000,
98 2, Registers::PIXFMT, 0x55,
99 3, Registers::FRMCTR1, 0x00, 0x1b,
100 3, Registers::DFUNCTR, 0x0a, 0xa2,
101 3, 0xf6, 0x01, 0x1d,
102 2, Registers::ENBLABLE3G, 0x00,
103 2, Registers::GAMMASET, 0x01,
104 16, Registers::GMCTRP1, 0x0f, 0x3f, 0x2f, 0x0c, 0x10, 0x0a, 0x53, 0xd5, 0x40, 0x0a, 0x13, 0x03, 0x08, 0x03, 0x00,
105 16, Registers::GMCTRN1, 0x00, 0x00, 0x10, 0x03, 0x0f, 0x05, 0x2c, 0xa2, 0x3f, 0x05, 0x0e, 0x0c, 0x37, 0x3c, 0x0F,
106 1, Registers::INVON,
107 1, Registers::SLPOUT,
108 0};
109 nssPin.write(1);
110 spiPort.init();
111 setBacklight(BACKLIGHT_OFF);
112 vTaskDelay(INIT_DELAY_SHORT);
113 resetDevice();
114 nssPin.write(0);
115 for (const uint8_t * seq = INIT_SEQUENCE; *seq; seq += seq[0] + 1) {
116 const uint8_t dataCount = seq[0] - 1;
117 sendCommand(seq[1]);
118 for (uint8_t i = 0; i < dataCount; ++i) sendByte(seq[2 + i]);
119 }
120 vTaskDelay(INIT_DELAY_LONG);
121 sendCommand(Registers::DISPON);
122 vTaskDelay(INIT_DELAY_FINAL);
123 setBacklight(BACKLIGHT_ON);
124 return true;
125 }

127 /** Transmits a block of pixel data to the display at specified coordinates.
128 *
129 * Configures the display address window to the specified region and transfers
130 * pixel data via SPI DMA. Pixel data is expected in RGB565 format.
131 *
132 * @param pixelBuffer Pointer to pixel data in RGB565 format.
133 * @param x Horizontal position (column) of the block.
134 * @param y Vertical position (row) of the block.
135 * @param width Block width in pixels.
136 * @param height Block height in pixels.
137 * @pre pixelBuffer must be non-null and valid.
138 * @pre width and height must be non-zero.
139 * @pre width * height must not exceed 65535 (16-bit limit).
140 * @note This method initiates DMA transfer; completion is notified via callback.
141 * @see setAddressWindow
142 */
143 virtual void transmitBlock(const void * pixelBuffer, uint32_t x, uint32_t y, size_t width, size_t height) noexcept override {
144 assert(pixelBuffer);
145 assert(width);
146 assert(height);
147 assert(width * height <= 0xffff);
148 const auto * pixelData = static_cast<const uint16_t *>(pixelBuffer);
149 auto * spiHandle = spiPort.getHandle();
150 configureSpi(spiHandle, SPI_DATASIZE_8BIT);
151 setAddressWindow(x, y, width, height);
152 sendCommand(Registers::RAMWR);
153 configureSpi(spiHandle, SPI_DATASIZE_16BIT);
154 setTransferActive(1);
155 spiPort.sendBlock(pixelData, width * height);
156 }

158 /** Configures the display rotation mode.
159 *
160 * Sets the display memory access control register (MADCTL) to enable or disable
161 * 90-degree rotation of the display output.
162 *
163 * @param isRotated True to enable 90-degree rotation, false for normal orientation.
164 * @pre Display must be initialized via setup().
165 * @note Changes take effect immediately on next pixel transmission.
166 */
167 virtual void setRotation(bool isRotated) noexcept {
168 sendCommand(Registers::MADCTL);
169 sendByte(isRotated ? 0b10111000 : 0b01111000);
170 }

292 /** Sends data buffer via SPI DMA.
293 *

303 /** Sends data buffer via SPI DMA.
304 *
305 * @param dataBuffer Pointer to data buffer.
306 * @param bufferSize Size of data buffer in bytes.
307 * @param timeout Timeout for transmission (not used in DMA mode).
308 * @return True if transmission initiated successfully, false otherwise.
309 */
310 bool sendData(const void * dataBuffer, size_t bufferSize, uint32_t timeout = 0) noexcept override {
311 (void)timeout;
312 const bool result = HAL_SPI_Transmit_DMA(
313 getHandle(),
314 static_cast<uint8_t *>(const_cast<void *>(dataBuffer)),
315 static_cast<uint16_t>(bufferSize)) == HAL_OK;
316 assert(result);
317 return result;
318 }

326 /** Configures SPI data size and reinitializes peripheral if needed.
327 *
328 * Dynamically adjusts SPI data size between 8-bit and 16-bit modes based on
329 * communication requirements (8-bit for commands, 16-bit for pixel data).
330 *
331 * @param spiHandle SPI peripheral handle.
332 * @param dataSize SPI data size (SPI_DATASIZE_8BIT or SPI_DATASIZE_16BIT).
333 */
334 void configureSpi(SPI_HandleTypeDef * spiHandle, uint32_t dataSize) noexcept {
335 if (spiHandle->Init.DataSize != dataSize) {
336 spiHandle->Init.DataSize = dataSize;
337 HAL_SPI_Init(spiHandle);
338 }
339 }

341 /** Sets the display address window (drawing area) for pixel transmission.
342 *
343 * Configures column (X) and page (Y) address registers to define the rectangular
344 * region where subsequent pixel data will be written.
345 *
346 * @param x Starting column address.
347 * @param y Starting row address.
348 * @param width Window width in pixels.
349 * @param height Window height in pixels.
350 * @see setPosition
351 */
352 void setAddressWindow(uint32_t x, uint32_t y, size_t width, size_t height) noexcept {
353 setPosition(Registers::CASET, x, x + width - 1);
354 setPosition(Registers::PASET, y, y + height - 1);
355 }
356
357 /** Sets address range in a specific position register (column or page).
358 *
359 * Configures either column (CASET) or row (PASET) address registers with start
360 * and end addresses to define the drawing region boundaries.
361 *
362 * @param registerAddress Register address (CASET for columns, PASET for rows).
363 * @param startAddress Start position value.
364 * @param endAddress End position value.
365 */
366 void setPosition(uint8_t registerAddress, uint32_t startAddress, uint32_t endAddress) noexcept {
367 const uint8_t dataBuffer[4] = {HI_BYTE(startAddress), LO_BYTE(startAddress), HI_BYTE(endAddress), LO_BYTE(endAddress)};
368 sendCommand(registerAddress);
369 for (const uint8_t dataByte : dataBuffer) {
370 spiPort.sendByte(dataByte);
371 }
372 }
373
374 /** Sends a command byte to the ILI9341 controller via SPI.
375 *
376 * Sets Data/Command pin to command mode (low), transmits the command byte,
377 * and returns to data mode (high).
378 *
379 * @param commandByte Command code to transmit.
380 */
381 void sendCommand(uint8_t commandByte) noexcept {
382 dcPin.write(0);
383 sendByte(commandByte);
384 dcPin.write(1);
385 }
386
387 /** Sends a single data byte via SPI to the display controller. */
388 void sendByte(uint8_t dataByte) noexcept { spiPort.sendByte(dataByte); }
389
390 /** Performs hardware reset of the ILI9341 display controller.
391 *
392 * Executes a power-on reset sequence: asserts reset signal (low),
393 * waits for specified duration, deasserts reset signal (high),
394 * and waits for device recovery and stabilization.
395 *
396 * @see RESET_DELAY_LOW, RESET_DELAY_HIGH
397 */
398 void resetDevice() noexcept {
399 nrstPin.write(0);
400 vTaskDelay(RESET_DELAY_LOW);
401 nrstPin.write(1);
402 vTaskDelay(RESET_DELAY_HIGH);
403 }
