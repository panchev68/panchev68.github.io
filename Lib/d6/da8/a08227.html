Lib::Helper::Convert::Hex example

Shows the conversions of Lib::Helper::Convert::Hex for 'uint8_t', 'uint16_t', 'uint32_t' and 'uint64_t': hexadecimal string, packed BCD value and decimal string.

The helpers are 'static constexpr' and do not touch the RTOS, so they are safe from tasks and ISRs. Every string result is a fixed, NUL-terminated std::array, so it is printed through 'data()'. Lib::Helper::Convert::Hex::toBcd() is available only for types up to 32 bits.

Usage

Eight-bit values

#include <Lib/Helper/Convert/Hex.hpp>
#include <cstdint>
#include <iostream>
void printHexU8() {
using Lib::Helper::Convert::Hex;
const uint8_t v8 = 0xAB; // 171 decimal
std::cout << "Hex<uint8_t>::toString(0xAB) = " << Hex<uint8_t>::toString(v8).data() << '\n'; // "AB"
std::cout << "Hex<uint8_t>::toBcd(171) = 0x" << std::hex << Hex<uint8_t>::toBcd(171) << std::dec << '\n'; // 0x171
std::cout << "Hex<uint8_t>::toBcdString(5, true) = " << Hex<uint8_t>::toBcdString(5, true).data() << '\n'; // "005"
}

Sixty-four-bit values

Lib::Helper::Convert::Hex::toBcd() is not available for 'uint64_t'; use Lib::Helper::Convert::Hex::toBcdString() for the decimal form.

#include <Lib/Helper/Convert/Hex.hpp>
#include <cstdint>
#include <iostream>
void printHexU64() {
using Lib::Helper::Convert::Hex;
const uint64_t v64 = 0x0123456789ABCDEFULL;
std::cout << "Hex<uint64_t>::toString(0x0123456789ABCDEF) = " << Hex<uint64_t>::toString(v64).data() << '\n';
std::cout << "Hex<uint64_t>::toBcdString(UINT64_MAX) = " << Hex<uint64_t>::toBcdString(UINT64_MAX).data() << '\n';
}
