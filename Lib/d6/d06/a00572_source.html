24 /** StreamBuffer class.
25 *
26 * This class provides a C++ wrapper around FreeRTOS stream buffers for safe and efficient data streaming.
27 *
28 * @note This class is not thread-safe for multiple writers/readers; use appropriate synchronization.

87 /** Opens the stream buffer.
88 *
89 * Creates a new FreeRTOS stream buffer with the specified size and trigger level.
90 *
91 * @param bufferSize The size of the buffer in bytes.
92 * @param triggerLevel The trigger level for notifications.
93 * @return True if the buffer was opened successfully, false otherwise.
94 */
95 bool open(size_t bufferSize, size_t triggerLevel) {
96 if (handle || bufferSize == 0 || triggerLevel == 0 || triggerLevel > bufferSize)
97 return false;
98 handle = xStreamBufferCreate(bufferSize, triggerLevel);
99 if (handle) capacity = bufferSize;
100 return handle != nullptr;
101 }

103 /** Closes the stream buffer.
104 *
105 * Deletes the FreeRTOS stream buffer and resets the handle.
106 */
107 void close() {
108 if (handle) {
109 vStreamBufferDelete(handle);
110 handle = nullptr;
111 capacity = 0;
112 }
113 }

115 /** Sends data to the stream buffer.

122 * @return Number of bytes sent.

124 size_t send(const void * source, size_t length, TickType timeout = 0) {

139 /** Sends data from a container.
140 *
141 * Sends data from a container that supports data() and size() methods.

144 * @param c The container to send.

151 /** Sends data from ISR.
152 *
153 * Sends data from a raw buffer within an ISR context.
154 *
155 * @param source Pointer to the data to send.
156 * @param length Length of the data in bytes.
157 * @return Number of bytes sent.
158 */
159 size_t sendFromISR(const void * source, size_t length) {
160 if (!handle || !source || length == 0) return 0;
161 BaseType_t isWoken = pdFALSE;
162 const size_t bytesSent = xStreamBufferSendFromISR(handle, source, length, &isWoken);
163 portYIELD_FROM_ISR(isWoken);
164 return bytesSent;
165 }

175 template <typename Container, typename = decltype(std::declval<const Container>().data()), typename = decltype(std::declval<const Container>().size())>
