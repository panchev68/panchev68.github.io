23 /** Ring buffer template class.
24 *
25 * Implements a fixed-size circular buffer for FIFO operations. Elements are stored in a contiguous
26 * array, and indices wrap around when reaching the end. Push operations fail if the buffer is full,
27 * and pop operations fail if empty.
28 *
29 * @tparam T The type of elements stored in the buffer.
30 * @tparam T_SIZE The fixed capacity of the buffer (must be greater than 0).
31 */
32 template <typename T, size_t T_SIZE> class RingBuffer {
33 public:
34 /** Capacity of the ring buffer. */
35 static constexpr size_t CAPACITY = T_SIZE;
36
37 static_assert(T_SIZE > 0, "RingBuffer capacity must be greater than 0");
38
39 /** Default constructor. */
40 constexpr RingBuffer() = default;
41
42 /** Destructor. */
43 ~RingBuffer() = default;
44
45 /** Push a value into the buffer (copy).
46 *
47 * Adds a copy of the value to the end of the buffer if space is available.
48 *
49 * @param value The value to push.
50 * @return True if the value was pushed successfully, false if the buffer is full.
51 */
52 bool push(const T & value) {
53 if (isFull())
54 return false;
55 buffer[writePos] = value;
56 writePos = (writePos + 1) % CAPACITY;
57 ++count;
58 return true;
59 }
60
61 /** Push a value into the buffer (move).
62 *
63 * Moves the value to the end of the buffer if space is available.
64 *
65 * @param value The value to push (moved).
66 * @return True if the value was pushed successfully, false if the buffer is full.
67 */
68 bool push(T && value) {
69 if (isFull())
70 return false;
71 buffer[writePos] = std::move(value);
72 writePos = (writePos + 1) % CAPACITY;
73 ++count;
74 return true;
75 }
76
77 /** Pop a value from the buffer.
78 *
79 * Removes and returns the front value from the buffer if available.
80 *
81 * @param value Reference to store the popped value.
82 * @return True if a value was popped successfully, false if the buffer is empty.
83 */
84 bool pop(T & value) {
85 if (isEmpty())
86 return false;
87 value = std::move(buffer[readPos]);
88 readPos = (readPos + 1) % CAPACITY;
89 --count;
90 return true;
91 }
92
93 /** Access the front element.
94 *
95 * Returns a reference to the front element without removing it.
96 *
97 * @return Reference to the front element.
98 * @pre Buffer must not be empty.
99 */
100 T & front() {
101 assert(!isEmpty());
102 return buffer[readPos];
103 }
104
105 /** Access the front element (const).
106 *
107 * Returns a const reference to the front element without removing it.
108 *
109 * @return Const reference to the front element.
110 * @pre Buffer must not be empty.
111 */
112 const T & front() const {
113 assert(!isEmpty());
114 return buffer[readPos];
115 }
116
117 /** Access the back element.
118 *
119 * Returns a reference to the back element without removing it.
120 *
121 * @return Reference to the back element.
122 * @pre Buffer must not be empty.
123 */
124 T & back() {
125 assert(!isEmpty());
126 return buffer[(writePos + CAPACITY - 1) % CAPACITY];
127 }
128
129 /** Access the back element (const).
130 *
131 * Returns a const reference to the back element without removing it.
132 *
133 * @return Const reference to the back element.
134 * @pre Buffer must not be empty.
135 */
136 const T & back() const {
137 assert(!isEmpty());
138 return buffer[(writePos + CAPACITY - 1) % CAPACITY];
139 }
140
141 /** Clear the buffer.
142 *
143 * Resets the buffer to empty state, discarding all elements.
144 */
145 void clear() {
146 readPos = 0;
147 writePos = 0;
148 count = 0;
149 }
150
151 /** Get the number of available elements.
152 *
153 * @return The number of elements currently in the buffer.
154 */
155 size_t available() const { return count; }
156
157 /** Check if the buffer is empty.
158 *
159 * @return True if the buffer contains no elements, false otherwise.
160 */
161 bool isEmpty() const { return count == 0; }
162
163 /** Check if the buffer is full.
164 *
165 * @return True if the buffer is at capacity, false otherwise.
166 */
167 bool isFull() const { return count == CAPACITY; }
168
169 private:
170 std::array<T, T_SIZE> buffer; /*!< Internal buffer storage. */
171 size_t writePos{0}; /*!< Current write position. */
172 size_t readPos{0}; /*!< Current read position. */
173 size_t count{0}; /*!< Number of elements in the buffer. */
174 };
