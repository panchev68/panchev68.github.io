146 /** Calculate Serial Number from Device UID
147 *
148 * Computes a unique serial number by reading the device's UID from memory,
149 * rearranging the bytes for consistency, and hashing the result using CityHash.
150 * This ensures a deterministic and unique identifier for the device.
151 *
152 * @return The computed serial number as a uint32_t value.
153 *
154 * @note Performs memory operations on UID_BASE; assumes valid pointer.
155 *
156 * @pre UID_BASE must be initialized and point to valid UID memory.
157 * @post Returns a hashed value based on the UID.
158 */
159 static uint32_t calculate() {
160 std::array<char, UID_TOTAL_SIZE> uid{};
161 assert(UID_BASE != nullptr);
162 std::memcpy(uid.data() + 8, reinterpret_cast<const void *>(UID_BASE), UID_PART_SIZE);
163 std::memcpy(uid.data() + 4, reinterpret_cast<const void *>(UID_BASE + UID_PART_SIZE), UID_PART_SIZE);
164 std::memcpy(uid.data(), reinterpret_cast<const void *>(UID_BASE + 2 * UID_PART_SIZE), UID_PART_SIZE);
165 return Lib::Helper::CityHash(uid.data(), UID_TOTAL_SIZE);
166 }
