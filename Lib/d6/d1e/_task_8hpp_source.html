113 /** Starts the FreeRTOS task using the stored configuration.
114 * @param function Task entry point function (optional).
115 * @return Task handle if creation succeeded, nullptr otherwise.
116 */
117 Handle start(std::function<void()> function = nullptr) {
118 if (config.stackSize == 0) config.stackSize = configMINIMAL_STACK_SIZE;
119 if (function != nullptr) this->function = function;
120 const static auto taskEntry = [](void * owner) {
121 auto * task = static_cast<Task *>(owner);
122 if (task != nullptr && task->function != nullptr) {
123 task->function();
124 }
125 // Delete the currently running task
126 vTaskDelete(nullptr);
127 };
128
129 assert(this->function != nullptr);
130 if (this->function == nullptr) return nullptr;
131
132 const BaseType_t result = xTaskCreate(taskEntry, config.name.data(), static_cast<uint16_t>(config.stackSize), this, static_cast<UBaseType_t>(config.priority), &handle);
133 if (result != pdPASS) {
134 handle = nullptr;
135 return nullptr;
136 }
137 return handle;
138 }
