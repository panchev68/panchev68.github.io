165 /** Starts the task with the given function.
166 * @param func Function to execute in the task.
167 * @return Task handle.
168 */
169 template<typename Callable>
170 Handle start(Callable && func) noexcept {
171 if (data.handle) return data.handle;
172 auto callable = std::make_unique<std::decay_t<Callable>>(std::forward<Callable>(func));
173 auto taskEntry = [](void * param) {
174 auto callable = std::unique_ptr<std::decay_t<Callable>>(static_cast<std::decay_t<Callable> *>(param));
175 if (callable) (*callable)();
176 vTaskDelete(nullptr);
177 };
178 const BaseType_t result = xTaskCreate(
179 taskEntry,
180 data.name.empty() ? "Task" : data.name.c_str(),
181 data.stackSize / sizeof(StackType_t),
182 callable.get(),
183 data.priority,
184 &data.handle);
185 if (result == pdPASS) {
186 callable.release();
187 return data.handle;
188 }
189 data.handle = nullptr;
190 return nullptr;
191 }
