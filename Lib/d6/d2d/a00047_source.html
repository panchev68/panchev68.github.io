30 /** Timer-based incremental encoder with button event support.
31 *
32 * Provides event-driven handling for encoder rotation and button actions using hardware timer and GPIO.
33 */
34 class EncoderTimer {
35 public:
36 /** EncoderTimer configuration.
37 *
38 * Contains timer and button pin handles required for initialization.
39 */
40 struct Config {
41 TIM_HandleTypeDef * timHandle; /*!< Timer handle. */
42 PinHandle button; /*!< Button pin handle. */
43 };

45 /** Encoder and button event codes.
46 *
47 * Enumerates possible events detected by the encoder and button.
48 */
49 enum class EventCode : int8_t {
50 BUTTON_PUSH, /*!< Button pressed. */
51 BUTTON_PULL, /*!< Button released. */
52 BUTTON_HOLD, /*!< Button held. */
53 ENCODER_CW, /*!< Encoder rotated clockwise. */
54 ENCODER_CCW, /*!< Encoder rotated counterclockwise. */
55 NONE = -1 /*!< No event. */
56 };

74 /** Initializes encoder timer and event queues.
75 *
76 * @return `true` if initialization succeeds, `false` otherwise.
77 * @post If successful, timer and queues are ready for operation.
78 */
79 virtual bool open() {
80 auto handleButtonPressEvent = [this](const Lib::HAL::GPIO::ExtInterrupt * exti) {
81 const bool pin = !exti->read();
82 queButton.overwriteFromISR(&pin);
83 };
84 auto handleTimerValueChangedEvent = [this](uint32_t value) {
85 const bool success = quePosition.overwriteFromISR(static_cast<int32_t>(value));
86 assert(success);
87 };
88
89 const bool success = timer.open() && quePosition.open() && queButton.open() && queueSet.open(2, {quePosition, queButton});
90 if (!success) {
91 assert(false);
92 return false;
93 }
94
95 timer.onValueChanged = handleTimerValueChangedEvent;
96 button.bind(handleButtonPressEvent);
97 rawPosition = 0;
98 initialized = true;
99 return true;
100 }
101
102 /** Reads encoder and button events.
103 *
104 * @return Current event code.
105 * @pre Encoder must be initialized.
106 */
107 virtual EventCode read() {
108 assert(initialized);
109 const auto selectedQueue = queueSet.select(0);
110
111 if (selectedQueue == quePosition) {
112 int32_t pos = 0;
113 if (quePosition.receive(pos, 0)) {
114 if (pos < rawPosition) {
115 rawPosition = pos;
116 return EventCode::ENCODER_CW;
117 }
118 if (pos > rawPosition) {
119 rawPosition = pos;
120 return EventCode::ENCODER_CCW;
121 }
122 }
123 } else if (selectedQueue == queButton) {
124 bool pinState = false;
125 if (queButton.receive(pinState, 0)) {
126 if (pinState) {
127 if (eventCode != EventCode::BUTTON_PUSH) {
128 eventCode = EventCode::BUTTON_PUSH;
129 timeoutButtonHold.set(BUTTON_HOLD_TIME);
130 }
131 } else eventCode = EventCode::BUTTON_PULL;
132 return eventCode;
133 }
134 }
135 if (eventCode == EventCode::BUTTON_PUSH && timeoutButtonHold.isExpired()) {
136 eventCode = EventCode::BUTTON_HOLD;
137 return EventCode::BUTTON_HOLD;
138 }
139 return EventCode::NONE;
140 }
141
142 private:
143 static constexpr uint32_t BUTTON_HOLD_TIME = 2000; /*!< Button hold time in ms. */
144
145 Lib::HAL::TimerEncoder timer; /*!< Encoder timer instance. */
146 Lib::HAL::GPIO::ExtInterrupt button; /*!< Button interrupt handler. */
147 RTOS::Queue<int32_t> quePosition; /*!< Queue for encoder position events. */
148 RTOS::Queue<bool> queButton; /*!< Queue for button state events. */
149 RTOS::QueueSet queueSet; /*!< Queue set for event synchronization. */
150 RTOS::Timeout timeoutButtonHold; /*!< Timeout for button hold detection. */
151 EventCode eventCode{EventCode::NONE}; /*!< Current button event code. */
152 int32_t position{0}; /*!< Absolute encoder position. */
153 int32_t rawPosition{0}; /*!< Last raw encoder position. */
154 bool initialized{false}; /*!< True after successful open(). */
155 };
