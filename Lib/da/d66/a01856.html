#Example: Converting BCD values with Lib::Helper::Convert::Bcd.

#Example: Converting BCD values with Lib::Helper::Convert::Bcd.Demonstrates converting a 16-bit BCD value to string and numeric forms and converting a single BCD digit to ASCII.

#include "Lib/Helper/Convert/Bcd.hpp"
#include <iostream>
int main() {
using Lib::Helper::Convert::Bcd;
uint16_t bcdValue = 0x1234; // BCD for digits 1,2,3,4
// Convert BCD to string
std::string s = Bcd::toString(bcdValue);
std::cout << "BCD to string: " << s << '\n'; // Output: 1234
// Convert BCD to decimal number
uint16_t dec = Bcd::toHex(bcdValue);
std::cout << "BCD to decimal: " << dec << '\n'; // Output: 1234
// Convert single BCD digit to ASCII
char c = Bcd::toAsc(0x05);
std::cout << "BCD digit to ASCII: " << c << '\n'; // Output: 5
return 0;
}
