1#/** Interrupt-driven SPI port specialization for HAL layer.
2 *
3 * Interrupt-driven `SpiPort` specialization using HAL IT (interrupt) APIs.
4 * This header defines `SpiPort<DriverMode::IT>` which performs non-blocking
5 * SPI transfers using HAL interrupt-based functions and relies on the base
6 * class event synchronization to signal completion to callers.
7 *
8 * Purpose and usage:
9 * - Use this specialization for low-latency or small-buffer transfers where
10 * interrupt-driven handling is preferred over DMA.
11 * - Requires the HAL SPI module enabled in the build configuration
12 * (see `HAL_SPI_MODULE_ENABLED`).
13 *
14 * Dependencies:
15 * - `SpiPortBase.hpp` for common SPI port plumbing and event synchronization.
16 * - Platform HAL interrupt callbacks must propagate events to the base class.
17 *
18 * @date 17 Jan 2026
19 * @version 3.04
20 * @file SpiPort-IT.hpp
21 */

31 /** SpiPort specialization for interrupt-driven SPI communication.
32 *
33 * Implements non-blocking transmit, receive and full-duplex operations
34 * using HAL interrupt APIs. Completion is signaled via the base class
35 * event mechanism.
36 */
37 template <> class SpiPort<DriverMode::IT> : public SpiPortBase {
38 public:
39 /** Construct an interrupt-mode SPI port.
40 *
41 * @param handle Pointer to `SPI_HandleTypeDef` provided by HAL.
42 * @pre `handle` must be a valid, initialized SPI handle.
43 */
44 explicit SpiPort(SPI_HandleTypeDef * handle) : SpiPortBase(handle) {}
45
46 /** Destructor for interrupt-mode SPI port. */
47 virtual ~SpiPort() = default;
48
49 /** Abort an ongoing interrupt-driven transfer and wait for completion.
50 *
51 * @param timeout RTOS ticks to wait for abort completion.
52 * @return `true` if the abort was requested and the completion event occurred, `false` otherwise.
53 */
54 virtual bool abort(uint32_t timeout) override {
55 return HAL_SPI_Abort_IT(handle.getHandle()) == HAL_OK && waitForEvent(ABORT_COMPLETED, timeout);
56 }
57
58 private:
59 /** Start an interrupt-based SPI reception and wait for completion.
60 *
61 * @param buffer Destination buffer where received bytes will be stored.
62 * @param size Number of bytes to receive (must be > 0).
63 * @param timeout RTOS ticks to wait for completion.
64 * @return `true` if the receive was started and the completion event occurred, `false` otherwise.
65 * @pre `buffer` must point to valid memory and `size` must be positive.
66 */
67 virtual bool receiveData(void * buffer, size_t size, uint32_t timeout) override {
68 return HAL_SPI_Receive_IT(handle.getHandle(), static_cast<uint8_t *>(buffer), size) == HAL_OK &&
69 waitForEvent(RX_COMPLETED, timeout);
70 }
71
72 /** Start an interrupt-based SPI transmission and wait for completion.
73 *
74 * @param buffer Source buffer containing bytes to transmit.
75 * @param size Number of bytes to transmit (must be > 0).
76 * @param timeout RTOS ticks to wait for completion.
77 * @return `true` if the transmit was started and the completion event occurred, `false` otherwise.
78 * @pre `buffer` must point to valid memory and `size` must be positive.
79 */
80 virtual bool sendData(const void * buffer, size_t size, uint32_t timeout) override {
81 return HAL_SPI_Transmit_IT(handle.getHandle(), static_cast<const uint8_t *>(buffer), size) == HAL_OK &&
82 waitForEvent(TX_COMPLETED, timeout);
83 }
84
85 /** Start an interrupt-based full-duplex SPI transfer and wait for completion.
86 *
87 * @param txBuffer Source buffer with data to transmit.
88 * @param rxBuffer Destination buffer where received bytes will be stored.
89 * @param size Number of bytes to exchange in both directions (must be > 0).
90 * @param timeout RTOS ticks to wait for completion.
91 * @return `true` if the transfer was started and the completion event occurred, `false` otherwise.
92 * @pre `txBuffer` and `rxBuffer` must point to valid memory and `size` must be positive.
93 */
94 virtual bool sendReceiveData(const void * txBuffer, void * rxBuffer, size_t size, uint32_t timeout) override {
95 return HAL_SPI_TransmitReceive_IT(handle.getHandle(), static_cast<const uint8_t *>(txBuffer),
96 static_cast<uint8_t *>(rxBuffer), size) == HAL_OK &&
97 waitForEvent(TX_RX_COMPLETED, timeout);
98 }
99 };
