1/** XPT2046 touchscreen controller driver with SPI interface and debouncing.
2 *
3 * This file contains the XPT2046 class for interfacing with the XPT2046 touchscreen controller.
4 *
5 * ## Overview
6 * Provides a complete driver for the XPT2046 capacitive touch controller, a 4-wire resistive
7 * touch panel controller with integrated 12-bit ADC. The driver uses bit-banged SPI for
8 * communication via GPIO pins (no hardware SPI required) and runs touch detection on a dedicated
9 * FreeRTOS task for non-blocking operation.
10 *
11 * ## Key Features
12 * - 12-bit ADC resolution for precise touch position detection
13 * - Configurable calibration ranges for X and Y axes with automatic scaling
14 * - Moving average integration (10-sample window) for noise reduction and stability
15 * - Debounce filtering to eliminate transient touch noise
16 * - Support for multiple display orientations with rotation capability (normal and flipped)
17 * - Event-driven interrupt handling via active-low NIRQ signal with RTOS queue signaling
18 * - Non-blocking operation with dedicated FreeRTOS task (IDLE priority)
19 *
20 * ## Hardware Interface
21 * - NIRQ: Interrupt pin (active-low touch detection signal)
22 * - MOSI: Serial data output to controller
23 * - MISO: Serial data input from controller ADC
24 * - SCK: Serial clock signal
25 * - NSS: Chip select (active-low)
26 *
27 * ## Notes
28 * - Touch coordinates are automatically clamped and scaled to display dimensions
29 * - All SPI communication is performed via GPIO bit-banging (not hardware SPI)
30 * - Integrator window must fill before reliable coordinates are available
31 * - RTOS task automatically starts during construction
32 *
33 * @file XPT2046.hpp
34 */

46 /** XPT2046 touchscreen controller driver with SPI interface and debouncing.
47 *
48 * Provides direct SPI communication with the XPT2046 capacitive touch controller using GPIO pins
49 * for all signals (MOSI, MISO, SCK, NSS). Includes automatic debouncing, calibration support,
50 * and coordinate transformation with rotation capability. Runs touch detection on a dedicated
51 * FreeRTOS task for non-blocking operation.
52 *
53 * Features:
54 * - 12-bit ADC resolution touch position detection.
55 * - Configurable calibration ranges for X and Y axes.
56 * - Moving average integration for noise reduction.
57 * - Support for multiple display orientations with rotation.
58 * - Debounce filtering to eliminate touch noise.
59 * - Event-driven interrupt handling with RTOS queue signaling.
60 *
61 * @note Uses bit-banged SPI communication instead of hardware SPI module.
62 * @note Touch coordinates are clamped to calibration bounds and scaled to display dimensions.
63 * @note Interrupt pin (NIRQ) must be active-low to trigger touch detection.
64 */

93 /** Initializes the XPT2046 touchscreen controller with SPI pins and calibration.
94 *
95 * Configures interrupt handler, SPI pins, scale multipliers, integrators,
96 * and starts the touch detection task.
97 *
98 * @param config SPI and interrupt pin handles.
99 * @param parameters Calibration ranges and display configuration.
100 * @pre Pin handles and parameters must be valid; xMax>xMin, yMax>yMin, width>0, height>0.
101 * @post Interrupt handler bound and touch detection task running.
102 */
103 explicit XPT2046(const Config & config, const Parameters & parameters)
104 : TaskWorker({"XPT2046", 128, RTOS::Task::Priority::IDLE}),
105 interface{
106 ExtInterrupt(config.nirq),
107 PinOutput(config.mosi),
108 PinOutput(config.sck),
109 PinOutput(config.nss),
110 PinInput(config.miso)},
111 parameters(parameters),
112 position{-1, -1},
113 rotation(0),
114 isPressed(false) {
115 const int32_t xRange = parameters.xMax - parameters.xMin;
116 const int32_t yRange = parameters.yMax - parameters.yMin;
117 assert(xRange > 0 && "X range must be positive.");
118 assert(yRange > 0 && "Y range must be positive.");
119 assert(parameters.width > 0 && "Display width must be positive.");
120 assert(parameters.height > 0 && "Display height must be positive.");
121 scaleMultiplierX = static_cast<int32_t>((static_cast<int64_t>(parameters.width) << 16) / xRange);
122 scaleMultiplierY = static_cast<int32_t>((static_cast<int64_t>(parameters.height) << 16) / yRange);
123 interface.nirq.bind([this](const Lib::HAL::GPIO::ExtInterrupt *) { queueButtonPressed.overwriteFromISR(true); });
124 interface.sck.write(1);
125 interface.nss.write(1);
126 const bool queueCreated = queueButtonPressed.open();
127 assert(queueCreated && "Failed to create button press queue.");
128 }

192 static constexpr uint8_t CMD_READ_X = 0x90; /*!< SPI command: read X position from ADC. */
193 static constexpr uint8_t CMD_READ_Y = 0xD0; /*!< SPI command: read Y position from ADC. */

208 /** Main touch detection task function.
209 *
210 * Runs in an infinite loop, waiting for touch press interrupts.
211 * On touch press, applies debounce filtering, reads ADC values,
212 * updates position with calibration and scaling, and sets pressed state.
213 */
214 void taskFunction() override {
215 while (true) {
216 handleTouchPress();
217 handleTouchWhilePressed();
218 }
219 }

221 /** Waits for touch interrupt and applies debounce filtering.
222 *
223 * Blocks until NIRQ signals touch, then validates with debounce timeout
224 * before resetting integrators and position data.
225 */
226 void handleTouchPress() {
227 isPressed = false;
228 bool unused = false;
229 queueButtonPressed.receive(unused, portMAX_DELAY);
230 RTOS::delay(5);
231
232 integratorX.resetWindow();
233 integratorY.resetWindow();
234 position.x = -1;
235 position.y = -1;
236
237 timeoutPinDebounce.set(parameters.debounceMs);
238 while (!timeoutPinDebounce.isExpired()) {
239 if (interface.nirq.read()) timeoutPinDebounce.set(parameters.debounceMs);
240 RTOS::Task::delay(1);
241 }
242 }

244 /** Transmits a byte via bit-banged SPI.
245 *
246 * @param data Byte value to transmit.
247 * @pre NSS must be asserted before calling.
248 */
249 void writeSpiByte(uint8_t data) {
250 for (uint8_t bit = 0x80; bit > 0; bit >>= 1) {
251 interface.mosi.write(data & bit);
252 interface.sck.write(0);
253 __NOP();
254 interface.sck.write(1);
255 __NOP();
256 }
257 }
258
259 /** Reads 12-bit ADC value from specified channel via SPI.
260 *
261 * @param command ADC channel select command (CMD_READ_X or CMD_READ_Y).
262 * @return 12-bit ADC value (0-4095).
263 */
264 uint16_t readAdc(uint8_t command) {
265 uint16_t adc = 0;
266 interface.nss.write(0);
267 writeSpiByte(command);
268 for (uint8_t bit = 12; bit > 0; --bit) {
269 interface.sck.write(0);
270 __NOP();
271 adc = (adc << 1) | interface.miso.read();
272 interface.sck.write(1);
273 __NOP();
274 }
275 interface.nss.write(1);
276 return adc;
277 }

279 /** Applies coordinate rotation based on orientation setting.
280 *
281 * Updates position in-place with rotated values.
282 */
283 void applyRotation() {
284 const auto width = static_cast<int32_t>(parameters.width);
285 const auto height = static_cast<int32_t>(parameters.height);
286 switch (rotation) {
287 case 1:
288 position.x = width - position.x;
289 break;
290 case 2:
291 case 4:
292 position.x = width - position.x;
293 position.y = height - position.y;
294 break;
295 case 3:
296 position.y = height - position.y;
297 break;
298 case 0:
299 default:
300 break;
301 }
302 }

304 /** Samples ADC channels and updates moving average integrators.
305 *
306 * @return True when integration window complete; false if accumulating.
307 */
308 bool readAndIntegrateAdc() {
309 rawPosition.x = integratorX.processValue(readAdc(CMD_READ_X));
310 rawPosition.y = integratorY.processValue(readAdc(CMD_READ_Y));
311 if (!integratorX.isWindowFull()) {
312 RTOS::Task::delay(5);
313 return false;
314 }
315 return true;
316 }

318 /** Clamps raw ADC values and scales to display pixel coordinates.
319 *
320 * Applies calibration bounds and fixed-point scaling transformation.
321 */
322 void clampAndScalePosition() {
323 clampValue(rawPosition.x, parameters.xMin, parameters.xMax);
324 clampValue(rawPosition.y, parameters.yMin, parameters.yMax);
325
326 constexpr int32_t SCALE_FACTOR = 16;
327 rawPosition.x -= parameters.xMin;
328 rawPosition.y -= parameters.yMin;
329 position.x = static_cast<int32_t>(
330 (static_cast<int64_t>(rawPosition.x) * scaleMultiplierX) >> SCALE_FACTOR);
331 position.y = static_cast<int32_t>(
332 (static_cast<int64_t>(rawPosition.y) * scaleMultiplierY) >> SCALE_FACTOR);
333 }
334
335 /** Samples, integrates, scales, and rotates touch coordinates.
336 *
337 * Updates position and marks press state as valid.
338 */
339 void processTouch() {
340 if (!readAndIntegrateAdc()) return;
341 clampAndScalePosition();
342 isPressed = true;
343 applyRotation();
344 RTOS::Task::delay(40);
345 }
346
347 /** Clamps value within min/max bounds. */
348 static void clampValue(int32_t & value, int32_t minVal, int32_t maxVal) {
349 if (value < minVal) {
350 value = minVal;
351 } else if (value > maxVal) {
352 value = maxVal;
353 }
354 }

356 /** Continuously processes touch data while NIRQ is asserted.
357 *
358 * Loops until touch release, updating position throughout.
359 */
360 void handleTouchWhilePressed() {
361 while (!interface.nirq.read()) { processTouch(); }
362 }
