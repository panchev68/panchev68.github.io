54 /** Converts value to BCD string.
55 *
56 * @param number Value to convert.
57 * @param showZero If false, leading zeros are omitted.
58 * @return BCD string representation.
59 */
60 static std::string toBcdString(uint8_t number, bool showZero = false) {
61 uint8_t bcd = toBcd(number);
62 char ascii[3];
63 ascii[0] = HexHelper::toAscii((bcd >> 4) & 0x0F);
64 ascii[1] = HexHelper::toAscii(bcd & 0x0F);
65 ascii[2] = '\0';
66 char * result = ascii;
67 if (!showZero)
68 while (*result == '0' && *(result + 1)) ++result;
69 return std::string(result);
70 }

85 * @param number Value to convert.
86 * @return BCD representation as uint32_t.
87 */
88 static constexpr uint32_t toBcd(uint16_t number) {
89 uint32_t result = 0;
90 uint32_t multiplier = 1;
91 for (uint16_t temp = number; temp; temp /= 10, multiplier <<= 4) result += (temp % 10) * multiplier;
92 return result;
93 }
