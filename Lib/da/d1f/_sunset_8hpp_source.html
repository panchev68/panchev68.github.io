239 /** Calculates the moon phase index for a given epoch time.
240 *
241 * This method computes the moon phase based on the lunar cycle, returning an index from 0 to 29 representing
242 * the phase of the moon at the specified epoch time.
243 *
244 * @param epochTime Epoch time in seconds.

246 * @pre epochTime must be a valid epoch time in seconds.
247 * @post Returns the moon phase index corresponding to the given time.

250 // Reference epoch for moon phase calculation.
251 constexpr int MOON_EPOCH{614100};
252 // Duration of lunar cycle in seconds.
253 constexpr int LUNAR_CYCLE_SECONDS{2551443};
254 // Number of seconds per day.
255 constexpr double SECONDS_PER_DAY{24.0 * 3600.0};
256 // Number of days in a simple lunar cycle approximation.
257 constexpr int LUNAR_CYCLE_DAYS{30};
258 int phase{(epochTime - MOON_EPOCH) % LUNAR_CYCLE_SECONDS};

345 /** Calculates the Julian date for a given calendar date.
346 *
347 * This method converts a Gregorian calendar date to its corresponding Julian date number,
348 * which is used in astronomical calculations.
349 *
350 * @param year Year value.
351 * @param month Month value.
352 * @param day Day value.
353 * @return Julian date for the specified date.
354 * @pre year, month, day must form a valid Gregorian date.
355 * @post Returns the calculated Julian date.
356 */
357 double calculateJulianDate(int year, int month, int day) const noexcept {
358 constexpr int FEBRUARY{2};
359 constexpr int MONTHS_TO_ADD{12};
360 constexpr double JULIAN_OFFSET{1524.5};
361 if (month <= FEBRUARY) {
362 year -= 1;
363 month += MONTHS_TO_ADD;
364 }
365 auto A{std::floor(year / 100.0)};
366 auto B{2.0 - A + std::floor(A / 4.0)};
367 return std::floor(365.25 * (year + 4716)) + std::floor(30.6001 * (month + 1)) + day + B - JULIAN_OFFSET;
368 }

370 /** Calculates the absolute sun time (sunrise or sunset) for a given sun angle.
371 *
372 * This method performs the core astronomical calculation for sunrise or sunset time using
373 * Julian centuries, equation of time, and solar declination.
374 *
375 * @param angle Sun angle for event calculation.
376 * @param isSunrise True for sunrise, false for sunset.
377 * @return Sun time in minutes from midnight UTC.
378 * @pre angle must be a valid sun angle.
379 * @pre Geographic position and current date must be set.
380 * @post Returns the calculated absolute sun time in UTC.
381 */
382 double calculateAbsoluteSunTime(double angle, bool isSunrise) const noexcept {
383 constexpr double MINUTES_PER_DEGREE{4.0};
384 constexpr double NOON_MINUTES{720.0};
385 constexpr double MINUTES_PER_DAY{1440.0};
386 auto t{calculateTimeJulianCenturies(julianDate)};
387 auto eqTime{calculateEquationOfTime(t)};
388 auto solarDec{calculateSunDeclination(t)};
389 auto hourAngle{isSunrise ? computeHourAngleSunrise(solarDec, angle) : computeHourAngleSunset(solarDec, angle)};
390 auto delta{geoLongitude + radiansToDegrees(hourAngle)};
391 auto timeDiff{MINUTES_PER_DEGREE * delta};
392 auto timeUTC{NOON_MINUTES - timeDiff - eqTime};
393 auto newt{calculateTimeJulianCenturies(calculateJulianDateFromCenturies(t) + timeUTC / MINUTES_PER_DAY)};
394 auto newSolarDec{calculateSunDeclination(newt)};
395 auto newHourAngle{isSunrise ? computeHourAngleSunrise(newSolarDec, angle) : computeHourAngleSunset(newSolarDec, angle)};
396 auto newDelta{geoLongitude + radiansToDegrees(newHourAngle)};
397 return NOON_MINUTES - MINUTES_PER_DEGREE * newDelta - calculateEquationOfTime(newt);
398 }

411 /** Calculates the hour angle for sunrise using stable formula with cached latitude.
412 *
413 * @param solarDeclination Solar declination in degrees.
414 * @param angle Sun angle for event calculation.
415 * @return Hour angle for sunrise in radians.
416 */
417 double computeHourAngleSunrise(double solarDeclination, double angle) const noexcept {
418 const auto declRad{degreesToRadians(solarDeclination)};
419 const auto zenithRad{degreesToRadians(angle)};
420 const auto cosZenith{std::cos(zenithRad)};
421 const auto sinDec{std::sin(declRad)};
422 const auto cosDec{std::cos(declRad)};
423 const auto numerator{cosZenith - (sinLat * sinDec)};
424 const auto denominator{cosLat * cosDec};
425 const auto cosH{numerator / denominator};
426 return std::acos(std::min(std::max(cosH, -1.0), 1.0));
427 }

440 /** Calculates the hour angle for sunrise for a given latitude, declination, and sun angle.
441 *
442 * @param latitude Geographic latitude in degrees.
443 * @param solarDeclination Solar declination in degrees.
444 * @param angle Sun angle for event calculation.
445 * @return Hour angle for sunrise in radians.

447 static double calculateHourAngleSunrise(double latitude, double solarDeclination, double angle) {
448 auto latRad{degreesToRadians(latitude)};
449 auto declRad{degreesToRadians(solarDeclination)};
450 auto cosH{std::cos(degreesToRadians(90.0 + angle)) / (std::cos(latRad) * std::cos(declRad)) - std::tan(latRad) * std::tan(declRad)};

479 /** Calculates the equation of time for a given Julian centuries value.
480 *
481 * @param julianCenturies Julian centuries since J2000.0.
482 * @return Equation of time in minutes.
483 */
484 double calculateEquationOfTime(double julianCenturies) const noexcept {
485 auto epsilon{23.439292 - 0.000013 * julianCenturies};
486 auto l0{std::fmod(280.46646 + 36000.76983 * julianCenturies, 360.0)};
487 auto e{0.016708634 - 0.000042037 * julianCenturies};
488 auto m{std::fmod(357.52911 + 35999.05029 * julianCenturies, 360.0)};
489 auto y{std::pow(std::tan(degreesToRadians(epsilon) / 2.0), 2)};
490 auto eqTime{
491 y * std::sin(2.0 * degreesToRadians(l0)) - 2.0 * e * std::sin(degreesToRadians(m)) + 4.0 * e * y * std::sin(degreesToRadians(m)) * std::cos(2.0 * degreesToRadians(l0)) - 0.5 * y * y * std::sin(4.0 * degreesToRadians(l0)) - 1.25 * e * e * std::sin(2.0 * degreesToRadians(m))};
492 return radiansToDegrees(eqTime) * 4.0;
493 }

495 /** Calculates the solar declination for a given Julian centuries value.
496 *
497 * @param julianCenturies Julian centuries since J2000.0.
498 * @return Solar declination in degrees.
499 */
500 double calculateSunDeclination(double julianCenturies) const noexcept {
501 auto epsilon{23.439292 - 0.000013 * julianCenturies};
502 auto l0{std::fmod(280.46646 + 36000.76983 * julianCenturies, 360.0)};
503 auto m{std::fmod(357.52911 + 35999.05029 * julianCenturies, 360.0)};
504 auto lambda{
505 l0 + (1.914602 - 0.004817 * julianCenturies - 0.000014 * julianCenturies * julianCenturies) * std::sin(degreesToRadians(m)) + (0.019993 - 0.000101 * julianCenturies) * std::sin(2.0 * degreesToRadians(m)) + 0.000289 * std::sin(3.0 * degreesToRadians(m))};
506 return radiansToDegrees(std::asin(std::sin(degreesToRadians(epsilon)) * std::sin(degreesToRadians(lambda))));
507 }
