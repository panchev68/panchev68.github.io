81 /** Helper to parse build date and time macros to string.
82 * @return Build date string.
83 */
84 static std::string_view buildDateString() {
85 static char buffer[11]; // YYYY-MM-DD\0
86 auto getDigit = [](const char * ptr, uint32_t offset) -> uint8_t {
87 assert(ptr);
88 return static_cast<uint8_t>(ptr[offset] - '0');
89 };
90 uint16_t year = 1000 * getDigit(__DATE__, 7) + 100 * getDigit(__DATE__, 8) + 10 * getDigit(__DATE__, 9) + getDigit(__DATE__, 10);
91
92 uint8_t month = 0;
93 switch (__DATE__[0]) {
94 case 'J': month = (__DATE__[1] == 'a') ? 1 : (__DATE__[2] == 'n' ? 6 : 7); break;
95 case 'F': month = 2; break;
96 case 'M': month = (__DATE__[2] == 'r') ? 3 : 5; break;
97 case 'A': month = (__DATE__[1] == 'p') ? 4 : 8; break;
98 case 'S': month = 9; break;
99 case 'O': month = 10; break;
100 case 'N': month = 11; break;
101 case 'D': month = 12; break;
102 default: month = 0; break;
103 }
104
105 uint8_t day = 10 * ((__DATE__[4] == ' ') ? 0 : getDigit(__DATE__, 4)) + getDigit(__DATE__, 5);
106
107 // Format to buffer
108 buffer[0] = '0' + (year / 1000) % 10;
109 buffer[1] = '0' + (year / 100) % 10;
110 buffer[2] = '0' + (year / 10) % 10;
111 buffer[3] = '0' + year % 10;
112 buffer[4] = '-';
113 buffer[5] = '0' + month / 10;
114 buffer[6] = '0' + month % 10;
115 buffer[7] = '-';
116 buffer[8] = '0' + day / 10;
117 buffer[9] = '0' + day % 10;
118 buffer[10] = '\0';
119
120 return buffer;
121 }
