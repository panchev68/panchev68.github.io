1/***************************************************************************************
2 * TimeZone.hpp - Time zone helper and conversion utilities.
3 *
4 * This header declares the Lib::Helper::TimeZone class and related types used to model
5 * a time zone identifier, a fixed offset from UTC, and optional calendar-based time
6 * change rules (for example daylight saving transitions).
7 *
8 * The API is intended to be lightweight and allocation-free: the time zone name and
9 * rule name fields are stored as std::string_view, so the caller must ensure the
10 * referenced character data remains valid for the lifetime of the TimeZone instance.
11 *
12 * The conversion helper uses the C library time facilities (std::tm + mktime) to turn
13 * local calendar fields into a time_t value and then adjusts the result with an explicit
14 * offset to obtain minutes since the epoch in UTC.
15 *
16 * @file TimeZone.hpp
17 * @date 16 Jan 2026
18 ***************************************************************************************/

28 /** TimeZone class.
29 *
30 * Encapsulates a time zone name, offset in minutes from UTC and a list of rules that
31 * describe time changes (for example daylight saving transitions). The class exposes
32 * simple accessors and a conversion helper to obtain UTC minutes from local calendar

84 /** Convert local calendar time to UTC minutes since the epoch.
85 *
86 * The function fills a local std::tm structure using the provided calendar
87 * fields, converts it to a time_t value with mktime (which treats the
88 * structure as local time) and adjusts the result by subtracting offsetSeconds
89 * to obtain UTC. The returned value is the number of minutes since the epoch
90 * in UTC, or -1 if the conversion failed.
91 *
92 * @param year Calendar year (for example 2024).
93 * @param month Calendar month in the range [1..12].
94 * @param day Calendar day of month in the range [1..31].
95 * @param hour Hour of day in the range [0..23].
96 * @param minute Minute in the range [0..59].
97 * @param offsetSeconds Offset in seconds to subtract from the local time to obtain UTC.
98 * @return Number of minutes since the epoch in UTC on success; -1 on error.
99 */
100 int convertLocalToUtcMinutes(int year, int month, int day, int hour, int minute, time_t offsetSeconds) const {
101 std::tm timeInfo{};
102 timeInfo.tm_year = year - 1900;
103 timeInfo.tm_mon = month - 1;
104 timeInfo.tm_mday = day;
105 timeInfo.tm_hour = hour;
106 timeInfo.tm_min = minute;
107 time_t localTime = mktime(&timeInfo);
108 return (localTime == -1) ? -1 : static_cast<int>((localTime - offsetSeconds) / 60);
109 }
110
111 private:
112 std::vector<TimeChangeRule> changeRules; /*!< List of time change rules. */
113 std::string_view name; /*!< Name of the time zone. */
114 int offsetMinutes; /*!< Offset in minutes from UTC. */
115 };
