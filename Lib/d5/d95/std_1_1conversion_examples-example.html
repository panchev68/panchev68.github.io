std::conversion_examples

Conversion utilities usage.

This example demonstrates basic usage of the lightweight conversion helpers defined in this header, including string-to-number conversions, number-to-string conversions, and value clamping.

Example usage:
#include "Lib/Helper/stdlib.hpp"
#include <iostream>
int main() {
// Convert string to long
const char * sint = " -1234 ";
long v = std::atol(sint);
std::cout << "atol: " << v << '\n';
// Convert string to double with end pointer
const char * sdouble = "3.1415e0";
char * end = nullptr;
double d = std::strtod(sdouble, &end);
std::cout << "strtod: " << d << " (next char='" << (end ? *end : '\0') << "')" << '\n';
// Integer to string
char buf[16];
char * p = std::itoa(static_cast<int>(v), buf, sizeof(buf), 10);
if (p) std::cout << "itoa: " << p << '\n';
// Clamp a value to range
int clamped = std::trip<int>(static_cast<int>(v), -100, 100);
std::cout << "trip: " << clamped << '\n';
return 0;
}
