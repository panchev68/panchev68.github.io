59 /** Extract value from vector.
60 * Extracts a value of type `T` from a vector of bytes at the specified offset.
61 * @tparam T Type of the value.
62 * @param bytes Source vector.
63 * @param offset Offset in the vector.
64 * @return Extracted value.
65 */
66 template <typename T> static T fromVec(const std::vector<uint8_t> & bytes, uint32_t offset = 0) {
67 assert(offset <= bytes.size());
68 T value{};
69 const size_t bytesToCopy = std::min(bytes.size() - offset, sizeof(T));
70 std::memcpy(&value, bytes.data() + offset, bytesToCopy);

105 Value64Handle & operator=(uint64_t value) {
106 if (status == Status::COMPLETED && valuePtr != nullptr) {
107 uint64_t & sourceValue = *valuePtr;
108 if (value != sourceValue) {
109 previousValue = sourceValue;
110 sourceValue = value;
111 status = Status::CHANGED;
112 }
113 }
114 return *this;
115 }

117 /** Update the status of the handle.
118 * @return Updated status.
119 */
120 Status update() {
121 if (valuePtr == nullptr) { return (status = Status::NO_INIT); }
122 if (status == Status::IN_PROGRESS) { return Status::IN_PROGRESS; }
123 return (status = (*valuePtr != previousValue) ? Status::CHANGED : Status::COMPLETED);
124 }
125
126 /** Mark the handle as completed. */
127 void complete() {
128 if (valuePtr != nullptr) {
129 previousValue = *valuePtr;
130 status = Status::COMPLETED;
131 }
132 }

136
137 /** Bind the handle to an external value.
138 * @param ref Reference to the external value.
139 * @post Changes are tracked to the external variable.

141 void bind(uint64_t & ref) {
142 valuePtr = &ref;
143 status = Status::CHANGED;
144 }

146 private:
