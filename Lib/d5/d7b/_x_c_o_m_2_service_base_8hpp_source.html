52 /** Convert value to byte array.
53 *
54 * Converts a value of type `T` to an array of bytes using memory copy semantics.
55 * Useful for serializing typed values into byte sequences for transmission.
56 *
57 * @tparam T Type of the value to convert.
58 * @tparam SIZE Size of the value in bytes (defaults to `sizeof(T)`).
59 * @param value Value to convert to bytes.
60 * @return Array of bytes representing the value in memory layout.
61 *
62 * @post Result contains `SIZE` bytes copied from the value.
63 */
64 template <typename T, size_t SIZE = sizeof(T)> static std::array<uint8_t, SIZE> toVec(const T & value) {
65 std::array<uint8_t, SIZE> bytes;
66 std::memcpy(bytes.data(), &value, SIZE);
67 return bytes;
68 }

70 /** Append value bytes to vector.
71 *
72 * Appends the bytes of a value of type `T` to a vector of bytes, resizing the
73 * vector as necessary. Useful for building byte sequences by appending multiple
74 * typed values in sequence.
75 *
76 * @tparam T Type of the value.
77 * @tparam SIZE Size of the value in bytes (defaults to `sizeof(T)`).
78 * @param bytes Vector to append to (modified in-place).
79 * @param value Value to append to the vector.
80 *
81 * @post Vector size is increased by `SIZE` bytes, containing the appended value.
82 */
83 template <typename T, size_t SIZE = sizeof(T)> static void appendVec(std::vector<uint8_t> & bytes, const T & value) {
84 const size_t oldSize = bytes.size();
85 bytes.resize(oldSize + SIZE);
86 std::memcpy(bytes.data() + oldSize, &value, SIZE);
87 }

89 /** Extract value from byte vector.
90 *
91 * Extracts a value of type `T` from a vector of bytes at the specified offset.
92 * If insufficient bytes are available, the remaining bytes are zero-filled.
93 * Uses bounds checking with assertion to ensure valid offset.
94 *
95 * @tparam T Type of the value to extract (must be trivially copyable).
96 * @param bytes Source vector of bytes.
97 * @param offset Offset in the vector to start extraction (defaults to 0).
98 * @return Extracted value, with zero-fill for missing bytes.
99 *
100 * @pre `offset <= bytes.size()`; asserts if violated.
101 * @post Result is a valid value of type `T` constructed from bytes at offset.
102 */
103 template <typename T> static T fromVec(const std::vector<uint8_t> & bytes, uint32_t offset = 0) {
104 assert(offset <= bytes.size());
105 T value{};
106 const size_t bytesToCopy = std::min(bytes.size() - offset, sizeof(T));
107 std::memcpy(&value, bytes.data() + offset, bytesToCopy);
108 return value;
109 }

112 /** Handle for tracking changes to 64-bit values.
113 *
114 * The `Value64Handle` class provides change tracking for 64-bit values through
115 * binding to external variables. It maintains status (NO_INIT, CHANGED, IN_PROGRESS,
116 * COMPLETED) and tracks the previous value to detect modifications. Useful for
117 * detecting when bound values change and managing operation state transitions.
118 */
119 class Value64Handle {
120 public:
121 /** Status of the `Value64Handle`.
122 *
123 * Tracks the state of the handle through initialization, change detection,
124 * operation progress, and completion.
125 */
126 enum class Status {
127 NO_INIT, /*!< Handle not initialized or not bound to a value. */
128 CHANGED, /*!< Bound value has changed since last update. */
129 IN_PROGRESS, /*!< Operation using this handle is in progress. */
130 COMPLETED /*!< Operation completed and value synchronized. */
131 };
132
133 /** Constructor of the `Value64Handle` class.
134 *
135 * Initializes the handle in NO_INIT status with no bound value.
136 */
137 Value64Handle() = default;
138
139 /** Destructor of the `Value64Handle` class. */
140 virtual ~Value64Handle() = default;
141
142 /** Implicit conversion to const reference of tracked value.
143 *
144 * Returns a const reference to the tracked value if the handle is bound,
145 * otherwise returns the previous value stored in the handle.
146 *
147 * @return Const reference to tracked value or previous value if not bound.
148 */
149 operator const uint64_t &() const { return (valuePtr != nullptr) ? *valuePtr : previousValue; }
150
151 /** Set the tracked value.
152 *
153 * Updates the bound value if the handle is in COMPLETED status and bound
154 * to a value. If the new value differs from the current value, marks the
155 * handle as CHANGED and stores the previous value for tracking.
156 *
157 * @param value New value to set in the bound variable.
158 * @return Reference to this handle for chaining.
159 *
160 * @post Handle status is CHANGED if value was different, COMPLETED otherwise.
161 */
162 Value64Handle & operator=(uint64_t value) {
163 if (status == Status::COMPLETED && valuePtr != nullptr) {
164 uint64_t & sourceValue = *valuePtr;
165 if (value != sourceValue) {
166 previousValue = sourceValue;
167 sourceValue = value;
168 status = Status::CHANGED;
169 }
170 }
171 return *this;
172 }
173
174 /** Update the status of the handle.
175 *
176 * Checks the current state of the bound value (if any) and updates the
177 * handle status accordingly. If the handle is not bound, returns NO_INIT.
178 * If in-progress, maintains IN_PROGRESS. Otherwise, compares the current
179 * value with the previous value to determine CHANGED or COMPLETED status.
180 *
181 * @return Updated status after checking bound value state.
182 *
183 * @post Status reflects current state: NO_INIT (unbound), IN_PROGRESS (unchanged),
184 * CHANGED (value differs from previous), or COMPLETED (value unchanged).
185 */
186 Status update() {
187 if (valuePtr == nullptr) { return (status = Status::NO_INIT); }
188 if (status == Status::IN_PROGRESS) { return Status::IN_PROGRESS; }
189 return (status = (*valuePtr != previousValue) ? Status::CHANGED : Status::COMPLETED);
190 }
191
192 /** Mark the handle as completed.
193 *
194 * Updates the previous value to the current bound value and sets status
195 * to COMPLETED. This is called after processing the change to reset the
196 * change tracking state.
197 *
198 * @post Previous value synchronized with current bound value; status is COMPLETED.
199 */
200 void complete() {
201 if (valuePtr != nullptr) {
202 previousValue = *valuePtr;
203 status = Status::COMPLETED;
204 }
205 }
206
207 /** Mark the handle as in progress.
208 *
209 * Sets the status to IN_PROGRESS, indicating that an operation using this
210 * handle is currently in progress. This prevents status updates during the
211 * operation.
212 *
213 * @post Status is IN_PROGRESS.
214 */
215 void inProgress() { status = Status::IN_PROGRESS; }
216
217 /** Bind the handle to an external value.
218 *
219 * Associates this handle with an external 64-bit variable, enabling change
220 * tracking and value management. After binding, the handle can detect when
221 * the external variable is modified and manage its lifecycle.
222 *
223 * @param ref Reference to the external 64-bit value to track.
224 *
225 * @pre `ref` must remain valid for the lifetime of the handle.
226 * @post Handle is bound to the external variable; status is CHANGED.
227 */
228 void bind(uint64_t & ref) {
229 valuePtr = &ref;
230 status = Status::CHANGED;
231 }
232
233 private:
234 uint64_t previousValue = static_cast<uint64_t>(-1); /*!< Previous value for detecting changes. */
235 uint64_t * valuePtr = nullptr; /*!< Pointer to the externally bound value. */
236 Status status = Status::NO_INIT; /*!< Current status of the change tracking. */
237 };
