57 /** Calculate moon phase for a date.
58 *
59 * Computes the moon phase index and name based on the provided date using astronomical formulas.
60 *
61 * @param date The date for calculation.
62 * @return The phase index (0-7).
63 */
64 uint32_t calculate(const Date & date) {
65 constexpr double DAYS_PER_YEAR = 365.25;
66 constexpr double DAYS_PER_MONTH = 30.6;
67 constexpr double JULIAN_EPOCH_OFFSET = 694039.09;
68 constexpr double LUNAR_CYCLE_DAYS = 29.5305882;
69 constexpr int MARCH_MONTH = 3;
70 constexpr int MONTHS_IN_YEAR = 12;
71 constexpr int PHASE_COUNT = 8;
72 constexpr double ROUNDING_OFFSET = 0.5;
73
74 const int year = date.year;
75 const int month = date.month;
76 const int day = date.day;
77
78 const int adjustedYear = (month < MARCH_MONTH) ? (year - 1) : year;
79 const int adjustedMonth = (month < MARCH_MONTH) ? (month + MONTHS_IN_YEAR) : month;
80
81 double julianDay = DAYS_PER_YEAR * adjustedYear + DAYS_PER_MONTH * (adjustedMonth + 1) + day - JULIAN_EPOCH_OFFSET;
82 julianDay /= LUNAR_CYCLE_DAYS;
83
84 const double fractionalPart = julianDay - static_cast<int>(julianDay);
85 const int idx = static_cast<int>(fractionalPart * PHASE_COUNT + ROUNDING_OFFSET) & (PHASE_COUNT - 1);
86
87 phaseIndex = static_cast<uint32_t>(idx);
88 phaseName = PHASE_NAMES[idx];
89 return phaseIndex;
90 }
