Lib::System::ProduceDate

Example usage of ProduceDate class.

Demonstrates reading and writing production date to OTP memory.

Example usage:

#include "Lib/System/ProduceDate.hpp"
#include <iostream>
int main() {
Lib::System::ProduceDate produceDate(0x00); // Offset 0x00 in OTP
Lib::System::ProduceDate::Date date;
// Write production date
date.day = 15;
date.month = 6;
date.year = 2024;
if (produceDate.write(date) == Lib::System::ProduceDate::Status::OK) {
std::cout << "Production date written successfully." << std::endl;
} else {
std::cout << "Failed to write production date." << std::endl;
}
// Read production date
if (produceDate.read(date) == Lib::System::ProduceDate::Status::OK) {
std::cout << "Production Date: " << date.toString() << std::endl;
} else {
std::cout << "Failed to read production date." << std::endl;
}
return 0;
}
