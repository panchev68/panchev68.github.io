30 /** Provides a foundation for managing system parameters with persistent storage and integrity checks.
31 *
32 * The ParametersBase class encapsulates functionality to read, write, and validate
33 * a contiguous parameter blob stored in a Storage implementation. A CRC16 is
34 * appended to the data to allow integrity checks. It uses an internal buffer
35 * for data and CRC storage, and provides methods for setup, writing, and validation.
36 *
37 * Role within the System module: Base class for parameter management in embedded systems.
38 *
39 * Important design decisions: Uses abstract StorageBase for backend flexibility,
40 * includes CRC for data integrity, and requires derived classes to implement
41 * default value restoration.
42 *
43 * @note Derived classes must implement restoreByDefault() to set default values.
44 */
45 class ParametersBase {
46 public:
47 /** Status enumeration for operation results.
48 *
49 * Defines the possible outcomes of parameter operations.
50 */
51 enum class Status {
52 OK, /*!< Operation successful. */
53 ERROR /*!< Operation failed. */
54 };

56 /** Abstract interface for storage backends.
57 *
58 * The StorageBase class defines the interface that must be implemented by
59 * concrete storage backends. Implementations perform the actual open/close/read/write
60 * operations on the medium used to persist the parameters.
61 *
62 * Role: Provides abstraction for different storage types (e.g., EEPROM, flash).
63 *
64 * Important design decisions: Pure virtual methods ensure all operations are implemented.
65 */
66 class StorageBase {
67 public:
68 /** Virtual destructor for StorageBase.
69 *
70 * Ensures proper cleanup of derived storage implementations.
71 */
72 virtual ~StorageBase() = default;
73
74 /** Open the storage for operations.
75 *
76 * Prepares the storage medium for read/write operations.
77 *
78 * @return true if successful, false otherwise.
79 */
80 virtual bool open() = 0;
81
82 /** Close the storage.
83 *
84 * Finalizes and closes the storage medium.
85 *
86 * @return true if successful, false otherwise.
87 */
88 virtual bool close() = 0;
89
90 /** Read data from storage.
91 *
92 * Reads a specified number of bytes from the storage into the provided buffer.
93 *
94 * @param buffer Pointer to the buffer to read into.
95 * @param size Number of bytes to read.
96 * @return true if successful, false otherwise.
97 */
98 virtual bool read(void * buffer, size_t size) = 0;
99
100 /** Write data to storage.
101 *
102 * Writes a specified number of bytes from the provided buffer to the storage.
103 *
104 * @param buffer Pointer to the buffer to write from.
105 * @param size Number of bytes to write.
106 * @return true if successful, false otherwise.
107 */
108 virtual bool write(const void * buffer, size_t size) = 0;
109 };

121 * Initializes the parameter manager with a storage backend and user data buffer.
122 *
123 * @param storage Pointer to the storage interface implementation.
124 * @param dataBuffer Pointer to the user-provided data buffer.

127 explicit ParametersBase(StorageBase * storage, void * dataBuffer, size_t dataSize)

139 /** Set up parameters from storage.
140 *
141 * Reads the parameter blob from storage, validates the CRC, and restores
142 * defaults and rewrites storage if the stored blob is invalid. Copies
143 * valid data to the user buffer.
144 *
145 * @return Status of the setup operation (OK or ERROR).
146 */
147 Status setup() {
148 if (!storage->open() || !storage->read(buffer.get(), size + CRC_SIZE)) {
149 assert(false);
150 return Status::ERROR;
151 }
152 if (!isValid()) {
153 restoreByDefault();
154 if (write() != Status::OK) {
155 assert(false);
156 return Status::ERROR;
157 }
158 }
159 std::memcpy(data, buffer.get(), size);
160 return Status::OK;
161 }
162
163 /** Write parameters to storage.
164 *
165 * Finalizes the internal buffer (copies user data and appends CRC) and
166 * writes the blob to the storage backend.
167 *
168 * @return Status of the write operation (OK or ERROR).
169 */
170 Status write() {
171 finalize();
172 bool result = storage->write(buffer.get(), size + CRC_SIZE);
173 assert(result);
174 return result ? Status::OK : Status::ERROR;
175 }
176
177 /** Check stored data CRC validity.
178 *
179 * Verifies the integrity of the stored parameter data by comparing the
180 * calculated CRC16 with the stored CRC value.
181 *
182 * @return true if CRC matches and data is valid, false otherwise.
183 */
184 bool isValid() const { return crc16(buffer.get(), size) == *getCrc(); }
185
186 protected:
187 /** Restore parameters to their default values.
188 *
189 * Must be implemented by derived classes to fill the user data buffer
190 * with appropriate default values.
191 */
192 virtual void restoreByDefault() = 0;
193
194 private:
195 /** Finalize the internal buffer.
196 *
197 * Copies the user data to the internal buffer and calculates the CRC.
198 */
199 void finalize() {
200 std::memcpy(buffer.get(), data, size);
201 *getCrc() = crc16(buffer.get(), size);
202 }
203
204 /** Get pointer to CRC field.
205 *
206 * Returns a pointer to the CRC16 value stored after the data blob in the internal buffer.
207 *
208 * @return Pointer to the CRC16 value.
209 */
210 uint16_t * getCrc() const { return reinterpret_cast<uint16_t *>(buffer.get() + size); }
211 };
