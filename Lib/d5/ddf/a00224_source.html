85 /** Task function for periodic updates and error state checks. */
86 void handleTickEvent() override {
87 if (!timeout.isExpired()) return;
88 timeout.set(UPDATE_PERIOD);
89
90 bool hasError = false;
91 uint32_t clearedErrorCount = 0;
92
93 for (auto & error : errorVector) {
94 if (error.checkForStateChanged()) {
95 auto state = error.getState();
96 if (state > Error::State::NO_ERROR) hasError = true;
97 else if (state == Error::State::CLEARED) clearedErrorCount++;
98 }
99 }
100
101 if (hasError) {
102 if (statusChangedHook) statusChangedHook(Status::STATUS_CHANGED);
103 } else if (clearedErrorCount != lastClearedErrorCount) {
104 lastClearedErrorCount = clearedErrorCount;
105 if (statusChangedHook) statusChangedHook(Status::ALL_ERRORS_CLEARED);
106 }
107 }
108 };
