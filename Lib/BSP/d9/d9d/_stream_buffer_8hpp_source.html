59 /** Create the stream buffer.
60 * @param bufferSize Size of the buffer.
61 * @param triggerLevel Trigger level for notifications.
62 * @return `true` if creation was successful, `false` otherwise.
63 */
64 bool create(size_t bufferSize, size_t triggerLevel) {
65 assert(handle == nullptr);
66 assert(bufferSize > 0);
67 assert(triggerLevel > 0);
68 assert(triggerLevel <= bufferSize);
69 if (handle || bufferSize == 0 || triggerLevel == 0 || triggerLevel > bufferSize) return false;
70 handle = xStreamBufferCreate(bufferSize, triggerLevel);
71 if (handle) capacity = bufferSize;
72 return handle != nullptr;
