61 /** Extract value from vector.

63 * @tparam T Type of the value.
64 * @param bytes Source vector.

68 template<typename T>

70 assert(offset <= bytes.size());
71 T value{};

112 uint64_t & sourceValue = *valuePtr;
113 if (value != sourceValue) {

122 /** Update the status of the handle.
123 * @return Updated status.
124 */
125 Status update() noexcept {
126 if (valuePtr == nullptr) {
127 return (status = Status::NO_INIT);
128 }
129 if (status == Status::IN_PROGRESS) {
130 return Status::IN_PROGRESS;
131 }
132 return (status = (*valuePtr != previousValue) ? Status::CHANGED : Status::COMPLETED);
133 }

141 }
