Usage example for the OTP class:

#include "Lib/HAL/OTP.hpp"
int main() {
Lib::HAL::OTP &otp = Lib::HAL::OTP::instance();
// Unlock OTP memory
if (otp.unlock()) {
// Write a 32-bit value to the beginning of OTP
otp.write<uint32_t>(Lib::HAL::OTP::beginAddr(), 0x12345678U);
// Lock OTP memory
otp.lock();
}
// Read the value back
uint32_t value = otp.read<uint32_t>(Lib::HAL::OTP::beginAddr());
return 0;
}

OTP (One-Time Programmable) memory management interface.
/** OTP (One-Time Programmable) memory management interface.
* This file provides the OTP class for managing STM32 OTP memory, including read, write, lock, and unlock
* operations. It supports singleton access, address range validation, and type-safe memory access for embedded
* applications. The design ensures robust integration with STM32 OTP hardware, supporting flexible configuration
* and efficient resource management in embedded systems. The OTP class promotes type safety, maintainability,
* and extensibility for various STM32 hardware requirements, facilitating seamless adaptation to evolving system
* needs.
* @file OTP.hpp
*/
#pragma once
#if defined(FLASH_OTP_BASE) || defined(__INTELLISENSE__) || defined(__DOXYGEN__)
#include "stm32f4xx_hal_def.h"
#include "stm32f4xx_hal_flash.h"
#include <cassert>
#include <cstddef>
#include <cstdint>
#include <limits>
namespace Lib::HAL {
/** Driver class for managing STM32 OTP memory.
* The OTP class provides singleton access and methods to lock, unlock, read, and write OTP memory. It ensures
* type safety and robust memory management for STM32 embedded applications, supporting address range validation
* and type-safe memory access. The class is designed for maintainability and extensibility in embedded system
* development.
*/
class OTP {
public:
/** Type alias for OTP address. */
using AddressType = uint32_t;
/** Size of OTP memory in bytes. */
static constexpr size_t OTP_SIZE = 512;
/** Number of OTP blocks. */
static constexpr size_t OTP_BLOCKS = 16;
/** Locks the OTP memory.
* @return `true` if lock was successful, `false` otherwise.
*/
bool lock() { return HAL_FLASH_Lock() == HAL_OK; }
/** Unlocks the OTP memory.
* @return `true` if unlock was successful, `false` otherwise.
*/
bool unlock() { return HAL_FLASH_Unlock() == HAL_OK; }
/** Writes a value of type T to the specified OTP address.
* @tparam T Type of value to write.
* @param addr OTP address.
* @param value Value to write.
* @return `true` if write was successful, `false` otherwise.
*/
template<typename T> bool write(AddressType addr, T value);
/** Writes a uint8_t value to the specified OTP address.
* @param addr OTP address.
* @param value Value to write.
* @return `true` if write was successful, `false` otherwise.
*/
template<> bool write<uint8_t>(AddressType addr, uint8_t value) {
if (!isValidRange(addr, sizeof(uint8_t))) {
assert(0);
return false;
}
return HAL_FLASH_Program(FLASH_TYPEPROGRAM_BYTE, addr, value) == HAL_OK;
}
/** Writes a uint16_t value to the specified OTP address.
* @param addr OTP address.
* @param value Value to write.
* @return `true` if write was successful, `false` otherwise.
*/
template<> bool write<uint16_t>(AddressType addr, uint16_t value) {
if (!isValidRange(addr, sizeof(uint16_t))) {
assert(0);
return false;
}
return HAL_FLASH_Program(FLASH_TYPEPROGRAM_HALFWORD, addr, value) == HAL_OK;
}
/** Writes a uint32_t value to the specified OTP address.
* @param addr OTP address.
* @param value Value to write.
* @return `true` if write was successful, `false` otherwise.
*/
template<> bool write<uint32_t>(AddressType addr, uint32_t value) {
if (!isValidRange(addr, sizeof(uint32_t))) {
assert(0);
return false;
}
return HAL_FLASH_Program(FLASH_TYPEPROGRAM_WORD, addr, value) == HAL_OK;
}
/** Writes a uint64_t value to the specified OTP address.
* @param addr OTP address.
* @param value Value to write.
* @return `true` if write was successful, `false` otherwise.
*/
template<> bool write<uint64_t>(AddressType addr, uint64_t value) {
if (!isValidRange(addr, sizeof(uint64_t))) {
assert(0);
return false;
}
return HAL_FLASH_Program(FLASH_TYPEPROGRAM_DOUBLEWORD, addr, value) == HAL_OK;
}
/** Reads a value of type T from the specified OTP address.
* @tparam T Type of value to read.
* @param addr OTP address.
* @return Value read from OTP.
*/
template<typename T> T read(AddressType addr);
/** Reads a uint8_t value from the specified OTP address.
* @param addr OTP address.
* @return Value read from OTP.
*/
template<> uint8_t read<uint8_t>(AddressType addr) {
if (!isValidRange(addr, sizeof(uint8_t))) {
assert(0);
return 0;
}
return *reinterpret_cast<uint8_t *>(addr);
}
/** Reads a uint16_t value from the specified OTP address.
* @param addr OTP address.
* @return Value read from OTP.
*/
template<> uint16_t read<uint16_t>(AddressType addr) {
if (!isValidRange(addr, sizeof(uint16_t))) {
assert(0);
return 0;
}
return *reinterpret_cast<uint16_t *>(addr);
}
/** Reads a uint32_t value from the specified OTP address.
* @param addr OTP address.
* @return Value read from OTP.
*/
template<> uint32_t read<uint32_t>(AddressType addr) {
if (!isValidRange(addr, sizeof(uint32_t))) {
assert(0);
return 0;
}
return *reinterpret_cast<uint32_t *>(addr);
}
/** Reads a uint64_t value from the specified OTP address.
* @param addr OTP address.
* @return Value read from OTP.
*/
template<> uint64_t read<uint64_t>(AddressType addr) {
if (!isValidRange(addr, sizeof(uint64_t))) {
assert(0);
return 0;
}
return *reinterpret_cast<uint64_t *>(addr);
}
/** Returns the beginning address of OTP memory.
* @return Beginning address.
*/
static constexpr AddressType beginAddr() { return FLASH_OTP_BASE; }
/** Returns the ending address of OTP memory.
* @return Ending address.
*/
static constexpr AddressType endAddr() { return FLASH_OTP_END; }
/** Returns a pointer to the beginning of OTP memory.
* @return Pointer to beginning.
*/
static constexpr uint8_t * begin() { return reinterpret_cast<uint8_t *>(beginAddr()); }
/** Returns a pointer to the end of OTP memory.
* @return Pointer to end.
*/
static constexpr uint8_t * end() { return reinterpret_cast<uint8_t *>(endAddr()); }
/** Returns the size of OTP memory.
* @return Size in bytes.
*/
static constexpr size_t getSize() { return end() - begin(); }
/** Retrieves the singleton instance of the OTP class.
* @return Reference to the singleton OTP instance.
*/
static OTP & instance() {
static OTP instance;
return instance;
}
private:
/** Constructor of the OTP class. */
OTP() = default;
/** Destructor of the OTP class. */
~OTP() = default;
/** Checks if the address range is valid within OTP memory.
* @param addr Address to check.
* @param size Size in bytes.
* @return `true` if range is valid, `false` otherwise.
*/
bool isValidRange(AddressType addr, size_t size) const {
return addr >= beginAddr() && (addr + size) <= endAddr();
}
};
} // namespace Lib::HAL
#endif
