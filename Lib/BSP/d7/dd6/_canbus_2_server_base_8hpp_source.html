106 /** Scans for active CAN bus client nodes. */
107 void scan() {
108 RxMessage rxMsg;
109 std::array<Node, NUM_OF_CANBUS_CLIENT_NODES> scannedNodes;
110 for (uint8_t index = 0U; index < NUM_OF_CANBUS_CLIENT_NODES; ++index) {
111 auto & node = scannedNodes[index];
112 node.activity = false;
113 uint8_t nodeId = static_cast<uint8_t>(index + NODE_ID_OFFSET);
114 if (send(nodeId) && ServiceBase::receive(rxMsg, RECEIVE_TIMEOUT) && rxMsg.getId() == nodeId) node.activity = true;
115 }
116 postScanCallback(scannedNodes);
117 }
