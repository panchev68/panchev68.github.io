34 /** SPI interface for LAN9252 EtherCAT Slave Controller.
35 *
36 * Encapsulates SPI and NSS pin operations with mutex protection for thread-safe communication.
37 */
38 class Interface : SpiPort {
39 public:
40 /** SPI configuration for LAN9252 interface.
41 *
42 * Holds SPI handle and NSS pin for initialization.
43 */
44 struct Config {
45 SPI_HandleTypeDef * spiHandle; /*!< SPI peripheral handle. */

49 static constexpr uint32_t SPI_TIMEOUT = 100; /*!< SPI timeout in milliseconds. */

51 /** Constructor of Interface class.

53 * @param config SPI and NSS pin configuration.
54 * @pre `config.spiHandle` must not be `nullptr`.
55 * @post Interface is ready for initialization.

59 /** Initializes SPI and mutex.

61 * @post If successful, interface is ready for communication.

63 bool open() override {
64 SpiPort::open();

70 /** Deinitializes SPI and destroys mutex.

81 /** Selects device (NSS low) and locks mutex.
82 *
83 * @post Device is selected and mutex is locked.

85 void select() {

90 /** Deselects device (NSS high) and unlocks mutex.

94 void deselect() {
95 pinNss.write(true);

99 /** SPI full-duplex transfer.
100 *
101 * @param source Source buffer.
102 * @param sourceSize Source length in bytes.
103 * @param[out] destination Destination buffer.
104 * @param destinationSize Destination length in bytes.
105 * @return `true` if successful, `false` otherwise.
106 * @pre `source` must be valid if `sourceSize` > 0.
107 * @pre `destination` must be valid if `destinationSize` > 0.
108 * @pre `sourceSize + destinationSize` must not exceed `MAX_BUFFER_SIZE`.
109 * @post If successful, destination contains received bytes.
110 */
111 bool readWrite(const void * source, size_t sourceSize, void * destination, size_t destinationSize) {
112 const size_t totalLength = sourceSize + destinationSize;
113
114 if (totalLength > MAX_BUFFER_SIZE) return false;
115
116 std::fill(txBuffer.begin(), txBuffer.begin() + totalLength, 0xFF);
117
118 if (source && sourceSize > 0) {
119 const auto * sourceBytes = static_cast<const uint8_t *>(source);
120 std::copy(sourceBytes, sourceBytes + sourceSize, txBuffer.begin());
121 }
122
123 const bool success = SpiPort::sendReceive(txBuffer.data(), rxBuffer.data(), totalLength, SPI_TIMEOUT);
124
125 if (success && destination && destinationSize > 0) {
126 // Copy received data to destination buffer
127 std::copy(rxBuffer.begin() + sourceSize, rxBuffer.begin() + totalLength, static_cast<uint8_t *>(destination));
128 }
129
130 return success;
131 }
132
133 /** Writes data to device.
134 *
135 * @param source Source data pointer.
136 * @param length Data length in bytes.
137 * @return `true` if successful, `false` otherwise.
138 * @pre `source` must not be `nullptr`.
139 * @pre `length` must be greater than 0.
140 * @post If successful, data is written to device.
141 */
142 bool write(const void * source, size_t length) {
143 assert(source != nullptr && length > 0);
144 return SpiPort::send(source, length, SPI_TIMEOUT);
145 }
146
147 /** Writes single byte to device.
148 *
149 * @param value Byte to write.
150 * @return `true` if successful, `false` otherwise.
151 * @post If successful, byte is written to device.
152 */
153 bool write(uint8_t value) { return write(&value, sizeof(value)); }
154
155 /** Reads data from device.
156 *
157 * @param[out] destination Destination buffer.
158 * @param length Data length in bytes.
159 * @return `true` if successful, `false` otherwise.
160 * @pre `destination` must not be `nullptr`.
161 * @pre `length` must be greater than 0.
162 * @post If successful, data is read from device into destination.
163 */
164 bool read(void * destination, size_t length) {
165 assert(destination != nullptr && length > 0);
166 return SpiPort::receive(destination, length, SPI_TIMEOUT);
167 }
168
169 /** Reads single byte from device.
170 *
171 * @param[out] value Reference to store byte.
172 * @return `true` if successful, `false` otherwise.
173 * @post If successful, byte is read from device.
174 */
175 bool read(uint8_t & value) { return read(&value, sizeof(value)); }
176
177 private:

179
180 PinOutput pinNss; /*!< NSS (chip select) pin. */
181 RTOS::Mutex mutexBlock; /*!< Mutex for thread-safe resource access. */
182 std::array<uint8_t, MAX_BUFFER_SIZE> txBuffer; /*!< Temporary transmit buffer. */
183 std::array<uint8_t, MAX_BUFFER_SIZE> rxBuffer; /*!< Temporary receive buffer. */
184 };
