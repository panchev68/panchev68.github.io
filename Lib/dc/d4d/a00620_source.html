84 /** Opens the serial port and starts communication tasks.
85 *
86 * Initializes the UART ports, starts the receive and transmit tasks, and sets the reset pin.
87 *
88 * @return Status::OK if successful, Status::ERROR otherwise.

96 txPort->open();

187 /** Task for receiving data from UART.
188 *
189 * Continuously receives data and buffers it until a complete message is received.
190 *
191 * @pre UART receive port must be initialized.
192 * @post Received data is buffered and handled via callback.
193 */
194 void receiveTask() {
195 size_t bufferIndex = 0;
196 uint8_t value = 0;
197 for (;;) {
198 if (rxPort->receive(&value, sizeof(uint8_t), 1) == sizeof(uint8_t)) {
199 if (bufferIndex < receiveData.size()) receiveData[bufferIndex++] = value;
200 else bufferIndex = 0;
201
202 } else if (bufferIndex > 0) {
203 handlePostReceivedEvent(receiveData.data(), bufferIndex);
204 bufferIndex = 0;
205 }
206 }
207 }

209 /** Task for transmitting data via UART.
210 *
211 * Waits for data in the message buffer and transmits it through the UART port.
212 *
213 * @pre UART transmit port and message buffer must be initialized.
214 * @post Data is transmitted and post-transmit callback is invoked.
215 */
216 void transmitTask() {
217 for (;;) {
218 auto length = messageBuffer.receive(transmitData.data(), transmitData.size(), portMAX_DELAY);
219 if (!txPort->send(transmitData.data(), length, 100)) { assert(false); }
220 handlePostTransmitEvent();
221 }
222 }
