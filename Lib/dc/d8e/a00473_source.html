151 *
152 * Encapsulates date components with comprehensive validation including leap year
153 * calculation and month-specific day range verification. Provides day of week calculation,
154 * string conversion (DD.MM.YYYY format), and comparison operators. Years are stored as
155 * absolute values starting from FIRST_YEAR (2000) for compatibility with 16-bit representation.

163 /** Day of week calculation and string representation utilities.
164 *
165 * Provides Zeller's congruence algorithm for day of week calculation, along with
166 * helper functions for short and long day name lookups. All operations are constexpr
167 * for compile-time evaluation.
168 *
169 * @note Uses ISO week convention (Monday=0, Sunday=6).
170 */
171 class DayOfWeek {
172 public:
173 /** Day of week enumeration.
174 *
175 * ISO week convention: Monday (MON)=0 through Sunday (SUN)=6.
176 */
177 enum class Name : uint8_t {
178 MON = 0, /*!< Monday */
179 TUE, /*!< Tuesday */
180 WED, /*!< Wednesday */
181 THU, /*!< Thursday */
182 FRI, /*!< Friday */
183 SAT, /*!< Saturday */
184 SUN /*!< Sunday */
185 };
186
187 /** Calculates the day of week for a given date using Zeller's congruence algorithm.
188 *
189 * @param date The date to calculate day of week for.
190 * @return The day of week enumeration (MON=0 through SUN=6).
191 */
192 static constexpr Name get(const Date & date) {
193 if (!date.isValid()) return Name::MON;
194 const uint32_t y = date.year - (date.month < 3U ? 1U : 0U);
195 const uint8_t mo = monthData[date.month - 1U].second;
196 const uint32_t wi = (y + y / 4U - y / 100U + y / 400U + mo + date.day) % 7U;
197 return static_cast<Name>((wi + 6U) % 7U);
198 }
199
200 /** Gets the short day name for the given day of week.
201 *
202 * @param name The day of week enumeration.
203 * @return Short string representation (e.g., "MON", "TUE").
204 */
205 static constexpr std::string_view getShortString(Name name) {
206 const size_t index = static_cast<size_t>(name);
207 return std::string_view(daysOfWeek[index].first);
208 }
209
210 /** Gets the long day name for the given day of week.
211 *
212 * @param name The day of week enumeration.
213 * @return Long string representation (e.g., "Monday", "Tuesday").
214 */
215 static constexpr std::string_view getLongString(Name name) {
216 const size_t index = static_cast<size_t>(name);
217 return std::string_view(daysOfWeek[index].second);
218 }

220 private:

412 /** Packs DateTime into a compact 32-bit unsigned integer.
413 *
414 * Bit layout (32 bits total):
415 * - Bits 0-5 (6 bits): seconds (0-59)
416 * - Bits 6-11 (6 bits): minutes (0-59)
417 * - Bits 12-16 (5 bits): hours (0-23)
418 * - Bits 17-21 (5 bits): day (1-31)
419 * - Bits 22-25 (4 bits): month (1-12)
420 * - Bits 26-31 (6 bits): year offset from FIRST_YEAR (2000-2063)
421 *
422 * @param dateTime The DateTime object to pack.
423 * @return Packed 32-bit representation for storage or transmission.
424 * @see DateTime::unpack()
425 */
426 static constexpr uint32_t pack(const DateTime & dateTime) {
427 const uint32_t y = (dateTime.date.year >= Date::FIRST_YEAR) ? (dateTime.date.year - Date::FIRST_YEAR) : 0U;
428 return (dateTime.time.seconds & 0x3FU) | ((dateTime.time.minutes & 0x3FU) << 6U) | ((dateTime.time.hours & 0x1FU) << 12U) |
429 ((dateTime.date.day & 0x1FU) << 17U) | ((dateTime.date.month & 0x0FU) << 22U) | ((y & 0x3FU) << 26U);
430 }
431
432 /** Unpacks DateTime from a 32-bit compressed representation.
433 *
434 * Reverses the packing operation by extracting date and time components from
435 * their bit fields. See pack() for bit layout details.
436 *
437 * @param value The packed 32-bit representation.
438 * @return Unpacked DateTime object with reconstructed date and time components.
439 * @see DateTime::pack()
440 */
441 static constexpr DateTime unpack(uint32_t value) {
442 return DateTime(Date(Date::FIRST_YEAR + ((value >> 26U) & 0x3FU), (value >> 22U) & 0x0FU, (value >> 17U) & 0x1FU),
443 Time((value >> 12U) & 0x1FU, (value >> 6U) & 0x3FU, value & 0x3FU));
444 }
