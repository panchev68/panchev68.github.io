21/** SPI interface for LAN9252 EtherCAT Slave Controller.
22 * Encapsulates SPI and NSS pin operations with mutex protection.
23 */
24class Interface {
25public:
26 /** SPI configuration for LAN9252 interface.
27 * Holds SPI handle and NSS pin.
28 */
29 struct Config {
30 SPI_HandleTypeDef * spiHandle; /*!< SPI handle. */
31 PinHandle nssPin; /*!< NSS pin handle. */
32 };
33
34 static constexpr uint32_t SPI_TIMEOUT = 100; /*!< SPI timeout in ms. */
35
36 /** Constructor of Interface class.
37 * @param config SPI and NSS pin configuration.
38 * @pre `config.spiHandle` must not be `nullptr`.
39 * @post Interface is ready for initialization.
40 */
41 explicit Interface(const Config & config) : spi(config.spiHandle), pinNss(config.nssPin) {}
42
43 /** Initializes SPI and mutex.
44 * @post If successful, interface is ready for communication.
45 */
46 void open() {
47 spi.init();
48 const bool success = mutexBlock.create();
49 assert(success);
50 }
51
52 /** Deinitializes SPI and destroys mutex.
53 * @post Interface is no longer usable.
54 */
55 void close() {
56 spi.deinit();
57 mutexBlock.destroy();
58 }
59
60 /** Selects device (NSS low) and locks mutex.
61 * @post Device is selected and mutex is locked.
62 */
63 void select() {
64 mutexBlock.lock();
65 pinNss.write(false);
66 }
67
68 /** Deselects device (NSS high) and unlocks mutex.
69 * @post Device is deselected and mutex is released.
70 */
71 void deselect() {
72 pinNss.write(true);
73 mutexBlock.unlock();
74 }
75
76 /** SPI full-duplex transfer.
77 * @param source Source buffer.
78 * @param sourceLength Source length.
79 * @param[out] destination Destination buffer.
80 * @param destinationLength Destination length.
81 * @return `true` if successful, `false` otherwise.
82 * @pre `source` must be valid if `sourceLength` > 0.
83 * @pre `destination` must be valid if `destinationLength` > 0.
84 * @pre `sourceLength + destinationLength` must not exceed `MAX_BUFFER_SIZE`.
85 * @post If successful, destination contains received bytes.
86 */
87 bool readWrite(const void * source, size_t sourceLength, void * destination, size_t destinationLength) {
88 const size_t totalLength = sourceLength + destinationLength;
89 if (totalLength > MAX_BUFFER_SIZE) return false;
90
91 std::fill(tempTxBuffer.begin(), tempTxBuffer.begin() + totalLength, 0xFF);
92
93 if (source && sourceLength > 0) {
94 const auto * sourceBytes = static_cast<const uint8_t *>(source);
95 std::copy(sourceBytes, sourceBytes + sourceLength, tempTxBuffer.begin());
96 }
97
98 const bool success = spi.sendReceive(tempTxBuffer.data(), tempRxBuffer.data(), totalLength, SPI_TIMEOUT);
99
100 if (success && destination && destinationLength > 0) {
101 std::copy(tempRxBuffer.begin() + sourceLength, tempRxBuffer.begin() + totalLength, static_cast<uint8_t *>(destination));
102 }
103
104 return success;
105 }
106
107 /** Writes data to device.
108 * @param source Source data pointer.
109 * @param length Data length.
110 * @return `true` if successful, `false` otherwise.
111 * @pre `source` must not be `nullptr`.
112 * @pre `length` must be greater than 0.
113 * @post If successful, data is written to device.
114 */
115 bool write(const void * source, size_t length) {
116 assert(source != nullptr && length > 0);
117 return spi.send(source, length, SPI_TIMEOUT);
118 }
119
120 /** Writes single byte to device.
121 * @param value Byte to write.
122 * @return `true` if successful, `false` otherwise.
123 * @post If successful, byte is written to device.
124 */
125 bool write(uint8_t value) { return write(&value, sizeof(value)); }
126
127 /** Reads data from device.
128 * @param[out] destination Destination buffer.
129 * @param length Data length.
130 * @return `true` if successful, `false` otherwise.
131 * @pre `destination` must not be `nullptr`.
132 * @pre `length` must be greater than 0.
133 * @post If successful, data is read from device into destination.
134 */
135 bool read(void * destination, size_t length) {
136 assert(destination != nullptr && length > 0);
137 return spi.receive(destination, length, SPI_TIMEOUT);
138 }
139
140 /** Reads single byte from device.
141 * @param[out] value Reference to store byte.
142 * @return `true` if successful, `false` otherwise.
143 * @post If successful, byte is read from device.
144 */
145 bool read(uint8_t & value) { return read(&value, sizeof(value)); }
146
147private:
148 static constexpr size_t MAX_BUFFER_SIZE = 256; /*!< Maximum buffer size for SPI transfers. */
149 Lib::HAL::SpiPort spi; /*!< SPI port. */
150 PinOutput pinNss; /*!< NSS pin. */
151 RTOS::Mutex mutexBlock; /*!< Mutex for resource access. */
152 std::array<uint8_t, MAX_BUFFER_SIZE> tempTxBuffer; /*!< Temporary TX buffer. */
153 std::array<uint8_t, MAX_BUFFER_SIZE> tempRxBuffer; /*!< Temporary RX buffer. */
154};
