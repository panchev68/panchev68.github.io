61 /** Mutex class for synchronization.
62 *
63 * Provides a wrapper around FreeRTOS semaphores for mutex functionality, supporting normal and recursive mutexes.
64 * Ensures thread-safe access to shared resources in multi-tasking embedded applications.
65 *
66 * The Mutex class encapsulates FreeRTOS semaphore operations to provide:
67 * - Normal mutexes for basic mutual exclusion
68 * - Recursive mutexes for nested locking scenarios
69 * - Timeout-based locking with configurable wait periods
70 * - Move semantics for efficient resource management
71 * - RAII-based locking through the LockGuard class
72 *
73 * Usage patterns:
74 * - Use normal mutexes for simple critical sections
75 * - Use recursive mutexes when a task may need to acquire the same mutex multiple times
76 * - Use LockGuard for automatic lock management and exception safety
77 * - Check mutex validity before operations using isValid()
78 *
79 * @note Mutex operations are not ISR-safe; use from tasks only.
80 * @note Recursive mutexes have higher memory and performance overhead.
81 * @note Always check FreeRTOS configuration for mutex support.
82 *
83 * @attention Do not use mutexes in ISR contexts; use binary semaphores instead.
84 * @attention Avoid long-held locks to prevent priority inversion and deadlocks.
85 */

97 * Defines the type of mutex: normal or recursive.

103 RECURSIVE = 1 /*!< Recursive mutex type. Allows nested locking by the same task. */

118 /** LockGuard class for RAII locking.
119 *
120 * Automatically locks the mutex on construction and unlocks on destruction.
121 * This RAII (Resource Acquisition Is Initialization) wrapper ensures that mutexes
122 * are properly unlocked even if exceptions occur or early returns happen.
123 *
124 * The LockGuard attempts to acquire the mutex during construction with an optional timeout.
125 * If the lock is acquired, it will be automatically released when the LockGuard goes out of scope.
126 * The lock status can be checked using isLocked() or the bool conversion operator.
127 *
128 * Usage example:
129 * @code
130 * {
131 * MutexGuard guard(myMutex, 100); // Try to lock for 100 ticks
132 * if (guard) {
133 * // Critical section - mutex is locked
134 * // ...
135 * } // Mutex automatically unlocked here
136 * }
137 * @endcode
138 *
139 * @note LockGuard is non-copyable and non-movable to prevent accidental resource management issues.
140 * @note If lock acquisition fails, the guard can still be used but will not unlock on destruction.
141 */
142 class LockGuard {
143 public:
144 /** Constructor of LockGuard class.
145 *
146 * Attempts to acquire the mutex lock with the specified timeout.
147 * If the lock is successfully acquired, it will be automatically released
148 * when the LockGuard is destroyed.
149 *
150 * @param mutex Reference to the mutex to lock.
151 * @param timeout Timeout for locking in ticks. Use Mutex::WAIT_FOREVER for infinite wait.
152 *
153 * @note If timeout is 0, the constructor performs a non-blocking lock attempt.
154 * @note The mutex must be valid (created) before using LockGuard.
155 */
156 explicit LockGuard(Mutex & mutex, TickType timeout = WAIT_FOREVER) noexcept : mutex(mutex), lockAcquired(mutex.lock(timeout)) {}
157
158 /** Destructor of LockGuard class.
159 *
160 * Automatically unlocks the mutex if it was successfully acquired during construction.
161 * This ensures proper resource cleanup and prevents deadlocks.
162 *
163 * @note Only unlocks if the lock was acquired; safe to call even if lock failed.
164 * @note Destructor is noexcept to ensure it never throws exceptions.
165 */
166 ~LockGuard() noexcept {
167 if (lockAcquired) mutex.unlock();
168 }
169
170 /** Check if the lock was acquired.
171 *
172 * Returns whether the mutex lock was successfully obtained during construction.
173 *
174 * @return `true` if the lock was acquired, `false` otherwise.
175 *
176 * @note This method is const and noexcept for safe status checking.
177 */
178 bool isLocked() const noexcept { return lockAcquired; }
179
180 /** Conversion operator to bool.
181 *
182 * Allows the LockGuard to be used directly in boolean contexts to check lock status.
183 * Equivalent to calling isLocked().
184 *
185 * @return `true` if the lock was acquired, `false` otherwise.
186 *
187 * @note Explicit conversion prevents accidental implicit conversions.
188 */
189 explicit operator bool() const noexcept { return lockAcquired; }
190
191 private:
192 Mutex & mutex; /*!< Reference to the associated mutex. */
193 const bool lockAcquired; /*!< Flag indicating if the lock was acquired. */
194
195 LockGuard(const LockGuard &) = delete;
196 LockGuard & operator=(const LockGuard &) = delete;
197 LockGuard(LockGuard &&) = delete;
198 LockGuard & operator=(LockGuard &&) = delete;
199 };

223 /** Create the mutex.
224 *
225 * Allocates and initializes the FreeRTOS semaphore based on the mutex type.
226 * This method creates the underlying FreeRTOS semaphore and stores the handle.
227 *
228 * @return `true` if creation was successful, `false` otherwise.
229 *
230 * @note Fails if the mutex is already created or if FreeRTOS configuration doesn't support the mutex type.
231 * @note For recursive mutexes, configUSE_RECURSIVE_MUTEXES must be 1.
232 * @note For normal mutexes, configUSE_MUTEXES must be 1.
233 *
234 * @attention Ensure sufficient heap memory is available for semaphore creation.
235 */
236 bool open() noexcept {
237 if (handle) return false;
238 if (mutexType == MutexType::RECURSIVE) {
239#if configUSE_RECURSIVE_MUTEXES == 1
240 handle = xSemaphoreCreateRecursiveMutex();
241#else
242 assert(false);
243 handle = nullptr;
244#endif
245 } else {
246#if configUSE_MUTEXES == 1
247 handle = xSemaphoreCreateMutex();
248#else
249 assert(false);
250 handle = nullptr;
251#endif
252 }
253 assert(handle != nullptr);
254 return handle != nullptr;
255 }

257 /** Destroy the mutex.
258 *
259 * Deletes the FreeRTOS semaphore and resets the handle to null.
260 * After calling this method, the mutex can be reopened with open().
261 *
262 * @note Safe to call multiple times or on uninitialized mutexes.
263 * @note Any tasks waiting on this mutex will be unblocked.
264 *
265 * @attention Do not call while the mutex is locked by another task.
266 */
267 void close() noexcept {
268 if (handle) {

306 /** Lock the mutex.
307 *
308 * Attempts to take the mutex with a timeout. Blocks the calling task
309 * until the mutex is available or the timeout expires.
310 *
311 * @param timeout Timeout in ticks. Use NO_WAIT for non-blocking, WAIT_FOREVER for infinite wait.
312 * @return `true` if the lock was acquired, `false` if timeout occurred or mutex invalid.
313 *
314 * @note Must be called from a task context, not ISR.
315 * @note For recursive mutexes, allows nested locking by the same task.
316 *
317 * @attention Avoid long timeouts in critical system paths.
318 */
319 bool lock(TickType timeout = WAIT_FOREVER) noexcept {
320 if (!handle) {
321 assert(false);
322 return false;
323 }
324 if (mutexType == MutexType::RECURSIVE) {
325#if configUSE_RECURSIVE_MUTEXES == 1
326 return xSemaphoreTakeRecursive(handle, timeout) == pdTRUE;
327#else
328 assert(false);
329 return false;
330#endif
331 }
332#if configUSE_MUTEXES == 1
333 return xSemaphoreTake(handle, timeout) == pdTRUE;
334#else
335 assert(false);
336 return false;
337#endif
338 }

340 /** Try to lock the mutex without waiting.
341 *
342 * Attempts to acquire the mutex immediately without blocking.
343 * Returns immediately regardless of success or failure.
344 *
345 * @return `true` if the lock was acquired, `false` if already locked or invalid.
346 *
347 * @note Equivalent to lock(NO_WAIT).
348 * @note Useful for polling-style mutex acquisition.

350 bool tryLock() noexcept { return lock(NO_WAIT); }

352 /** Unlock the mutex.
353 *
354 * Releases the mutex, allowing other tasks to acquire it.
355 * For recursive mutexes, decrements the lock count.
356 *
357 * @note Safe to call even if mutex is not locked (no-op).
358 * @note For recursive mutexes, must be called the same number of times as lock().
359 *
360 * @attention Ensure the calling task actually owns the lock.
361 */
362 void unlock() noexcept {
363 if (!handle) return;
364 if (mutexType == MutexType::RECURSIVE) {
365#if configUSE_RECURSIVE_MUTEXES == 1
366 xSemaphoreGiveRecursive(handle);
367#else
368 assert(false);
369#endif
370 return;
371 }
372#if configUSE_MUTEXES == 1
373 xSemaphoreGive(handle);
374#else
375 assert(false);
376#endif
377 }
