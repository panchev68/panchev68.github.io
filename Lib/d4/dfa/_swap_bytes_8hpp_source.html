20 template<typename U>
21 constexpr U bswapUnsigned(U value) {
22 static_assert(std::is_integral<U>::value, "U must be integral");
23 static_assert(std::is_unsigned<U>::value, "U must be unsigned");
24 static_assert(sizeof(U) == 1U || sizeof(U) == 2U || sizeof(U) == 4U || sizeof(U) == 8U,
25 "Unsupported integer width for byte swap");
26
27 U result = 0;
28 for (size_t i = 0U; i < sizeof(U); ++i) {
29 const U byte = (value >> static_cast<U>(8U * i)) & static_cast<U>(0xFFU);
30 result |= (byte << static_cast<U>(8U * (sizeof(U) - 1U - i)));
31 }
32 return result;
33 }

36 /** Swaps the byte order of a value.
37 * @tparam T Type of the value.
38 * @param value Value whose byte order is to be swapped.
39 * @return Value with its byte order swapped.
40 */
41 template<typename T>
42 constexpr T swapBytes(T value) {
43 static_assert(std::is_integral<T>::value, "swapBytes requires integral type");
44 using U = typename std::make_unsigned<T>::type;
45 const U u = static_cast<U>(value);
46 const U r = Detail::bswapUnsigned<U>(u);
47 return static_cast<T>(r);
48 }

61 /** Swaps the byte order of a memory buffer.
62 * Supports in-place reversal when `src == dst`.
63 * @param src Pointer to the source buffer.
64 * @param dst Pointer to the destination buffer (can be same as `src`).
65 * @param size Length of the buffer in bytes.
66 */
67 inline void swapBytes(const void * src, void * dst, size_t size) {
68 assert(src != nullptr);
69 assert(dst != nullptr);
70
71 if (src == dst) {
72 auto * p = static_cast<uint8_t *>(dst);
73 if (size == 0U) return;
74 size_t i = 0U;
75 size_t j = size - 1U;
76 while (i < j) {
77 const uint8_t tmp = p[i];
78 p[i] = p[j];
79 p[j] = tmp;
80 ++i;
81 --j;
82 }
83 return;
84 }
85
86 const auto * srcPtr = static_cast<const uint8_t *>(src);
87 auto * dstPtr = static_cast<uint8_t *>(dst);
88 for (size_t index = 0U; index < size; ++index) dstPtr[index] = srcPtr[size - 1U - index];
89 }
