87 /** Sets up the parameters by reading from storage and validating.
88 * @return Status of the setup operation.
89 */
90 Status setup() noexcept {
91 if (!storage->open() || !storage->read(buffer.get(), size + CRC_SIZE)) {
92 assert(false);
93 return Status::ERROR;
94 }
95 if (!isValid()) {
96 restoreByDefault();
97 if (write() != Status::OK) {
98 assert(false);
99 return Status::ERROR;
100 }
101 }
102 std::memcpy(data, buffer.get(), size);
103 return Status::OK;
104 }
105
106 /** Writes the current parameters to storage with CRC.
107 * @return Status of the write operation.
108 */
109 Status write() noexcept {
110 finalize();
111 bool result = storage->write(buffer.get(), size + CRC_SIZE);
112 assert(result);
113 return result ? Status::OK : Status::ERROR;
114 }
115
116 /** Checks if the stored data is valid using CRC.
117 * @return true if valid, false otherwise.
118 */
119 bool isValid() const noexcept { return crc16(buffer.get(), size) == *getCrc(); }

125 virtual void restoreByDefault() = 0;
