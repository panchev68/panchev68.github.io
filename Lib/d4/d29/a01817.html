Lib::Helper::Convert::dBM

Converting dB values to percentages using the dBM class.

This example demonstrates how to use the dBM class to convert dB (decibel) values to percentage representations for signal strength indicators. The class provides both linear and lookup-table based conversions.

Example usage:
#include "Lib/Helper/Convert/dBM.hpp"
#include <iostream>
int main() {
// Convert dB values to linear percentages
int16_t dbValue1 = -50; // -50 dB
uint8_t percent1 = Lib::Helper::Convert::dBM::toPercent(dbValue1);
std::cout << "dB -50 to percent: " << static_cast<int>(percent1) << "%" << std::endl; // Output: 50%
int8_t dbValue2 = -20; // -20 dB
uint8_t percent2 = Lib::Helper::Convert::dBM::toPercent(dbValue2);
std::cout << "dB -20 to percent: " << static_cast<int>(percent2) << "%" << std::endl; // Output: 80%
// Convert using digital percent lookup table
int8_t dbValue3 = 10; // 10 dB
uint8_t digiPercent = Lib::Helper::Convert::dBM::toDigiPercent(dbValue3);
std::cout << "dB 10 to digital percent: " << static_cast<int>(digiPercent) << "%" << std::endl; // Output: 100%
int8_t dbValue4 = 50; // 50 dB
uint8_t digiPercent2 = Lib::Helper::Convert::dBM::toDigiPercent(dbValue4);
std::cout << "dB 50 to digital percent: " << static_cast<int>(digiPercent2) << "%" << std::endl; // Output: 50%
return 0;
}
