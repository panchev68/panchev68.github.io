2 * I2CPort class definition and I2C HAL abstraction for STM32.
3 * This file defines the `I2CPort` class, which provides a type-safe, RTOS-integrated abstraction
4 * for STM32 HAL I2C peripherals. It manages device registration, callback setup, event signaling,
5 * and provides high-level methods for I2C memory and master transfers, error/status handling,
6 * and device readiness checks. The class is designed for use with FreeRTOS and TouchGFX, and
7 * integrates with the DeviceManagerBase for device tracking and resource management.

66 /** Event bits for I2C transfer completion and error signaling. */

88 /** Initializes the I2C port and registers callbacks.
89 * @return `true` if initialization was successful, `false` otherwise.
90 */
91 bool init() {
92 if (!eventGroup.isCreated() && !eventGroup.create(false)) {
93 assert(false);
94 return false;
95 }
96 if (HAL_I2C_Init(getHandle()) != HAL_OK) {
97 assert(false);
98 return false;
99 }
100 static constexpr std::array<std::pair<CallbackID, void (*)(I2C_HandleTypeDef *)>, 6> callbacks{{{CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID, handleInterrupt<CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID>},
101 {CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID, handleInterrupt<CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID>},
102 {CallbackID::HAL_I2C_MEM_RX_COMPLETE_CB_ID, handleInterrupt<CallbackID::HAL_I2C_MEM_RX_COMPLETE_CB_ID>},
103 {CallbackID::HAL_I2C_MEM_TX_COMPLETE_CB_ID, handleInterrupt<CallbackID::HAL_I2C_MEM_TX_COMPLETE_CB_ID>},
104 {CallbackID::HAL_I2C_ABORT_CB_ID, handleInterrupt<CallbackID::HAL_I2C_ABORT_CB_ID>},
105 {CallbackID::HAL_I2C_ERROR_CB_ID, handleInterrupt<CallbackID::HAL_I2C_ERROR_CB_ID>}}};
106 for (const auto & cb : callbacks) {
107 if (HAL_I2C_RegisterCallback(getHandle(), cb.first, cb.second) != HAL_OK) {
108 assert(false);
109 return false;
110 }
111 }
112 return true;

120 /** Receives data from an I2C device in master mode.
121 * @param deviceAddress I2C device address.
122 * @param[out] destination Pointer to the destination buffer.
123 * @param size Number of bytes to receive.
124 * @param timeout Timeout in milliseconds.
125 * @return `true` if receive was successful, `false` otherwise.
126 * @pre `destination` must not be `nullptr` and `size` must be greater than 0.
127 */
128 bool receive(uint32_t deviceAddress, void * destination, size_t size, uint32_t timeout = 0) {
129 assert(destination && size > 0);
130 return HAL_I2C_Master_Receive_IT(getHandle(), static_cast<uint16_t>(deviceAddress), static_cast<uint8_t *>(destination), static_cast<uint16_t>(size)) == HAL_OK && waitForComplete<EventBit::RX_COMPLETED>(timeout);
131 }
132
133 /** Sends data to an I2C device in master mode.
134 * @param deviceAddress I2C device address.
135 * @param source Pointer to the source buffer.
136 * @param size Number of bytes to send.
137 * @param timeout Timeout in milliseconds.
138 * @return `true` if send was successful, `false` otherwise.
139 * @pre `source` must not be `nullptr` and `size` must be greater than 0.
140 */
141 bool send(uint32_t deviceAddress, const void * source, size_t size, uint32_t timeout = 0) {
142 assert(source && size > 0);
143 return HAL_I2C_Master_Transmit_IT(getHandle(), static_cast<uint16_t>(deviceAddress), const_cast<uint8_t *>(static_cast<const uint8_t *>(source)), static_cast<uint16_t>(size)) == HAL_OK && waitForComplete<EventBit::TX_COMPLETED>(timeout);
144 }

146 /** Receives data from an I2C device memory.
147 * @param deviceAddress I2C device address.
148 * @param addressSize Memory address size.
149 * @param memoryAddress Memory address to read from.
150 * @param[out] destination Pointer to the destination buffer.
151 * @param size Number of bytes to receive.
152 * @param timeout Timeout in milliseconds.
153 * @return `true` if receive was successful, `false` otherwise.
154 */
155 bool receive(uint32_t deviceAddress, MemAddrSize addressSize, uint32_t memoryAddress, void * destination, size_t size, uint32_t timeout = 0) {
156 return HAL_I2C_Mem_Read_IT(getHandle(), static_cast<uint16_t>(deviceAddress), static_cast<uint16_t>(memoryAddress), static_cast<uint32_t>(addressSize), static_cast<uint8_t *>(destination), static_cast<uint16_t>(size)) == HAL_OK && waitForComplete<EventBit::MEM_RX_COMPLETED>(timeout);
157 }

159 /** Sends data to an I2C device memory.
160 * @param deviceAddress I2C device address.
161 * @param addressSize Memory address size.
162 * @param memoryAddress Memory address to write to.

164 * @param size Number of bytes to send.
165 * @param timeout Timeout in milliseconds.

168 bool send(uint32_t deviceAddress, MemAddrSize addressSize, uint32_t memoryAddress, const void * source, size_t size, uint32_t timeout = 0) {
169 return HAL_I2C_Mem_Write_IT(getHandle(), static_cast<uint16_t>(deviceAddress), static_cast<uint16_t>(memoryAddress), static_cast<uint32_t>(addressSize), const_cast<uint8_t *>(static_cast<const uint8_t *>(source)), static_cast<uint16_t>(size)) == HAL_OK && waitForComplete<EventBit::MEM_TX_COMPLETED>(timeout);
170 }

172 /** Aborts an ongoing I2C transfer.

191 /** Waits for transfer completion or error.
192 * @tparam eventBit The event bit to wait for.
193 * @param timeout Timeout in milliseconds.
194 * @return `true` if the event occurred, `false` on timeout or error.
195 */
196 template<EventBit eventBit> bool waitForComplete(uint32_t timeout) {
197 const auto result = eventGroup.waitBits(static_cast<uint32_t>(eventBit) | static_cast<uint32_t>(EventBit::ERROR), true, false, timeout);
198 return (result & static_cast<uint32_t>(eventBit)) != 0;
199 }

201 /** Interrupt handler template.
202 * @tparam callbackID The callback ID.
203 * @param handle Pointer to the I2C handle.
204 */
205 template<CallbackID callbackID> static void handleInterrupt(I2C_HandleTypeDef * handle) {
206 if (handle && handle->Instance) {
207 auto * port = deviceManager.getDevice(handle);
208 if (port) {
209 if constexpr (callbackID == CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID) {
210 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::RX_COMPLETED));
211 } else if constexpr (callbackID == CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID) {
212 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::TX_COMPLETED));
213 } else if constexpr (callbackID == CallbackID::HAL_I2C_MEM_RX_COMPLETE_CB_ID) {
214 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::MEM_RX_COMPLETED));
215 } else if constexpr (callbackID == CallbackID::HAL_I2C_MEM_TX_COMPLETE_CB_ID) {
216 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::MEM_TX_COMPLETED));
217 } else if constexpr (callbackID == CallbackID::HAL_I2C_ABORT_CB_ID) {
218 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::ABORT_COMPLETED));
219 } else if constexpr (callbackID == CallbackID::HAL_I2C_ERROR_CB_ID) {
220 port->eventGroup.setBits(static_cast<uint32_t>(EventBit::ERROR));
221 }
222 }
223 }
