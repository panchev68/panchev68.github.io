37 /** Convert string to uppercase.
38 *
39 * @param input Input string view to convert.
40 * @return New string with all characters converted to uppercase.
41 *
42 * @pre input.data() is valid pointer when size() > 0.
43 * @post Returns string with all characters converted to uppercase.

45 static std::string toUpperCase(std::string_view input) {
46 std::string result(input.data(), input.size());
47 std::transform(result.begin(), result.end(), result.begin(),
48 [](unsigned char c) { return static_cast<char>(std::toupper(c)); });

53 *
54 * @param input Input string view to convert.

56 *

67 /** Trim leading and trailing whitespace.
68 *
69 * @param input Input string view to trim.
70 * @return String view with leading and trailing whitespace removed.
71 *
72 * @pre input.data() is valid pointer when size() > 0.
73 * @post Returns string view with whitespace removed from both ends.
74 */
75 static std::string_view trim(std::string_view input) {
76 if (input.empty()) return std::string_view();
77 size_t start = 0;
78 size_t end = input.size();
79 while (start < end && std::isspace(static_cast<unsigned char>(input[start]))) ++start;
80 while (end > start && std::isspace(static_cast<unsigned char>(input[end - 1]))) --end;
81 return input.substr(start, end - start);
82 }

84 /** Find index of a C-string in a null-terminated array.
85 *
86 * @param target String view to search for.
87 * @param array Null-terminated array of C-strings.
88 * @return Index of the target string if found, or -1 if not found.
89 *
90 * @pre array is not null pointer.
91 * @post Returns index of found string or -1 if not found.
92 */
93 static int32_t findInArray(std::string_view target, const char * const * array) {
94 constexpr int32_t NOT_FOUND = -1;
95 assert(array);
96 int32_t index = 0;
97 while (*array) {
98 if (target == std::string_view(*array)) return index;
99 ++array;
100 ++index;
101 }
102 return NOT_FOUND;
103 }
