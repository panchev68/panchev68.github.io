68 static uint32_t stringToU32(const char * versionString) {
69 uint8_t major = 0, minor = 0;
70 const char * p = versionString;
71 // Parse major version
72 while (*p && *p != '.') {
73 if (*p >= '0' && *p <= '9') {
74 major = static_cast<uint8_t>(major * 10 + (*p - '0'));
75 } else {
76 return 0; // Invalid character
77 }
78 ++p;
79 }
80 if (*p == '.') ++p;
81 // Parse minor version
82 while (*p && *p != '.') {

84 minor = static_cast<uint8_t>(minor * 10 + (*p - '0'));
85 } else {
86 return 0; // Invalid character
87 }
88 ++p;
89 }
90 return (static_cast<uint32_t>(major) << 8) | minor;
