29 /** Semaphore class for RTOS synchronization.
30 *
31 * The Semaphore class encapsulates FreeRTOS binary semaphore functionality, providing
32 * a high-level interface for thread synchronization. It supports operations like taking
33 * and giving semaphores with configurable timeouts, ISR-safe operations, and state queries.
34 *
35 * Role: Provides synchronization primitives for RTOS tasks and interrupt service routines.
36 *
37 * Important design decisions: Inherits from SemaphoreBase for common semaphore operations
38 * and uses RAII principles for resource management.
39 *
40 */
41 class Semaphore : public SemaphoreBase {
42 public:
43 /** Enumeration of semaphore states.
44 *
45 * Defines the possible states of a semaphore for easy querying and debugging.
46 *
47 */
48 enum class State : uint8_t {
49 AVAILABLE = 0, /*!< Semaphore is available and can be taken. */
50 TAKEN = 1, /*!< Semaphore is currently taken. */
51 INVALID = 2 /*!< Semaphore handle is invalid or not initialized. */
52 };

171 /** Take the semaphore with second timeout.
172 *
173 * Attempts to take the semaphore within the specified seconds.
174 *
175 * @param waitSec Maximum seconds to wait.
176 * @return True if taken successfully, false on timeout.
177 *
178 */
179 bool takeSeconds(uint32_t waitSec) noexcept {
180 const uint64_t ms = static_cast<uint64_t>(waitSec) * 1000ULL;
181 const uint32_t clampedMs = (ms > UINT32_MAX) ? UINT32_MAX : static_cast<uint32_t>(ms);
182 return takeMs(clampedMs);
183 }

279 }

295 /** Factory method to create a semaphore.
296 *
297 * Creates a unique pointer to a Semaphore instance.
298 *
299 * @param isAutoCreate If true, automatically opens the semaphore.
300 * @return Unique pointer to the semaphore, or nullptr on failure.
301 *
302 */
303 static std::unique_ptr<Semaphore> make(bool isAutoCreate = true) {
304 auto instance = std::make_unique<Semaphore>(isAutoCreate);
305 return (isAutoCreate && !instance->isOpen()) ? nullptr : std::move(instance);
306 }
307
308 /** Factory method to create a semaphore with initial state.
309 *
310 * Creates a semaphore and optionally gives it initially.
311 *
312 * @param isInitiallyGiven If true, gives the semaphore after creation.
313 * @return Unique pointer to the semaphore.
314 *
315 */
316 static std::unique_ptr<Semaphore> makeWithState(bool isInitiallyGiven = false) {
317 auto instance = make(true);
318 if (instance && isInitiallyGiven) instance->give();
319 return instance;
320 }
321
322 /** Factory method to create a mutex semaphore.
323 *
324 * Creates a semaphore that starts in the given state (mutex).
325 *
326 * @return Unique pointer to the mutex semaphore.

328 */
329 static std::unique_ptr<Semaphore> makeMutex() { return makeWithState(true); }

331 /** Factory method to create a signal semaphore.

338 static std::unique_ptr<Semaphore> makeSignal() { return makeWithState(false); }
