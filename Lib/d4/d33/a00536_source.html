1/** RAII wrapper for FreeRTOS event groups with type-safe bit management.
2 *
3 * Defines `RTOS::EventGroup<N_BITS>`, a C++17 move-only RAII wrapper around
4 * FreeRTOS event groups. Provides type-safe helpers for setting and waiting on
5 * event bits, supports both template- and bitset-based interfaces, and includes
6 * ISR-safe helpers that call the appropriate FreeRTOS ISR APIs.
7 *
8 * The wrapper manages event group lifecycle through RAII semantics, automatically
9 * releasing resources in the destructor. Supports variadic template bit operations
10 * for compile-time type checking and flexible bitset-based operations for dynamic
11 * bit selection. All ISR-safe operations perform context switching when needed.
12 *
13 * Important dependencies include FreeRTOS.h, event_groups.h, and Timeout.hpp.
14 * Template parameter `N_BITS` must not exceed the bit width of `EventBits_t` on
15 * the target platform. The class is move-only to prevent accidental double-release.
16 *
17 * @note Designed for real-time embedded applications with interrupt handling.
18 * @note Thread-safe ISR operations with automatic context switch support.
19 *

23 */

35 /** RAII wrapper for FreeRTOS event groups.
36 *
37 * Owns a FreeRTOS event group handle and releases it in the destructor.
38 * The type is move-only to prevent accidental double-release of the underlying
39 * handle. Provides both template-based and bitset-based bit operations for
40 * flexibility and type safety in embedded real-time systems.
41 *
42 * @tparam N_BITS Number of event bits supported (1-EventBits_t width).
43 *
44 * @note Move semantics transfer handle ownership to prevent dangling references.
45 * @note Not copyable; use move semantics for resource management.
46 */
47 template <size_t N_BITS> class EventGroup final {
48 public:
49 using EventBits = std::bitset<N_BITS>;
50
51 /** Construct an EventGroup in an uninitialized state.
52 *
53 * Initializes the event group handle to null. The underlying FreeRTOS event
54 * group is not created until `open()` is called. Static assertions validate
55 * template parameter at compile-time.
56 */
57 EventGroup() : handle(nullptr) {
58 static_assert(N_BITS > 0 && N_BITS <= (sizeof(EventBits_t) * 8),
59 "N_BITS must be between 1 and EventBits_t bit width");
60 }
61
62 /** Destructor. Releases the FreeRTOS event group handle if present. */
63 ~EventGroup() {
64 if (handle) vEventGroupDelete(handle);
65 }
66
67 /** Move constructor transfers ownership of the underlying handle.
68 *
69 * Transfers handle ownership from the source object to this instance.
70 * The source object's handle is set to null to prevent double-release.
71 *
72 * @param other Source EventGroup to move from.
73 */
74 EventGroup(EventGroup && other) : handle(other.handle) {
75 other.handle = nullptr;
76 }
77
78 /** Create the FreeRTOS event group.
79 *
80 * Allocates and initializes a FreeRTOS event group from the heap.
81 * After successful creation, the event group is ready for use with
82 * `waitForBits()` and `setBitsFromISR()` operations.
83 *
84 * @return `true` if creation succeeded, `false` if heap allocation failed.
85 *
86 * @pre Event group must not already be open.
87 * @post If successful, the event group is ready for use.
88 */
89 bool open() {
90 handle = xEventGroupCreate();
91 return handle != nullptr;
92 }
93
94 /** Destroy the event group and release associated resources.
95 *
96 * Deletes the FreeRTOS event group handle and frees associated memory.
97 * Safe to call multiple times or on an uninitialized event group.
98 */
99 void close() {
100 if (handle) {
101 vEventGroupDelete(handle);
102 handle = nullptr;
103 }
104 }

106 /** Wait for the specified bit indices to be set (template variant).
107 *
108 * Blocks until any of the specified bits are set or timeout expires.
109 * Compile-time validation ensures bit indices are within range.
110 *
111 * @tparam BITS Variadic bit indices to wait for.
112 * @param timeout Timeout duration in ticks.
113 * @return Bitset representing the event group bits when the wait completed.
114 *
115 * @pre The event group handle must be valid (created via `open()`).
116 */
117 template <std::size_t... BITS> EventBits waitForBits(uint32_t timeout) {
118 static_assert(((BITS < N_BITS) && ...), "Bit index out of range");
119 return EventBits(xEventGroupWaitBits(handle, ((1U << BITS) | ...), pdFALSE, pdFALSE, timeout));
120 }
121
122 /** Wait for the specified bits to be set (bitset variant).
123 *
124 * Blocks until any of the bits in the provided bitset are set or timeout
125 * expires. Converts the bitset to the native FreeRTOS event bits type.
126 *
127 * @param bitsToWait Bitset of bits to wait for.
128 * @param timeout Timeout duration in ticks.
129 * @return Bitset representing the event group bits when the wait completed.
130 *
131 * @pre The event group handle must be valid (created via `open()`).
132 */
133 EventBits waitForBits(const EventBits & bitsToWait, uint32_t timeout) {
134 return EventBits(xEventGroupWaitBits(handle, static_cast<EventBits_t>(bitsToWait.to_ulong()), pdFALSE,
135 pdFALSE, timeout));
136 }
137
138 /** Set event bits from ISR context (bitset variant).
139 *
140 * Sets the provided bits and triggers a context switch if a higher priority
141 * task was woken. Safe to call from interrupt context.
142 *
143 * @param bits Bitset of bits to set.
144 *
145 * @pre Must be called from an ISR context.
146 */
147 void setBitsFromISR(const EventBits & bits) {
148 BaseType_t isWoken = pdFALSE;
149 xEventGroupSetBitsFromISR(handle, static_cast<EventBits_t>(bits.to_ulong()), &isWoken);
150 portYIELD_FROM_ISR(isWoken);
151 }
152
153 /** Set the specified bit indices from ISR context (template variant).
154 *
155 * Sets the bits corresponding to the provided indices and triggers a context
156 * switch if a higher priority task was woken. Compile-time validation ensures
157 * indices are within range.
158 *
159 * @tparam BITS Variadic bit indices to set.
160 *
161 * @pre Must be called from an ISR context.
162 */
163 template <std::size_t... BITS> void setBitsFromISR() {
164 BaseType_t isWoken = pdFALSE;
165 xEventGroupSetBitsFromISR(handle, ((1U << BITS) | ...), &isWoken);
166 portYIELD_FROM_ISR(isWoken);
167 }
168
169 /** Return whether the event group has been created and is ready.
170 *
171 * Checks if the event group has been successfully created via `open()`.
172 *
173 * @return `true` if created and ready to use, `false` otherwise.
174 */
175 bool isOpen() const {
176 return handle != nullptr;
177 }
178
179 private:
180 EventGroupHandle_t handle; /*!< FreeRTOS event group handle. */
181
182 EventGroup(const EventGroup &) = delete;
183 EventGroup & operator=(const EventGroup &) = delete;
184 EventGroup & operator=(EventGroup &&) = delete;
185 };

188 *
189 * Comprehensive example demonstrating EventGroup usage in embedded applications.
190 * This example shows initialization, ISR signaling, and task-side waiting patterns
191 * for real-time event synchronization using FreeRTOS event groups.
192 *
193 * @par Example usage:
194 *
195 * @code{.cpp}
196 * #include "EventGroup.hpp"
197 * #include <bitset>
198 *
199 * // Define application event bit roles for readability
200 * enum Event : size_t { EVT_INIT = 0, EVT_FRAME = 1, EVT_BUTTON = 2, EVT_ERROR = 3 };
201 *
202 * // EventGroup instance (RAII wrapper) - supports up to 8 event bits
203 * static RTOS::EventGroup<8> systemEvents;
204 *
205 * // Initialize the event group during system startup
206 * void initializeSystemEvents() {
207 * if (!systemEvents.open()) {
208 * // Handle allocation failure (log, retry, safe-fail)
209 * // In embedded systems, this might trigger system reset
210 * }
211 * }
212 *
213 * // Example ISR: signal frame synchronization event from display interrupt
214 * extern "C" void onDisplayVSyncISR() {
215 * // Use template variant for compile-time type safety
216 * systemEvents.setBitsFromISR<EVT_FRAME>();
217 * }
218 *
219 * // Example ISR: signal button press event from GPIO interrupt
220 * extern "C" void onButtonPressISR() {
221 * // Use template variant for single bit signaling
222 * systemEvents.setBitsFromISR<EVT_BUTTON>();
223 * }
224 *
225 * // Task that waits for frame events with timeout handling
226 * void frameProcessingTask(uint32_t timeoutTicks) {
227 * if (!systemEvents.isOpen()) return; // Ensure created
228 *
229 * // Template variant: wait for EVT_FRAME only
230 * auto result = systemEvents.waitForBits<EVT_FRAME>(timeoutTicks);
231 * if (result[EVT_FRAME]) {
232 * // Handle frame event (render, process, update display)
233 * processNextFrame();
234 * } else {
235 * // Timeout occurred; handle accordingly
236 * handleFrameTimeout();
237 * }
238 * }
239 *
240 * // Task that waits for multiple conditions using bitset
241 * void userInterfaceTask(uint32_t timeoutTicks) {
242 * if (!systemEvents.isOpen()) return;
243 *
244 * std::bitset<8> waitMask;
245 * waitMask[EVT_BUTTON] = 1; // Wait for button press
246 * waitMask[EVT_INIT] = 1; // Or initialization complete
247 *
248 * auto result = systemEvents.waitForBits(waitMask, timeoutTicks);
249 * if (result.any()) {
250 * if (result[EVT_BUTTON]) {
251 * // Handle button interaction
252 * processButtonEvent();
253 * }
254 * if (result[EVT_INIT]) {
255 * // Handle initialization completion
256 * finalizeSystemSetup();
257 * }
258 * } else {
259 * // Timeout - no events occurred
260 * handleUIIdleState();
261 * }
262 * }
263 *
264 * // Error handling task using bitset for multiple error conditions
265 * void errorMonitoringTask() {
266 * if (!systemEvents.isOpen()) return;
267 *
268 * std::bitset<8> errorMask;
269 * errorMask[EVT_ERROR] = 1; // Monitor error events
270 *
271 * while (true) {
272 * auto result = systemEvents.waitForBits(errorMask, portMAX_DELAY);
273 * if (result[EVT_ERROR]) {
274 * // Handle system error (log, reset, recovery)
275 * performErrorRecovery();
276 * }
277 * }
278 * }
279 *
280 * // Clean-up when shutting down (optional in embedded systems)
281 * void shutdownSystemEvents() {
282 * if (systemEvents.isOpen()) {
283 * systemEvents.close();
284 * }
285 * }
286 * @endcode
287 *
288 * @par Notes:
289 * - The `setBitsFromISR` helpers are safe to call from ISR context and will
290 * perform context switching if a higher priority task is woken.
291 * - Template variants provide compile-time type safety and avoid runtime errors.
292 * - Bitset variants offer flexibility for dynamic event combinations.
293 * - Always check `isOpen()` before using the event group.
294 * - Use appropriate timeouts to prevent task blocking in real-time systems.
