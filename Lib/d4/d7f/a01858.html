Lib::Helper::Convert::dBM

#Example: Using Lib::Helper::Convert::dBM utilities.

#Example: Using Lib::Helper::Convert::dBM utilities.Demonstrates converting dBm values to percentage representations using both the linear mapping and the digitized lookup table.

#include "Lib/Helper/Convert/dBM.hpp" #include <iostream>
int main() { using Lib::Helper::Convert::dBM;
int16_t lvl16 = -50;
uint8_t pct16 = dBM::toPercent(lvl16);
std::cout << "dBm " << lvl16 << " -> percent: " << static_cast<int>(pct16) << '\n';
int8_t lvl8 = -30;
uint8_t pct8 = dBM::toPercent(lvl8);
std::cout << "dBm " << static_cast<int>(lvl8) << " -> percent: " << static_cast<int>(pct8) << '\n';
// Use digitized lookup for small non-negative indices
int8_t idx = 10;
uint8_t digi = dBM::toDigiPercent(idx);
std::cout << "dBm index " << static_cast<int>(idx) << " -> digitized percent: " << static_cast<int>(digi) << '\n';
return 0;
}
