1/***************************************************************************************
2 * TimePeriod.hpp - Time period measurement using FreeRTOS ticks.
3 *
4 * This header defines the Lib::Helper::TimePeriod class used to measure elapsed time
5 * in FreeRTOS tick units. The class provides start/stop timing and a non-destructive
6 * read of the elapsed tick count using RTOS::TickCounter::get().
7 *
8 * The implementation is designed for embedded use: it stores only the start tick and
9 * a running flag, relies on unsigned arithmetic to handle tick counter wrap-around,
10 * and performs no dynamic allocation.
11 *
12 * @file TimePeriod.hpp
13 * @date 16 Jan 2026
14 ***************************************************************************************/
15#pragma once

20namespace Lib::Helper {

22 /** Measures time periods using FreeRTOS TickCounter.
23 * Enables starting and stopping a time measurement, returning the elapsed ticks.
24 */
25 class TimePeriod {
26 public:
27 /** Constructor of TimePeriod class. */
28 explicit TimePeriod() = default;
29
30 /** Destructor of TimePeriod class. */
31 ~TimePeriod() = default;
32
33 /** Starts the time measurement.
34 * @post running == true, startTick == RTOS::TickCounter::get()
35 */
36 void start() {
37 startTick = RTOS::TickCounter::get();
38 running = true;
39 }
40
41 /**
42 * Stops the time measurement and stores the elapsed ticks.
43 *
44 * @param[out] ticks Reference to receive the elapsed tick count.
45 * @pre The timer is running.
46 * @post The timer is stopped and `startTick` is cleared.
47 * @return true if the timer was active and `ticks` was set; false otherwise.
48 */
49 bool stop(uint32_t & ticks) {
50 if (!running) return false;
51 ticks = RTOS::TickCounter::get() - startTick; // unsigned arithmetic handles wrap-around
52 running = false;
53 startTick = 0U;
54 return true;
55 }
56
57 /** Returns true if a measurement is currently running. */
58 bool isRunning() const {
59 return running;
60 }
61
62 /**
63 * Returns the elapsed ticks since start. If the timer is not running, returns 0.
64 * This is a non-destructive read and does not stop the measurement.
65 */
66 uint32_t elapsed() const {
67 if (!running) return 0U;
68 return RTOS::TickCounter::get() - startTick;
69 }
70
71 /** Reset the timer state. After reset the timer is not running and startTick is zero. */
72 void reset() {
73 running = false;
74 startTick = 0U;
75 }
76
77 private:
78 uint32_t startTick{0U}; /*!< The start time of the measurement in ticks. */
79 bool running{false}; /*!< Indicates whether a measurement is currently active. */
80 };

82} // namespace Lib::Helper
