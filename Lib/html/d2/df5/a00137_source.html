136 virtual bool start(bool pauseAfterStart) {
137 assert(state == State::NO_INIT && "start() must only be called once, from the NO_INIT state");
138
139 auto & serviceManager = Service::Manager::instance();
140 serviceManager.add(&channelScanner);

143 serviceManager.add(&spectrumAnalyzer);

145 if (!module.open()) {
146 state = State::ERROR;
147 return false;

149 state = State::INIT;
150 if (!modemStatus.open()) {
151 state = State::ERROR;
152 return false;
153 }

155 RTOS::Semaphore semaphoreModemStatusHwReset;
156 if (!semaphoreModemStatusHwReset.open()) {
157 state = State::ERROR;
158 return false;
159 }

161 modemStatus.onHardwareReset = [&semaphoreModemStatusHwReset]() {
162 static_cast<void>(semaphoreModemStatusHwReset.give());
163 };
164
165 module.responseEvents.onReceivePacket16bit = [this](const Lib::Digi::Core::Frame0x81 & frame) {
166 handleReceivedPacked16BitsEvent(frame);
167 };
168
169 module.responseEvents.onTransmitStatus = [this](const Lib::Digi::Core::Frame0x89 & frame) {
170 handleTransmitStatusUpdate(frame);
171 };
172
173 module.reset();
174 isModulePresent = semaphoreModemStatusHwReset.take(HW_RESET_TIMEOUT);
175
176 modemStatus.onHardwareReset = nullptr;
177 semaphoreModemStatusHwReset.close();
178
179 if (!isModulePresent) {
180 state = State::ERROR;
181 return false;
182 }
183
184 RTOS::delay(POST_RESET_SETTLE_DELAY);
185 if (!queueRxMessage.open() || !queueTransmitDeliveryStatus.open()) {
186 assert(!"Message queue creation failed - insufficient resources");
187 state = State::ERROR;
188 return false;
189 }
190 parameters.read();
191
192 if (task.start([this]() { taskEntry(); }) == nullptr) {
193 state = State::ERROR;
194 return false;
195 }
196
197 state = State::RUNNING;
198 if (pauseAfterStart) pause();
199 return true;
200 }

303 void handleReceivedPacked16BitsEvent(const Lib::Digi::Core::Frame0x81 & frame) {
304 if (state != State::RUNNING) return;
305
306 const uint32_t rssi = std::min<uint32_t>(frame.rssi, MAX_RSSI);
307 connection.info.signalStrength = static_cast<uint8_t>(((MAX_RSSI - rssi) * 100) / MAX_RSSI);
308
309 diagnostic.receivedDataMessages++;
310 Core::Message msg(frame.rfData);
311 const bool queued = queueRxMessage.overwrite(msg);
312 assert(queued && "ApplicationBase: queueRxMessage overwrite failed - queue not open");
313 }

324 if (frame.frameId == messageFrameId) {
325 messageFrameId = (messageFrameId == MAX_MESSAGE_FRAME_ID) ? 1 : messageFrameId + 1;
326 }
327 const bool queued = queueTransmitDeliveryStatus.overwrite(frame.deliveryStatus);
328 assert(queued && "ApplicationBase: queueTransmitDeliveryStatus overwrite failed - queue not open");
329 }
