108 constexpr uint16_t BRNG_32V = 1U << 13U;

111 constexpr uint16_t SADC_12BIT_128SAMPLES = 15U << 3U;

115 operation = BRNG_32V | PGA_DIV_8_320MV | BADC_12BIT_128SAMPLES | SADC_12BIT_128SAMPLES | MODE_SANDB_CONTINUOUS;

139 [[nodiscard]] float readVoltage() noexcept {
140 uint16_t rawValue = 0U;
141 const bool isReadOk = readRegister(Register::BUS_VOLTAGE, rawValue);

236 [[nodiscard]] bool readRegister(Register reg, uint16_t & value) noexcept {
237 const auto regAddr = static_cast<uint8_t>(reg);
238
239 if (!i2cPort.masterTransmit(i2cAddr, std::span{&regAddr, 1U}, I2C_TIMEOUT)) {
240 assert(!"INA219: I2C transmission failure during register address selection");
241 return false;
242 }
243
244 std::array<uint8_t, 2U> buffer{};
245 if (!i2cPort.masterReceive(i2cAddr, buffer, I2C_TIMEOUT)) {
246 assert(!"INA219: I2C burst receive transaction contents failure");
247 return false;
248 }
249
250 value = static_cast<uint16_t>((buffer[0U] << 8U) | buffer[1U]);
251 return true;
252 }

261 [[nodiscard]] bool writeRegister(Register reg, uint16_t value) noexcept {
262 const std::array<uint8_t, 3U> packet{
263 static_cast<uint8_t>(reg),
264 static_cast<uint8_t>(value >> 8U),
265 static_cast<uint8_t>(value & 0xFFU),
266 };
267
268 if (!i2cPort.masterTransmit(i2cAddr, packet, I2C_TIMEOUT)) {
269 assert(!"INA219: I2C write transaction failure during register updates");
270 return false;
271 }
272
273 return true;
274 }
