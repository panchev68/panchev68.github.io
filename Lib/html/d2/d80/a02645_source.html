330 [[nodiscard]] bool writeInternal(uint32_t addr, const void * buffer, size_t size, uint8_t memAddSize) {
331 if (!isReady) {
332 assert(!"Driver: write requires an opened device");
333 return false;
334 }
335 if (!isAddressRangeValid(addr, size)) {
336 assert(!"Driver: write exceeds the EEPROM address range");
337 return false;
338 }
339
340 const auto * data = static_cast<const uint8_t *>(buffer);
341 uint32_t currentAddr = addr;
342 size_t remaining = size;
343
344 while (remaining > 0U) {
345 const size_t pageOffset = currentAddr % pageSize;
346 const size_t writeLen = std::min(remaining, static_cast<size_t>(pageSize) - pageOffset);
347
348 if (!i2cPort->send(deviceAddr, memAddSize, currentAddr, data, writeLen, timeoutValue())) {
349 assert(!"Driver: page write failed on the I2C bus");
350 return false;
351 }
352
353 RTOS::delay(WRITE_CYCLE_TIME);
354 currentAddr += writeLen;
355 data += writeLen;
356 remaining -= writeLen;
357 }
358 return true;
359 }
