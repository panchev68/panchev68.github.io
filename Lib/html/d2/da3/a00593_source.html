105 [[nodiscard]] double update(const Measurement & currentMeas) {
106 if (!isInitialized) [[unlikely]] {
107 previousMeasurement = currentMeas;
108 isInitialized = true;
109 estimatedMaxPower = 0.0;
110 voltageSetpoint = params.mppVoltageRef * START_VOLTAGE_FRACTION;
111 return voltageSetpoint;
112 }
113
114 const double temperatureDelta = currentMeas.temperature - previousMeasurement.temperature;
115 const double currentDelta = currentMeas.current - previousMeasurement.current;
116 const double voltageDelta = currentMeas.voltage - previousMeasurement.voltage;
117 const double currentMeasuredPower = currentMeas.voltage * currentMeas.current;
118
119 estimatedMaxPower = maxPowerEstimator.isBound() ? maxPowerEstimator(currentMeas.irradiance, temperatureDelta, estimatedMaxPower)
120 : estimateMaxPower(currentMeas.irradiance, temperatureDelta);
121
122 if (currentMeasuredPower >= (estimatedMaxPower - tolerance)) {
123 voltageSetpoint = currentDelta * mppReferenceImpedance;
124 } else if (std::fabs(currentDelta) <= CURRENT_DELTA_EPSILON) {
125 voltageSetpoint = params.mppVoltageRef * START_VOLTAGE_FRACTION;
126 } else if ((voltageDelta * currentDelta) > 0.0) {
127 voltageSetpoint = params.mppVoltageRef * START_VOLTAGE_FRACTION;
128 } else {
129 voltageSetpoint = -currentDelta * mppReferenceImpedance;
130 }
131
132 previousMeasurement = currentMeas;
133 return voltageSetpoint;
134 }

197 constexpr void initializeCoefficients() noexcept {
198 tolerance = params.toleranceFactor * params.maxPowerRef;
199
200 if (params.referenceIrradiance > 0.0) {
201 irradianceCoefficient = params.maxPowerRef / params.referenceIrradiance;
202
203 if (params.temperatureCoefficient != 0.0) {
204 efficiencyCoefficient = 1.0 / (params.referenceIrradiance * params.temperatureCoefficient);
205 }
206 }
207 if (params.mppCurrentRef > 0.0) {
208 mppReferenceImpedance = params.mppVoltageRef / params.mppCurrentRef;
209 }
210 }
211
219 [[nodiscard]] constexpr double estimateMaxPower(double irradiance, double temperatureDelta) const noexcept {
220 return irradiance * irradianceCoefficient * (1.0 - (TEMPERATURE_DERATING * temperatureDelta));
221 }
