200 inline bool MFRC522::stopTask() {
201 if (!task.isCreated()) {
202 return false;
203 }
204
205 stopRequested.store(true);
206
207 for (std::chrono::milliseconds waited{0}; task.isCreated(); waited += STOP_POLL_DELAY) {
208 if (waited >= STOP_TIMEOUT) {
209 assert(!"MFRC522: the polling task did not stop within the timeout");
210 return false;
211 }
212 RTOS::delay(STOP_POLL_DELAY);
213 }
214
215 chipset.setAntenna(false);
216 return true;
217 }

219 inline bool MFRC522::stop() {

223 inline bool MFRC522::start() {
224 constexpr std::chrono::milliseconds READY_POLL_DELAY{10};
225 constexpr std::chrono::milliseconds READY_TIMEOUT{100};
226
227 if (task.isCreated()) {
228 assert(!"MFRC522: start() called while the polling task is already running");
229 return false;
230 }
231 stopRequested.store(false);
232
233 if (!io.initialize()) {
234 assert(!"MFRC522: failed to open SPI transport");
235 return false;
236 }
237 io.hardwareReset();
238
239 const uint8_t version = chipset.readRegister(Chipset::Register::VERSION);
240
241 if (version == VERSION_ALL_ZEROES || version == VERSION_ALL_ONES) {
242 assert(!"MFRC522: communication failed; check wiring and SPI power");
243 return false;
244 }
245
246 if (!chipset.softReset()) {
247 assert(!"MFRC522: soft reset failed; check hardware and power supply");
248 return false;
249 }
250
251 chipset.toIdleState();
252 for (std::chrono::milliseconds waited{0}; !chipset.isReady(); waited += READY_POLL_DELAY) {
253 if (waited >= READY_TIMEOUT) {
254 assert(!"MFRC522: device did not become ready within timeout");
255 return false;
256 }
257 RTOS::delay(READY_POLL_DELAY);
258 }
259
260 chipset.initTimer();
261 chipset.applyInitialGain();
262 chipset.setAntenna(true);
263
264 return task.start([this]() { taskEntry(); });
265 }

267 inline void MFRC522::taskEntry() {
268 constexpr std::chrono::milliseconds POLL_DELAY_WITH_CARD{500};
269 constexpr std::chrono::milliseconds POLL_DELAY_NO_CARD{100};
270
271 while (!stopRequested.load()) {
272 if (!chipset.isCardPresent()) {
273 RTOS::delay(POLL_DELAY_NO_CARD);
274 continue;
275 }
276
277 std::array<uint8_t, Chipset::UID_LENGTH> uid{};
278 if (chipset.readUID(uid) && onCardDetected.isBound()) {
279 onCardDetected(std::span<const uint8_t>(uid));
280 }
281
282 chipset.clearCardDetection();
283 card.halt();
284 RTOS::delay(POLL_DELAY_WITH_CARD);
285 }
