90 [[nodiscard]] static size_t build(const Frame & frame, std::span<uint8_t> output) {
91 if (frame.payload.size() > MAX_PAYLOAD_SIZE) {
92 return 0U;
93 }
94
95 std::array<uint8_t, MAX_RAW_FRAME_SIZE> rawFrame{};
96 const size_t payloadSize = frame.payload.size();
97
98 rawFrame[START_OFFSET] = START_DELIMITER;
99 rawFrame[LENGTH_OFFSET] = static_cast<uint8_t>(payloadSize);
100 rawFrame[COMMAND_OFFSET] = frame.command;
101 std::copy_n(frame.payload.begin(), payloadSize, rawFrame.begin() + HEADER_SIZE);
102
103 const size_t checkedLength = HEADER_SIZE + payloadSize;
104 const uint16_t crc = calculateChecksum(std::span<const uint8_t>(rawFrame).first(checkedLength));
105 rawFrame[checkedLength] = static_cast<uint8_t>(crc & BYTE_MASK);
106 rawFrame[checkedLength + 1U] = static_cast<uint8_t>(crc >> BITS_PER_BYTE);
107
108 const size_t rawSize = checkedLength + CRC_SIZE;
109 if (output.empty()) {
110 return 0U;
111 }
112
113 // The leading delimiter is what marks the start of a frame, so it is the one byte that is never escaped.
114 output[0] = START_DELIMITER;
115 size_t written = 1U;
116
117 for (size_t index = 1U; index < rawSize; ++index) {
118 uint8_t value = rawFrame[index];
119 const size_t needed = isEsc(value) ? 2U : 1U;
120
121 if ((written + needed) > output.size()) {
122 return 0U;
123 }
124 if (isEsc(value)) {
125 output[written++] = ESC_SUBST;
126 value ^= PRECODE_MASK;
127 }
128 output[written++] = value;
129 }
130 return written;
131 }

144 [[nodiscard]] static Status parse(std::span<const uint8_t> input, std::span<uint8_t> workspace, Frame & outputFrame) {
145 if (input.size() < (HEADER_SIZE + CRC_SIZE) || input.size() > MAX_FRAME_SIZE || input[0] != START_DELIMITER || workspace.empty()) {
146 return Status::ERROR;
147 }
148
149 workspace[0] = START_DELIMITER;
150 size_t decodedSize = 1U;
151 bool isEscape = false;
152
153 for (size_t index = 1U; index < input.size(); ++index) {
154 uint8_t value = input[index];
155
156 if (isEscape) {
157 value ^= PRECODE_MASK;
158 isEscape = false;
159 } else if (value == ESC_SUBST) {
160 isEscape = true;
161 continue;
162 }
163 if (decodedSize >= workspace.size()) {
164 return Status::ERROR;
165 }
166 workspace[decodedSize++] = value;
167 }
168
169 // A frame that ends on an escape marker was truncated in transit; its last byte never arrived.
170 if (isEscape || decodedSize < (HEADER_SIZE + CRC_SIZE)) {
171 return Status::ERROR;
172 }
173
174 const size_t payloadLength = workspace[LENGTH_OFFSET];
175 const size_t checkedLength = HEADER_SIZE + payloadLength;
176
177 if ((checkedLength + CRC_SIZE) != decodedSize) {
178 return Status::ERROR;
179 }
180
181 const std::span<const uint8_t> decoded = std::span<const uint8_t>(workspace).first(decodedSize);
182 const uint16_t crcCalculated = calculateChecksum(decoded.first(checkedLength));
183 const uint16_t crcReceived = static_cast<uint16_t>(decoded[checkedLength]) |
184 static_cast<uint16_t>(static_cast<uint16_t>(decoded[checkedLength + 1U]) << BITS_PER_BYTE);
185
186 if (crcCalculated != crcReceived) {
187 return Status::ERROR;
188 }
189
190 outputFrame.command = decoded[COMMAND_OFFSET];
191 outputFrame.payload = decoded.subspan(HEADER_SIZE, payloadLength);
192 return Status::OK;
193 }
