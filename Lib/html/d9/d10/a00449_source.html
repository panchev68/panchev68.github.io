257 [[nodiscard]] static constexpr uint32_t pack(const DateTime & dateTime) noexcept {
258 assert(dateTime.date.year >= Date::FIRST_YEAR && "Date::year is before Date::FIRST_YEAR");
259 const uint32_t yearDelta = (dateTime.date.year >= Date::FIRST_YEAR) ? static_cast<uint32_t>(dateTime.date.year - Date::FIRST_YEAR) : 0U;
260 assert(yearDelta <= YEAR_DELTA_MASK && "Date::year exceeds the packable range");
261 return ((dateTime.time.seconds & SECONDS_MASK) << SECONDS_SHIFT) | ((dateTime.time.minutes & MINUTES_MASK) << MINUTES_SHIFT) |
262 ((dateTime.time.hours & HOURS_MASK) << HOURS_SHIFT) | ((dateTime.date.day & DAY_MASK) << DAY_SHIFT) | ((dateTime.date.month & MONTH_MASK) << MONTH_SHIFT) |
263 ((yearDelta & YEAR_DELTA_MASK) << YEAR_DELTA_SHIFT);
264 }

272 [[nodiscard]] static constexpr DateTime unpack(uint32_t packed) noexcept {
273 return DateTime(Date(static_cast<uint16_t>(Date::FIRST_YEAR + ((packed >> YEAR_DELTA_SHIFT) & YEAR_DELTA_MASK)),

278 static_cast<uint8_t>((packed >> SECONDS_SHIFT) & SECONDS_MASK)));
