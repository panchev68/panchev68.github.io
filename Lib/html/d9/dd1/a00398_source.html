92 // registration still leaves the remaining ones attempted.
93 [[maybe_unused]] bool isCallbacksRegistered = (HAL_OSPI_RegisterCallback(handle, HAL_OSPI_CMD_CPLT_CB_ID, cmdCompleteISR) == HAL_OK);
