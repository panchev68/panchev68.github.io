166 [[nodiscard]] bool startTimeoutInterrupt(uint32_t timeoutMs) {
167 if (!isInitialized) {
168 assert(!"Cannot start timeout; ensure peripheral is initialized before use");
169 return false;
170 }
171
172 uint32_t clockFrequencyHz;
173 switch (__HAL_RCC_GET_LPTIM1_SOURCE()) {
174 case RCC_LPTIM1CLKSOURCE_LSE:
175 clockFrequencyHz = LSE_CLOCK_FREQUENCY_HZ;
176 break;
177 case RCC_LPTIM1CLKSOURCE_LSI:
178 clockFrequencyHz = LSI_CLOCK_FREQUENCY_HZ;
179 break;
180 case RCC_LPTIM1CLKSOURCE_MSIK:
181 clockFrequencyHz = MSIK_CLOCK_FREQUENCY_HZ;
182 break;
183 default:
184 assert(!"Unrecognized LPTIM1 clock source; update LPTimer's clock frequency table to match the "
185 "current CubeMX Clock Configuration");
186 clockFrequencyHz = LSI_CLOCK_FREQUENCY_HZ;
187 break;
188 }

190 static constexpr uint32_t PRESCALER_DIVIDERS[] = {1U, 2U, 4U, 8U, 16U, 32U, 64U, 128U};
197 static constexpr uint32_t PRESCALER_MACROS[] = {LPTIM_PRESCALER_DIV1,
198 LPTIM_PRESCALER_DIV2,
199 LPTIM_PRESCALER_DIV4,
200 LPTIM_PRESCALER_DIV8,
201 LPTIM_PRESCALER_DIV16,
202 LPTIM_PRESCALER_DIV32,
203 LPTIM_PRESCALER_DIV64,
204 LPTIM_PRESCALER_DIV128};
205 static constexpr uint32_t PRESCALER_COUNT = sizeof(PRESCALER_DIVIDERS) / sizeof(PRESCALER_DIVIDERS[0]);
206
207 uint32_t prescalerIndex = 0U;
208 uint64_t rawTicks = 0U;
209 for (; prescalerIndex < PRESCALER_COUNT; ++prescalerIndex) {
210 rawTicks = (static_cast<uint64_t>(timeoutMs) * clockFrequencyHz) / (1000ULL * PRESCALER_DIVIDERS[prescalerIndex]);
211 if (rawTicks <= MAX_TIMER_TICKS) {
212 break;
213 }
214 }
215 if (prescalerIndex >= PRESCALER_COUNT) {
216 prescalerIndex = PRESCALER_COUNT - 1U;
217 }
218
219 uint32_t ticks = (rawTicks > MAX_TIMER_TICKS) ? MAX_TIMER_TICKS : static_cast<uint32_t>(rawTicks);
220 ticks = (ticks < MIN_TIMER_TICKS) ? MIN_TIMER_TICKS : ticks;
221
222 activeClockFrequencyHz = clockFrequencyHz / PRESCALER_DIVIDERS[prescalerIndex];
223
224 handle->Init.Clock.Prescaler = PRESCALER_MACROS[prescalerIndex];
225 handle->Init.Period = ticks;
226 if (HAL_LPTIM_Init(handle) != HAL_OK) {
227 assert(!"Failed to reconfigure LPTIM period; check handle state and configuration");
228 return false;
229 }
230
231 LPTIM_OC_ConfigTypeDef ocConfig{};
232 ocConfig.Pulse = ticks;
233 ocConfig.OCPolarity = LPTIM_OCPOLARITY_HIGH;
234 if (HAL_LPTIM_OC_ConfigChannel(handle, &ocConfig, channel) != HAL_OK) {
235 assert(!"Failed to configure LPTIM channel; check handle state and configuration");
236 return false;
237 }
238
239 eventGroup.clearAllEvents();
240
241 handle->State = HAL_LPTIM_STATE_READY;
242 handle->ChannelState[0] = HAL_LPTIM_CHANNEL_STATE_READY;
243
244 __HAL_LPTIM_CLEAR_FLAG(handle, LPTIM_FLAG_ARRM);
245
246 handle->Instance->DIER = 0U;
247
248 if (HAL_LPTIM_SetOnce_Start_IT(handle, channel) != HAL_OK) {
249 assert(!"Failed to start timeout interrupt; check handle state and configuration");
250 return false;
251 }
252
253 __HAL_LPTIM_DISABLE_IT(handle, LPTIM_IT_CMP1OK | LPTIM_IT_CC1 | LPTIM_IT_ARROK | LPTIM_IT_UPDATE);
254
255 __HAL_LPTIM_CLEAR_FLAG(handle, LPTIM_FLAG_CMP1OK | LPTIM_FLAG_ARROK | LPTIM_FLAG_UPDATE);
256 HAL_NVIC_ClearPendingIRQ(LPTIM1_IRQn);
257
258 return true;
259 }

276 [[nodiscard]] uint32_t getElapsedRtosTicks(uint32_t armedTimeoutMs) const {
277 if (handle == nullptr || activeClockFrequencyHz == 0U) {

288 return (rtosTicks64 > UINT32_MAX) ? UINT32_MAX : static_cast<uint32_t>(rtosTicks64);
