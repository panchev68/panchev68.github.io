208 [[nodiscard]] bool write(uint32_t addr, const void * data, size_t size) {
209 if (!isChipPresent || data == nullptr || size == 0) return false;
210
211 const Info & chipInfo = getChipInfo();
212 const size_t pageSize = chipInfo.pageSize ? chipInfo.pageSize : PAGE_SIZE;
213
214 const uint8_t * bytes = static_cast<const uint8_t *>(data);
215 while (size > 0) {
216 const size_t pageOffset = addr & (pageSize - 1);
217 const size_t writeLength = std::min(pageSize - pageOffset, size);
218
219 if (!writeEnable()) return false;
220
221 OSPI_RegularCmdTypeDef cmd{};
222 cmd.InstructionMode = HAL_OSPI_INSTRUCTION_1_LINE;
223 cmd.InstructionSize = HAL_OSPI_INSTRUCTION_8_BITS;
224 cmd.Instruction = static_cast<uint32_t>(Register::Command::QUAD_PAGE_PROGRAM);
225 cmd.AddressMode = HAL_OSPI_ADDRESS_1_LINE;
226 cmd.AddressSize = HAL_OSPI_ADDRESS_24_BITS;
227 cmd.Address = addr;
228 cmd.DataMode = HAL_OSPI_DATA_4_LINES;
229 cmd.NbData = static_cast<uint32_t>(writeLength);
230
231 if (!ospi.command(cmd, OSPI_TIMEOUT) || !ospi.transmit(bytes, OSPI_TIMEOUT)) return false;
232 if (!waitForWriteComplete(MAX_WRITE_TIME)) return false;
233
234 addr += static_cast<uint32_t>(writeLength);
235 bytes += writeLength;
236 size -= writeLength;
237 }
