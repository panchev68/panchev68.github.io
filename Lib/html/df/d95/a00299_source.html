242 Socket::Error errorCode;
243 size_t messageSize;
244 DataHandle<> receivedDataHandle;
245 DataHandle<> outgoingDataHandle;
246
247 setConnectionState(ConnectionState::INIT);
248 initCallback();
249
250 while (true) {
251 FreeRTOS::delay(1);
252 while (!readLinkState()) {
253 if (getConnectionState() == ConnectionState::CONNECTED) {
254 setConnectionState(ConnectionState::DISCONNECTED);
255 if (sockets[0].close() != Socket::Error::OK)
256 assert(0);
257 }
258 FreeRTOS::delay(100);
259 }
260 if (getConnectionState() != ConnectionState::CONNECTED) {
261 if (sockets[0].open(Socket::Protocol::TCP, remote.port, 0) == Socket::Error::OK) {
262 if (sockets[0].connect(remote) == Socket::Error::OK)
263 setConnectionState(ConnectionState::CONNECTED);
264 else {
265 if (sockets[0].close() != Socket::Error::OK)
266 assert(0);
267 FreeRTOS::delay(100);
268 }
269 } else
270 assert(0);
271 } else {
272 if ((messageSize = sockets[0].getReceivedSize()) > 0) {
273 updateRxLedCallback(true);
274 diagnostic.rxMessageCounter++;
275 receivedDataHandle(new uint8_t[messageSize], messageSize);
276 errorCode = sockets[0].receive(receivedDataHandle, messageSize);
277 if (errorCode == Socket::Error::OK)
278 handleIncomingMessage(receivedDataHandle);
279 else if (errorCode != Socket::Error::ERR_DATALEN) {
280 if (sockets[0].close() != Socket::Error::OK)
281 assert(0);
282 }
283 delete receivedDataHandle.data();
284 }
285 if (handleOutgoingMessage(outgoingDataHandle)) {
286 if (sockets[0].send(outgoingDataHandle, messageSize) == Socket::Error::OK) {
287 updateTxLedCallback(true);
288 diagnostic.txMessageCounter++;
289 } else {
290 if (sockets[0].close() != Socket::Error::OK)
291 assert(0);
292 setConnectionState(ConnectionState::DISCONNECTED);
293 }
294 }
295 }
296 taskYIELD();
