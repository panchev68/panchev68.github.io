196 bool processSensorData(uint16_t * data, size_t length) {
197 auto calcChecksum = [](const uint8_t * data) -> uint8_t {
198 assert(data);
199 return (data[0] + data[1] + data[2] + data[3]) & 0xfe;
200 };
201 auto getRaw = [](const uint8_t * data) -> int16_t {
202 assert(data);
203 return data[0] << 8 | data[1];
204 };
205
206 uint8_t sensorData[5] = {0};
207 uint32_t byteIndex = 0;
208 uint32_t bitPosition = 0;
209 auto lastTimestamp = data[0];
210
211 for (size_t edgeIndex = 0; edgeIndex < length; ++edgeIndex) {
212 auto previousTimestamp = lastTimestamp;
213 lastTimestamp = data[edgeIndex];
214 data[edgeIndex] -= previousTimestamp;
215
216 if (edgeIndex >= 3) {
217 size_t bitIndex = edgeIndex - 3;
218 if (data[edgeIndex] > TIMER_THRESHOLD) {
219 byteIndex = bitIndex / 8;
220 bitPosition = bitIndex % 8;
221 sensorData[byteIndex] |= (0x80U >> bitPosition);
222 }
223 }
224 }
225 const uint8_t & frameChecksum = sensorData[4];
226 auto checksum = calcChecksum(sensorData);
227 diagnostic.frameChecksum = frameChecksum;
228 diagnostic.calculatedChecksum = checksum;
229
230 if (frameChecksum != checksum) {
231 diagnostic.invalidFrameChecksum++;
232 return false;
233 }
234
235 auto rawTemperature = getRaw(&sensorData[2]);
236 auto rawHumidity = getRaw(&sensorData[0]);
237
238 rawTemperature = ((rawTemperature & 0x8000) ? -(rawTemperature & 0x7fff) : rawTemperature);
239
240 if ((rawTemperature <= MIN_TEMPERATURE) || (rawTemperature >= MAX_TEMPERATURE)) {
241 diagnostic.invalidTemperature++;
242 return false;
243 }
244
245 if ((rawHumidity <= MIN_HUMIDITY) || (rawHumidity >= MAX_HUMIDITY)) {
246 diagnostic.invalidHumidity++;
247 return false;
248 }
249
250 this->temperature = static_cast<float>(rawTemperature) / 10.0F;
251 this->humidity = static_cast<float>(rawHumidity) / 10.0F;
252 isNewData = true;
253 return true;
254 }
