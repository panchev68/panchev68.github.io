114 [[nodiscard]] std::optional<float> readTemperatureCelsius() {
115 if (!reset()) return std::nullopt;
116 writeByte(static_cast<uint8_t>(Command::SKIP_ROM));
117 writeByte(static_cast<uint8_t>(Command::CONVERT_T));
118
119 RTOS::delay(CONVERSION_TIME);
120
121 if (!reset()) return std::nullopt;
122 writeByte(static_cast<uint8_t>(Command::SKIP_ROM));
123 writeByte(static_cast<uint8_t>(Command::READ_SCRATCHPAD));
124
125 std::array<uint8_t, SCRATCHPAD_SIZE> scratchpad{};
126 for (uint8_t & byte : scratchpad) {
127 byte = readByte();
128 }
129
130 const std::span<const uint8_t> payload(scratchpad.data(), SCRATCHPAD_SIZE - 1);
131 if (static_cast<uint8_t>(Lib::Helper::crc8(payload)) != scratchpad[SCRATCHPAD_SIZE - 1]) return std::nullopt;
132
133 const int16_t rawTemperature = static_cast<int16_t>((static_cast<uint16_t>(scratchpad[1]) << 8) | scratchpad[0]);
134 return static_cast<float>(rawTemperature) / TEMPERATURE_UNITS_PER_DEGREE;
135 }
