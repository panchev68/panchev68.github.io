192 [[nodiscard]] static constexpr int32_t computeScaleMultiplier(int32_t outputRange, int32_t rawMin, int32_t rawMax) {
193 const int32_t rawSpan = std::max(rawMax - rawMin, 1);
194 return static_cast<int32_t>((static_cast<int64_t>(outputRange) << SCALE_FACTOR) / rawSpan);
195 }

203 [[nodiscard]] uint16_t transfer12(uint8_t command) {
204 uint16_t data = 0;
205 interface.nss.write(false);
206
207 for (int i = 7; i >= 0; --i) {
208 interface.mosi.write((command >> i) & 1);
209 interface.sck.write(false);
210 __NOP();
211 interface.sck.write(true);
212 __NOP();
213 }
214
215 for (int i = 0; i < 12; ++i) {
216 interface.sck.write(false);
217 __NOP();
218 data = (data << 1) | (interface.miso.read() ? 1 : 0);
219 interface.sck.write(true);
220 __NOP();
221 }
222
223 interface.nss.write(true);
224 return data;
225 }

233 void taskEntry() {
234 interface.nirq.bindCallback([this] { task.notify.fireEvent(TOUCH_NOTIFY_BIT); });
235
236 while (true) {
237 static_cast<void>(task.notify.waitForBits(TOUCH_NOTIFY_BIT, portMAX_DELAY));
238 RTOS::Task::delay(driverParameters.debounceDuration);
239
240 integratorX.resetWindow();
241 integratorY.resetWindow();
242
243 while (true) {
244 if (interface.nirq.isHigh()) [[unlikely]] {
245 RTOS::Task::delay(STABILITY_DELAY_TICKS);
246 if (interface.nirq.isHigh()) {
247 break;
248 }
249 }
250
251 const uint16_t sampleRawX = transfer12(CMD_READ_X);
252 const uint16_t sampleRawY = transfer12(CMD_READ_Y);
253
254 if (integratorX.isWindowFull()) [[likely]] {
255 const int32_t rawX = std::clamp<int32_t>(integratorX.processValue(sampleRawX), driverParameters.xMin, driverParameters.xMax);
256 const int32_t rawY = std::clamp<int32_t>(integratorY.processValue(sampleRawY), driverParameters.yMin, driverParameters.yMax);
257
258 const int32_t mappedX = ((rawX - driverParameters.xMin) * scaleMultiplierX) >> SCALE_FACTOR;
259 const int32_t mappedY = ((rawY - driverParameters.yMin) * scaleMultiplierY) >> SCALE_FACTOR;
260
261 {
262 RTOS::MutexGuard guard(mutex);
263 switch (currentRotation) {
264 case Rotation::FLIP_X:
265 position.x = driverParameters.width - mappedX;
266 position.y = mappedY;
267 break;
268
269 case Rotation::FLIP_XY:
270 position.x = driverParameters.width - mappedX;
271 position.y = driverParameters.height - mappedY;
272 break;
273
274 case Rotation::FLIP_Y:
275 position.x = mappedX;
276 position.y = driverParameters.height - mappedY;
277 break;
278
279 case Rotation::DEFAULT:
280 default:
281 position.x = mappedX;
282 position.y = mappedY;
283 break;
284 }
285 isPressed = true;
286 }
287 } else [[unlikely]] {
288 integratorX.addValue(sampleRawX);
289 integratorY.addValue(sampleRawY);
290 }
291 RTOS::Task::delay(driverParameters.debounceDuration);
292 }
293
294 task.notify.clearState();
295 {
296 RTOS::MutexGuard guard(mutex);
297 isPressed = false;
298 }
299 }
