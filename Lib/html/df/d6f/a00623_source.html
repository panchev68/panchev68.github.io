205 /*
206 * These fields are shared between task context and the idle hook, so they are atomic rather than volatile:
207 * 'volatile' orders nothing and makes no read-modify-write indivisible, it only stops the compiler from caching
208 * the value. Every type here is lock-free on Cortex-M, so an access still compiles to a plain load or store
209 * plus the barriers the ordering asks for.

231 [[nodiscard]] static bool registerCallback(Registry & registry, std::atomic<uint32_t> & count, PeripheralCallback callback) noexcept {
232 if (callback == nullptr) [[unlikely]] {
233 assert(!"PowerManager: cannot register a null callback");
234 return false;
235 }
236
237 // FIX: two concurrent registrations could claim the same slot; the mask-saving guard works pre-scheduler.
238 const Kernel::CriticalSectionFromISR guard;
239
240 const uint32_t slot = count.load(std::memory_order_relaxed);
241 if (slot >= MAX_CALLBACKS) [[unlikely]] {
242 assert(!"PowerManager: callback registry is full");
243 return false;
244 }
245
246 registry[slot] = callback;
247 count.store(slot + 1U, std::memory_order_release);
248 return true;
249 }

257 static void invokeCallbacks(const Registry & registry, const std::atomic<uint32_t> & count) noexcept {
258 // Acquire: pairs with the release store in registerCallback() so every counted slot is visible here.
259 const uint32_t published = count.load(std::memory_order_acquire);
260
261 for (uint32_t index = 0U; index < published; ++index) {
262 if (registry[index] != nullptr) {
263 registry[index]();
264 }
265 }

279 void handlePortSuppressTicksAndSleep(uint32_t expectedIdleTime) noexcept {
280 if (expectedIdleTime < MIN_TICKS_FOR_SLEEP_MODE) [[unlikely]] {
281 return;
282 }
283
284 __disable_irq();
285 __DSB();
286 __ISB();
287
288 if (eTaskConfirmSleepModeStatus() == eAbortSleep) [[unlikely]] {
289 __enable_irq();
290 return;
291 }
292
293 if (canEnterDeepSleep() && enterToSleep(expectedIdleTime)) {
294 exitFromSleep();
295 } else {
296 enterLightSleep();
297 }
298
299 // FIX: unmasked only once post-sleep callbacks ran, so wake-up ISRs never see unrestored peripherals.
300 __enable_irq();
301 __ISB();
302 }

313 [[nodiscard]] bool enterToSleep(uint32_t expectedIdleTime) noexcept {
314 Lib::HAL::LPTimer * const device = timer.load(std::memory_order_relaxed);
315
316 if (!isInitialized.load(std::memory_order_acquire) || device == nullptr) [[unlikely]] {
317 assert(!"PowerManager::enterToSleep() called before successful setup()");
318 return false;
319 }
320
321 if (expectedIdleTime <= MIN_TICKS_FOR_STOP_MODE || !isDeepSleepAllowed.load(std::memory_order_relaxed)) {
322 return false;
323 }
324
325 const uint32_t clampedIdleTime = std::min(expectedIdleTime, MAX_IDLE_TIME_BEFORE_OVERFLOW);
326 // Truncating keeps the wake-up at or before the next unblock time, never after it.
327 const uint32_t timeoutMs = (clampedIdleTime * MILLISECONDS_PER_SECOND) / configTICK_RATE_HZ;
328
329 HAL_SuspendTick();
330 if (!device->startTimeoutInterrupt(timeoutMs)) [[unlikely]] {
331 // FIX: the HAL time base used to stay suspended when arming the LPTIM failed.
332 HAL_ResumeTick();
333 return false;
334 }
335
336 invokeCallbacks(preSleepRegistry, preSleepCallbackCount);
337
338 __HAL_RCC_PWR_CLK_ENABLE();
339 HAL_PWREx_EnterSTOP1Mode(PWR_STOPENTRY_WFI);
340
341 const uint32_t elapsedRtosTicks = std::min(device->getElapsedRtosTicks(timeoutMs), clampedIdleTime);
342 if (elapsedRtosTicks > 0U) {
343 vTaskStepTick(elapsedRtosTicks);
344 }
345
346 SystemClock_Config();
347 static_cast<void>(HAL_InitTick(TICK_INT_PRIORITY));
348 return true;
349 }
