155 void receiveTask() {
156 size_t bufferIndex = 0;
157 uint8_t value = 0;
158 for (;;) {
159 if (rxPort->receive(&value, sizeof(uint8_t), 1) == sizeof(uint8_t)) {
160 if (bufferIndex < receiveHandler.size()) receiveHandler[bufferIndex++] = value;
161 else bufferIndex = 0;
162 } else if (bufferIndex > 0) {
163 handlePostReceivedEvent(receiveHandler.data(), bufferIndex);
164 bufferIndex = 0;
165 }
166 }
167 }

169 void transmitTask() {
170 for (;;) {
171 auto length = messageBuffer.receive(transmitData.data(), transmitData.size(), portMAX_DELAY);
172 if (!txPort->send(transmitData.data(), length, 100)) assert(false && "UART transmit failed");
173 handlePostTransmitEvent();
