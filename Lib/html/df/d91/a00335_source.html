119 [[nodiscard]] bool registerDevice(uint16_t pinNumber, ExtInterrupt * device) noexcept {
120 const int index = getPinIndex(pinNumber);
121 if (index < 0 || index >= static_cast<int>(MAX_PINS)) {
122 assert(!"Invalid pin index");
123 return false;
124 }
125
126 RTOS::Kernel::CriticalSection guard;
127 if (registry[index] != nullptr) {
128 assert(!"Pin already registered");
129 return false;
130 }
131 registry[index] = device;
132
133 // Lazy-initialize the single, shared worker thread upon first device registration
134 if (!isTaskCreated) {
135 task.setup({"EXTI", WORKER_STACK_SIZE, RTOS::Task::Priority::HIGH});
136 if (task.start([this] { taskEntry(); }) == nullptr) {
137 assert(!"Failed to start EXTI worker task");
138 registry[index] = nullptr; // Roll back registration on failure
139 return false;
140 }
141 isTaskCreated = true;
142 }
143 return true;
144 }

219 void taskEntry() {
220 while (true) {
221 // Clear nothing on entry (catches events fired before the wait began), clear every bit on exit so
222 // the next iteration starts from a clean notification value.
223 const uint32_t activeBits = static_cast<uint32_t>(task.notify.wait(0U, 0xFFFFFFFFU, portMAX_DELAY).to_ulong());
224
225 // Directly target only the active bits without doing standard linear registry sweeps
226 uint32_t remainingBits = activeBits;
227 while (remainingBits != 0U) {
228 const uint32_t currentBit = remainingBits & (~remainingBits + 1U); // Isolate lowest set bit
229 const int index = getPinIndex(static_cast<uint16_t>(currentBit));
230
231 if (index >= 0) {
232 if (auto * device = registry[index]) {
233 if (device->userCallback) {
234 device->userCallback();
235 }
236 }
237 }
238 remainingBits &= ~currentBit; // Clear processed bit
239 }
240 }
