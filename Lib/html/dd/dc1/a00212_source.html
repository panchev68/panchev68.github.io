150 void handleReceivedData(const Frame0x88 & frame) noexcept {
151 assert(frame.frameId == FRAME_ID && "Response frame ID mismatch - expected 0x08");
152 if (frame.frameId == FRAME_ID && sendFrame.atCommand == frame.atCommand) {
153 if (frame.data.size() >= responseMessage.DATA_SIZE) [[unlikely]] {
154 assert(!"Response data exceeds message buffer");
155 return;

158 std::copy(frame.data.begin(), frame.data.end(), responseMessage.data.begin());
159 if (!channel.enqueueResponse(responseMessage, SEND_TIMEOUT)) assert(!"Failed to queue local AT command response - queue full or timeout");
