555 [[nodiscard]] Handle start() noexcept {

564 auto taskEntry = [](void * ownerPtr) {
565 auto * t = static_cast<Task *>(ownerPtr);
566 if (t != nullptr) {
567 Function localFunction = t->function;
568
569 if (localFunction) {
570 localFunction();
571 }
572
573 t->handle = nullptr;
574 t->function = Function();
575 }
576 vTaskDelete(nullptr);

580 Handle taskHandle = nullptr;
581
582 // Suspend the scheduler across creation so the new task cannot start running (and read 'handle') before it
583 // is assigned below.
584 vTaskSuspendAll();
585 const auto res = xTaskCreate(taskEntry, config.name.data(), static_cast<uint16_t>(stack), this, static_cast<UBaseType_t>(config.priority), &taskHandle);
