246 [[nodiscard]] static constexpr size_t slotOf(size_t pos) noexcept {
247 if constexpr (IS_POWER_OF_TWO) {
248 return pos & (T_SIZE - 1U);
249 } else {
250 return (pos >= T_SIZE) ? (pos - T_SIZE) : pos;

260 [[nodiscard]] static constexpr size_t next(size_t pos) noexcept {
261 if constexpr (IS_POWER_OF_TWO) {
262 return pos + 1U; // Unsigned wrap at 2^32 is a multiple of T_SIZE, so the mask stays consistent.
263 } else {
264 return (pos + 1U == INDEX_RANGE) ? 0U : (pos + 1U);
