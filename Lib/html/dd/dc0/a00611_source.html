77 [[nodiscard]] static void * allocate(size_t size, Status * status = nullptr) noexcept {
78 assert(!Kernel::inISR() && "MemoryManager::allocate() must be called from task context");
79 if (!isValidSize(size)) [[unlikely]] {
80 setStatus(status, (size == 0U) ? Status::INVALID_SIZE : Status::SIZE_TOO_LARGE);
81 recordFailure();
82 return nullptr;
83 }
84 void * const raw = pvPortMalloc(size + HEADER_SIZE);
85 if (raw == nullptr) [[unlikely]] {
86 assert(!"MemoryManager: heap allocation failed - out of memory");
87 setStatus(status, Status::OUT_OF_MEMORY);
88 recordFailure();
89 return nullptr;

303 static void recordAlloc(size_t size) noexcept {
304 /*
305 * FIX: counters are shared by every allocating task, so the read-modify-write needs a critical section. The
306 * mask-saving variant is used because it is valid in any context and, unlike the nesting task-level pair,
307 * does not leave interrupts masked when called before the scheduler starts.
308 */
309 const Kernel::CriticalSectionFromISR guard;
310 ++diagnostic.allocated;
311 ++diagnostic.active;
312 diagnostic.currentUsage += size;
313 diagnostic.peakUsage = std::max(diagnostic.peakUsage, diagnostic.currentUsage);
314 }
315
321 static void recordDealloc(size_t size) noexcept {
322 const Kernel::CriticalSectionFromISR guard;
323 diagnostic.currentUsage = (diagnostic.currentUsage >= size) ? (diagnostic.currentUsage - size) : 0U;
324 ++diagnostic.deallocated;
325 if (diagnostic.active > 0U) {
326 --diagnostic.active;
327 }
328 }
