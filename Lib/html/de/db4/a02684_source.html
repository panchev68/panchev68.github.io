175 [[nodiscard]] bool read(uint32_t addr, void * buffer, size_t size) {
176 if (!isChipPresent || buffer == nullptr || size == 0) return false;
177
178 uint8_t * bytes = static_cast<uint8_t *>(buffer);
179 while (size > 0) {
180 const size_t readLength = std::min(size, static_cast<size_t>(BLOCK_READ_SIZE));
181
182 OSPI_RegularCmdTypeDef cmd{};
183 cmd.InstructionMode = HAL_OSPI_INSTRUCTION_1_LINE;
184 cmd.InstructionSize = HAL_OSPI_INSTRUCTION_8_BITS;
185 cmd.Instruction = static_cast<uint32_t>(Register::Command::FAST_READ_QUAD_IO);
186 cmd.AddressMode = HAL_OSPI_ADDRESS_4_LINES;
187 cmd.AddressSize = HAL_OSPI_ADDRESS_24_BITS;
188 cmd.Address = addr;
189 cmd.DummyCycles = FAST_READ_QUAD_IO_DUMMY_CYCLES;
190 cmd.DataMode = HAL_OSPI_DATA_4_LINES;
191 cmd.NbData = static_cast<uint32_t>(readLength);
192
193 if (!ospi.command(cmd, OSPI_TIMEOUT) || !ospi.receive(std::span<uint8_t>(bytes, readLength), OSPI_TIMEOUT)) return false;
194
195 addr += static_cast<uint32_t>(readLength);
196 bytes += readLength;
197 size -= readLength;
198 }
199 return true;
200 }

210 [[nodiscard]] bool write(uint32_t addr, const void * data, size_t size) {
211 if (!isChipPresent || data == nullptr || size == 0) return false;
212
213 const Info & chipInfo = getChipInfo();
214 const size_t pageSize = chipInfo.pageSize ? chipInfo.pageSize : PAGE_SIZE;
215
216 const uint8_t * bytes = static_cast<const uint8_t *>(data);
217 while (size > 0) {
218 const size_t pageOffset = addr & (pageSize - 1);
219 const size_t writeLength = std::min(pageSize - pageOffset, size);
220
221 if (!writeEnable()) return false;
222
223 OSPI_RegularCmdTypeDef cmd{};
224 cmd.InstructionMode = HAL_OSPI_INSTRUCTION_1_LINE;
225 cmd.InstructionSize = HAL_OSPI_INSTRUCTION_8_BITS;
226 cmd.Instruction = static_cast<uint32_t>(Register::Command::QUAD_PAGE_PROGRAM);
227 cmd.AddressMode = HAL_OSPI_ADDRESS_1_LINE;
228 cmd.AddressSize = HAL_OSPI_ADDRESS_24_BITS;
229 cmd.Address = addr;
230 cmd.DataMode = HAL_OSPI_DATA_4_LINES;
231 cmd.NbData = static_cast<uint32_t>(writeLength);
232
233 if (!ospi.command(cmd, OSPI_TIMEOUT) || !ospi.transmit(std::span<const uint8_t>(bytes, writeLength), OSPI_TIMEOUT)) return false;
234 if (!waitForWriteComplete(MAX_WRITE_TIME)) return false;
235
236 addr += static_cast<uint32_t>(writeLength);
237 bytes += writeLength;
238 size -= writeLength;
239 }
240 return true;
241 }

311 [[nodiscard]] bool waitForWriteComplete(std::chrono::milliseconds timeout) {
312 uint8_t status = 0;
313 const uint32_t start = HAL_GetTick();
314 do {
315 if (!readStatusRegister(status)) return false;
316 if (!(status & STATUS_BUSY_MASK)) return true;
317 } while ((HAL_GetTick() - start) < static_cast<uint32_t>(timeout.count()));
318 return false;
319 }
