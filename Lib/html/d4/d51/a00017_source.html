415 void updateTxBuffer() {
416 auto txPorts = reinterpret_cast<MC33879::TxPort *>(txBuffer.data() + 1);
417 const size_t ports = portCount();
418 for (size_t i = 0; i < pinCount; ++i) {
419 const auto & map = mappingFields[i];
420 const size_t p = static_cast<size_t>(map.port);
421 if (p >= ports) continue;
422 if ((value & (1ULL << i)) != 0) txPorts[p].value |= map.mask;
423 else txPorts[p].value &= static_cast<uint8_t>(~map.mask);
424 }
425 }
426
432 void performSpiTransfer() {
433 txBuffer[0] = diagCounter;
434 nssPin.write(0);
435 const bool spiSuccess = spiPort.sendReceive(std::span<const uint8_t>(txBuffer.data(), bufferSize), std::span<uint8_t>(rxBuffer.data(), bufferSize), OUTPUT_SPI_TIMEOUT);
436 nssPin.write(1);
437 RTOS::Task::yield();
438 if (!spiSuccess) return;
439 if (diagCounter == 0) oePin.write(1);
440 if (++diagCounter == 0) diagCounter = 1;
441 }

448 void processRxBuffer() {
449 if (!isSpiError && diagCounter > 0 && rxBuffer[bufferSize - 1] != txBuffer[0]) {
450 isSpiError = true;
451 if (onSpiError) onSpiError();
452 }
453 uint64_t newStatus = 0;
454 auto rxPorts = reinterpret_cast<MC33879::RxPort *>(rxBuffer.data() + 1);
455 const size_t ports = portCount();
456 for (size_t i = 0; i < pinCount; ++i) {
457 const auto & map = mappingFields[i];
458 const size_t p = static_cast<size_t>(map.port);
459 if (p < ports && (rxPorts[p].status & map.mask) != 0) newStatus |= (1ULL << i);
460 }
461 if (status != newStatus) {
462 status = newStatus;
