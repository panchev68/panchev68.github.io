100 const double eqTime = calculateEquationOfTime(t);

113 double computeHourAngle(double dec, double angle) const {
114 const double decRad = degToRad(dec);
115 const double cosH = (std::cos(degToRad(angle)) - (sinLat * std::sin(decRad))) / (cosLat * std::cos(decRad));
116 return std::acos(std::clamp(cosH, -1.0, 1.0));
117 }

122 double calculateEquationOfTime(double t) const {
