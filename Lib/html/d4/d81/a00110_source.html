83 bool init() {
84 uint8_t regAddress = REG_CONFIG;
85 if (!i2cPort.masterTransmit(slaveAddress, &regAddress, sizeof(regAddress), STANDARD_TIMEOUT_DURATION)) {
86 assert(!"Failed to send config register address during init");

117 bool readTemperature(float & outTemperature) {
118 uint8_t buffer[2U] = {0U, 0U};
119 uint8_t regAddress = REG_TEMPERATURE;
120 if (!i2cPort.masterTransmit(slaveAddress, &regAddress, 1U, STANDARD_TIMEOUT_DURATION)) {
121 return false;
122 }
123
124 RTOS::delay(1U);
125 if (!i2cPort.masterReceive(slaveAddress, buffer, 2U, STANDARD_TIMEOUT_DURATION)) {
126 return false;
127 }
128
129 int16_t rawValue = static_cast<int16_t>((buffer[0U] << 4U) | (buffer[1U] >> 4U));
130 if ((rawValue & 0x0800) != 0) {
131 rawValue |= static_cast<int16_t>(0xF000);
132 }
133
134 outTemperature = static_cast<float>(rawValue) * 0.0625f;
135 return true;
