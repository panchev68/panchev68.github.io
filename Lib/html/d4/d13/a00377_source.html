87 [[nodiscard]] bool open() {
88 if (!events.open()) {
89 return false;
90 }
91 if (!Base::open()) {
92 return false;

94 I2C_HandleTypeDef * const handle = this->getHandle();
95
96 // Register HAL callbacks to route events into this instance. '&=' never short-circuits, so a failing
97 // registration still leaves the remaining ones attempted.
98 [[maybe_unused]] bool isCallbacksRegistered =
99 (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID, txCompleteISR) == HAL_OK);
100 isCallbacksRegistered &= (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID, rxCompleteISR) == HAL_OK);
101 isCallbacksRegistered &= (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_ERROR_CB_ID, errorISR) == HAL_OK);
102 assert(isCallbacksRegistered && "I2CPort<DMA>: failed to register one or more HAL I2C callbacks");
103
104 if (!DeviceManager<I2CPort, I2C_HandleTypeDef>::instance().registerDevice(this)) {
105 assert(!"I2CPort<DMA>: failed to register I2C device with DeviceManager");
106 return false;
107 }
108 return true;
109 }

117 [[nodiscard]] bool close() {
118 I2C_HandleTypeDef * const handle = this->getHandle();
119
120 DeviceManager<I2CPort, I2C_HandleTypeDef>::instance().unregisterDevice(this);
121
122 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID);
123 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID);
124 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_ERROR_CB_ID);
125
126 const bool result = Base::close();
127 events.close();
128 return result;
129 }

139 [[nodiscard]] bool masterTransmit(uint16_t devAddress, std::span<const uint8_t> data, std::chrono::milliseconds timeout) {
140 if (data.empty()) [[unlikely]] {
141 assert(!"I2CPort<DMA>: masterTransmit() requires a non-empty source buffer");
142 return false;
143 }
144 if (data.size() > static_cast<size_t>((std::numeric_limits<uint16_t>::max)())) [[unlikely]] {
145 assert(!"I2CPort<DMA>: masterTransmit() payload exceeds the 16-bit length the HAL accepts");
146 return false;
147 }
148 static_cast<void>(events.clearBits(EventBits(Event::ALL)));
149 if (!masterSendHandler(devAddress, data)) {
150 return false;
151 }
152 const uint32_t result = waitForEvent(Event::TX_COMPLETE | Event::ERROR, RTOS::TickCounter::msToTicks(static_cast<uint32_t>(timeout.count())));
153 return ((result & Event::TX_COMPLETE) != 0U) && ((result & Event::ERROR) == 0U);
154 }

198 [[nodiscard]] bool abort(uint16_t devAddress) {
199 I2C_HandleTypeDef * const handle = this->getHandle();
200
201 if (handle == nullptr) [[unlikely]] {
202 assert(!"I2CPort<DMA>: abort() requires a valid native I2C handle");
203 return false;
204 }
205 // Report the abort as an error event so a task blocked in waitForEvent() wakes up immediately instead of
206 // stalling until its timeout expires.
207 const bool result = (HAL_I2C_Master_Abort_IT(handle, devAddress) == HAL_OK);
208 static_cast<void>(events.setBits(EventBits(Event::ERROR)));
