168 [[nodiscard]] bool lock(TickType timeout = WAIT_FOREVER) noexcept {
169 if (mutexHandle == nullptr) [[unlikely]] {
170 assert(!"Mutex::lock() called before open()");
171 return false;
172 }
173 // FIX: A mutex has an owner task and priority inheritance, so taking it from an ISR is illegal.
174 assert((xPortIsInsideInterrupt() == pdFALSE) && "Mutex::lock() called from an ISR");
175 if (mutexType == Type::RECURSIVE) {
176#if (configUSE_RECURSIVE_MUTEXES == 1)
177 return xSemaphoreTakeRecursive(mutexHandle, timeout) == pdTRUE;
178#else
179 return false;
180#endif
181 }
182 return xSemaphoreTake(mutexHandle, timeout) == pdTRUE;
