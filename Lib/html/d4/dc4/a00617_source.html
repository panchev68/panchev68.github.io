73 static Lib::Helper::DateTime stringToDateTime(std::string_view timeString, std::string_view dateString) {
74 assert(timeString.data() != nullptr && "timeString pointer is null");
75 assert(dateString.data() != nullptr && "dateString pointer is null");
76
77 auto getDigit = [](std::string_view str, size_t offset) -> uint8_t {
78 assert(offset < str.size() && "offset exceeds string size");
79 return static_cast<uint8_t>(str[offset] - '0');
80 };
81
82 Lib::Helper::DateTime dateTime;
83 dateTime.time.hours = 10 * getDigit(timeString, 0) + getDigit(timeString, 1);
84 dateTime.time.minutes = 10 * getDigit(timeString, 3) + getDigit(timeString, 4);
85 dateTime.time.seconds = 10 * getDigit(timeString, 6) + getDigit(timeString, 7);
86 dateTime.date.year = 1000 * getDigit(dateString, 7) + 100 * getDigit(dateString, 8) + 10 * getDigit(dateString, 9) + getDigit(dateString, 10);
87 dateTime.date.month = parseMonth(dateString);
88 dateTime.date.day = 10 * (dateString[4] == ' ' ? 0 : getDigit(dateString, 4)) + getDigit(dateString, 5);
89 return dateTime;
90 }

102 static uint32_t stringToU32(std::string_view versionString) {
103 std::array<uint8_t, 2> versionParts{0U, 0U};
104 size_t begin = 0U;
105
106 for (size_t i = 0U; i < versionParts.size(); ++i) {
107 size_t found = versionString.find('.', begin);
108 if (found == std::string_view::npos) {
109 found = versionString.size();
110 }
111
112 /*
113 * Parsed digit by digit rather than through std::stoul: that function reports a malformed field by
114 * throwing, and this target is built with exceptions disabled, so the failure would terminate
115 * instead of being handled. A non-digit simply ends the field here.
116 */
117 uint32_t value = 0U;
118 for (size_t index = begin; index < found; ++index) {
119 const char character = versionString[index];
120 if ((character < '0') || (character > '9')) {
121 break;
122 }
123 value = (value * 10U) + static_cast<uint32_t>(character - '0');
124 }
125 versionParts[i] = static_cast<uint8_t>(value & 0xFFU);
126
127 begin = found + 1U;
128 if (found == versionString.size()) {
129 break;
130 }
131 }
132
133 return (static_cast<uint32_t>(versionParts[0]) << 8U) | static_cast<uint32_t>(versionParts[1]);
134 }

218 [[nodiscard]] static constexpr size_t appendNumber(VersionString & buffer, size_t offset, uint8_t value, size_t minimumDigits) {
219 std::array<char, MAX_FIELD_DIGITS> digits{};
220 size_t count = 0U;
221 uint8_t remaining = value;
222
223 do {
224 digits[count++] = static_cast<char>('0' + (remaining % 10U));
225 remaining = static_cast<uint8_t>(remaining / 10U);
226 } while (remaining > 0U);
227
228 for (size_t pad = count; pad < minimumDigits; ++pad) {
229 buffer[offset++] = '0';
230 }
231 for (size_t index = count; index > 0U; --index) {
232 buffer[offset++] = digits[index - 1U];
233 }
234 return offset;
235 }

246 static uint8_t parseMonth(std::string_view dateString) {
247 switch (dateString[2]) {
248 case 'b':
249 return 2; // Feb
250 case 'y':
251 return 5; // May
252 case 'l':
253 return 7; // Jul
254 case 'g':
255 return 8; // Aug
256 case 'p':
257 return 9; // Sep
258 case 't':
259 return 10; // Oct
260 case 'v':
261 return 11; // Nov
262 case 'c':
263 return 12; // Dec
264 case 'n':
265 return dateString[1] == 'a' ? 1 : 6; // Jan or Jun
266 default:
267 return dateString[1] == 'a' ? 3 : 4; // Mar or Apr
268 }
