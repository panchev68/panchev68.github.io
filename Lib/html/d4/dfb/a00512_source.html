174 T compute(T setpoint, T input) {
175 if (!autoMode) return state.lastOutput;
176
177 T error = (setpoint - input) * (reverseDirection ? T{-1} : T{1});
178
179 if (state.firstRun) {
180 state.lastInput = input;
181 state.firstRun = false;
182 }
183
184 T proportional = params.kp * error;
185
186 state.integralSum += params.ki * error * params.sampleTime;
187 state.integralSum = std::clamp(state.integralSum, params.outputMin - proportional, params.outputMax - proportional);
188
189 T derivative = params.kd * (state.lastInput - input) / params.sampleTime;
190
191 T output = std::clamp(proportional + state.integralSum + derivative, params.outputMin, params.outputMax);
192
193 state.lastInput = input;
194 state.lastOutput = output;
195
196 return output;
197 }
