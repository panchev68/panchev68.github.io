118 [[nodiscard]] bool registerDevice(uint16_t pinNumber, ExtInterrupt * device) noexcept {
119 const int index = getPinIndex(pinNumber);
120 if (index < 0 || index >= static_cast<int>(MAX_PINS)) {
121 assert(!"Invalid pin index");
122 return false;
123 }
124
125 RTOS::Kernel::CriticalSection guard;
126 if (registry[index] != nullptr) {
127 assert(!"Pin already registered");
128 return false;
129 }
130 registry[index] = device;
131
132 // Lazy-initialize the single, shared worker thread upon first device registration
133 if (!isTaskCreated) {
134 task.setup({"EXTI", WORKER_STACK_SIZE, RTOS::Task::Priority::HIGH});
135 if (task.start([this] { taskEntry(); }) == nullptr) {
136 assert(!"Failed to start EXTI worker task");
137 registry[index] = nullptr; // Roll back registration on failure
138 return false;
139 }
140 isTaskCreated = true;
141 }
142 return true;
143 }

218 void taskEntry() {
219 while (true) {
220 // Clear nothing on entry (catches events fired before the wait began), clear every bit on exit so
221 // the next iteration starts from a clean notification value.
222 const uint32_t activeBits = static_cast<uint32_t>(task.notify.wait(0U, 0xFFFFFFFFU, portMAX_DELAY).to_ulong());
223
224 // Directly target only the active bits without doing standard linear registry sweeps
225 uint32_t remainingBits = activeBits;
226 while (remainingBits != 0U) {
227 const uint32_t currentBit = remainingBits & (~remainingBits + 1U); // Isolate lowest set bit
228 const int index = getPinIndex(static_cast<uint16_t>(currentBit));
229
230 if (index >= 0) {
231 if (auto * device = registry[index]) {
232 if (device->userCallback) {
233 device->userCallback();
234 }
235 }
236 }
237 remainingBits &= ~currentBit; // Clear processed bit
238 }
239 }
