393 template<typename Rep, typename Period>
394 [[nodiscard]] constexpr Kernel::TickType durationToTicks(std::chrono::duration<Rep, Period> duration) noexcept {
395 using WideTicks = std::chrono::duration<int64_t, Kernel::TickDuration::period>;
396
397 if (duration <= duration.zero()) {
398 return 0U;
399 }
400 // FIX: rounds straight to ticks; the former millisecond step truncated sub-millisecond durations to zero.
401 const int64_t ticks = std::chrono::ceil<WideTicks>(duration).count();
402 return (ticks >= static_cast<int64_t>(portMAX_DELAY)) ? portMAX_DELAY : static_cast<Kernel::TickType>(ticks);
403 }
