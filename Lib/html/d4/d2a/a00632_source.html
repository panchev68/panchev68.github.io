147 [[nodiscard]] bool overwrite() noexcept {
148 if (semaphoreHandle == nullptr) [[unlikely]] {
149 assert(!"Semaphore: overwrite() requires an open semaphore");
150 return false;
151 }
152 assert((xPortIsInsideInterrupt() == pdFALSE) && "Semaphore: overwrite() called from an ISR");
153 /* FIX: The former take-then-give pair left a window in which the semaphore looked taken to a preempting
154 * task. A binary give only fails when the semaphore is already available, which is the requested state. */
155 static_cast<void>(xSemaphoreGive(semaphoreHandle));
156 return true;
157 }

169 return false;

171 // FIX: Same single-give collapse as overwrite(); the take/give pair was not atomic either.
172 static_cast<void>(xSemaphoreGiveFromISR(semaphoreHandle, &higherPriorityTaskWoken));
