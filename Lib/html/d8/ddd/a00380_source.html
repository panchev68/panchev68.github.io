86 [[nodiscard]] bool open() {

95 // Register HAL callbacks to route events into this instance. '&=' never short-circuits, so a failing
96 // registration still leaves the remaining ones attempted.
97 [[maybe_unused]] bool isCallbacksRegistered = (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID, txCompleteISR) == HAL_OK);
98 isCallbacksRegistered &= (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID, rxCompleteISR) == HAL_OK);
99 isCallbacksRegistered &= (HAL_I2C_RegisterCallback(handle, CallbackID::HAL_I2C_ERROR_CB_ID, errorISR) == HAL_OK);
100 assert(isCallbacksRegistered && "I2CPort<IT>: failed to register one or more HAL I2C callbacks");
101
102 if (!DeviceManager<I2CPort, I2C_HandleTypeDef>::instance().registerDevice(this)) {
103 assert(!"I2CPort<IT>: failed to register I2C device with DeviceManager");
104 return false;
105 }
106 return HAL_I2C_EnableListen_IT(handle) == HAL_OK;
107 }

115 [[nodiscard]] bool close() {
116 I2C_HandleTypeDef * const handle = this->getHandle();
117
118 static_cast<void>(HAL_I2C_DisableListen_IT(handle));
119 DeviceManager<I2CPort, I2C_HandleTypeDef>::instance().unregisterDevice(this);
120
121 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_MASTER_TX_COMPLETE_CB_ID);
122 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_MASTER_RX_COMPLETE_CB_ID);
123 HAL_I2C_UnRegisterCallback(handle, CallbackID::HAL_I2C_ERROR_CB_ID);

125 const bool result = Base::close();

138 [[nodiscard]] bool masterTransmit(uint16_t devAddress, std::span<const uint8_t> data, std::chrono::milliseconds timeout) {
139 if (data.empty()) [[unlikely]] {
140 assert(!"I2CPort<IT>: masterTransmit() requires a non-empty source buffer");
141 return false;
142 }
143 if (data.size() > static_cast<size_t>((std::numeric_limits<uint16_t>::max)())) [[unlikely]] {
144 assert(!"I2CPort<IT>: masterTransmit() payload exceeds the 16-bit length the HAL accepts");
145 return false;
146 }
147 static_cast<void>(events.clearBits(EventBits(Event::ALL)));
