67 {RCC_FLAG_LPWRRST, State::LOW_POWER},
68 {RCC_FLAG_WWDGRST, State::WINDOW_WATCHDOG},
69 {RCC_FLAG_IWDGRST, State::INDEPENDENT_WATCHDOG},
70 {RCC_FLAG_SFTRST, State::SOFTWARE},
71 {RCC_FLAG_PORRST, State::POWER_ON},
72 {RCC_FLAG_PINRST, State::PIN},
73 {RCC_FLAG_BORRST, State::BROWNOUT},
74 }};
75
76 State state = State::UNKNOWN;
77 for (const auto & [flag, candidate] : FLAG_TO_STATE) {
78 if (__HAL_RCC_GET_FLAG(flag)) {
79 state = candidate;
80 break;
81 }
82 }
83
84 __HAL_RCC_CLEAR_RESET_FLAGS();
