158 [[nodiscard]] bool readWrite(const void * source, size_t sourceSize, void * destination, size_t destinationSize) {
159 const size_t totalLength = sourceSize + destinationSize;
160
161 if (totalLength == 0U || totalLength > MAX_BUFFER_SIZE) {
162 assert(!"LAN9252 Interface: readWrite() length is zero or exceeds MAX_BUFFER_SIZE");
163 return false;
164 }
165 if ((sourceSize > 0U && source == nullptr) || (destinationSize > 0U && destination == nullptr)) {
166 assert(!"LAN9252 Interface: readWrite() received a null buffer for a non-zero length");
167 return false;
168 }

170 std::fill_n(txBuffer.begin(), totalLength, PAD_BYTE);

172 if (sourceSize > 0U) {
173 const auto * sourceBytes = static_cast<const uint8_t *>(source);
174 std::copy_n(sourceBytes, sourceSize, txBuffer.begin());

177 if (!SpiPortIT::sendReceive(std::span<const uint8_t>(txBuffer.data(), totalLength), std::span<uint8_t>(rxBuffer.data(), totalLength), SPI_TIMEOUT)) {
