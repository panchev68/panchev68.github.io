219 void enterToSleep(uint32_t expectedIdleTime) {
220 HAL_SuspendTick();
221
222 Lib::HAL::LPTimer * device = timer.load(std::memory_order_relaxed);
223
224 if (!isInitialized.load(std::memory_order_acquire) || device == nullptr) [[unlikely]] {
225 assert(!"PowerManager::enterToSleep() called before successful setup()");
226 enterLightSleep();
227 return;
228 }
229
230 if (expectedIdleTime > MIN_TICKS_FOR_STOP_MODE && isDeepSleepAllowed.load(std::memory_order_relaxed)) {
231 uint32_t clampedIdleTime = (expectedIdleTime > MAX_IDLE_TIME_BEFORE_OVERFLOW) ? MAX_IDLE_TIME_BEFORE_OVERFLOW : expectedIdleTime;
232 uint32_t timeoutMs = (clampedIdleTime * MILLISECONDS_PER_SECOND) / configTICK_RATE_HZ;
233
234 if (device->startTimeoutInterrupt(timeoutMs)) {
235 wasDeepSleepEntered.store(true, std::memory_order_relaxed);
236
237 invokeCallbacks(preSleepRegistry, preSleepCallbackCount);
238
239 __HAL_RCC_PWR_CLK_ENABLE();
240 HAL_PWREx_EnterSTOP1Mode(PWR_STOPENTRY_WFI);
241
242 uint32_t elapsedRtosTicks = device->getElapsedRtosTicks(timeoutMs);
243 elapsedRtosTicks = (elapsedRtosTicks > clampedIdleTime) ? clampedIdleTime : elapsedRtosTicks;
244
245 if (elapsedRtosTicks > 0U) {
246 vTaskStepTick(elapsedRtosTicks);
247 }
248
249 SystemClock_Config();
250 HAL_InitTick(TICK_INT_PRIORITY);
251 __enable_irq();
252 __ISB();
253 __NOP();
254 __NOP();
255 } else {
256 enterLightSleep();
257 }
258 } else {
259 enterLightSleep();
260 }
261 }

266 void enterLightSleep() noexcept {
267 wasDeepSleepEntered.store(false, std::memory_order_relaxed);
268 SCB->SCR &= ~static_cast<uint32_t>(SCB_SCR_SLEEPDEEP_Msk);
269 __WFI();
270 HAL_ResumeTick();
271 __enable_irq();
272 }

277 void exitFromSleep() {
278 Lib::HAL::LPTimer * device = timer.load(std::memory_order_relaxed);
279 assert(isInitialized.load(std::memory_order_acquire) && device != nullptr && "Cannot exit low power mode; invalid state");
280
281 if (!wasDeepSleepEntered.load(std::memory_order_relaxed) || device == nullptr) {
282 return;
283 }
284
285 device->stopInterrupt();
286 invokeCallbacks(postSleepRegistry, postSleepCallbackCount);
287 }

300 [[nodiscard]] static bool registerCallback(std::array<PeripheralCallback, MAX_CALLBACKS> & registry, std::atomic<uint32_t> & count,
301 PeripheralCallback callback) noexcept {
302 if (callback == nullptr) {
303 assert(!"PowerManager: cannot register a null callback");
304 return false;
305 }
306
307 const uint32_t slot = count.load(std::memory_order_relaxed);
308 if (slot >= MAX_CALLBACKS) {
309 assert(!"PowerManager: callback registry is full");
310 return false;
311 }
312
313 registry[slot] = callback;
314 count.store(slot + 1U, std::memory_order_release);
315 return true;
316 }

324 static void invokeCallbacks(const std::array<PeripheralCallback, MAX_CALLBACKS> & registry, const std::atomic<uint32_t> & count) noexcept {
325 // Acquire: pairs with the release store in registerCallback() so every counted slot is visible here.
326 const uint32_t published = count.load(std::memory_order_acquire);
327
328 for (uint32_t index = 0U; index < published; ++index) {
329 if (registry[index] != nullptr) {
330 registry[index]();
331 }
332 }
333 }

340 void handlePortSuppressTicksAndSleep(uint32_t expectedIdleTime) {
341 if (expectedIdleTime < MIN_TICKS_FOR_SLEEP_MODE) [[unlikely]] {
342 return;
343 }
344
345 __disable_irq();
346 __DSB();
347 __ISB();
348
349 auto sleepStatus = eTaskConfirmSleepModeStatus();
350
351 if (sleepStatus == eAbortSleep) [[unlikely]] {
352 __enable_irq();
353 } else {
354 if (canEnterDeepSleep()) {
355 enterToSleep(expectedIdleTime);
356 exitFromSleep();
357 } else {
358 enterLightSleep();
359 }
360 }
361 }
