155 [[nodiscard]] bool getButtonState(uint32_t stableSamples = DEFAULT_STABLE_SAMPLES) {
156 if (stableSamples == 0U) {
157 return !button.read();
158 }
159
160 bool state = button.read();
161 uint32_t matches = 0U;
162
163 for (uint32_t attempt = 0U; attempt < (stableSamples * MAX_SAMPLE_ATTEMPTS); ++attempt) {
164 RTOS::delay(SAMPLE_PERIOD);
165 const bool currentState = button.read();
166
167 if (currentState != state) {
168 state = currentState;
169 matches = 0U;
170 continue;
171 }
172 matches++;
173 if (matches >= stableSamples) {
174 break;
175 }
176 }
177 return !state;
178 }

323 buttonHoldPeriod = ticksToMilliseconds(xTaskGetTickCountFromISR() - buttonStartTick);

326
327 const bool isRotationEvent = (state == EventState::CCW || state == EventState::CW);
328 if (!isRotationEvent || (++rotationEventCounter <= pulseDivider)) {
329 return;
330 }
