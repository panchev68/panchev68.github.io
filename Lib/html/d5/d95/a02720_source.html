125 static bool isValid(const ApiFrameBuffer & apiFrame) {
126 size_t msgSize = apiFrame.getSize();
127 if (msgSize < sizeof(Frame::Header) + 1) return false;
128 const auto & header = *reinterpret_cast<const Frame::Header *>(apiFrame.getData());
129 if (header.startDelimiter != START_DELIMITER) return false;
130 size_t headerLen = static_cast<size_t>((header.lengthMSB << 8) | header.lengthLSB);
131 if (headerLen < 1) [[unlikely]] return false;
132 size_t crcLen = sizeof(Frame::Header) + headerLen;
133 if (crcLen > msgSize - 1) return false;
134 return Checksum(apiFrame) == apiFrame.getData()[crcLen];
135 }

168 static const ApiFrameBuffer & build(const Frame::Data & frameData) {
169 static ApiFrameBuffer msg;
170 assert(API_MODE == ApiMode::API1);
171 size_t frameSize = frameData.vector.size() + 1;
172 msg.clear();
173 msg.append(START_DELIMITER);
174 msg.append(static_cast<uint8_t>(frameSize >> 8));
175 msg.append(static_cast<uint8_t>(frameSize & 0xff));
176 msg.append(frameData.type);
177 msg.append(frameData.vector.data(), frameData.vector.size());
178 msg.append(Checksum(msg));
179 return msg;
180 }
