326 [[nodiscard]] static constexpr uint32_t pack(const DateTime & dateTime) noexcept {
327 assert(dateTime.date.year >= Date::FIRST_YEAR && "Date::year is before Date::FIRST_YEAR");
328 const uint32_t yearDelta = (dateTime.date.year >= Date::FIRST_YEAR) ? static_cast<uint32_t>(dateTime.date.year - Date::FIRST_YEAR) : 0U;
329 assert(yearDelta <= YEAR_DELTA_MASK && "Date::year exceeds the packable range");
330 return ((dateTime.time.seconds & SECONDS_MASK) << SECONDS_SHIFT) | ((dateTime.time.minutes & MINUTES_MASK) << MINUTES_SHIFT) |
331 ((dateTime.time.hours & HOURS_MASK) << HOURS_SHIFT) | ((dateTime.date.day & DAY_MASK) << DAY_SHIFT) |
332 ((dateTime.date.month & MONTH_MASK) << MONTH_SHIFT) | ((yearDelta & YEAR_DELTA_MASK) << YEAR_DELTA_SHIFT);
333 }
334
341 [[nodiscard]] static constexpr DateTime unpack(uint32_t packed) noexcept {
342 return DateTime(Date(static_cast<uint16_t>(Date::FIRST_YEAR + ((packed >> YEAR_DELTA_SHIFT) & YEAR_DELTA_MASK)),
343 static_cast<uint8_t>((packed >> MONTH_SHIFT) & MONTH_MASK),
344 static_cast<uint8_t>((packed >> DAY_SHIFT) & DAY_MASK)),
345 Time(static_cast<uint8_t>((packed >> HOURS_SHIFT) & HOURS_MASK),
346 static_cast<uint8_t>((packed >> MINUTES_SHIFT) & MINUTES_MASK),
347 static_cast<uint8_t>((packed >> SECONDS_SHIFT) & SECONDS_MASK)));
348 }
