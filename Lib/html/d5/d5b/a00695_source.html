288 int8_t controlCallback(uint8_t command, uint8_t * buffer, uint16_t length) {
289 struct LineCoding {
290 uint32_t baudRate;
291 uint8_t charFormat;
292 uint8_t parityType;
293 uint8_t dataBits;
294 };
295
296 static LineCoding lineCoding{115200, 0, 0, 8};
297 switch (command) {
298 case CDC_SET_LINE_CODING:
299 std::memcpy(&lineCoding, buffer, sizeof(LineCoding));
300 SerialPort::setBaudRate(lineCoding.baudRate);
301 break;
302 case CDC_GET_LINE_CODING:
303 std::memcpy(buffer, &lineCoding, sizeof(LineCoding));
304 break;
305 default:
306 break;
307 }
