173 void processResponse() noexcept {
174 const size_t count = std::min(response.size(), SPECTRUM_SIZE);
175 for (size_t i = 0; i < count; ++i) {
176 const uint8_t valPercent = Lib::Helper::Convert::dBM::toDigiPercent(response[i]);
177 auto & channel = data[i];
178
179 channel.current = static_cast<uint8_t>((valPercent + channel.current) / 2);

181 if (channel.average > channel.current) {
182 channel.average -= 1;
183 } else if (channel.average < channel.current) {
