141 [[nodiscard]] EventCode read() {
142 if (!isInitialized) {
143 assert(!"EncoderTimer: read() requires a successfully opened driver");
144 return EventCode::NONE;
145 }
146
147 if (const auto pinState = buttonQueue.receive(0U)) {
148 if (!*pinState) {
149 eventCode = EventCode::BUTTON_PULL;
150 } else if (eventCode != EventCode::BUTTON_PUSH) {
151 eventCode = EventCode::BUTTON_PUSH;
152 buttonHoldTimeout.set(RTOS::millisecondsToTicks(BUTTON_HOLD_TIME.count()));
153 }
154 return eventCode;
155 }
156
157 const uint32_t position = timer.read();
158 if (position != rawPosition) {
159 /*
160 * The hardware direction bit is authoritative: comparing counter magnitudes would misreport the step
161 * where the counter wraps between 0 and the auto-reload value.
162 */
163 const bool isClockwise = timer.isClockwise();
164 rawPosition = position;
165 return isClockwise ? EventCode::ENCODER_CW : EventCode::ENCODER_CCW;
166 }
167
168 if (eventCode == EventCode::BUTTON_PUSH && buttonHoldTimeout.isExpired()) {
169 eventCode = EventCode::BUTTON_HOLD;
170 return EventCode::BUTTON_HOLD;
171 }
172 return EventCode::NONE;
173 }
