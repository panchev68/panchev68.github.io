187 [[nodiscard]] bool writeImpl(uint32_t addr, const void * data, size_t size) {
188 if (!isChipPresent || !data || size == 0) return false;
189
190 const auto & chipInfo = getChipInfo();
191 const size_t pageSize = chipInfo.pageSize ? chipInfo.pageSize : PAGE_SIZE;
192
193 const uint8_t * bytes = static_cast<const uint8_t *>(data);
194 while (size > 0) {
195 const size_t pageOffset = addr & (pageSize - 1);
196 const size_t writeLength = std::min(pageSize - pageOffset, size);
197
198 if (!writeEnable()) return false;
199
200 select();
201 const bool success = sendCommandA24(static_cast<uint8_t>(Register::Command::PAGE_PROGRAM), addr) &&
202 spiPort.send(std::span<const uint8_t>(bytes, writeLength), std::chrono::milliseconds(SPI_TIMEOUT));
203 deselect();
204
205 if (!success || !waitForWriteComplete(MAX_WRITE_TIME)) return false;
206
207 addr += writeLength;
208 bytes += writeLength;
