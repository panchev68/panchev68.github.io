111 void handleTickEvent() {
112 bool hasError = false;
113 uint32_t clearedErrorCount = 0;
114
115 for (auto & error : errorVector) {
116 if (error.checkForStateChanged()) {
117 const auto state = error.getState();
118 if (state > Error::State::NO_ERROR) {
119 hasError = true;
120 } else if (state == Error::State::CLEARED) {
121 clearedErrorCount++;
122 }
123 }
124 }
125
126 if (hasError) {
127 if (statusChangedHook) {
128 statusChangedHook(Status::STATUS_CHANGED);
129 }
130 } else if (clearedErrorCount != lastClearedErrorCount) {
131 lastClearedErrorCount = clearedErrorCount;
132 if (statusChangedHook) {
133 statusChangedHook(Status::ALL_ERRORS_CLEARED);
134 }
135 }
