111 template<typename Rep, typename Period>
112 static void waitFor(std::chrono::duration<Rep, Period> duration) noexcept {
113 using Nanoseconds64 = std::chrono::duration<uint64_t, std::nano>;
114 constexpr uint64_t NANOSECONDS_PER_SECOND = 1'000'000'000ULL;
115
116 if (duration <= std::chrono::duration<Rep, Period>::zero()) {
117 return;
118 }
119 enable();
120 // FIX: Computed in 64 bits and rounded up; us * cycles-per-us overflowed 32 bits after about 25 s and
121 // truncated both a fractional MHz clock and sub-cycle nanosecond requests to a shorter delay.
122 const uint64_t totalNanoseconds = std::chrono::duration_cast<Nanoseconds64>(duration).count();
123 const uint64_t clock = SystemCoreClock;
124 const uint64_t wholeSeconds = totalNanoseconds / NANOSECONDS_PER_SECOND;
125 const uint64_t fraction = totalNanoseconds % NANOSECONDS_PER_SECOND;
126 // Split into whole seconds and a fraction so that the product with the clock cannot overflow 64 bits.
127 waitCycles(wholeSeconds * clock + (fraction * clock + NANOSECONDS_PER_SECOND - 1ULL) / NANOSECONDS_PER_SECOND);
128 }

136 static void enable() noexcept {
137 if ((CoreDebug->DEMCR & CoreDebug_DEMCR_TRCENA_Msk) == 0U || (DWT->CTRL & DWT_CTRL_CYCCNTENA_Msk) == 0U) [[unlikely]] {
138 CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
139 // FIX: CYCCNT is no longer reset to zero, which corrupted concurrent users of the cycle counter.
140 DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
141 }
142 }

149 static void waitCycles(uint64_t cycles) noexcept {
150 // FIX: Long waits are split into chunks; one wait of 2^32 cycles or more wrapped CYCCNT and ended early.
151 while (cycles > 0U) {
152 const uint32_t chunk = (cycles > MAX_CHUNK_CYCLES) ? MAX_CHUNK_CYCLES : static_cast<uint32_t>(cycles);
153 const uint32_t start = DWT->CYCCNT;
154 while (static_cast<uint32_t>(DWT->CYCCNT - start) < chunk) {
155 }
156 cycles -= chunk;
157 }
158 }
