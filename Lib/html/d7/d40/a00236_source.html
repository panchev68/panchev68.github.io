175 void handleReceivedFrameEvent(const Frame0x97 & frame) {
176 if ((frame.frameId == 0x17) && (sendFrame.atCommand == frame.atCmd)) {
177 if ((frame.status != 0) && (frame.status != 4)) {
178 assert(!"Remote AT command failed - invalid status code");
179 } else if (frame.parameters.size() >= responseMessage.DATA_SIZE) [[unlikely]] {
180 assert(!"Response parameters exceed maximum size");
181 } else {
182 responseMessage.size = static_cast<uint8_t>(frame.parameters.size());
183 std::memcpy(responseMessage.data.data(), frame.parameters.data(), responseMessage.size);
184 if (!channel.enqueueResponse(responseMessage, SEND_TIMEOUT)) assert(!"Failed to queue remote AT command response - queue full or timeout");
