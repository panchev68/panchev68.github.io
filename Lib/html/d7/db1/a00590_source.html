606 if (instance != nullptr && instance->callback) instance->callback();
607 };

609 handle = xTimerCreate(config.name.empty() ? "Timer" : config.name.data(),
610 config.period,
611 (config.mode == Mode::AUTO_RELOAD) ? pdTRUE : pdFALSE,
612 reinterpret_cast<void *>(this),
