94 [[nodiscard]] bool open() {
95 if (!events.open()) {
96 return false;
97 }
98 SpiHandle * handle = this->getHandle();
99
100 if (HAL_SPI_Init(handle) != HAL_OK) {
101 assert(!"Failed to initialize SPI peripheral");
102 return false;
103 }
104
105 // Register HAL callbacks to route events into this instance. '&=' never short-circuits, so a failing
106 // registration still leaves the remaining ones attempted.
107 [[maybe_unused]] bool isCallbacksRegistered = (HAL_SPI_RegisterCallback(handle, HAL_SPI_TX_COMPLETE_CB_ID, txCompleteISR) == HAL_OK);
108 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, HAL_SPI_RX_COMPLETE_CB_ID, rxCompleteISR) == HAL_OK);
109 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, HAL_SPI_TX_RX_COMPLETE_CB_ID, txRxCompleteISR) == HAL_OK);
110 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, HAL_SPI_ERROR_CB_ID, errorISR) == HAL_OK);
111 assert(isCallbacksRegistered && "Failed to register one or more HAL SPI callbacks");
112
113 if (!DeviceManager<T_DEVICE, SpiHandle>::instance().registerDevice(static_cast<T_DEVICE *>(this))) {
114 assert(!"Failed to register SPI device with DeviceManager");
115 return false;
116 }
117 return true;
118 }
119
125 [[nodiscard]] bool close() {
126 DeviceManager<T_DEVICE, SpiHandle>::instance().unregisterDevice(static_cast<T_DEVICE *>(this));
127
128 SpiHandle * handle = this->getHandle();
129 HAL_SPI_UnRegisterCallback(handle, HAL_SPI_TX_COMPLETE_CB_ID);
130 HAL_SPI_UnRegisterCallback(handle, HAL_SPI_RX_COMPLETE_CB_ID);
131 HAL_SPI_UnRegisterCallback(handle, HAL_SPI_TX_RX_COMPLETE_CB_ID);
132 HAL_SPI_UnRegisterCallback(handle, HAL_SPI_ERROR_CB_ID);
133
134 events.close();
135 return HAL_SPI_DeInit(handle) == HAL_OK;
136 }

145 [[nodiscard]] bool send(std::span<const uint8_t> data, uint32_t timeout) {
146 if (data.size() > static_cast<size_t>((std::numeric_limits<uint16_t>::max)())) {
147 assert(!"send() payload exceeds the 16-bit length the HAL accepts");
148 return false;
149 }
150
151 if (onBeginTransaction) {
152 onBeginTransaction();
153 }
154
155 static_cast<void>(events.clearBits(EventBits(Event::TX_COMPLETE | Event::ERROR)));
156 if (!sendHandler(data)) {
157 assert(!"sendHandler failed to initiate transmission");
158 if (onEndTransaction) {
159 onEndTransaction();
160 }
161 return false;
162 }
163
164 const uint32_t event = waitForEvent(Event::TX_COMPLETE | Event::ERROR, timeout);
165 const bool result = ((event & Event::TX_COMPLETE) != 0U) && ((event & Event::ERROR) == 0U);
166 if (onEndTransaction) {
167 onEndTransaction();
168 }
169
170 return result;
171 }
