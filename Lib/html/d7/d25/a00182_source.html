215 void handleReceivedCharEvent(uint8_t data) {
216 size_t pos = rxFrame.getSize();
217
218 if (rxState.inProgress && rxState.timeout.isExpired()) {
219 diagnostic.ignoredMessagesByTimeout++;
220 rxFrame.clear();
221 rxState.remainingBytes = -1;
222 rxState.inProgress = false;
223 }
224
225 if (pos == offsetof(Frame::Header, Frame::Header::startDelimiter)) {
226 if (data == Protocol::START_DELIMITER) {
227 if (rxState.inProgress) assert(!"Message overlap detected - new frame before previous completion");
228 self().emitBeginMessageReceive();
229 rxState.inProgress = true;
230 rxFrame.clear();
231 rxState.remainingBytes = -1;
232 rxFrame.append(data);
233 rxState.timeout.set(FRAME_TIMEOUT_MS);
234 }
235 rxState.timeout.set(FRAME_TIMEOUT_MS);
236 return;
237 }
238
239 rxFrame.append(data);
240
241 if (pos == offsetof(Frame::Header, Frame::Header::lengthMSB)) {
242 rxState.payloadLength = static_cast<size_t>(data) << 8;
243 } else if (pos == offsetof(Frame::Header, Frame::Header::lengthLSB)) {
244 rxState.payloadLength |= data;
245 if (rxState.payloadLength >= MAX_FRAME_SIZE) {
246 diagnostic.maxMsgLenghtErrors++;
247 rxFrame.clear();
248 self().emitEndMessageReceived();
249 rxState.inProgress = false;
250 rxState.remainingBytes = -1;
251 return;
252 }
253 rxState.remainingBytes = rxState.payloadLength + sizeof(uint8_t);
254 } else {
255 if (rxState.remainingBytes > 0) --rxState.remainingBytes;
256
257 if (rxState.remainingBytes == 0) {
258 if (Protocol::isValid(rxFrame)) {
259 Protocol::processMessage(rxFrame);
260 } else {
261 diagnostic.invalidMessages++;
262 assert(!"Invalid message - checksum or format validation failed");
263 }
264 rxFrame.clear();
265 self().emitEndMessageReceived();
266 rxState.inProgress = false;
267 rxState.remainingBytes = -1;
268 }
