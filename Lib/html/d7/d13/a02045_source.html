125 [[nodiscard]] bool waitForEvent(uint32_t eventMask, uint32_t * activeBits, uint32_t timeout) {
126 if (activeBits == nullptr) {

134 *activeBits = static_cast<uint32_t>(received.to_ulong());

230 template<CallbackId Id> static void dispatchCallback(CAN_HandleTypeDef * hcan) {
231 if (hcan == nullptr) {
232 assert(!"CAN HAL callback invoked with a null handle");
233 return;
234 }
235
236 auto * device = DeviceManager<T_DEVICE, CAN_HandleTypeDef>::instance().getDevice(hcan);
237 if (device == nullptr) {
238 return;
239 }
240
241 if constexpr (Id == CallbackId::TX_COMPLETE) {
242 device->handleTxInterrupt();
243 } else if constexpr (Id == CallbackId::RX_FIFO0_PENDING) {
244 device->handleRxFifo0Interrupt();
245 } else if constexpr (Id == CallbackId::RX_FIFO1_PENDING) {
246 device->handleRxFifo1Interrupt();
247 } else if constexpr (Id == CallbackId::ERROR_OCCURRED) {
248 device->handleErrorInterrupt();
249 }
