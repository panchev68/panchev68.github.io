420 void updateTxBuffer() {
421 auto txPorts = reinterpret_cast<MC33879::TxPort *>(txBuffer.data() + 1);
422 const size_t ports = portCount();
423 for (size_t i = 0; i < pinCount; ++i) {
424 const auto & map = mappingFields[i];
425 const size_t p = static_cast<size_t>(map.port);
426 if (p >= ports) continue;
427 if ((value & (1ULL << i)) != 0) txPorts[p].value |= map.mask;
428 else txPorts[p].value &= static_cast<uint8_t>(~map.mask);
429 }
430 }
431
436 void performSpiTransfer() {
437 txBuffer[0] = diagCounter;
438 nssPin.write(0);
439 const bool spiSuccess = spiPort.sendReceive(std::span<const uint8_t>(txBuffer.data(), bufferSize), std::span<uint8_t>(rxBuffer.data(), bufferSize),
440 std::chrono::milliseconds(OUTPUT_SPI_TIMEOUT));
441 nssPin.write(1);
442 RTOS::Task::yield();
443 if (!spiSuccess) return;
444 if (diagCounter == 0) oePin.write(1);
445 if (++diagCounter == 0) diagCounter = 1;
446 }
447
453 void processRxBuffer() {
454 if (!isSpiError && diagCounter > 0 && rxBuffer[bufferSize - 1] != txBuffer[0]) {
455 isSpiError = true;
456 if (onSpiError) onSpiError();
457 }
458 uint64_t newStatus = 0;
459 auto rxPorts = reinterpret_cast<MC33879::RxPort *>(rxBuffer.data() + 1);
460 const size_t ports = portCount();
461 for (size_t i = 0; i < pinCount; ++i) {
462 const auto & map = mappingFields[i];
463 const size_t p = static_cast<size_t>(map.port);
464 if (p < ports && (rxPorts[p].status & map.mask) != 0) newStatus |= (1ULL << i);
465 }
466 if (status != newStatus) {
467 status = newStatus;
468 if (onStatusChanged) onStatusChanged(status);
