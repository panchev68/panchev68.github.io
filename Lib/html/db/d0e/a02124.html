Example showing common usage of the conversion helpers provided by Lib::Helper::Convert::dBM. The example demonstrates converting a signed dBm-like value and a discrete index into percentage values suitable for UI presentation.

#include <iostream>
#include <cstdint>
#include <Lib/Helper/Convert/dBM.hpp>
void example_dBM_usage() {
// Convert a floating dBm-like value to percent
double dbm_value = -45.0;
auto percent = Lib::Helper::Convert::dBM::toPercent(dbm_value);
std::cout << "dBm " << dbm_value << " -> " << int(percent) << "%\n";
// Convert a discrete index (0..99) using lookup table
int8_t index = 10;
auto lookup_percent = Lib::Helper::Convert::dBM::toDigiPercent(index);
std::cout << "index " << int(index) << " -> " << int(lookup_percent) << "%\n";
}
