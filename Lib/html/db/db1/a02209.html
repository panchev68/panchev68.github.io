Lib::Helper::Convert::dBM

Demonstrates converting dBm-like values and discrete indices to percentage values using Lib::Helper::Convert::dBM helpers.

Demonstrates converting dBm-like values and discrete indices to percentage values using Lib::Helper::Convert::dBM helpers.This example shows how to use toPercent and toDigiPercent for UI-ready percentage values.

#include <cstdint>
#include <iostream>
#include "Lib/Helper/Convert/dBM.hpp"
using Lib::Helper::Convert::dBM;
void example_dBM_usage() {
// Convert a floating dBm-like value to percent
const double dbmValue = -45.0;
const uint8_t percent = dBM::toPercent(dbmValue);
std::cout << "dBm " << dbmValue << " -> " << int(percent) << "%\n"; // 55
// Edge cases
std::cout << "dBm -120 -> " << int(dBM::toPercent(-120.0)) << "%\n"; // 0
std::cout << "dBm 0 -> " << int(dBM::toPercent(0.0)) << "%\n"; // 100
// Convert a discrete index (0..99) using the lookup table; any integer type is accepted
const uint8_t index = 40;
const uint8_t lookupPercent = dBM::toDigiPercent(index);
std::cout << "index " << int(index) << " -> " << int(lookupPercent) << "%\n"; // 88
// Discrete index edge cases
std::cout << "index -1 -> " << int(dBM::toDigiPercent(-1)) << "%\n"; // 100
std::cout << "index 100 -> " << int(dBM::toDigiPercent(100)) << "%\n"; // 1
std::cout << "index 200 -> " << int(dBM::toDigiPercent(static_cast<uint8_t>(200))) << "%\n"; // 1
}
