116 Error parse(const uint8_t * data, size_t size, Command & command, uint8_t & option) {
117 if (!data || size == 0) return Error::EMPTY_MESSAGE;
118
119 size_t offset = 0;
120 while (size - offset >= sizeof(MessageHeader)) {
121 const auto & header = *reinterpret_cast<const MessageHeader *>(data + offset);
122 if (header.delimiter != START_DELIMITER) return Error::INVALID_START_DELIMITER;
123
124 handleIncomingMessageHeader(header);
125 command = static_cast<Command>(header.command);
126 option = header.option;
127
128 size_t payloadLength = header.length;
129 if (size - offset < sizeof(MessageHeader) + payloadLength) return Error::INVALID_PAYLOAD_LENGTH;
130
131 const uint8_t * payloadData = data + offset + sizeof(MessageHeader);
132 Error result = processCommand(command, payloadLength, payloadData);
133 if (result != Error::OK) return result;
134 offset += sizeof(MessageHeader) + payloadLength;
135 }
136
137 if (offset != size) return Error::INVALID_PAYLOAD_LENGTH;
138 return Error::OK;
139 }

177 Error processCommand(Command command, size_t payloadLength, const uint8_t * payloadData) {
178 switch (command) {
179 case Command::ECHO:
180 echoCallback();
181 break;
182 case Command::INPUTS:
183 if (payloadLength % sizeof(uint64_t) != 0) return Error::INVALID_PAYLOAD_LENGTH;
184 processNodeData(payloadLength, payloadData, true);
185 break;
186 case Command::OUTPUTS:
187 if (payloadLength % sizeof(uint64_t) != 0) return Error::INVALID_PAYLOAD_LENGTH;
188 processNodeData(payloadLength, payloadData, false);
189 break;
190 default:
191 return Error::COMMAND_NOT_SUPPORT;
192 }
193 return Error::OK;
194 }

205 void processNodeData(size_t payloadLength, const uint8_t * payloadData, bool isInputs) {
206 size_t nodesCount = payloadLength / sizeof(uint64_t);
207 updateNodeActivityCallback(nodesCount);
208 for (size_t i = 0; i < nodesCount; ++i) {
209 uint64_t value = 0;
210 std::memcpy(&value, payloadData + i * sizeof(uint64_t), sizeof(uint64_t));
211 if (isInputs) inputsUpdateCallback(static_cast<uint8_t>(i), value);
212 else handleOutputsUpdate(static_cast<uint8_t>(i), value);
213 }
