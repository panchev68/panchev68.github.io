Hexadecimal and decimal ASCII conversions for an unsigned integer type.

Hexadecimal and decimal ASCII conversions for an unsigned integer type."std::to_chars" is not used for the digits: it is not constexpr before C++23, emits lowercase and does not pad, so the fixed-width uppercase form would need a second pass anyway.

Template Parameters
TUnsigned integer type to convert, such as "uint8_t", "uint16_t", "uint32_t" or "uint64_t"; "bool" is rejected.

#include "Ascii.hpp"
#include "HexHelper.hpp"
#include <algorithm>
#include <array>
#include <concepts>
#include <cstddef>
#include <cstdint>
#include <limits>
namespace Lib::Helper::Convert {
template<std::unsigned_integral T> requires(!std::same_as<T, bool>) class Hex final {
public:
static constexpr uint32_t NIBBLE_BITS = 4U;
static constexpr uint32_t HEX_DIGIT_COUNT = sizeof(T) * 2U;
static constexpr uint32_t DECIMAL_DIGIT_COUNT = std::numeric_limits<T>::digits10 + 1U;
static constexpr uint32_t BCD_MAX_VALUE = HexHelper::BCD_MAX_VALUE;
using HexDigits = std::array<char, HEX_DIGIT_COUNT + 1U>;
using DecimalDigits = std::array<char, DECIMAL_DIGIT_COUNT + 1U>;
#pragma region Conversions
[[nodiscard]] static constexpr uint32_t toBcd(T number) noexcept requires(sizeof(T) <= sizeof(uint32_t)) {
return HexHelper::toBcd(number);
}
[[nodiscard]] static constexpr HexDigits toString(T number) noexcept {
HexDigits digits{};
T remaining = number;
for (size_t index = HEX_DIGIT_COUNT; index > 0U; --index) {
digits[index - 1U] = Ascii::fromHex(static_cast<uint8_t>(remaining & Ascii::NIBBLE_MASK));
remaining = static_cast<T>(remaining >> NIBBLE_BITS);
}
return digits;
}
[[nodiscard]] static constexpr DecimalDigits toBcdString(T number, bool showZero = false) noexcept {
constexpr T DECIMAL_BASE = 10U;
DecimalDigits digits{};
const auto end = digits.begin() + DECIMAL_DIGIT_COUNT;
auto first = end;
T remaining = number;
do {
*--first = static_cast<char>('0' + (remaining % DECIMAL_BASE));
remaining = static_cast<T>(remaining / DECIMAL_BASE);
} while (remaining != 0U);
if (showZero) {
std::fill(digits.begin(), first, '0');
} else {
const auto length = end - first;
std::shift_left(digits.begin(), end, first - digits.begin());
std::fill(digits.begin() + length, end, '\0');
}
return digits;
}
#pragma endregion
private:
#pragma region Deleted constructors and operators
Hex() = delete;
~Hex() = delete;
Hex(const Hex &) = delete;
Hex & operator=(const Hex &) = delete;
#pragma endregion
};
} // namespace Lib::Helper::Convert
