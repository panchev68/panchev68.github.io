149 [[nodiscard]] bool resetChip() {
150 pinReset.write(false);
151 RTOS::delay(RESET_LOW_TIME);
152 pinReset.write(true);
153
154 RTOS::Timeout timeout;
155 timeout.set(RTOS::millisecondsToTicks(RESET_TIMEOUT.count()));
156
157 while (!timeout.isExpired()) {
158 if (verifyChipId()) {
159 chipState = State::READY;
160 RTOS::delay(READY_DELAY);
161 return true;
162 }
163 RTOS::delay(POLL_PERIOD);
164 }
165
166 assert(!"LAN9252 Base: the chip did not report the expected identifier after reset");
167 chipState = State::ERROR;
168 return false;
169 }

198 const std::array<uint8_t, 3> command{Registers::CMD_SERIAL_READ, hiByte(address), loByte(address)};

328 if (buffer == nullptr || length == 0U || length > CSR_MAX_LENGTH) {
329 assert(!"LAN9252 Base: readCSR() requires a valid buffer and a length of 1..4");

385 [[nodiscard]] bool readPRAM(uint16_t address, uint8_t * buffer, uint16_t length) {
386 if (buffer == nullptr || length == 0U) {
387 assert(!"LAN9252 Base: readPRAM() requires a valid buffer and a non-zero length");
388 return false;
389 }
390
391 if (!writeU32(Registers::PRAM_READ_CMD_REG, Registers::PRAM_RW_ABORT_MASK) || !waitPramIdle(Registers::PRAM_READ_CMD_REG)) {
392 return false;
393 }
394
395 const std::array<uint8_t, TRANSFER_SETUP_SIZE> setup = makeTransferSetup(address, length);
396 if (!writeBuffer(Registers::PRAM_READ_ADDR_LEN_REG, setup.data(), setup.size()) || !waitPramSpace(Registers::PRAM_READ_CMD_REG)) {
397 return false;
398 }
399
400 uint32_t firstWord = 0U;
401 if (!readU32(Registers::PRAM_READ_FIFO_REG, firstWord)) {
402 return false;
403 }
404
405 /*
406 * The first FIFO word only carries the bytes from the requested offset onwards, so the leading partial
407 * word is copied out before the aligned burst loop takes over.
408 */
409 const std::array<uint8_t, WORD_SIZE> firstBytes = Helper::toBytes(firstWord);
410 const size_t bytePos = address & WORD_OFFSET_MASK;
411 size_t remaining = length;
412 size_t copied = std::min(WORD_SIZE - bytePos, remaining);
413 std::copy_n(firstBytes.begin() + static_cast<ptrdiff_t>(bytePos), copied, buffer);
414 remaining -= copied;
415
416 if (remaining == 0U) {
417 return true;
418 }
419
420 if (!spi.select()) {
421 return false;
422 }
423 bool isTransferred = spi.write(Registers::CMD_FAST_READ) && sendAddress(Registers::PRAM_READ_FIFO_REG) && spi.write(Registers::CMD_FAST_READ_DUMMY);
424
425 while (isTransferred && remaining > 0U) {
426 uint32_t word = 0U;
427 isTransferred = readBurstMode(word);
428 if (!isTransferred) {
429 break;
430 }
431 const std::array<uint8_t, WORD_SIZE> wordBytes = Helper::toBytes(word);
432 const size_t partLength = std::min(WORD_SIZE, remaining);
433 std::copy_n(wordBytes.begin(), partLength, buffer + copied);
434 copied += partLength;
435 remaining -= partLength;
436 }
437 spi.deselect();
438
439 if (!isTransferred) {
440 assert(!"LAN9252 Base: readPRAM() burst transfer failed");
441 }
442 return isTransferred;
443 }

458 [[nodiscard]] bool writePRAM(const uint8_t * buffer, uint16_t address, uint16_t length) {
459 if (buffer == nullptr || length == 0U) {
460 assert(!"LAN9252 Base: writePRAM() requires a valid buffer and a non-zero length");
461 return false;
462 }
463
464 if (!writeU32(Registers::PRAM_WRITE_CMD_REG, Registers::PRAM_RW_ABORT_MASK) || !waitPramIdle(Registers::PRAM_WRITE_CMD_REG)) {
465 return false;
466 }
467
468 const std::array<uint8_t, TRANSFER_SETUP_SIZE> setup = makeTransferSetup(address, length);
469 if (!writeBuffer(Registers::PRAM_WRITE_ADDR_LEN_REG, setup.data(), setup.size()) || !waitPramSpace(Registers::PRAM_WRITE_CMD_REG)) {
470 return false;
471 }
472
473 uint32_t status = 0U;
474 if (!readU32(Registers::PRAM_WRITE_CMD_REG, status)) {
475 return false;
476 }
477 uint8_t available = static_cast<uint8_t>(status & Registers::PRAM_SPACE_AVBL_COUNT_MASK);
478
479 const size_t bytePos = address & WORD_OFFSET_MASK;
480 size_t remaining = length;
481 size_t consumed = std::min(WORD_SIZE - bytePos, remaining);
482
483 std::array<uint8_t, WORD_SIZE> word{};
484 std::copy_n(buffer, consumed, word.begin() + static_cast<ptrdiff_t>(bytePos));
485 if (!writeU32(Registers::PRAM_WRITE_FIFO_REG, Helper::fromBytes<uint32_t>(word))) {
486 return false;
487 }
488 available = (available > 0U) ? static_cast<uint8_t>(available - 1U) : 0U;
489 remaining -= consumed;
490
491 if (remaining == 0U) {
492 return true;
493 }
494
495 if (!spi.select()) {
496 return false;
497 }
498 bool isTransferred = spi.write(Registers::CMD_SERIAL_WRITE) && sendAddress(Registers::PRAM_WRITE_FIFO_REG);
499
500 while (isTransferred && remaining > 0U) {
501 if (available == 0U) {
502 /*
503 * The FIFO is full. Release the bus so the space counter can be re-read through a framed register
504 * access, then resume the burst once the device reports room again.
505 */
506 spi.deselect();
507 if (!waitPramSpace(Registers::PRAM_WRITE_CMD_REG) || !readU32(Registers::PRAM_WRITE_CMD_REG, status)) {
508 return false;
509 }
510 available = static_cast<uint8_t>(status & Registers::PRAM_SPACE_AVBL_COUNT_MASK);
511 if (available == 0U) {
512 assert(!"LAN9252 Base: the PRAM write FIFO reported no free space");
513 return false;
514 }
515 if (!spi.select()) {
516 return false;
517 }
518 isTransferred = spi.write(Registers::CMD_SERIAL_WRITE) && sendAddress(Registers::PRAM_WRITE_FIFO_REG);
519 continue;
520 }
521
522 const size_t partLength = std::min(WORD_SIZE, remaining);
523 word.fill(0U);
524 std::copy_n(buffer + consumed, partLength, word.begin());
525 isTransferred = writeBurstMode(Helper::fromBytes<uint32_t>(word));
526 consumed += partLength;
527 remaining -= partLength;
528 available = static_cast<uint8_t>(available - 1U);
529 }
530 spi.deselect();
531
532 if (!isTransferred) {
533 assert(!"LAN9252 Base: writePRAM() burst transfer failed");
534 }
535 return isTransferred;
536 }

583 [[nodiscard]] bool waitCsrIdle() {
584 RTOS::Timeout timeout;
585 timeout.set(RTOS::millisecondsToTicks(REGISTER_TIMEOUT.count()));
586
587 for (;;) {
588 uint32_t status = 0U;
589 if (!readU32(Registers::ESC_CSR_CMD_REG, status)) {
590 return false;
591 }
592 if ((status & Registers::ESC_CSR_BUSY) == 0U) {
593 return true;
594 }
595 if (timeout.isExpired()) {
596 assert(!"LAN9252 Base: the ESC CSR interface stayed busy");
597 return false;
598 }
599 }
