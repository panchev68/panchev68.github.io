Lib::Helper::Convert::Hex

Example demonstrating conversions provided by Lib::Helper::Convert::Hex for uint8_t, uint16_t, uint32_t and uint64_t.

Example demonstrating conversions provided by Lib::Helper::Convert::Hex for uint8_t, uint16_t, uint32_t and uint64_t. Shows hex string, packed BCD value and decimal string output. Every string result is a fixed, NUL-terminated std::array, so it is printed through data().

#include <cstdint>
#include <iostream>
#include "Lib/Helper/Convert/Hex.hpp"
using namespace Lib::Helper::Convert;
void example_Hex_usage() {
// uint8_t: hex string, packed BCD and decimal string
const uint8_t v8 = 0xAB; // 171 decimal
std::cout << "Hex<uint8_t>::toString(0xAB) = " << Hex<uint8_t>::toString(v8).data() << '\n';
std::cout << "Hex<uint8_t>::toBcd(171) = 0x" << std::hex << Hex<uint8_t>::toBcd(171) << std::dec << '\n';
std::cout << "Hex<uint8_t>::toBcdString(5, true) = " << Hex<uint8_t>::toBcdString(5, true).data() << '\n';
// uint16_t: 4-char hex and decimal string
const uint16_t v16 = 0x1234;
std::cout << "Hex<uint16_t>::toString(0x1234) = " << Hex<uint16_t>::toString(v16).data() << '\n';
std::cout << "Hex<uint16_t>::toBcdString(1023) = " << Hex<uint16_t>::toBcdString(1023).data() << '\n';
// uint32_t: 8-char hex and decimal string
const uint32_t v32 = 0xDEADBEEF;
std::cout << "Hex<uint32_t>::toString(0xDEADBEEF) = " << Hex<uint32_t>::toString(v32).data() << '\n';
std::cout << "Hex<uint32_t>::toBcdString(1234567890) = " << Hex<uint32_t>::toBcdString(1234567890).data()
<< '\n';
// uint64_t: 16-char hex and decimal string
const uint64_t v64 = 0x0123456789ABCDEFULL;
std::cout << "Hex<uint64_t>::toString(0x0123456789ABCDEF) = " << Hex<uint64_t>::toString(v64).data() << '\n';
std::cout << "Hex<uint64_t>::toBcdString(UINT64_MAX) = " << Hex<uint64_t>::toBcdString(UINT64_MAX).data()
<< '\n';
}
