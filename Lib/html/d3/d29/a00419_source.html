89 [[nodiscard]] bool open() noexcept {
90 if (!events.open()) {
91 return false;
92 }
93 if (!Base::open()) {
94 return false;
95 }
96 SPI_HandleTypeDef * const handle = this->getHandle();
97
98 // Register HAL callbacks to route events into this instance. '&=' never short-circuits, so a failing
99 // registration still leaves the remaining ones attempted.
100 [[maybe_unused]] bool isCallbacksRegistered = (HAL_SPI_RegisterCallback(handle, CallbackID::HAL_SPI_TX_COMPLETE_CB_ID, txCompleteISR) == HAL_OK);
101 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, CallbackID::HAL_SPI_RX_COMPLETE_CB_ID, rxCompleteISR) == HAL_OK);
102 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, CallbackID::HAL_SPI_TX_RX_COMPLETE_CB_ID, txRxCompleteISR) == HAL_OK);
103 isCallbacksRegistered &= (HAL_SPI_RegisterCallback(handle, CallbackID::HAL_SPI_ERROR_CB_ID, errorISR) == HAL_OK);
104 assert(isCallbacksRegistered && "SPIPort<DMA>: failed to register one or more HAL SPI callbacks");
105
106 if (!DeviceManager<SPIPort, SPI_HandleTypeDef>::instance().registerDevice(this)) {
107 assert(!"SPIPort<DMA>: failed to register SPI device with DeviceManager");
108 return false;
109 }
110 return true;
111 }

119 [[nodiscard]] bool close() noexcept {
120 SPI_HandleTypeDef * const handle = this->getHandle();
121
122 DeviceManager<SPIPort, SPI_HandleTypeDef>::instance().unregisterDevice(this);
123
124 HAL_SPI_UnRegisterCallback(handle, CallbackID::HAL_SPI_TX_COMPLETE_CB_ID);
125 HAL_SPI_UnRegisterCallback(handle, CallbackID::HAL_SPI_RX_COMPLETE_CB_ID);
126 HAL_SPI_UnRegisterCallback(handle, CallbackID::HAL_SPI_TX_RX_COMPLETE_CB_ID);
127 HAL_SPI_UnRegisterCallback(handle, CallbackID::HAL_SPI_ERROR_CB_ID);

129 const bool result = Base::close();
130 events.close();

150 [[nodiscard]] bool abort() noexcept {
151 SPI_HandleTypeDef * const handle = this->getHandle();
152
153 if (handle == nullptr) [[unlikely]] {
154 assert(!"SPIPort<DMA>: abort() requires a valid native SPI handle");
155 return false;

157 // Report the abort as an error event so a task blocked in waitForEvent() wakes up immediately instead of

160 static_cast<void>(events.setBits(EventBits(Event::ERROR)));

199 [[nodiscard]] bool sendHandler(std::span<const uint8_t> data, std::chrono::milliseconds timeout) noexcept {
200 SPI_HandleTypeDef * const handle = this->getHandle();
201
202 if (handle == nullptr) [[unlikely]] {
203 assert(!"SPIPort<DMA>: sendHandler() requires an open native SPI handle");
204 return false;
205 }
206 static_cast<void>(events.clearBits(EventBits(Event::TX_COMPLETE | Event::ERROR)));
207 if (HAL_SPI_Transmit_DMA(handle, const_cast<uint8_t *>(data.data()), static_cast<uint16_t>(data.size())) != HAL_OK) {
208 return false;
209 }
210 const uint32_t result = waitForEvent(Event::TX_COMPLETE | Event::ERROR, RTOS::TickCounter::msToTicks(static_cast<uint32_t>(timeout.count())));
211 return ((result & Event::TX_COMPLETE) != 0U) && ((result & Event::ERROR) == 0U);
212 }
