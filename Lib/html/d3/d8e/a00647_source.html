107 [[nodiscard]] bool setup(TickType period = DEFAULT_PERIOD, bool autoStart = true) noexcept {
108 // FIX: A zero period trips configASSERT() inside the kernel, so reject it here.
109 if (period == 0U) [[unlikely]] {
110 assert(!"TickClient: period must be greater than zero");
111 return false;
112 }
113 if (handle == nullptr) {
114 handle = xTimerCreate("TickClient", period, pdTRUE, static_cast<void *>(this), &timerCallback);
115 if (handle == nullptr) [[unlikely]] {
116 assert(!"TickClient: failed to create native timer");
117 return false;
118 }
119 const bool isStarted = autoStart ? start() : true;
120 assert(isStarted && "TickClient: failed to start timer");
121 return isStarted;
122 }
123
124 if (!changePeriod(period)) {
125 return false;
126 }
127 // 'xTimerChangePeriod()' implicitly (re)starts an inactive timer as a side effect, so it must be stopped
128 // again here to honor a caller-requested 'autoStart == false' on an already-created timer.
129 return autoStart ? true : stop();

293 [[nodiscard]] static TickType getCommandTimeout() noexcept {
294 // FIX: The timer service task cannot drain its own command queue, so blocking on it from a timer callback
295 // deadlocks once the queue is full.
296 if (xTaskGetSchedulerState() == taskSCHEDULER_NOT_STARTED) {
297 return TIMER_CMD_TIMEOUT;
298 }
299 return (xTaskGetCurrentTaskHandle() == xTimerGetTimerDaemonTaskHandle()) ? TickType{0U} : TIMER_CMD_TIMEOUT;
300 }
