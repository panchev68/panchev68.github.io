113 void modemStatusHandler(const Frame0x8a & frame) {
114 using ModemStatusEnum = Frame0x8a::ModemStatus;
115 auto emitEvent = [](std::function<void()> & event) {
116 if (event) event();
117 };
118 switch (frame.modemStatus) {
119 case ModemStatusEnum::HARDWARE_RESET:
120 emitEvent(onHardwareReset);
121 break;
122 case ModemStatusEnum::WATCHDOG_TIMER_RESET:
123 emitEvent(onWatchdogTimerReset);
124 assert(!"Watchdog timer reset triggered");
125 break;
126 case ModemStatusEnum::JOINED_NETWORK:
127 emitEvent(onJoinedNetwork);
128 break;
129 case ModemStatusEnum::LEFT_NETWORK:
130 emitEvent(onLeftNetwork);
131 break;
132 case ModemStatusEnum::COORDINATOR_STARTED:
133 emitEvent(onCoordinatorStarted);
134 break;
135 case ModemStatusEnum::NETWORK_SECURITY_KEY_WAS_UPDATED:
136 emitEvent(onNetworkSecurityKeyWasUpdated);
137 break;
138 case ModemStatusEnum::NETWORK_WOKE_UP:

140 break;
141 case ModemStatusEnum::NETWORK_WENT_TO_SLEEP:
142 emitEvent(onNetworkWentToSleep);
143 break;
144 case ModemStatusEnum::VOLTAGE_SUPPLY_LIMIT_EXCEEDED:
145 emitEvent(onVoltageSupplyLimitExceeded);
146 break;
147 case ModemStatusEnum::REMOTE_MANAGER_CONNECTED:
148 emitEvent(onRemoteManagerConnected);
149 break;
150 case ModemStatusEnum::REMOTE_MANAGER_DISCONNECTED:
151 emitEvent(onRemoteManagerDisconnected);
152 break;
153 case ModemStatusEnum::MODEM_CONFIGURATION_CHANGED_WHILE_JOIN_IN_PROGRESS:
154 emitEvent(onModemConfigurationChangedWhileJoinInProgress);
155 break;
156 case ModemStatusEnum::ACCESS_FAULT:
157 emitEvent(onAccessFault);
158 assert(!"Access fault occurred");
159 break;
160 case ModemStatusEnum::FATAL_ERROR:
161 emitEvent(onFatalError);
162 assert(!"Fatal error occurred");
163 break;
164 case ModemStatusEnum::SECURE_SESSION_SUCCESSFULLY_ESTABLISHED:
165 emitEvent(onSecureSessionSuccessfullyEstablished);

167 case ModemStatusEnum::SECURE_SESSION_ENDED:
168 emitEvent(onSecureSessionEnded);
169 break;
170 case ModemStatusEnum::SECURE_SESSION_AUTHENTICATION_FAILED:
171 emitEvent(onSecureSessionAuthenticationFailed);
172 break;
173 case ModemStatusEnum::COORDINATOR_DETECTED_A_PAN_ID_CONFLICT_BUT_TOOK_NO_ACTION:
174 emitEvent(onCoordinatorDetectedAPanIdConflictButTookNoAction);

194 emitEvent(onCellularComponentUpdateFailed);
195 break;
196 case ModemStatusEnum::CELLULAR_COMPONENT_UPDATE_COMPLETED:
197 emitEvent(onCellularComponentUpdateCompleted);
198 break;
199 case ModemStatusEnum::XBEE_FIRMWARE_UPDATE_STARTED:
200 emitEvent(onXbeeFirmwareUpdateStarted);
201 break;

203 emitEvent(onXbeeFirmwareUpdateFailed);
