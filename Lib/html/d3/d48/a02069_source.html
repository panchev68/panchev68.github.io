159 void update() override {
160 using State = TaskFuncData::State;
161 using DeliveryStatus = TaskFuncData::DeliveryStatus;
162
163 auto & d = taskData;
164
165 auto isConnected = [this]() {
166 return connection.state == Connection::State::CONNECTED;
167 };
168 auto disconnect = [this, &d] {
169 d.timeoutConnection.set(0);
170 connection.state = Connection::State::DISCONNECTED;
171 };
172 auto connect = [this] {
173 connection.state = Connection::State::CONNECTED;
174 };
175 auto refreshTimeout = [&d] {
176 d.timeoutConnection.set(ApplicationBase::CONNECTION_TIMEOUT);
177 };
178
179 while (true) {
180 switch (d.state) {
181 case State::BEGIN_TRANSACTION:
182 if (!d.timeoutTransaction.isExpired()) break;
183 d.timeoutTransaction.set(TRANSACTION_PACING);
184 d.state = State::SEND_MESSAGE;
185 continue;
186
187 case State::SEND_MESSAGE:
188 if (isConnected()) connection.info.transmitPeriod.beginMeasurement();
189 handleOutgoingMessage(d.txMessageBuffer);
190
191 if (queueTransmitDeliveryStatus.receive(0)) {
192 connection.info.deletedTransmitStatus++;
193 }
194
195 if (!sendHandler(d.txMessageBuffer)) {
196 d.state = State::ERROR;
197 break;
198 }
199 d.state = State::WAIT_FOR_SENDING_COMPLETED;
200 continue;
201
202 case State::WAIT_FOR_SENDING_COMPLETED:
203 if (const auto status = queueTransmitDeliveryStatus.receive(MAX_RESPONSE_TIME / 2)) {
204 d.transmitDeliveryStatus = *status;
205 d.isError = (d.transmitDeliveryStatus != DeliveryStatus::SUCCESS);
206 if (d.isError) {
207 if (d.transmitDeliveryStatus == DeliveryStatus::NO_ACK_RECEIVED) {
208 disconnect();
209 RTOS::delay(ApplicationBase::PAUSE_BETWEEN_MESSAGES_IN_DISCONNECTED);
210 } else {
211 connection.info.unsuccessTransmitStatus++;
212 }
213 }
214 } else {
215 connection.info.noTransmitStatus++;
216 d.isError = true;
217 }
218
219 if (d.isError) {
220 d.state = State::ERROR;
221 break;
222 }
223
224 if (queueRxMessage.receive(0)) {
225 connection.info.deletedMessages++;
226 }
227 d.state = State::WAITING_FOR_RECEIVE_INCOMING_MESSAGE;
228 continue;
229
230 case State::WAITING_FOR_RECEIVE_INCOMING_MESSAGE:
231 if (const auto rxMessage = queueRxMessage.receive(ApplicationBase::MAX_RESPONSE_TIME)) {
232 d.rxMessage = *rxMessage;
233 if (handleIncomingMessage(d.rxMessage)) {
234 if (!isConnected()) connect();
235 connection.info.transmitPeriod.endMeasurement();
236 outputsUpdatedSucceessfulCallback();
237 refreshTimeout();
238 d.isError = false;
239 } else {
240 d.isError = true;
241 }
242 } else {
243 d.isError = true;
244 if (isConnected()) connection.info.retriesCounter++;
245 else RTOS::delay(ApplicationBase::PAUSE_BETWEEN_MESSAGES_IN_DISCONNECTED);
246 }
247
248 if (d.isError) {
249 d.state = State::ERROR;
250 break;
251 }
252 d.state = State::END_TRANSACTION;
253 continue;
254
255 case State::END_TRANSACTION:
256 if (d.timeoutConnection.isExpired() && isConnected()) {
257 connection.info.signalStrength = 0;
258 connection.info.transmitPeriod.reset();
259 connection.info.reconnectionCounter++;
260 connection.state = Connection::State::DISCONNECTED;
261 }
262 d.state = State::BEGIN_TRANSACTION;
263 break;
264
265 case State::ERROR:
266 d.state = State::END_TRANSACTION;
267 break;
268 }
