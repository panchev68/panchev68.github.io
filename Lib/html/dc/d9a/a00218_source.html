115 void modemStatusHandler(const Frame0x8a & frame) {
116 using ModemStatusEnum = Frame0x8a::ModemStatus;
117 auto emitEvent = [](FunctionRef<void()> & event) {
118 if (event) event();
119 };
120 switch (frame.modemStatus) {
121 case ModemStatusEnum::HARDWARE_RESET:
122 emitEvent(onHardwareReset);
123 break;
124 case ModemStatusEnum::WATCHDOG_TIMER_RESET:
125 emitEvent(onWatchdogTimerReset);
126 assert(!"Watchdog timer reset triggered");
127 break;
128 case ModemStatusEnum::JOINED_NETWORK:
129 emitEvent(onJoinedNetwork);
130 break;
131 case ModemStatusEnum::LEFT_NETWORK:
132 emitEvent(onLeftNetwork);
133 break;
134 case ModemStatusEnum::COORDINATOR_STARTED:
135 emitEvent(onCoordinatorStarted);
136 break;
137 case ModemStatusEnum::NETWORK_SECURITY_KEY_WAS_UPDATED:
138 emitEvent(onNetworkSecurityKeyWasUpdated);
139 break;
140 case ModemStatusEnum::NETWORK_WOKE_UP:
141 emitEvent(onNetworkWokeUp);
142 break;
143 case ModemStatusEnum::NETWORK_WENT_TO_SLEEP:
144 emitEvent(onNetworkWentToSleep);
145 break;
146 case ModemStatusEnum::VOLTAGE_SUPPLY_LIMIT_EXCEEDED:
147 emitEvent(onVoltageSupplyLimitExceeded);
148 break;
149 case ModemStatusEnum::REMOTE_MANAGER_CONNECTED:
150 emitEvent(onRemoteManagerConnected);
151 break;
152 case ModemStatusEnum::REMOTE_MANAGER_DISCONNECTED:
153 emitEvent(onRemoteManagerDisconnected);
154 break;
155 case ModemStatusEnum::MODEM_CONFIGURATION_CHANGED_WHILE_JOIN_IN_PROGRESS:
156 emitEvent(onModemConfigurationChangedWhileJoinInProgress);
157 break;
158 case ModemStatusEnum::ACCESS_FAULT:
159 emitEvent(onAccessFault);
160 assert(!"Access fault occurred");
161 break;
162 case ModemStatusEnum::FATAL_ERROR:
163 emitEvent(onFatalError);
164 assert(!"Fatal error occurred");
165 break;
166 case ModemStatusEnum::SECURE_SESSION_SUCCESSFULLY_ESTABLISHED:
167 emitEvent(onSecureSessionSuccessfullyEstablished);

169 case ModemStatusEnum::SECURE_SESSION_ENDED:
170 emitEvent(onSecureSessionEnded);
171 break;
172 case ModemStatusEnum::SECURE_SESSION_AUTHENTICATION_FAILED:
173 emitEvent(onSecureSessionAuthenticationFailed);
174 break;
175 case ModemStatusEnum::COORDINATOR_DETECTED_A_PAN_ID_CONFLICT_BUT_TOOK_NO_ACTION:
176 emitEvent(onCoordinatorDetectedAPanIdConflictButTookNoAction);

186 emitEvent(onBLEDisconnect);
187 break;

193 emitEvent(onCellularComponentUpdateStarted);

196 emitEvent(onCellularComponentUpdateFailed);
197 break;
198 case ModemStatusEnum::CELLULAR_COMPONENT_UPDATE_COMPLETED:
199 emitEvent(onCellularComponentUpdateCompleted);
200 break;
201 case ModemStatusEnum::XBEE_FIRMWARE_UPDATE_STARTED:
202 emitEvent(onXbeeFirmwareUpdateStarted);
203 break;
204 case ModemStatusEnum::XBEE_FIRMWARE_UPDATE_FAILED:
205 emitEvent(onXbeeFirmwareUpdateFailed);
206 assert(!"XBee firmware update failed");
