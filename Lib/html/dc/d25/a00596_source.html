90 [[nodiscard]] constexpr float update(const Measurement & measurement) noexcept {
91 const float currentPower = measurement.voltage * measurement.current;
92
93 if (!isInitialized) [[unlikely]] {
94 previousPower = currentPower;
95 isInitialized = true;
96 voltageSetpoint = clampVoltage(voltageSetpoint);
97 return voltageSetpoint;
98 }
99
100 const float deltaPower = currentPower - previousPower;
101 const float deltaMagnitude = (deltaPower < 0.0F) ? -deltaPower : deltaPower;
102
103 // Change only if the difference exceeds the configured noise threshold.
104 if (deltaMagnitude > config.powerThreshold) {
105 // If power decreased, reverse direction; otherwise continue.
106 if (deltaPower < 0.0F) {
107 lastDirection = (lastDirection == Direction::UP) ? Direction::DOWN : Direction::UP;
108 }
109
110 voltageSetpoint = clampVoltage(voltageSetpoint + (directionSign(lastDirection) * config.stepSize));
111
112 /*
113 * The reference is refreshed only here, so it always holds the power measured before the most recent
114 * perturbation - which is exactly what the next delta must be compared against. Refreshing it on every
115 * call instead would reduce the delta to a sample-to-sample difference, and a drift slower than
116 * 'powerThreshold' per sample would then never accumulate enough to move the setpoint at all.
117 */
118 previousPower = currentPower;
119 }
120 return voltageSetpoint;
121 }
