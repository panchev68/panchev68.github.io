114 [[nodiscard]] Status update() noexcept {
115 const bool hasOffSensor = offSensor.isPresent();
116 const bool hasOnSensor = onSensor.isPresent();
117 const bool isOffActive = offSensor.read() == Sensor::State::ACTIVE;
118 const bool isOnActive = onSensor.read() == Sensor::State::ACTIVE;
119
120 switch (state) {
121 case State::UNKNOW:
122 output = false;
123 if (hasOffSensor && isOffActive && !(hasOnSensor && isOnActive)) {
124 state = State::OFF;
125 } else if (hasOnSensor && isOnActive && !(hasOffSensor && isOffActive)) {
126 state = State::ON;
127 }
128 break;
129 case State::OFF:
130 output = false;
131 if (hasOffSensor && !isOffActive) {
132 state = State::OFF_LEAVING;
133 calmingTimeout.set(calmingTicks);
134 }
135 break;
136 case State::OFF_LEAVING:
137 output = false;
138 if (isOffActive) {
139 state = State::OFF;
140 } else if (calmingTimeout.isExpired()) [[unlikely]] {
141 raiseError(Error::UNCOMMANDED_OFF);
142 }
143 break;
144 case State::OFF_CALMING:
145 output = false;
146 if (!isOffActive) {
147 state = State::MOVE_TO_OFF;
148 switchTimeout.set(switchTimeoutTicks);
149 } else if (calmingTimeout.isExpired()) {
150 state = State::OFF;
151 }
152 break;
153 case State::MOVE_TO_OFF:
154 output = false;
155 if (hasOffSensor) {
156 if (isOffActive) {
157 state = State::OFF_CALMING;
158 calmingTimeout.set(calmingTicks);
159 } else if (switchTimeout.isExpired()) [[unlikely]] {
160 raiseError(Error::TIMEOUT_OFF);
161 }
162 } else if (switchTimeout.isExpired()) {
163 state = State::OFF;
164 }
165 break;
166 case State::MOVE_TO_ON:
167 output = true;
168 if (hasOnSensor) {
169 if (isOnActive) {
170 state = State::ON_CALMING;
171 calmingTimeout.set(calmingTicks);
172 } else if (switchTimeout.isExpired()) [[unlikely]] {
173 raiseError(Error::TIMEOUT_ON);
174 }
175 } else if (switchTimeout.isExpired()) {
176 state = State::ON;
177 }
178 break;
179 case State::ON_CALMING:
180 output = true;
181 if (!isOnActive) {
182 state = State::MOVE_TO_ON;
183 switchTimeout.set(switchTimeoutTicks);
184 } else if (calmingTimeout.isExpired()) {
185 state = State::ON;
186 }
187 break;
188 case State::ON:
189 output = true;
190 if (hasOnSensor && !isOnActive) {
191 state = State::ON_LEAVING;
192 calmingTimeout.set(calmingTicks);
193 }
194 break;
195 case State::ON_LEAVING:
196 output = true;
197 if (isOnActive) {
198 state = State::ON;
199 } else if (calmingTimeout.isExpired()) [[unlikely]] {
200 raiseError(Error::UNCOMMANDED_ON);
201 }
202 break;
203 case State::ERROR:
204 output = outputLevel;
205 if (hasOffSensor && hasOnSensor) {
206 isErrorRecoverable = (isOffActive != isOnActive);
207 } else if (hasOffSensor) {
208 isErrorRecoverable = isOffActive;
209 } else {
210 isErrorRecoverable = isOnActive;
211 }
212 break;
213 }
214
215 outputLevel = output;
216 status = deriveStatus();
217 return status;
218 }

228 [[nodiscard]] bool command(const Command command) noexcept {
229 const bool isTargetOn = (command == Command::ON);
230 const State targetState = isTargetOn ? State::MOVE_TO_ON : State::MOVE_TO_OFF;
231 const bool isAlreadyThere = isTargetOn
232 ? (state == State::ON || state == State::ON_CALMING ||
233 state == State::MOVE_TO_ON)
234 : (state == State::OFF || state == State::OFF_CALMING ||
235 state == State::MOVE_TO_OFF);
236 if (isAlreadyThere) {
237 return true;
238 }
239 error = Error::NONE;
240 state = targetState;
241 switchTimeout.set(switchTimeoutTicks);
242 return true;
243 }
