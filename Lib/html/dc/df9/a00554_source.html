50 [[nodiscard]] static double calculateEquationOfTime(double t) noexcept {
51 const double meanLongitude = degToRad(calculateMeanLongitude(t));
52 const double meanAnomaly = degToRad(calculateMeanAnomaly(t));
53 const double eccentricity = 0.016708634 - t * (0.000042037 + 0.0000001267 * t);
54 // FIX: Uses the corrected obliquity of date instead of a fixed 23.439 degrees.
55 const double tanHalfObliquity = std::tan(degToRad(calculateObliquity(t)) / 2.0);
56 const double y = tanHalfObliquity * tanHalfObliquity;
57 const double sin2L0 = std::sin(2.0 * meanLongitude);
58 const double sinM = std::sin(meanAnomaly);
59 // FIX: The second-order terms in 4 L0 and 2 M of the NOAA formula were missing.
60 const double equation = y * sin2L0 - 2.0 * eccentricity * sinM + 4.0 * eccentricity * y * sinM * std::cos(2.0 * meanLongitude) -
61 0.5 * y * y * std::sin(4.0 * meanLongitude) - 1.25 * eccentricity * eccentricity * std::sin(2.0 * meanAnomaly);
62 return MINUTES_PER_DEGREE * radToDeg(equation);
63 }

71 [[nodiscard]] static double calculateSunDeclination(double t) noexcept {
72 const double meanAnomaly = degToRad(calculateMeanAnomaly(t));
73 // FIX: The equation of center lacked its 2 M and 3 M terms, and nutation and aberration were ignored.
74 const double center = std::sin(meanAnomaly) * (1.914602 - t * (0.004817 + 0.000014 * t)) +
75 std::sin(2.0 * meanAnomaly) * (0.019993 - 0.000101 * t) + std::sin(3.0 * meanAnomaly) * 0.000289;
76 const double omega = degToRad(125.04 - 1934.136 * t);
77 const double apparentLongitude = degToRad(calculateMeanLongitude(t) + center - 0.00569 - 0.00478 * std::sin(omega));
78 return radToDeg(std::asin(std::sin(degToRad(calculateObliquity(t))) * std::sin(apparentLongitude)));
79 }

187 [[nodiscard]] double computeHourAngle(double dec, double angle) const noexcept {
188 const double decRad = degToRad(dec);
189 const double cosH = (std::cos(degToRad(angle)) - (sinLat * std::sin(decRad))) / (cosLat * std::cos(decRad));
190 return std::acos(std::clamp(cosH, -1.0, 1.0));

278 [[nodiscard]] static double calculateObliquity(double t) noexcept {
279 const double seconds = 21.448 - t * (46.815 + t * (0.00059 - t * 0.001813));
280 const double meanObliquity = 23.0 + (26.0 + seconds / 60.0) / 60.0;
281 return meanObliquity + 0.00256 * std::cos(degToRad(125.04 - 1934.136 * t));

292 [[nodiscard]] double calculateSunTimeAt(double t, double angle, bool isSunrise) const noexcept {
293 const double hourAngle = computeHourAngle(calculateSunDeclination(t), angle);
294 const double delta = geoLongitude + radToDeg(isSunrise ? hourAngle : -hourAngle);
295 return SOLAR_NOON_UTC - (MINUTES_PER_DEGREE * delta) - calculateEquationOfTime(t);
