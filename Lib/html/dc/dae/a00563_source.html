191 [[nodiscard]] constexpr int32_t convertLocalToUtcMinutes(int32_t year, int32_t month, int32_t day, int32_t hour, int32_t minute,
192 std::time_t offsetSeconds) const noexcept {
193 using Minutes64 = std::chrono::duration<int64_t, std::ratio<60>>;
194
195 if (year < FIRST_YEAR || year > LAST_YEAR || month < 1 || month > 12 || day < 1 || hour < 0 || hour > 23 || minute < 0 ||
196 minute > 59) [[unlikely]] {
197 return INVALID_MINUTES;
198 }
199 const std::chrono::year_month_day date{std::chrono::year{year} / std::chrono::month{static_cast<unsigned>(month)} /
200 std::chrono::day{static_cast<unsigned>(day)}};
201 if (!date.ok()) [[unlikely]] {
202 // FIX: Out-of-range days (e.g. 31 April) were silently normalized by mktime() instead of rejected.
203 return INVALID_MINUTES;
204 }
205
206 // FIX: mktime() applied the C runtime zone and DST and ignored this zone's own offset entirely.
207 const std::chrono::seconds localTimeOfDay = std::chrono::hours{hour} + std::chrono::minutes{minute} - offset;
208 // Midnight is widened to 64-bit seconds first; a sum in 32-bit minutes would overflow in far-future years.
209 const std::chrono::sys_seconds utc = std::chrono::sys_seconds{std::chrono::sys_days{date}} + localTimeOfDay;
210 const std::chrono::seconds sinceEpoch = utc.time_since_epoch() - std::chrono::seconds{static_cast<int64_t>(offsetSeconds)};
211 // FIX: Flooring instead of truncating keeps results before the caller epoch on the correct minute.
212 const int64_t result = std::chrono::floor<Minutes64>(sinceEpoch).count();
213 if (result < std::numeric_limits<int32_t>::min() || result > std::numeric_limits<int32_t>::max()) [[unlikely]] {
214 return INVALID_MINUTES;
215 }
216 return static_cast<int32_t>(result);
217 }
