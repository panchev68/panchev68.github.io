163 [[nodiscard]] bool readWrite(const void * source, size_t sourceSize, void * destination, size_t destinationSize) {
164 const size_t totalLength = sourceSize + destinationSize;
165
166 if (totalLength == 0U || totalLength > MAX_BUFFER_SIZE) {
167 assert(!"LAN9252 Interface: readWrite() length is zero or exceeds MAX_BUFFER_SIZE");
168 return false;
169 }
170 if ((sourceSize > 0U && source == nullptr) || (destinationSize > 0U && destination == nullptr)) {
171 assert(!"LAN9252 Interface: readWrite() received a null buffer for a non-zero length");
172 return false;
173 }
174
175 std::fill_n(txBuffer.begin(), totalLength, PAD_BYTE);
176
177 if (sourceSize > 0U) {
178 const auto * sourceBytes = static_cast<const uint8_t *>(source);
179 std::copy_n(sourceBytes, sourceSize, txBuffer.begin());
180 }
181
182 if (!spiPort.sendReceive(std::span<const uint8_t>(txBuffer.data(), totalLength), std::span<uint8_t>(rxBuffer.data(), totalLength),
183 std::chrono::milliseconds(SPI_TIMEOUT))) {
184 return false;
185 }
186
187 if (destinationSize > 0U) {
188 std::copy_n(rxBuffer.begin() + static_cast<ptrdiff_t>(sourceSize), destinationSize, static_cast<uint8_t *>(destination));
189 }
190 return true;
191 }
