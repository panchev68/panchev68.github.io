198 static uint8_t parseHexByte(char high, char low) {
199 uint8_t highNibble = 0, lowNibble = 0;
200 if (high >= '0' && high <= '9') highNibble = high - '0';
201 else if (high >= 'A' && high <= 'F') highNibble = high - 'A' + 10;
202 else if (high >= 'a' && high <= 'f') highNibble = high - 'a' + 10;
203 else return UINT8_MAX;
204 if (low >= '0' && low <= '9') lowNibble = low - '0';
205 else if (low >= 'A' && low <= 'F') lowNibble = low - 'A' + 10;
206 else if (low >= 'a' && low <= 'f') lowNibble = low - 'a' + 10;
207 else return UINT8_MAX;
208 return (highNibble << 4) | lowNibble;
209 }
