Lib::Helper::Convert::Hex

Example demonstrating conversions provided by Lib::Helper::Convert::Hex specializations for uint8_t, uint16_t, uint32_t and uint64_t.

Example demonstrating conversions provided by Lib::Helper::Convert::Hex specializations for uint8_t, uint16_t, uint32_t and uint64_t. Shows hex string, BCD value and formatted BCD string output.

#include <iostream>
#include "Hex.hpp"
using namespace Lib::Helper::Convert;
void example_Hex_usage() {
// uint8_t: hex string, BCD and BCD string
uint8_t v8 = 0xAB; // 171 decimal
std::cout << "Hex<uint8_t>::toString(0xAB) = " << Hex<uint8_t>::toString(v8) << '\n';
std::cout << "Hex<uint8_t>::toBcd(171) = 0x" << Hex<uint8_t>::toString(static_cast<uint8_t>(Hex<uint8_t>::toBcd(171) >> 24)) << "..." << '\n';
std::cout << "Hex<uint8_t>::toBcdString(5) = " << Hex<uint8_t>::toBcdString(5, true) << '\n';
// uint16_t: 4-char hex and BCD string
uint16_t v16 = 0x1234;
std::cout << "Hex<uint16_t>::toString(0x1234) = " << Hex<uint16_t>::toString(v16) << '\n';
std::cout << "Hex<uint16_t>::toBcdString(1023) = " << Hex<uint16_t>::toBcdString(1023) << '\n';
// uint32_t: 8-char hex and BCD string
uint32_t v32 = 0xDEADBEEF;
std::cout << "Hex<uint32_t>::toString(0xDEADBEEF) = " << Hex<uint32_t>::toString(v32) << '\n';
std::cout << "Hex<uint32_t>::toBcdString(1234567890) = " << Hex<uint32_t>::toBcdString(1234567890) << '\n';
// uint64_t: 16-char hex
uint64_t v64 = 0x0123456789ABCDEFULL;
std::cout << "Hex<uint64_t>::toString(0x0123456789ABCDEF) = " << Hex<uint64_t>::toString(v64) << '\n';
}
