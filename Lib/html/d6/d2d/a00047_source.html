144 [[nodiscard]] EventCode read() {
145 if (!isInitialized) {
146 assert(!"EncoderTimer: read() requires a successfully opened driver");
147 return EventCode::NONE;
148 }
149
150 if (const auto pinState = buttonQueue.receive(0U)) {
151 if (!*pinState) {
152 eventCode = EventCode::BUTTON_PULL;
153 } else if (eventCode != EventCode::BUTTON_PUSH) {
154 eventCode = EventCode::BUTTON_PUSH;
155 buttonHoldTimeout.set(RTOS::millisecondsToTicks(BUTTON_HOLD_TIME.count()));
156 }
157 return eventCode;
158 }
159
160 const uint32_t position = timer.read();
161 if (position != rawPosition) {
162 /*
163 * The hardware direction bit is authoritative: comparing counter magnitudes would misreport the step
164 * where the counter wraps between 0 and the auto-reload value.
165 */
166 const bool isClockwise = timer.isClockwise();
167 rawPosition = position;
168 return isClockwise ? EventCode::ENCODER_CW : EventCode::ENCODER_CCW;
169 }
170
171 if (eventCode == EventCode::BUTTON_PUSH && buttonHoldTimeout.isExpired()) {
172 eventCode = EventCode::BUTTON_HOLD;
173 return EventCode::BUTTON_HOLD;
174 }
175 return EventCode::NONE;
176 }
