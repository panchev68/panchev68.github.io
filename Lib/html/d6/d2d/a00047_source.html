148 [[nodiscard]] bool getButtonState(uint32_t stableSamples = DEFAULT_STABLE_SAMPLES) {
149 if (stableSamples == 0U) {
150 return !button.read();
151 }
152
153 bool state = button.read();
154 uint32_t matches = 0U;
155
156 for (uint32_t attempt = 0U; attempt < (stableSamples * MAX_SAMPLE_ATTEMPTS); ++attempt) {
157 RTOS::delay(SAMPLE_PERIOD);
158 const bool currentState = button.read();
159
160 if (currentState != state) {
161 state = currentState;
162 matches = 0U;
163 continue;
164 }
165 matches++;
166 if (matches >= stableSamples) {
167 break;
168 }
169 }
170 return !state;
171 }

300 void write(EventState state) {
301 if (!isOpen()) {
302 return;
303 }
304
305 if (state != lastState) {
306 lastState = state;
307 rotationEventCounter = 0;
308 if (state == EventState::BUT_PUSH) {
309 buttonStartTick = xTaskGetTickCountFromISR();
310 } else if (state == EventState::BUT_PULL) {
311 buttonHoldPeriod = ticksToMilliseconds(xTaskGetTickCountFromISR() - buttonStartTick);
312 }
313 }
314
315 const bool isRotationEvent = (state == EventState::CCW || state == EventState::CW);
316 if (!isRotationEvent || (++rotationEventCounter <= pulseDivider)) {
317 return;
318 }
319 rotationEventCounter = 0;
320
321 if (reversePolarity) {
322 state = (state == EventState::CCW) ? EventState::CW : EventState::CCW;
323 }
324
325 BaseType_t higherPriorityTaskWoken = pdFALSE;
326 const bool isSent = queue.sendFromISR(state, higherPriorityTaskWoken);
327 assert(isSent && "EncoderPins: event queue full, dropped event");
328 portYIELD_FROM_ISR(higherPriorityTaskWoken);
