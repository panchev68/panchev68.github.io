Lib::Helper::TimeZone

Constructs a Lib::Helper::TimeZone, updates it and converts local calendar time to UTC minutes since the epoch.

Constructs a Lib::Helper::TimeZone, updates it and converts local calendar time to UTC minutes since the epoch.The conversion shifts the local time by the zone offset and does not use the C runtime time zone, so it runs in task or main context without the RTOS and is also usable at compile time. A result of TimeZone::INVALID_MINUTES reports invalid input. Daylight saving time is resolved separately through TimeZone::TimeChangeRule.

See also
Lib::Helper::TimeZone::convertLocalToUtcMinutes, Lib::Helper::TimeZone::TimeChangeRule
Example usage:
#include <Lib/Helper/TimeZone.hpp>
#include <chrono>
#include <cstdint>
#include <ctime>
void timezoneExample() {
using namespace std::chrono_literals;
// Central European Time, UTC+60 minutes
constexpr Lib::Helper::TimeZone tz("CET", 60);
// Local 2026-05-11 12:00 CET is 11:00 UTC; an epoch offset of 0 counts from 1970-01-01 00:00 UTC
constexpr std::time_t epochOffsetSeconds = 0;
const int32_t utcMinutes = tz.convertLocalToUtcMinutes(2026, 5, 11, 12, 0, epochOffsetSeconds);
if (utcMinutes == Lib::Helper::TimeZone::INVALID_MINUTES) {
// handle invalid input (bad date or time of day)
}
// Summer time begins on the last Sunday of March at 02:00 local time, UTC+120 minutes afterwards
using Rule = Lib::Helper::TimeZone::TimeChangeRule;
constexpr Rule summerTime{"CEST", Rule::LAST_WEEK, 0, 3, 2, 120};
const std::chrono::local_seconds summerStart = summerTime.getLocalTransition(std::chrono::year{2026});
(void)summerStart; // 2026-03-29 02:00 local time
// Copy the zone and switch it to summer time
Lib::Helper::TimeZone summerZone = tz;
summerZone.setName(summerTime.ruleName);
summerZone.setOffset(summerTime.getOffset());
const int32_t summerUtcMinutes = summerZone.convertLocalToUtcMinutes(2026, 5, 11, 12, 0, epochOffsetSeconds);
(void)summerUtcMinutes; // 10:00 UTC
}
