98 [[nodiscard]] bool getDateTime(Time & time, Date & date) const {
99 // getHandle() resolves to the const overload here; HAL_RTC_Get*() require a mutable pointer.
100 RTC_HandleTypeDef * handle = const_cast<RTC_HandleTypeDef *>(getHandle());
101 if (handle == nullptr) {
102 assert(!"Rtc: getDateTime() requires a valid native RTC handle");
103 return false;
104 }
105
106 RTC_TimeTypeDef halTime{};
107 RTC_DateTypeDef halDate{};
108
109 if (HAL_RTC_GetTime(handle, &halTime, RTC_FORMAT_BIN) != HAL_OK) {
110 return false;
111 }
112
113 // The shadow registers stay locked until the date is read as well, so this call is mandatory
114 if (HAL_RTC_GetDate(handle, &halDate, RTC_FORMAT_BIN) != HAL_OK) {
115 return false;
116 }
117
118 time.hours = halTime.Hours;
119 time.minutes = halTime.Minutes;
120 time.seconds = halTime.Seconds;
121
122 date.year = halDate.Year;
123 date.month = halDate.Month;
124 date.day = halDate.Date;
125 date.weekDay = halDate.WeekDay;

180 void startTickService(const std::function<void(const Time &, const Date &)> & callback) {
181 if (!callback) {
182 assert(!"Rtc: startTickService() requires a callable observer");
183 return;
184 }
185 if (isServiceRunning) {
186 return;
187 }
188
189 userCallback = callback;
190 isServiceRunning = true;
191
192 // Configure hardware Alarm A to trigger mask on every second change natively
193 setupHardwareSecondsAlarm();
194
195 tickTask.setup({"RtcEventWorker", TICK_TASK_STACK_SIZE, RTOS::Task::Priority::NORMAL});
196 if (tickTask.start([this] { taskMethod(); }) == nullptr) {
197 // Without a worker the alarm would keep firing into nothing, so undo the whole setup
198 assert(!"Rtc: failed to create the RTC tick worker task");
199 deactivateHardwareAlarm();
200 isServiceRunning = false;
201 userCallback = nullptr;
202 }
203 }

259 void setupHardwareSecondsAlarm() {
260 RTC_HandleTypeDef * handle = getHandle();
261 if (handle == nullptr) {
262 assert(!"Rtc: setupHardwareSecondsAlarm() requires a valid native RTC handle");
263 return;
264 }
265 RTC_AlarmTypeDef alarm{};
266 alarm.AlarmTime.Hours = 0U;
267 alarm.AlarmTime.Minutes = 0U;
268 alarm.AlarmTime.Seconds = 0U;
269 alarm.AlarmSubSecondMask = 0U;
270 alarm.AlarmTime.DayLightSaving = RTC_DAYLIGHTSAVING_NONE;
271 alarm.AlarmTime.StoreOperation = RTC_STOREOPERATION_RESET;
272
273 // Critical masks: ignore date, hours and minutes to match every minute, every second
274 alarm.AlarmMask = RTC_ALARMMASK_DATEWEEKDAY | RTC_ALARMMASK_HOURS | RTC_ALARMMASK_MINUTES;
275 alarm.AlarmSubSecondMask = RTC_ALARMSUBSECONDMASK_ALL;
276 alarm.AlarmDateWeekDaySel = RTC_ALARMDATEWEEKDAYSEL_DATE;
277 alarm.AlarmDateWeekDay = 1U;
278 alarm.Alarm = RTC_ALARM_A;
279
280 [[maybe_unused]] const bool isAlarmArmed = (HAL_RTC_SetAlarm_IT(handle, &alarm, RTC_FORMAT_BIN) == HAL_OK);
281 assert(isAlarmArmed && "Rtc: hardware refused the per-second Alarm A configuration");
282 }

298 void taskMethod() {
299 Time timeContext{};
300 Date dateContext{};
301
302 while (isServiceRunning) {
303 // PURE WAKE ON EVENT: Task sleeps efficiently until the hardware Alarm ISR triggers a notification
304 if (tickTask.notifyWait()) {
305 if (getDateTime(timeContext, dateContext) && userCallback) {
306 // Safe execution of user business logic inside Thread context
307 userCallback(timeContext, dateContext);
308 }
309 }
310 }
311 }
