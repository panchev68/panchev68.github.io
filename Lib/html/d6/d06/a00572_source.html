209 T_VALUE interpolate(T_INPUT x) {
210 assert(pointCount >= 2 && "at least 2 points required for interpolation");
211 ensureSplineReady();
212 if (x <= points[0].x) return points[0].y;
213 if (x >= points[pointCount - 1].x) return points[pointCount - 1].y;

227 T_VALUE getDerivative(T_INPUT x) {
228 assert(pointCount >= 2 && "at least 2 points required for derivative");
229 ensureSplineReady();
230 const bool isAtLeftEnd = x <= points[0].x;
231 if (isAtLeftEnd) return segments[0].b;
232 const bool isAtRightEnd = x >= points[pointCount - 1].x;
233 const size_t index = isAtRightEnd ? (pointCount - 2) : findInterval(x);
234 const T_VALUE dx = isAtRightEnd ? (points[pointCount - 1].x - points[index].x) : (x - points[index].x);
235 const T_VALUE c2 = 2 * segments[index].c;
236 const T_VALUE c3 = 3 * segments[index].d;
237 return segments[index].b + dx * (c2 + dx * c3);

281 void computeSplineCoefficients() {
282 if (isComputed || pointCount < 2) return;
283 if (pointCount == 2) {
284 const T_VALUE slope = (points[1].y - points[0].y) / (points[1].x - points[0].x);
285 segments[0] = SplineSegment(points[0].y, slope, T_VALUE{}, T_VALUE{});
286 isComputed = true;
287 return;
288 }
289 const size_t n = pointCount - 1;
290 std::array<T_VALUE, MAX_POINTS> h, alpha, l, mu, z, c;
291 for (size_t i = 0; i < n; ++i) {
292 h[i] = points[i + 1].x - points[i].x;
293 assert(h[i] > T_VALUE{} && "point x-coordinates must be strictly increasing");
294 }
295 for (size_t i = 1; i < n; ++i) alpha[i] = (3.0 / h[i]) * (points[i + 1].y - points[i].y) - (3.0 / h[i - 1]) * (points[i].y - points[i - 1].y);
296 if (boundaryType == BoundaryType::NATURAL) {
297 l[0] = T_VALUE{1};
298 mu[0] = T_VALUE{};
299 z[0] = T_VALUE{};
300 l[n] = T_VALUE{1};
301 z[n] = T_VALUE{};
302 c[n] = T_VALUE{};
303 } else {
304 l[0] = 2 * h[0];
305 mu[0] = T_VALUE{0.5};
306 alpha[0] = (3.0 / h[0]) * (points[1].y - points[0].y) - 3 * leftDerivative;
307 z[0] = alpha[0] / l[0];
308 l[n] = h[n - 1] * (2 - mu[n - 1]);
309 alpha[n] = 3 * rightDerivative - (3.0 / h[n - 1]) * (points[n].y - points[n - 1].y);
310 z[n] = (alpha[n] - h[n - 1] * z[n - 1]) / l[n];
311 c[n] = z[n];
312 }
313 for (size_t i = 1; i < n; ++i) {
314 l[i] = 2 * (points[i + 1].x - points[i - 1].x) - h[i - 1] * mu[i - 1];
315 mu[i] = h[i] / l[i];
316 z[i] = (alpha[i] - h[i - 1] * z[i - 1]) / l[i];
317 }
318 for (size_t j = n - 1; j > 0; --j) c[j] = z[j] - mu[j] * c[j + 1];
319 c[0] = boundaryType == BoundaryType::NATURAL ? T_VALUE{} : z[0] - mu[0] * c[1];
320 for (size_t j = 0; j < n; ++j) {
321 const T_VALUE a = points[j].y;
322 const T_VALUE b = (points[j + 1].y - points[j].y) / h[j] - h[j] * (c[j + 1] + 2 * c[j]) / 3;
323 const T_VALUE d = (c[j + 1] - c[j]) / (3 * h[j]);
324 segments[j] = SplineSegment(a, b, c[j], d);
325 }
326 isComputed = true;
327 }

338 size_t findInterval(T_INPUT x) const {
339 assert(pointCount >= 2 && "at least 2 points required");
340 assert(isSorted && "points must be sorted");
341 auto it = std::lower_bound(points.begin(), points.begin() + pointCount, x, [](const Point & point, const T_INPUT value) { return point.x < value; });
342 const size_t index = static_cast<size_t>(std::distance(points.begin(), it));
343 return (index > 0) ? (index - 1) : 0;
344 }

353 static constexpr T_VALUE evaluateSplineSegment(const SplineSegment & segment, T_VALUE dx) {
