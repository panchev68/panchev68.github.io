199 inline bool MFRC522::stopTask() {
200 if (!task.isCreated()) {
201 return false;
202 }
203
204 stopRequested.store(true);
205
206 for (std::chrono::milliseconds waited{0}; task.isCreated(); waited += STOP_POLL_DELAY) {
207 if (waited >= STOP_TIMEOUT) {
208 assert(!"MFRC522: the polling task did not stop within the timeout");
209 return false;
210 }
211 RTOS::delay(STOP_POLL_DELAY);
212 }
213
214 chipset.setAntenna(false);
215 return true;
216 }

222 inline bool MFRC522::start() {
223 constexpr std::chrono::milliseconds READY_POLL_DELAY{10};
224 constexpr std::chrono::milliseconds READY_TIMEOUT{100};
225
226 if (task.isCreated()) {
227 assert(!"MFRC522: start() called while the polling task is already running");
228 return false;
229 }
230 stopRequested.store(false);
231
232 if (!io.initialize()) {
233 assert(!"MFRC522: failed to open SPI transport");
234 return false;
235 }
236 io.hardwareReset();
237
238 const uint8_t version = chipset.readRegister(Chipset::Register::VERSION);
239
240 if (version == VERSION_ALL_ZEROES || version == VERSION_ALL_ONES) {
241 assert(!"MFRC522: communication failed; check wiring and SPI power");
242 return false;
243 }
244
245 if (!chipset.softReset()) {
246 assert(!"MFRC522: soft reset failed; check hardware and power supply");
247 return false;
248 }
249
250 chipset.toIdleState();
251 for (std::chrono::milliseconds waited{0}; !chipset.isReady(); waited += READY_POLL_DELAY) {
252 if (waited >= READY_TIMEOUT) {
253 assert(!"MFRC522: device did not become ready within timeout");
254 return false;
255 }
256 RTOS::delay(READY_POLL_DELAY);
257 }
258
259 chipset.initTimer();
260 chipset.applyInitialGain();
261 chipset.setAntenna(true);
262
263 return task.start([this]() { taskEntry(); });
264 }

266 inline void MFRC522::taskEntry() {
267 constexpr std::chrono::milliseconds POLL_DELAY_WITH_CARD{500};
268 constexpr std::chrono::milliseconds POLL_DELAY_NO_CARD{100};
269
270 while (!stopRequested.load()) {
271 if (!chipset.isCardPresent()) {
272 RTOS::delay(POLL_DELAY_NO_CARD);
273 continue;
274 }
275
276 std::array<uint8_t, Chipset::UID_LENGTH> uid{};
277 if (chipset.readUID(uid) && onCardDetected.isBound()) {
278 onCardDetected(std::span<const uint8_t>(uid));
279 }
280
281 chipset.clearCardDetection();
282 card.halt();
283 RTOS::delay(POLL_DELAY_WITH_CARD);
284 }
285 }
