149 void handleTickEvent() {
150 switch (state) {
151 case State::IDLE:
152 break;
153 case State::START:
154 startTimer();
155 timeout.set(duration);
156 state = State::WAIT_FOR_COMPLETED;
157 break;
158 case State::WAIT_FOR_COMPLETED:
159 if (!timeout.isExpired()) break;
160 stopTimer();
161 if (--count == 0) {

165 timeout.set(pause);
