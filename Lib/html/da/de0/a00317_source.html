170 [[nodiscard]] uint32_t read(std::chrono::milliseconds timeout) noexcept {
171 if (!start()) {
172 return 0U;
173 }

176 if (((result & Event::CONVERSION_COMPLETE) == 0U) || ((result & Event::ERROR) != 0U)) [[unlikely]] {
