195 State update() {
196 if (!connection || connection->state != Connection::State::CONNECTED) {
197 state = State::DISCONNECTED;
198 percents = 0;
199 return state;
200 }
201
202 if (timeout.isExpired()) {
203 timeout.set(RESET_CYCLE_PERIOD);
204 lastRetriesCounter = connection->info.retriesCounter;
205 if (mode == Mode::RUNTIME) {
206 if (retriesSubst > 0) retriesSubst--;
207 if (transmitTimeSubst > 0) transmitTimeSubst--;
208 if (rssiSubst > 0) rssiSubst--;
209 }
210 }
211
212 if (connection->info.retriesCounter > lastRetriesCounter) {
213 retriesSubst = std::min<int32_t>(100, retriesSubst + 50);
214 lastRetriesCounter = connection->info.retriesCounter;
215 }
216
217 uint32_t currentTransmit = connection->info.transmitPeriod.getRawValue();
218 if (lastTransmitPeriod != currentTransmit) {
219 lastTransmitPeriod = currentTransmit;
220 if (currentTransmit > TRANSMIT_TIME_THRESHOLD) {
221 transmitTimeSubst = std::min<int32_t>(100, transmitTimeSubst + 5);
222 }
223 }
224
225 uint8_t currentRssi = connection->info.signalStrength;
226 lastRssi = static_cast<uint8_t>(((lastRssi * 2) + currentRssi) / 3);
227
228 if (lastRssi < RSSI_MIN_THRESHOLD) {
229 rssiSubst = RSSI_MIN_THRESHOLD - lastRssi;
230 }
231
232 int32_t result = 100 - retriesSubst - transmitTimeSubst - rssiSubst;
233 percents = static_cast<uint8_t>(std::max<int32_t>(0, result));
234
235 if (percents > 90) state = State::AMAZING;
236 else if (percents > 75) state = State::VERY_GOOD;
237 else if (percents > 50) state = State::OKAY;
238 else if (percents > 30) state = State::NOT_GOOD;
239 else state = State::UNUSABLE;
240
241 if (onUpdate) onUpdate(this);
242 return state;
243 }
