144 bool writeHandler(uint32_t identifier, const uint8_t * data, uint8_t length) {
145 CAN_HandleTypeDef * nativeHandle = getHandle();

160 txHeader.StdId = (identifier < 0x800U) ? identifier : 0U;
161 txHeader.ExtId = (identifier >= 0x800U) ? identifier : 0U;
162 txHeader.IDE = (identifier < 0x800U) ? CAN_ID_STD : CAN_ID_EXT;
163 txHeader.RTR = CAN_RTR_DATA;
164 txHeader.DLC = length;
165 txHeader.TransmitGlobalTime = DISABLE;
166
167 uint32_t txMailbox = 0U;
168 return HAL_CAN_AddTxMessage(nativeHandle, &txHeader, const_cast<uint8_t *>(data), &txMailbox) == HAL_OK;
