Lib::Helper::Convert::Ascii

Demonstrates ASCII validation and hexadecimal conversion helpers provided by Lib::Helper::Convert::Ascii.

Demonstrates ASCII validation and hexadecimal conversion helpers provided by Lib::Helper::Convert::Ascii.Shows checking characters for ASCII range, decimal and hex digit tests, and converting between hex characters and numeric values using the static constexpr APIs.

Example usage:
#include "Lib/Helper/Convert/Ascii.hpp"
#include <cstdint>
void asciiExample() {
// Validate ASCII and printable range
const char c1 = 'A';
const bool isAscii = Lib::Helper::Convert::Ascii::isAscii(c1);
const bool isPrintable = Lib::Helper::Convert::Ascii::isPrintableAscii(c1);
// Decimal digit test
const char d = '7';
const bool isDigit = Lib::Helper::Convert::Ascii::isDigit(d);
// Hex digit tests and conversions
const char h1 = 'F';
const char h2 = 'a';
if (Lib::Helper::Convert::Ascii::isHexDigit(h1) && Lib::Helper::Convert::Ascii::isHexDigit(h2)) {
const uint8_t val1 = Lib::Helper::Convert::Ascii::toHex(h1); // 15
const uint8_t val2 = Lib::Helper::Convert::Ascii::toHex(h2); // 10
// Compose a byte from two hex characters (high nibble then low nibble)
const uint8_t byte = static_cast<uint8_t>((val1 << 4) | val2);
(void)byte; // use the computed value as needed
// Convert back to uppercase hex characters
const char outHigh = Lib::Helper::Convert::Ascii::fromHex((byte >> 4) & 0x0F);
const char outLow = Lib::Helper::Convert::Ascii::fromHex(byte & 0x0F);
(void)outHigh; (void)outLow;
}
(void)isAscii; (void)isPrintable; (void)isDigit;
}
