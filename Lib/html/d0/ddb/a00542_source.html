76 constexpr uint32_t calculate(const Date & date) noexcept {
77 assert(date.isValid() && "MoonPhase::calculate: date must be valid");
78
79 // FIX: The former Julian day formula mixed a March-based month index with the January-based Meeus terms and
80 // omitted the Gregorian correction, shifting the result by roughly 80 days; std::chrono is exact.
81 const int64_t noon = static_cast<int64_t>(date.toSysDays().time_since_epoch().count()) * MICRODAYS_PER_DAY + MICRODAYS_PER_DAY / 2;
82 int64_t age = (noon - REFERENCE_NEW_MOON) % SYNODIC_MONTH;
83 if (age < 0) {
84 age += SYNODIC_MONTH;
85 }
86
87 phaseIndex = static_cast<uint32_t>((age * PHASE_COUNT + SYNODIC_MONTH / 2) / SYNODIC_MONTH) % PHASE_COUNT;
88 phaseName = PHASE_NAMES[phaseIndex];
89 return phaseIndex;
90 }
