87 [[nodiscard]] bool open() noexcept {
88 if (!events.open()) {
89 return false;
90 }
91 if (!Base::open()) {
92 return false;
93 }
94 ADC_HandleTypeDef * const handle = this->getHandle();
95
96 // Register HAL callbacks to route events into this instance. '&=' never short-circuits, so a failing
97 // registration still leaves the remaining ones attempted.
98 [[maybe_unused]] bool isCallbacksRegistered =
99 (HAL_ADC_RegisterCallback(handle, CallbackID::HAL_ADC_CONVERSION_COMPLETE_CB_ID, conversionCompleteISR) == HAL_OK);
100 isCallbacksRegistered &= (HAL_ADC_RegisterCallback(handle, CallbackID::HAL_ADC_ERROR_CB_ID, errorISR) == HAL_OK);
101 assert(isCallbacksRegistered && "ADCPort<DMA>: failed to register one or more HAL ADC callbacks");
102
103 if (!DeviceManager<ADCPort, ADC_HandleTypeDef>::instance().registerDevice(this)) {
104 assert(!"ADCPort<DMA>: failed to register ADC device with DeviceManager");
105 return false;
106 }
107 return true;
108 }

116 [[nodiscard]] bool close() noexcept {
117 ADC_HandleTypeDef * const handle = this->getHandle();
118
119 DeviceManager<ADCPort, ADC_HandleTypeDef>::instance().unregisterDevice(this);
120
121 HAL_ADC_UnRegisterCallback(handle, CallbackID::HAL_ADC_CONVERSION_COMPLETE_CB_ID);
122 HAL_ADC_UnRegisterCallback(handle, CallbackID::HAL_ADC_ERROR_CB_ID);
123
124 const bool result = Base::close();
125 events.close();

135 [[nodiscard]] bool start(std::span<uint32_t> destination) noexcept {
136 ADC_HandleTypeDef * const handle = this->getHandle();
137 if (handle == nullptr) [[unlikely]] {
138 assert(!"ADCPort<DMA>: start() requires an open native ADC handle");
139 return false;
140 }
141 if (destination.empty()) [[unlikely]] {
142 assert(!"ADCPort<DMA>: start() requires a non-empty destination buffer");
143 return false;
144 }
145 static_cast<void>(events.clearBits(EventBits(Event::ALL)));
146 return HAL_ADC_Start_DMA(handle, destination.data(), static_cast<uint32_t>(destination.size())) == HAL_OK;
147 }
