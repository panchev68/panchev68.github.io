122 void taskEntry() {
123 Socket::Error error;
124 size_t length;
125 DataHandle<> incomingData;
126 DataHandle<> outgoingData;
127
128 setConnectionState(ConnectionState::INIT);
129 initCallback();
130
131 while (true) {
132 FreeRTOS::delay(1);
133 while (!readLinkState()) {
134 if (getConnectionState() == ConnectionState::CONNECTED) {
135 setConnectionState(ConnectionState::DISCONNECTED);
136 if ((error = sockets[0].close()) != Socket::Error::OK) assert(0);
137 }
138 FreeRTOS::delay(100);
139 }
140 if (getConnectionState() != ConnectionState::CONNECTED) {
141 if ((error = sockets[0].open(Socket::Protocol::TCP, remote.port, 0)) == Socket::Error::OK) {
142 if ((error = sockets[0].connect(remote)) == Socket::Error::OK) {
143 setConnectionState(ConnectionState::CONNECTED);
144 } else {
145 if ((error = sockets[0].close()) != Socket::Error::OK) assert(0);
146 FreeRTOS::delay(100);
147 }
148 } else {
149 assert(0);
150 }
151 } else {
152 if ((length = sockets[0].getReceivedSize()) > 0) {
153 updateRxLedCallback(true);
154 diagnostic.rxMessageCounter++;
155 incomingData(new uint8_t[length], length);
156 if ((error = sockets[0].receive(incomingData, length)) == Socket::Error::OK) {
157 handleIncomingMessage(incomingData);
158 } else if (error != Socket::Error::ERR_DATALEN) {
159 if ((error = sockets[0].close()) != Socket::Error::OK) assert(0);
160 }
161 delete incomingData.data();
162 }
163 if (handleOutgoingMessage(outgoingData)) {
164 if ((error = sockets[0].send(outgoingData, length)) == Socket::Error::OK) {
165 updateTxLedCallback(true);
166 diagnostic.txMessageCounter++;
167 } else {
168 if ((error = sockets[0].close()) != Socket::Error::OK) assert(0);
169 setConnectionState(ConnectionState::DISCONNECTED);
170 }
171 }
172 }
173 taskYIELD();
174 handleApplicationProcess();
175 }
