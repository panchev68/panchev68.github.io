Lib::Helper::Convert::HexHelper

Example showing common usage of the lightweight conversion helpers provided by Lib::Helper::Convert::HexHelper. The example demonstrates converting a nibble to its ASCII hex representation, packing a 0..99 value into an 8-bit BCD, packing a small decimal into a 32-bit BCD, and converting an ASCII hex digit back to its numeric nibble.

#include <iostream>
#include <cstdint>
#include "HexHelper.hpp"
void example_HexHelper_usage() {
// Convert low nibble 0xA to ASCII 'A'
char c = Lib::Helper::Convert::HexHelper::toAscii(0xAu);
std::cout << "Nibble 0xA -> " << c << "\n"; // Expected: A
// Pack decimal 42 into 8-bit BCD (0x42)
uint8_t bcd8 = Lib::Helper::Convert::HexHelper::toBcd8(42u);
std::cout << "BCD8(42) -> 0x" << std::hex << int(bcd8) << std::dec << "\n"; // Expected: 0x42
// Pack decimal 1234 into 32-bit packed BCD
uint32_t bcd32 = Lib::Helper::Convert::HexHelper::toBcd(static_cast<uint16_t>(1234u));
std::cout << "BCD32(1234) -> 0x" << std::hex << bcd32 << std::dec << "\n"; // Expected: packed BCD representing 1234
// Convert ASCII 'F' to numeric nibble (15)
uint8_t nib = Lib::Helper::Convert::HexHelper::fromAscii('F');
std::cout << "'F' -> " << int(nib) << "\n"; // Expected: 15
}
