157 void receiveTask() {
158 size_t bufferIndex = 0;
159 uint8_t value = 0;
160 for (;;) {
161 if (rxPort->receive(std::span<uint8_t>(&value, 1U), std::chrono::milliseconds(1))) {
162 if (bufferIndex < receiveHandler.size()) receiveHandler[bufferIndex++] = value;
163 else bufferIndex = 0;
164 } else if (bufferIndex > 0) {
165 handlePostReceivedEvent(receiveHandler.data(), bufferIndex);
