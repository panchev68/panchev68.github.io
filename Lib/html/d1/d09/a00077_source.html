147 void writeRegister(Register reg, uint8_t value) {
148 const uint8_t addr = (static_cast<uint8_t>(reg) << 1) & REGISTER_ADDRESS_WRITE_MASK;
149 const uint8_t txBuffer[2] = {addr, value};

160 void writeRegister(Register reg, uint8_t count, const uint8_t * values) {
161 if (values == nullptr || count == 0U) {
162 return;
163 }
164
165 const uint8_t addr = (static_cast<uint8_t>(reg) << 1) & REGISTER_ADDRESS_WRITE_MASK;
166 const uint8_t clampedCount = (count > MAX_MULTI_BYTE_COUNT) ? MAX_MULTI_BYTE_COUNT : count;
167
168 uint8_t txBuffer[BUFFER_SIZE] = {};
169 txBuffer[0] = addr;
170 std::copy(values, values + clampedCount, txBuffer + 1);
171
172 (void) io.send(txBuffer, clampedCount + 1U);
173 }

181 uint8_t readRegister(Register reg) {
182 const uint8_t addr = ((static_cast<uint8_t>(reg) << 1) | REGISTER_ADDRESS_READ_FLAG);
183 const uint8_t txBuffer[2] = {addr, 0x00};
184 uint8_t rxBuffer[2] = {};
185
186 if (!io.sendReceive(txBuffer, rxBuffer, sizeof(rxBuffer))) {
187 return 0x00;
188 }
189 return rxBuffer[1];
190 }

200 void readRegister(Register reg, uint8_t count, uint8_t * values, uint8_t rxAlign = 0U) {
201 if (values == nullptr || count == 0U) {
202 return;
203 }
204
205 // Per the MFRC522 SPI protocol, each transmitted byte carries the address for the NEXT byte to be
206 // clocked out. 'addr' must therefore be repeated for every byte requested; only the final byte sent is
207 // 0x00 (terminator). Sending 0x00 earlier would be read by the chip as a write to register 0, which is
208 // why a single leading 'addr' followed by zeroes previously produced garbage for count > 1.
209 const uint8_t addr = ((static_cast<uint8_t>(reg) << 1) | REGISTER_ADDRESS_READ_FLAG);
210 const uint8_t clampedCount = (count > MAX_MULTI_BYTE_COUNT) ? MAX_MULTI_BYTE_COUNT : count;
211
212 uint8_t txBuffer[BUFFER_SIZE] = {};
213 std::fill(txBuffer, txBuffer + clampedCount, addr);
214
215 uint8_t rxBuffer[BUFFER_SIZE] = {};
216 size_t rxLen = static_cast<size_t>(clampedCount) + 1U;
217
218 if (!io.sendReceive(txBuffer, rxBuffer, rxLen)) {
219 return;
220 }
221
222 uint8_t index = 0U;
223 if (rxAlign != 0U) {
224 const uint8_t mask = static_cast<uint8_t>(0xFFU << rxAlign);
225 values[0] = static_cast<uint8_t>((values[0] & static_cast<uint8_t>(~mask)) | (rxBuffer[1] & mask));
226 index = 1U;
227 }
228 for (; index < clampedCount; ++index) {
229 values[index] = rxBuffer[index + 1U];
230 }
231 }

258 bool softReset() {
259 constexpr std::chrono::milliseconds SOFT_RESET_SETTLE_DELAY{50};
260
261 writeRegister(Register::COMMAND, static_cast<uint8_t>(Command::SOFT_RESET));
262 RTOS::delay(SOFT_RESET_SETTLE_DELAY);
263 if ((readRegister(Register::COMMAND) & COMMAND_STATE_MASK) != static_cast<uint8_t>(Command::IDLE)) {
264 assert(!"MFRC522 did not reset to IDLE state after soft reset.");
265 return false;
266 }
267 return true;
268 }

316 bool isCardPresent() {

318

353 bool transceive(const uint8_t * txData, uint8_t txLength, uint8_t * rxData, uint8_t rxCapacity, uint8_t & rxLength, uint8_t txLastBits = 0U,
354 std::chrono::milliseconds timeout = std::chrono::milliseconds{30}) {
355 rxLength = 0U;
356
357 toIdleState();
358 writeRegister(Register::FIFO_LEVEL, FIFO_FLUSH);
359 writeRegister(Register::BIT_FRAMING, txLastBits & 0x07U);
360
361 if (txLength > 0U) {
362 writeRegister(Register::FIFO_DATA, txLength, txData);
363 }
364
365 writeRegister(Register::COMMAND, static_cast<uint8_t>(Command::TRANSCEIVE));
366 setRegisterBitMask(Register::BIT_FRAMING, BIT_FRAMING_START_SEND);
367
368 RTOS::delay(timeout);
369
370 if ((readRegister(Register::ERROR) & ERROR_MASK) != 0U) {
371 return false;
372 }
373
374 uint8_t count = readRegister(Register::FIFO_LEVEL);
375 if (count > rxCapacity) {
376 count = rxCapacity;
377 }
378 // readRegister()'s multi-byte overload silently caps a single transfer at MAX_MULTI_BYTE_COUNT bytes; clamp
379 // here too so 'rxLength' never overstates how many bytes were actually written into 'rxData'.
380 if (count > MAX_MULTI_BYTE_COUNT) {
381 count = MAX_MULTI_BYTE_COUNT;
382 }
383 if (count > 0U && rxData != nullptr) {
384 readRegister(Register::FIFO_DATA, count, rxData);
385 }
386 rxLength = count;
387 return true;
388 }

435 bool readUID(std::array<uint8_t, UID_LENGTH> & uid) {
436 constexpr std::chrono::milliseconds RESPONSE_WAIT{10};
437
438 writeRegister(Register::BIT_FRAMING, 0x00);
439
440 toIdleState();
441 writeRegister(Register::FIFO_LEVEL, FIFO_FLUSH);
442
443 writeRegister(Register::FIFO_DATA, PICC_CMD_SEL_CL1);
444 writeRegister(Register::FIFO_DATA, PICC_CMD_ANTICOLL_NVB);
445
446 writeRegister(Register::COMMAND, static_cast<uint8_t>(Command::TRANSCEIVE));
447 setRegisterBitMask(Register::BIT_FRAMING, BIT_FRAMING_START_SEND);
448
449 RTOS::delay(RESPONSE_WAIT);
450
451 if ((readRegister(Register::ERROR) & ERROR_MASK) != 0U) {
452 return false;
453 }
454
455 const uint8_t count = readRegister(Register::FIFO_LEVEL);
456
457 if (count != ANTICOLLISION_RESPONSE_LENGTH) {
458 return false;
459 }
460
461 std::array<uint8_t, ANTICOLLISION_RESPONSE_LENGTH> response = {};
462 for (uint8_t i = 0U; i < ANTICOLLISION_RESPONSE_LENGTH; ++i) {
463 response[i] = readRegister(Register::FIFO_DATA);
464 }
465
466 const uint8_t bcc = static_cast<uint8_t>(response[0] ^ response[1] ^ response[2] ^ response[3]);
467 if (bcc != response[UID_LENGTH]) {
468 return false;
469 }
470
471 std::copy(response.begin(), response.begin() + UID_LENGTH, uid.begin());
472 return true;
473 }

491 bool executeAuthentication(const uint8_t * authData, size_t authDataSize) {
492 constexpr std::chrono::milliseconds RESPONSE_WAIT{10};
493
494 if (authData == nullptr || authDataSize != AUTH_DATA_LENGTH) {
495 return false;
496 }
497
498 writeRegister(Register::COMMAND, static_cast<uint8_t>(Command::IDLE));
499 writeRegister(Register::FIFO_LEVEL, FIFO_FLUSH);
500
501 for (uint8_t i = 0U; i < AUTH_DATA_LENGTH; ++i) {
502 writeRegister(Register::FIFO_DATA, authData[i]);
503 }
504
505 writeRegister(Register::COMMAND, static_cast<uint8_t>(Command::MF_AUTHENT));
506
507 RTOS::delay(RESPONSE_WAIT);
508
509 return (readRegister(Register::STATUS2) & STATUS2_CRYPTO1_ON) != 0U;
510 }
