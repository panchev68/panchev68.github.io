96 bool detectCard() {
97 chipset.setRegisterBitMask(Chipset::Register::TX_CONTROL, ANTENNA_ENABLE);
98
99 const uint8_t command[] = {static_cast<uint8_t>(Command::REQA)};
100 uint8_t backData[RESPONSE_BUFFER_SIZE] = {};
101 uint8_t backLen = 0U;
102
103 const bool detected =
104 chipset.transceive(command, sizeof(command), backData, static_cast<uint8_t>(sizeof(backData)), backLen, REQA_BIT_FRAMING) && (backLen >= MIN_REQA_RESPONSE);
105 if (!detected) {
106 chipset.clearRegisterBitMask(Chipset::Register::TX_CONTROL, ANTENNA_ENABLE);
107 }
108 return detected;
109 }

121 bool selectCard() {
122 const uint8_t anticollisionCommand[] = {static_cast<uint8_t>(Command::SEL_CL1), static_cast<uint8_t>(Command::ANTICOLLISION_NVB)};
123 uint8_t anticollisionResponse[RESPONSE_BUFFER_SIZE] = {};
124 uint8_t anticollisionLen = 0U;
125
126 if (!chipset.transceive(anticollisionCommand, sizeof(anticollisionCommand), anticollisionResponse,
127 static_cast<uint8_t>(sizeof(anticollisionResponse)), anticollisionLen) ||
128 anticollisionLen < MIN_ANTICOLLISION_RESPONSE) {
129 return false;
130 }
131
132 const uint8_t bcc = static_cast<uint8_t>(anticollisionResponse[0] ^ anticollisionResponse[1] ^ anticollisionResponse[2] ^ anticollisionResponse[3]);
133 if (bcc != anticollisionResponse[UID_EXTRACT_SIZE]) {
134 return false;
135 }
136
137 uint8_t selectCommand[SELECT_COMMAND_SIZE] = {static_cast<uint8_t>(Command::SEL_CL1), static_cast<uint8_t>(Command::SELECT_NVB)};
138 std::copy(anticollisionResponse, anticollisionResponse + UID_EXTRACT_SIZE + 1U, selectCommand + 2);
139
140 uint8_t selectResponse[RESPONSE_BUFFER_SIZE] = {};
141 uint8_t selectLen = 0U;
142
143 chipset.enableCRC(true);
144 const bool selected = chipset.transceive(selectCommand, sizeof(selectCommand), selectResponse,
145 static_cast<uint8_t>(sizeof(selectResponse)), selectLen);
146 chipset.enableCRC(false);
147
148 if (!selected || selectLen < MIN_SELECT_RESPONSE) {
149 return false;
150 }
151
152 uid.size = UID_EXTRACT_SIZE;
153 std::copy(anticollisionResponse + UID_EXTRACT_OFFSET, anticollisionResponse + UID_EXTRACT_OFFSET + UID_EXTRACT_SIZE, uid.uidByte.begin());
154 uid.sak = selectResponse[0];
155 return true;
156 }

177 Type getCardType() const {
178 const uint8_t sak = uid.sak;
179
180 if ((sak & SAK_ISO14443_4_MASK) != 0U) {
181 return Type::ISO_14443_4;
182 }
183
184 switch (static_cast<Sak>(sak)) {
185 case Sak::MIFARE_MINI:
186 return Type::MIFARE_MINI;
187 case Sak::MIFARE_1K:
188 return Type::MIFARE_1K;
189 case Sak::MIFARE_4K:
190 return Type::MIFARE_4K;
191 case Sak::MIFARE_UL:
192 return Type::MIFARE_UL;
193 default:
194 return Type::UNKNOWN;
195 }
196 }

213 bool authenticate(uint8_t blockAddress, bool keyType, const uint8_t * key) {
214 if (key == nullptr) {
215 assert(!"Authentication key pointer must not be null");
216 return false;
217 }
218
219 const Type cardType = getCardType();
220 const uint16_t blockCount = (cardType == Type::MIFARE_1K) ? MIFARE_1K_BLOCKS : (cardType == Type::MIFARE_4K) ? MIFARE_4K_BLOCKS : MIFARE_MINI_BLOCKS;
221 const uint16_t maxBlockIndex = blockCount - 1U;
222 const uint8_t maxBlock = static_cast<uint8_t>(std::min(maxBlockIndex, static_cast<uint16_t>(std::numeric_limits<uint8_t>::max())));
223 const uint8_t clampedBlock = (blockAddress > maxBlock) ? maxBlock : blockAddress;
224
225 uint8_t authData[AUTH_DATA_SIZE] = {};
226 authData[0] = keyType ? static_cast<uint8_t>(Command::AUTH_KEY_A) : static_cast<uint8_t>(Command::AUTH_KEY_B);
227 authData[1] = clampedBlock;
228 std::copy(key, key + AUTH_KEY_SIZE, authData + 2);
229 std::copy(uid.uidByte.begin(), uid.uidByte.begin() + UID_EXTRACT_SIZE, authData + 2 + AUTH_KEY_SIZE);
230
231 return chipset.executeAuthentication(authData, AUTH_DATA_SIZE);
232 }

253 bool halt() {
254 const uint8_t haltCommand[] = {static_cast<uint8_t>(Command::HALT_1), static_cast<uint8_t>(Command::HALT_2)};
255 uint8_t haltResponse[RESPONSE_BUFFER_SIZE] = {};
256 uint8_t responseLen = 0U;
257
258 // HLTA is CRC protected per ISO/IEC 14443-3; the hardware appends the CRC_A while CRC is enabled.
259 chipset.enableCRC(true);
260 // The PICC does not acknowledge HALT; a timeout on this exchange is the expected, successful outcome.
261 chipset.transceive(haltCommand, sizeof(haltCommand), haltResponse, static_cast<uint8_t>(sizeof(haltResponse)), responseLen);
262 chipset.enableCRC(false);
263 chipset.clearRegisterBitMask(Chipset::Register::TX_CONTROL, ANTENNA_ENABLE);
264 return true;
265 }
