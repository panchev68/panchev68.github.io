Lib::Helper::Convert::Bcd

Demonstrates converting BCD values to string, ASCII digit and numeric decimal value.

Demonstrates converting BCD values to string, ASCII digit and numeric decimal value.This example shows typical uses of Bcd::toString, Bcd::toAsc, and Bcd::toHex.

#include <iostream>
#include "Bcd.hpp"
int main() {
// Example 1: Convert 0x1234 (BCD -> "1234")
uint16_t bcdValue = 0x1234;
std::string s = Lib::Helper::Convert::Bcd::toString(bcdValue);
std::cout << "toString(0x1234) = " << s << '\n';
// Example 2: Single digit to ASCII
char ch = Lib::Helper::Convert::Bcd::toAsc(0x05);
std::cout << "toAsc(0x05) = " << ch << '\n';
// Example 3: BCD to numeric decimal value
uint16_t num = Lib::Helper::Convert::Bcd::toHex(0x0102); // 0x01 0x02 -> 102
std::cout << "toHex(0x0102) = " << num << '\n';
return 0;
}
