156 if (!timeout.isExpired()) return;
157 timeout.set(updatePeriod);
158 uint16_t adcValue = 0;

160 adcValue = integrator.processValue(adcValue);
161 if (!integrator.isWindowFull()) {
162 restartConversion();
163 return;
164 }
165 float newVoltage = interpolator.interpolate(adcValue);
166 if (newVoltage != currentVoltage) {
167 currentVoltage = newVoltage;
168 handleValueChanged(newVoltage);

170 restartConversion();
