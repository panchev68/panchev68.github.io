Example usage of the CRC16 calculator.

#include <Lib/Helper/crc16.hpp>
#include <vector>
std::vector<uint8_t> bytes = {1, 2, 3, 4, 5, 6};
Lib::Helper::crc16 calc1(bytes.data(), bytes.size());
uint16_t result1 = calc1;
Lib::Helper::crc16 calc2(bytes);
uint16_t result2 = calc2; // result2 == result1
